    <canvas id="stage" width="960" height="540"></canvas>
    <script>
        const canvas = document.getElementById('stage');
        const ctx = canvas.getContext('2d');

        // Seeded random for deterministic fur/noise
        class SeededRandom {
            constructor(seed) { this.seed = seed; }
            next() {
                this.seed = (this.seed * 9301 + 49297) % 233280;
                return this.seed / 233280;
            }
        }

        // Vector math
        const Vec = {
            add: (a, b) => ({ x: a.x + b.x, y: a.y + b.y }),
            sub: (a, b) => ({ x: a.x - b.x, y: a.y - b.y }),
            mul: (v, s) => ({ x: v.x * s, y: v.y * s }),
            lerp: (a, b, t) => ({ x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t }),
            dist: (a, b) => Math.hypot(b.x - a.x, b.y - a.y),
            norm: (v) => {
                const len = Math.hypot(v.x, v.y);
                return len > 0 ? { x: v.x / len, y: v.y / len } : v;
            }
        };

        // Animation easing
        const ease = {
            linear: t => t,
            easeInQuad: t => t * t,
            easeOutQuad: t => 1 - (1 - t) * (1 - t),
            easeInOutQuad: t => t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t,
            easeOutCubic: t => 1 + (--t) * t * t,
            easeInOutCubic: t => t < 0.5 ? 4 * t * t * t : (t - 1) * (2 * (t - 2)) * (2 * (t - 2)) + 1,
            smoothstep: (t) => t * t * (3 - 2 * t)
        };

        // Pomeranian dog structure
        class Pomeranian {
            constructor() {
                this.pos = { x: 150, y: 380 };
                this.scale = 1;
                this.rotation = 0; // Radians
                this.bodyAngle = 0; // Spine tilt
                this.headTilt = 0;
                this.tailCurl = 0;
                this.legPhases = [0, 0.5, 1, 0.5]; // FL, HR, FR, HL
                this.eyesOpen = 1;
                this.mouthOpen = 0;
                this.boneInMouth = false;
                this.boneGrip = { x: 0, y: 0 };
            }

            draw(ctx, t) {
                ctx.save();
                ctx.translate(this.pos.x, this.pos.y);
                ctx.rotate(this.rotation);
                ctx.scale(this.scale, this.scale);

                // Draw body (ellipse with fur texture)
                this.drawBody(ctx, t);
                this.drawLegs(ctx, t);
                this.drawTail(ctx, t);
                this.drawHead(ctx, t);
                this.drawEars(ctx, t);

                ctx.restore();
            }

            drawBody(ctx, t) {
                // Main body ellipse
                ctx.save();
                ctx.rotate(this.bodyAngle);
                
                // Fur base
                const grad = ctx.createLinearGradient(-35, -15, -35, 15);
                grad.addColorStop(0, '#e8c59a');
                grad.addColorStop(0.5, '#d4a574');
                grad.addColorStop(1, '#c89658');
                
                ctx.fillStyle = grad;
                ctx.beginPath();
                ctx.ellipse(0, 0, 45, 22, 0, 0, Math.PI * 2);
                ctx.fill();

                // Chest ruff - distinctive Pom feature
                ctx.fillStyle = '#e8c59a';
                ctx.beginPath();
                for (let i = 0; i < 12; i++) {
                    const angle = (i / 12) * Math.PI * 2;
                    const x = Math.cos(angle) * 50;
                    const y = Math.sin(angle) * 30;
                    if (i === 0) ctx.moveTo(x, y);
                    else ctx.lineTo(x, y);
                }
                ctx.closePath();
                ctx.fill();

                // Fur texture details
                ctx.strokeStyle = 'rgba(0, 0, 0, 0.08)';
                ctx.lineWidth = 0.5;
                for (let i = 0; i < 20; i++) {
                    const rng = new SeededRandom(t * 100 + i);
                    const x = (rng.next() - 0.5) * 80;
                    const y = (rng.next() - 0.5) * 40;
                    const len = 3 + rng.next() * 4;
                    const angle = rng.next() * Math.PI * 2;
                    ctx.beginPath();
                    ctx.moveTo(x, y);
                    ctx.lineTo(x + Math.cos(angle) * len, y + Math.sin(angle) * len);
                    ctx.stroke();
                }

                ctx.restore();
            }

            drawLegs(ctx, t) {
                const legLength = 20;
                const legWidth = 6;
                const positions = [
                    { x: -15, y: 15 }, // Front left
                    { x: 15, y: 15 },  // Front right
                    { x: -20, y: 15 }, // Hind left
                    { x: 20, y: 15 }   // Hind right
                ];

                positions.forEach((pos, i) => {
                    ctx.save();
                    ctx.translate(pos.x, pos.y);
                    
                    const phase = this.legPhases[i];
                    const legBend = Math.sin(phase * Math.PI) * 8;
                    
                    // Upper leg
                    ctx.fillStyle = '#d4a574';
                    ctx.beginPath();
                    ctx.moveTo(-legWidth / 2, 0);
                    ctx.lineTo(legWidth / 2, 0);
                    ctx.lineTo(legWidth / 2 - 1, legLength - legBend);
                    ctx.lineTo(-legWidth / 2 + 1, legLength - legBend);
                    ctx.closePath();
                    ctx.fill();

                    // Paw
                    ctx.fillStyle = '#c89658';
                    ctx.beginPath();
                    ctx.ellipse(0, legLength - legBend, 4, 5, 0, 0, Math.PI * 2);
                    ctx.fill();

                    ctx.restore();
                });
            }

            drawTail(ctx, t) {
                ctx.save();
                
                const tailX = 40;
                const tailY = -8;
                ctx.translate(tailX, tailY);
                
                // Curl over the back - Pom characteristic
                const curl = Math.PI * 1.2 + this.tailCurl;
                const tailRadius = 20;
                
                ctx.strokeStyle = '#d4a574';
                ctx.lineWidth = 14;
                ctx.lineCap = 'round';
                ctx.lineJoin = 'round';
                ctx.beginPath();
                
                for (let i = 0; i <= 20; i++) {
                    const angle = (i / 20) * curl - Math.PI / 2;
                    const r = tailRadius * (1 + i / 20 * 0.3);
                    const x = Math.cos(angle) * r;
                    const y = Math.sin(angle) * r;
                    if (i === 0) ctx.moveTo(x, y);
                    else ctx.lineTo(x, y);
                }
                ctx.stroke();

                // Tail tip highlight
                ctx.strokeStyle = '#e8c59a';
                ctx.lineWidth = 7;
                ctx.beginPath();
                const endAngle = curl - Math.PI / 2;
                const endR = tailRadius * 1.3;
                ctx.moveTo(Math.cos(endAngle) * endR, Math.sin(endAngle) * endR);
                ctx.lineTo(Math.cos(endAngle) * (endR + 8), Math.sin(endAngle) * (endR + 8));
                ctx.stroke();

                ctx.restore();
            }

            drawHead(ctx, t) {
                ctx.save();
                ctx.translate(-45, -10);
                ctx.rotate(this.headTilt);

                // Head shape - rounded with small pointed muzzle
                const grad = ctx.createRadialGradient(0, 0, 5, 0, 0, 20);
                grad.addColorStop(0, '#e8c59a');
                grad.addColorStop(1, '#d4a574');

                ctx.fillStyle = grad;
                ctx.beginPath();
                ctx.ellipse(0, 0, 18, 16, 0, 0, Math.PI * 2);
                ctx.fill();

                // Muzzle - small pointed area
                ctx.fillStyle = '#d4a574';
                ctx.beginPath();
                ctx.ellipse(-20, 0, 10, 8, 0, 0, Math.PI * 2);
                ctx.fill();

                // Eyes - small, dark, expressive
                ctx.fillStyle = '#1a1a1a';
                ctx.beginPath();
                ctx.ellipse(-8, -6, 3 * this.eyesOpen, 4 * this.eyesOpen, 0, 0, Math.PI * 2);
                ctx.fill();

                ctx.fillStyle = '#ffffff';
                ctx.beginPath();
                ctx.ellipse(-7, -7, 1.2, 1.5, 0, 0, Math.PI * 2);
                ctx.fill();

                // Nose
                ctx.fillStyle = '#3a3a3a';
                ctx.beginPath();
                ctx.ellipse(-24, 0, 2.5, 2, 0, 0, Math.PI * 2);
                ctx.fill();

                // Mouth
                ctx.strokeStyle = '#3a3a3a';
                ctx.lineWidth = 0.8;
                ctx.beginPath();
                ctx.moveTo(-24, 0);
                ctx.lineTo(-24, 2 + this.mouthOpen * 3);
                ctx.stroke();

                ctx.restore();
            }

            drawEars(ctx, t) {
                // Small, upright triangular ears - Pom characteristic
                ctx.save();
                
                const earPositions = [
                    { x: -25, y: -20, flip: 1 },
                    { x: -15, y: -20, flip: -1 }
                ];

                earPositions.forEach(pos => {
                    ctx.save();
                    ctx.translate(pos.x, pos.y);
                    ctx.scale(pos.flip, 1);
                    ctx.rotate(this.headTilt * 0.3);

                    ctx.fillStyle = '#d4a574';
                    ctx.beginPath();
                    ctx.moveTo(0, 0);
                    ctx.lineTo(-5, -12);
                    ctx.lineTo(3, -10);
                    ctx.closePath();
                    ctx.fill();

                    ctx.fillStyle = '#c89658';
                    ctx.beginPath();
                    ctx.moveTo(-1, -2);
                    ctx.lineTo(-3, -10);
                    ctx.lineTo(1, -8);
                    ctx.closePath();
                    ctx.fill();

                    ctx.restore();
                });

                ctx.restore();
            }
        }

        // Bone object
        class Bone {
            constructor() {
                this.x = 750;
                this.y = 320;
                this.rotation = Math.PI / 6;
                this.carried = false;
            }

            draw(ctx) {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.rotation);

                // Main bone shaft
                ctx.fillStyle = '#d4a574';
                ctx.fillRect(-25, -3, 50, 6);

                // Left knob
                ctx.fillStyle = '#c89658';
                ctx.beginPath();
                ctx.ellipse(-28, 0, 6, 8, 0, 0, Math.PI * 2);
                ctx.fill();

                // Right knob
                ctx.beginPath();
                ctx.ellipse(28, 0, 6, 8, 0, 0, Math.PI * 2);
                ctx.fill();

                // Highlight
                ctx.strokeStyle = 'rgba(255, 255, 255, 0.3)';
                ctx.lineWidth = 1;
                ctx.beginPath();
                ctx.moveTo(-15, -2);
                ctx.lineTo(15, -2);
                ctx.stroke();

                ctx.restore();
            }
        }

        // Scene state
        const dog = new Pomeranian();
        const bone = new Bone();
        const puddle = {
            x: 400,
            y: 420,
            width: 120,
            height: 60
        };

        function renderFrame(t) {
            // Clamp time
            t = Math.max(0, Math.min(12, t));

            ctx.fillStyle = '#87ceeb';
            ctx.fillRect(0, 0, 960, 540);

            // Draw yard
            ctx.fillStyle = '#7cb342';
            ctx.fillRect(0, 430, 960, 110);

            // Draw ground line
            ctx.strokeStyle = 'rgba(0, 0, 0, 0.1)';
            ctx.lineWidth = 1;
            ctx.beginPath();
            ctx.moveTo(0, 430);
            ctx.lineTo(960, 430);
            ctx.stroke();

            // Draw puddle
            ctx.save();
            const pudGrad = ctx.createLinearGradient(puddle.x - puddle.width/2, puddle.y - puddle.height/2, 
                                                     puddle.x - puddle.width/2, puddle.y + puddle.height/2);
            pudGrad.addColorStop(0, 'rgba(100, 180, 220, 0.6)');
            pudGrad.addColorStop(1, 'rgba(80, 150, 200, 0.5)');
            ctx.fillStyle = pudGrad;
            ctx.beginPath();
            ctx.ellipse(puddle.x, puddle.y, puddle.width/2, puddle.height/2, 0, 0, Math.PI * 2);
            ctx.fill();

            // Puddle rim shadow
            ctx.strokeStyle = 'rgba(0, 0, 0, 0.15)';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.ellipse(puddle.x, puddle.y, puddle.width/2, puddle.height/2, 0, 0, Math.PI * 2);
            ctx.stroke();

            // Draw doghouse
            const dogHouseX = 750;
            const dogHouseY = 300;
            const dogHouseW = 150;
            const dogHouseH = 120;

            ctx.fillStyle = '#8b6914';
            ctx.fillRect(dogHouseX - dogHouseW/2, dogHouseY, dogHouseW, dogHouseH);

            // Roof
            ctx.fillStyle = '#6b4910';
            ctx.beginPath();
            ctx.moveTo(dogHouseX - dogHouseW/2 - 10, dogHouseY);
            ctx.lineTo(dogHouseX, dogHouseY - 40);
            ctx.lineTo(dogHouseX + dogHouseW/2 + 10, dogHouseY);
            ctx.closePath();
            ctx.fill();

            // Doorway
            const doorX = dogHouseX;
            const doorY = dogHouseY + 30;
            const doorW = 50;
            const doorH = 70;
            ctx.fillStyle = '#3a2a0a';
            ctx.fillRect(doorX - doorW/2, doorY, doorW, doorH);

            // Shadow in doorway
            const doorShadow = ctx.createLinearGradient(doorX - doorW/2, doorY, doorX - doorW/2, doorY + doorH);
            doorShadow.addColorStop(0, 'rgba(0, 0, 0, 0.3)');
            doorShadow.addColorStop(1, 'rgba(0, 0, 0, 0.7)');
            ctx.fillStyle = doorShadow;
            ctx.fillRect(doorX - doorW/2, doorY, doorW, doorH);

            // ============ ANIMATION TIMELINE ============

            if (t >= 0 && t < 1) {
                // 0-1s: Establish scene, dog notices bone
                const phase = t;
                dog.pos = { x: 150, y: 380 };
                dog.rotation = 0;
                dog.eyesOpen = 1;
                dog.legPhases = [0, 0.5, 1, 0.5];
            }
            else if (t >= 1 && t < 3) {
                // 1-3s: Run toward doghouse, jump puddle
                const phase = (t - 1) / 2;

                if (phase < 0.65) {
                    // Approach puddle
                    const dist = 400 - 150;
                    dog.pos.x = 150 + phase / 0.65 * (puddle.x - 80 - 150);
                    dog.pos.y = 380;

                    // Running gait
                    const gaitPhase = phase * 8 % 1;
                    dog.legPhases = [
                        (gaitPhase + 0) % 1,
                        (gaitPhase + 0.5) % 1,
                        (gaitPhase + 0.5) % 1,
                        (gaitPhase + 0) % 1
                    ];

                    // Head bob
                    dog.headTilt = Math.sin(gaitPhase * Math.PI * 2) * 0.1;
                    dog.bodyAngle = Math.sin(gaitPhase * Math.PI * 2) * 0.05;
                }
                else {
                    // Jump phase
                    const jumpPhase = (phase - 0.65) / 0.35;
                    const jumpArc = Math.sin(jumpPhase * Math.PI) * 60;
                    
                    dog.pos.x = puddle.x - 80 + jumpPhase * 160;
                    dog.pos.y = 380 - jumpArc;

                    // Tuck legs during jump
                    dog.legPhases = [0.5, 0.5, 0.5, 0.5];
                    dog.bodyAngle = (jumpPhase - 0.5) * 0.2;
                }
            }
            else if (t >= 3 && t < 4.5) {
                // 3-4.5s: Enter doghouse
                const phase = (t - 3) / 1.5;
                
                dog.pos.x = (puddle.x + 160) + phase * (dogHouseX - 30 - (puddle.x + 160));
                dog.pos.y = 380;

                // Slow gait
                const gaitPhase = phase * 3 % 1;
                dog.legPhases = [
                    (gaitPhase + 0) % 1,
                    (gaitPhase + 0.5) % 1,
                    (gaitPhase + 0.5) % 1,
                    (gaitPhase + 0) % 1
                ];

                if (phase > 0.6) {
                    // Entering doorway, head lowers
                    const enterPhase = (phase - 0.6) / 0.4;
                    dog.pos.y = 380 - enterPhase * 40;
                    dog.pos.x = dogHouseX - 30 + enterPhase * 20;
                }
            }
            else if (t >= 4.5 && t < 6) {
                // 4.5-6s: Retrieve bone inside
                const phase = (t - 4.5) / 1.5;

                // Dog inside, mostly hidden
                dog.pos.x = dogHouseX + 20;
                dog.pos.y = dogHouseY + 60;
                dog.pos.x += (Math.random() > 0.5 ? 1 : -1) * Math.sin(phase * Math.PI) * 5; // Shuffle

                // Head down searching
                dog.headTilt = -0.4 + Math.sin(phase * Math.PI * 2) * 0.1;
                dog.mouthOpen = Math.sin(phase * Math.PI) * 0.5;

                // Bone retrieval
                if (phase > 0.3) {
                    bone.carried = true;
                    const gripPhase = (phase - 0.3) / 0.7;
                    bone.x = 750 + gripPhase * (dogHouseX - 750 * 0.5);
                    bone.y = 320 - gripPhase * 70;
                    bone.rotation = Math.PI / 4 + gripPhase * 0.3;
                }
            }
            else if (t >= 6 && t < 8) {
                // 6-8s: Back out with bone
                const phase = (t - 6) / 2;

                // Back out
                dog.pos.x = dogHouseX + 20 - phase * (dogHouseX + 20 - (puddle.x + 200));
                dog.pos.y = dogHouseY + 60 - phase * (dogHouseY + 60 - 380);

                // Reverse walk
                const gaitPhase = phase * 4 % 1;
                dog.legPhases = [
                    (0.5 - gaitPhase + 0) % 1,
                    (0.5 - gaitPhase + 0.5) % 1,
                    (0.5 - gaitPhase + 0.5) % 1,
                    (0.5 - gaitPhase + 0) % 1
                ];

                dog.rotation = Math.PI + Math.sin(phase * Math.PI * 2) * 0.1;
                dog.headTilt = 0.3;
                dog.mouthOpen = 0.7;

                // Bone in mouth
                const mouthX = dog.pos.x - 40;
                const mouthY = dog.pos.y - 10;
                bone.x = mouthX;
                bone.y = mouthY;
                bone.rotation = Math.PI / 4;
            }
            else if (t >= 8 && t < 10) {
                // 8-10s: Turn to face camera
                const phase = (t - 8) / 2;

                const turnAngle = Math.PI + phase * Math.PI;
                
                dog.rotation = turnAngle;
                dog.pos.x = puddle.x + 250;
                dog.pos.y = 380;

                // Pivot steps
                const stepPhase = phase * 2 % 1;
                dog.legPhases = [
                    (stepPhase + 0.2) % 1,
                    (stepPhase + 0.7) % 1,
                    (stepPhase + 0.7) % 1,
                    (stepPhase + 0.2) % 1
                ];

                dog.headTilt = -0.2 * Math.sin(phase * Math.PI);
                dog.bodyAngle = 0.1 * Math.sin(phase * Math.PI * 2);

                // Bone follows
                const mouthX = dog.pos.x + Math.cos(turnAngle - Math.PI / 6) * 40;
                const mouthY = dog.pos.y - 10;
                bone.x = mouthX;
                bone.y = mouthY;
                bone.rotation = Math.PI / 4 - phase * 0.3;
            }
            else if (t >= 10) {
                // 10-12s: Hold proud front-facing pose
                const phase = (t - 10) / 2;

                dog.rotation = 0; // Face camera
                dog.pos.x = 540;
                dog.pos.y = 380;

                // Subtle breathing
                dog.bodyAngle = Math.sin(phase * Math.PI * 2) * 0.05;
                dog.headTilt = Math.sin(phase * Math.PI * 2 + 0.5) * 0.08;
                dog.tailCurl = Math.sin(phase * Math.PI * 2) * 0.1;
                dog.mouthOpen = 0.3 + Math.sin(phase * Math.PI * 2) * 0.1;

                // Bone in mouth
                bone.x = dog.pos.x - 20;
                bone.y = dog.pos.y - 5;
                bone.rotation = 0.1;
            }

            // Draw puddle reflection (only where puddle exists)
            ctx.save();
            ctx.globalAlpha = 0.35;
            
            // Create clipping region for puddle
            ctx.beginPath();
            ctx.ellipse(puddle.x, puddle.y, puddle.width/2, puddle.height/2, 0, 0, Math.PI * 2);
            ctx.clip();

            // Draw reflected dog (flipped vertically)
            ctx.save();
            const reflectDog = new Pomeranian();
            Object.assign(reflectDog, dog);
            reflectDog.pos.y = puddle.y + (puddle.y - dog.pos.y);
            reflectDog.scale *= -1; // Flip
            reflectDog.draw(ctx, t);
            ctx.restore();

            ctx.restore();

            // Draw dog
            dog.draw(ctx, t);

            // Draw bone
            if (!bone.carried || t < 4.5) {
                bone.draw(ctx);
            } else if (bone.carried && t >= 4.5) {
                bone.draw(ctx);
            }

            // Contact shadows
            ctx.fillStyle = 'rgba(0, 0, 0, 0.15)';
            ctx.beginPath();
            ctx.ellipse(dog.pos.x, 400, 30, 8, 0, 0, Math.PI * 2);
            ctx.fill();
        }

        // Initial render
        renderFrame(0);
        window.renderFrame = renderFrame;
    </script>
</body>

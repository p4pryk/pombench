    <canvas id="stage" width="960" height="540"></canvas>
    <script>
        const canvas = document.getElementById('stage');
        const ctx = canvas.getContext('2d');

        // Deterministic noise function
        function seededRandom(seed) {
            const x = Math.sin(seed) * 10000;
            return x - Math.floor(x);
        }

        // Dog skeleton and geometry
        class PomeranianDog {
            constructor() {
                this.bodyLength = 35;
                this.bodyHeight = 25;
                this.headRadius = 12;
                this.neckRuff = 5;
                this.legLength = 20;
                this.legThickness = 4;
                this.pawRadius = 3;
                this.earLength = 10;
                this.earWidth = 5;
                this.tailCurve = 0.7;
            }

            getHeadPosition(bodyX, bodyY, bodyRotation) {
                const headOffsetX = Math.cos(bodyRotation) * (this.bodyLength * 0.4 - 5);
                const headOffsetY = Math.sin(bodyRotation) * (this.bodyLength * 0.4 - 5);
                return {
                    x: bodyX + headOffsetX,
                    y: bodyY + headOffsetY - 3
                };
            }

            getLegPositions(bodyX, bodyY, bodyRotation, gaitPhase, isReverse) {
                const legSpacing = this.bodyLength * 0.25;
                const positions = [];
                
                const frontLeftX = bodyX + Math.cos(bodyRotation) * (legSpacing * 0.8);
                const frontLeftY = bodyY + Math.sin(bodyRotation) * (legSpacing * 0.8);
                
                const frontRightX = bodyX + Math.cos(bodyRotation) * (legSpacing * 0.8);
                const frontRightY = bodyY + Math.sin(bodyRotation) * (legSpacing * 0.8) + 8;
                
                const backLeftX = bodyX - Math.cos(bodyRotation) * (legSpacing * 0.5);
                const backLeftY = bodyY - Math.sin(bodyRotation) * (legSpacing * 0.5);
                
                const backRightX = bodyX - Math.cos(bodyRotation) * (legSpacing * 0.5);
                const backRightY = bodyY - Math.sin(bodyRotation) * (legSpacing * 0.5) + 8;

                // Gait animation
                const stride = Math.sin(gaitPhase * Math.PI) * 3;
                const bounce = Math.abs(Math.sin(gaitPhase * Math.PI)) * 2;

                positions.push({
                    x: frontLeftX + (isReverse ? stride : -stride),
                    y: frontLeftY - bounce
                });
                positions.push({
                    x: frontRightX - (isReverse ? stride : -stride),
                    y: frontRightY - bounce
                });
                positions.push({
                    x: backLeftX - (isReverse ? -stride : stride),
                    y: backLeftY - bounce * 0.5
                });
                positions.push({
                    x: backRightX + (isReverse ? -stride : stride),
                    y: backRightY - bounce * 0.5
                });

                return positions;
            }

            getTailPosition(bodyX, bodyY, bodyRotation, tailWag) {
                const tailBaseX = bodyX - Math.cos(bodyRotation) * this.bodyLength * 0.3;
                const tailBaseY = bodyY - Math.sin(bodyRotation) * this.bodyLength * 0.3;
                
                const tailCurveAmount = this.tailCurve + Math.sin(tailWag) * 0.2;
                const tailTipX = tailBaseX + Math.cos(bodyRotation - Math.PI * 0.5) * this.bodyLength * tailCurveAmount;
                const tailTipY = tailBaseY + Math.sin(bodyRotation - Math.PI * 0.5) * this.bodyLength * tailCurveAmount;
                
                return { baseX: tailBaseX, baseY: tailBaseY, tipX: tailTipX, tipY: tailTipY };
            }

            drawFur(ctx, x, y, rotation, alpha) {
                ctx.save();
                ctx.translate(x, y);
                ctx.rotate(rotation);
                ctx.globalAlpha = alpha;

                // Body fur - light orange/cream with texture
                const bodyGradient = ctx.createRadialGradient(-5, -3, 5, 0, 0, 25);
                bodyGradient.addColorStop(0, '#f5deb3');
                bodyGradient.addColorStop(0.5, '#ff9d6f');
                bodyGradient.addColorStop(1, '#ff8c42');
                ctx.fillStyle = bodyGradient;
                ctx.beginPath();
                ctx.ellipse(0, 0, this.bodyLength, this.bodyHeight + this.neckRuff, 0, 0, Math.PI * 2);
                ctx.fill();

                // Neck ruff
                ctx.fillStyle = '#ffc29c';
                ctx.beginPath();
                ctx.ellipse(this.bodyLength * 0.35, -2, this.bodyLength * 0.25, this.bodyHeight * 0.7, 0, 0, Math.PI * 2);
                ctx.fill();

                ctx.restore();
            }

            drawHead(ctx, x, y, bodyRotation, mouthOpen, boneInMouth, alpha) {
                ctx.save();
                ctx.globalAlpha = alpha;

                const headPos = this.getHeadPosition(x, y, bodyRotation);
                ctx.translate(headPos.x, headPos.y);
                ctx.rotate(bodyRotation + Math.PI * 0.1);

                // Head base
                const headGradient = ctx.createRadialGradient(-2, -2, 3, 0, 0, this.headRadius);
                headGradient.addColorStop(0, '#ffc29c');
                headGradient.addColorStop(1, '#ff8c42');
                ctx.fillStyle = headGradient;
                ctx.beginPath();
                ctx.ellipse(0, 0, this.headRadius, this.headRadius * 0.95, 0, 0, Math.PI * 2);
                ctx.fill();

                // Muzzle
                ctx.fillStyle = '#f5deb3';
                ctx.beginPath();
                ctx.ellipse(this.headRadius * 0.5, 0, this.headRadius * 0.4, this.headRadius * 0.35, 0, 0, Math.PI * 2);
                ctx.fill();

                // Nose
                ctx.fillStyle = '#333';
                ctx.beginPath();
                ctx.ellipse(this.headRadius * 0.65, -1, 2, 1.5, 0, 0, Math.PI * 2);
                ctx.fill();

                // Eyes
                ctx.fillStyle = '#000';
                ctx.beginPath();
                ctx.arc(-this.headRadius * 0.2, -this.headRadius * 0.25, 2, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = '#fff';
                ctx.beginPath();
                ctx.arc(-this.headRadius * 0.15, -this.headRadius * 0.3, 0.8, 0, Math.PI * 2);
                ctx.fill();

                // Ears
                ctx.fillStyle = '#ff8c42';
                for (let side of [-1, 1]) {
                    ctx.beginPath();
                    ctx.ellipse(side * this.headRadius * 0.3, -this.headRadius * 0.4, 
                               this.earWidth, this.earLength, 
                               Math.PI * 0.15 * side, 0, Math.PI * 2);
                    ctx.fill();
                }

                // Mouth opening
                if (mouthOpen > 0.1) {
                    ctx.strokeStyle = '#8b6914';
                    ctx.lineWidth = 1;
                    ctx.beginPath();
                    ctx.moveTo(this.headRadius * 0.5, 0);
                    ctx.quadraticCurveTo(this.headRadius * 0.7, mouthOpen * 3, this.headRadius * 0.5, mouthOpen * 2);
                    ctx.stroke();
                }

                ctx.restore();
            }

            drawLegs(ctx, bodyX, bodyY, bodyRotation, legPositions, alpha) {
                ctx.save();
                ctx.globalAlpha = alpha;
                ctx.strokeStyle = '#ff9d6f';
                ctx.lineWidth = this.legThickness;
                ctx.lineCap = 'round';

                for (let i = 0; i < legPositions.length; i++) {
                    const leg = legPositions[i];
                    ctx.beginPath();
                    const baseY = bodyY + (i < 2 ? 0 : 4);
                    ctx.moveTo(leg.x, baseY);
                    ctx.lineTo(leg.x, leg.y + this.legLength);
                    ctx.stroke();

                    // Paw
                    ctx.fillStyle = '#ff8c42';
                    ctx.beginPath();
                    ctx.ellipse(leg.x, leg.y + this.legLength, this.pawRadius, this.pawRadius * 0.8, 0, 0, Math.PI * 2);
                    ctx.fill();
                }

                ctx.restore();
            }

            drawTail(ctx, tailPos, bodyRotation, tailWag, alpha) {
                ctx.save();
                ctx.globalAlpha = alpha;

                // Tail fur - plumed effect
                const tailGradient = ctx.createLinearGradient(tailPos.baseX, tailPos.baseY, tailPos.tipX, tailPos.tipY);
                tailGradient.addColorStop(0, '#ffc29c');
                tailGradient.addColorStop(1, '#ff9d6f');

                ctx.strokeStyle = tailGradient;
                ctx.lineWidth = 12;
                ctx.lineCap = 'round';

                // Curved tail path
                const controlX = tailPos.baseX + Math.cos(bodyRotation - Math.PI * 0.4) * 15;
                const controlY = tailPos.baseY + Math.sin(bodyRotation - Math.PI * 0.4) * 15;

                ctx.beginPath();
                ctx.moveTo(tailPos.baseX, tailPos.baseY);
                ctx.quadraticCurveTo(controlX, controlY, tailPos.tipX, tailPos.tipY);
                ctx.stroke();

                ctx.restore();
            }
        }

        class Bone {
            constructor() {
                this.length = 18;
                this.thickness = 4;
                this.endRadius = 3;
            }

            draw(ctx, x, y, rotation, alpha) {
                ctx.save();
                ctx.translate(x, y);
                ctx.rotate(rotation);
                ctx.globalAlpha = alpha;

                // Main shaft
                ctx.fillStyle = '#d4a574';
                ctx.beginPath();
                ctx.moveTo(-this.length * 0.4, -this.thickness * 0.5);
                ctx.lineTo(this.length * 0.4, -this.thickness * 0.5);
                ctx.lineTo(this.length * 0.4, this.thickness * 0.5);
                ctx.lineTo(-this.length * 0.4, this.thickness * 0.5);
                ctx.fill();

                // Rounded ends
                ctx.fillStyle = '#c99a6e';
                ctx.beginPath();
                ctx.arc(-this.length * 0.4, 0, this.endRadius, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.arc(this.length * 0.4, 0, this.endRadius, 0, Math.PI * 2);
                ctx.fill();

                // Shadow detail
                ctx.strokeStyle = '#8b6914';
                ctx.lineWidth = 0.5;
                ctx.beginPath();
                ctx.moveTo(-this.length * 0.3, -this.thickness * 0.3);
                ctx.lineTo(this.length * 0.3, -this.thickness * 0.3);
                ctx.stroke();

                ctx.restore();
            }
        }

        class DogHouse {
            constructor() {
                this.width = 50;
                this.height = 45;
                this.doorWidth = 22;
                this.doorHeight = 28;
                this.roofHeight = 20;
            }

            draw(ctx, x, y, alpha) {
                ctx.save();
                ctx.globalAlpha = alpha;

                // Main body
                ctx.fillStyle = '#8b4513';
                ctx.fillRect(x - this.width / 2, y, this.width, this.height);

                // Roof
                ctx.fillStyle = '#654321';
                ctx.beginPath();
                ctx.moveTo(x - this.width / 2 - 3, y);
                ctx.lineTo(x + this.width / 2 + 3, y);
                ctx.lineTo(x, y - this.roofHeight);
                ctx.fill();

                // Door frame
                ctx.fillStyle = '#3d2817';
                ctx.fillRect(x - this.doorWidth / 2, y + 8, this.doorWidth, this.doorHeight);

                // Door opening (darker)
                ctx.fillStyle = '#1a0f0a';
                ctx.fillRect(x - this.doorWidth / 2 + 1, y + 9, this.doorWidth - 2, this.doorHeight - 2);

                // Door trim
                ctx.strokeStyle = '#5c3d2e';
                ctx.lineWidth = 2;
                ctx.strokeRect(x - this.doorWidth / 2, y + 8, this.doorWidth, this.doorHeight);

                // Wood texture
                ctx.strokeStyle = 'rgba(0,0,0,0.2)';
                ctx.lineWidth = 1;
                for (let i = 0; i < 4; i++) {
                    ctx.beginPath();
                    ctx.moveTo(x - this.width / 2, y + i * 12);
                    ctx.lineTo(x + this.width / 2, y + i * 12);
                    ctx.stroke();
                }

                ctx.restore();
            }

            getDoorThreshold() {
                return { x: this.width / 2, y: 8 + this.doorHeight };
            }
        }

        const dog = new PomeranianDog();
        const bone = new Bone();
        const doghouse = new DogHouse();

        function renderFrame(timeSeconds) {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Clamp time
            const t = Math.max(0, Math.min(12, timeSeconds));

            // Ground and scene
            ctx.fillStyle = '#90ee90';
            ctx.fillRect(0, 360, canvas.width, 180);

            // Grass texture
            ctx.fillStyle = 'rgba(0,100,0,0.1)';
            for (let i = 0; i < 30; i++) {
                const seed = i + Math.floor(t * 10);
                const x = (seed * 73) % canvas.width;
                const y = 360 + (seed * 97) % 150;
                ctx.fillRect(x, y, 2, 1);
            }

            // Shadow
            ctx.fillStyle = 'rgba(0,0,0,0.15)';
            ctx.beginPath();
            ctx.ellipse(canvas.width / 2, 385, 120, 20, 0, 0, Math.PI * 2);
            ctx.fill();

            // Doghouse position
            const dogHouseX = canvas.width * 0.7;
            const dogHouseY = 310;

            doghouse.draw(ctx, dogHouseX, dogHouseY, 1);

            // Bone initial position (on doghouse floor)
            const boneInitialX = dogHouseX - 5;
            const boneInitialY = dogHouseY + 25;

            let dogX, dogY, dogRotation, gaitPhase, isReverse, mouthOpen, boneX, boneY, boneRotation, inMouth;

            if (t < 1) {
                // Setup phase
                dogX = 150;
                dogY = 350;
                dogRotation = 0;
                gaitPhase = 0;
                isReverse = false;
                mouthOpen = 0;
                boneX = boneInitialX;
                boneY = boneInitialY;
                boneRotation = 0;
                inMouth = false;
            } else if (t < 3) {
                // Running toward doghouse
                const runProgress = (t - 1) / 2;
                dogX = 150 + runProgress * (dogHouseX - 80 - 150);
                dogY = 350;
                dogRotation = 0;
                gaitPhase = (t - 1) * 8;
                isReverse = false;
                mouthOpen = 0;
                boneX = boneInitialX;
                boneY = boneInitialY;
                boneRotation = 0;
                inMouth = false;
            } else if (t < 4.5) {
                // Entering doghouse
                const enterProgress = (t - 3) / 1.5;
                const doorX = dogHouseX - doghouse.doorWidth / 2 - 10;
                dogX = dogHouseX - 50 + enterProgress * 45;
                dogY = 350 - enterProgress * 30;
                dogRotation = 0;
                gaitPhase = (t - 3) * 4;
                isReverse = false;
                mouthOpen = 0;
                boneX = boneInitialX;
                boneY = boneInitialY;
                boneRotation = 0;
                inMouth = false;
            } else if (t < 6) {
                // Grabbing bone
                const grabProgress = (t - 4.5) / 1.5;
                dogX = dogHouseX - 5;
                dogY = 320;
                dogRotation = 0;
                gaitPhase = 0;
                isReverse = false;
                mouthOpen = grabProgress > 0.3 ? (grabProgress - 0.3) * 0.8 : 0;
                
                if (grabProgress > 0.5) {
                    boneX = dogHouseX - 10 + grabProgress * 5;
                    boneY = boneInitialY - grabProgress * 5;
                    inMouth = true;
                } else {
                    boneX = boneInitialX;
                    boneY = boneInitialY;
                    inMouth = false;
                }
                boneRotation = grabProgress * 0.3;
            } else if (t < 8) {
                // Backing out
                const backProgress = (t - 6) / 2;
                dogX = dogHouseX - 5 - backProgress * 40;
                dogY = 320 + backProgress * 30;
                dogRotation = 0;
                gaitPhase = (t - 6) * 4;
                isReverse = true;
                mouthOpen = 0.8;
                inMouth = true;
                boneX = dogHouseX - 10 - backProgress * 40;
                boneY = boneInitialY - 5 - backProgress * 30;
                boneRotation = 0.3;
            } else if (t < 10) {
                // Turning to face viewer
                const turnProgress = (t - 8) / 2;
                dogX = dogHouseX - 45 - turnProgress * 20;
                dogY = 350;
                dogRotation = turnProgress * Math.PI * 0.4;
                gaitPhase = 0;
                isReverse = false;
                mouthOpen = 0.6;
                inMouth = true;
                boneX = dogHouseX - 50 - turnProgress * 20;
                boneY = 330 - turnProgress * 10;
                boneRotation = turnProgress * Math.PI * 0.5;
            } else {
                // Final pose facing viewer
                dogX = dogHouseX - 65;
                dogY = 350;
                dogRotation = Math.PI * 0.4;
                gaitPhase = 0;
                isReverse = false;
                mouthOpen = 0.6;
                inMouth = true;
                boneX = dogHouseX - 70;
                boneY = 325;
                boneRotation = Math.PI * 0.5;
            }

            // Draw bone (not in mouth)
            if (!inMouth && t < 4.5) {
                bone.draw(ctx, boneX, boneY, boneRotation, 1);
            }

            // Calculate leg positions
            const legPositions = dog.getLegPositions(dogX, dogY, dogRotation, gaitPhase, isReverse);

            // Draw dog
            dog.drawFur(ctx, dogX, dogY, dogRotation, 1);
            dog.drawLegs(ctx, dogX, dogY, dogRotation, legPositions, 1);

            const tailPos = dog.getTailPosition(dogX, dogY, dogRotation, t * 3);
            dog.drawTail(ctx, tailPos, dogRotation, t * 3, 1);

            dog.drawHead(ctx, dogX, dogY, dogRotation, mouthOpen, inMouth, 1);

            // Draw bone in mouth
            if (inMouth) {
                const headPos = dog.getHeadPosition(dogX, dogY, dogRotation);
                const mouthX = headPos.x + Math.cos(dogRotation) * 8;
                const mouthY = headPos.y + Math.sin(dogRotation) * 4;
                bone.draw(ctx, mouthX, mouthY, boneRotation, 1);
            }
        }

        // Initial render
        renderFrame(0);

        // Expose for recorder
        window.renderFrame = renderFrame;
    </script>

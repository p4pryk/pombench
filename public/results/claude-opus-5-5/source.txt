// dog state
function dogState(t){
let s={x:0,y:0,dir:1,yaw:0,lift:0,phase:0,gait:0,jump:-1,head:0,mouth:0,pitch:0,crouch:0};
// path x vs time
const xs=[[0,160],[1,175],[2.2,PX-95],[2.55,PX+90],[3.3,HX-60],[4.5,HX+15],[6,HX+15],[8,HX-90],[10,HX-110],[12,HX-110]];
function px(t){for(let i=0;i<xs.length-1;i++){let[a,va]=xs[i],[b,vb]=xs[i+1];if(t<=b){let u=(t-a)/(b-a);if(i==1||i==3)u=u;else u=ss(0,1,u)*.6+u*.4;return lerp(va,vb,u)}}return xs[xs.length-1][1]}
t=cl(t,0,12);s.x=px(t);
let dy=t<3.3?0:t<4.5?lerp(0,-22,ss(3.3,4.5,t)):t<6?-22:t<8?lerp(-22,10,ss(6,8,t)):lerp(10,30,ss(8,10,t));
s.y=lerp(370,HY+2,ss(2.8,4.4,t))+ (t>6?lerp(0,dy+22,ss(6,10,t)):0)+(t>=4.5&&t<6?0:0);
s.y=t<2.8?370:t<4.5?lerp(370,HY-2,ss(2.8,4.4,t)):t<6?HY-2:t<8?lerp(HY-2,HY+8,ss(6,8,t)):lerp(HY+8,HY+40,ss(8,10,t));
s.gait=t<1?ss(.6,1,t):t<2.2?1:t<2.55?0:t<4.3?1-ss(3.9,4.4,t)*.6:t<6?0:t<8?.6:t<10?.35:0;
s.phase=t*9.5;
if(t>=6&&t<8)s.phase=-t*5;if(t>=8)s.phase=t*4;
if(t>1.9&&t<2.2)s.crouch=ss(1.9,2.2,t);
if(t>=2.2&&t<2.55){s.jump=(t-2.2)/.35;s.lift=Math.sin(s.jump*Math.PI)*48;s.pitch=-.35*Math.cos(s.jump*Math.PI);}
if(t>=2.55&&t<2.8)s.crouch=Math.sin((t-2.55)/.25*Math.PI)*.8;
s.yaw=t<8?0:ss(8.2,10,t)*Math.PI/2; // 0 side(right), pi/2 front

s.mouth=t<4.6?0:t<5.2?ss(4.6,5,t):0.35;
s.hasBone=t>=5.15;
s.breath=t>10?Math.sin(t*5)*.8:0;
s.t=t;return s;}

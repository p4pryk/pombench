function sr(n){var s=Math.sin(n*127.1+311.7)*43758.5453;return s-Math.floor(s);}
function clamp(v,a,b){return v<a?a:(v>b?b:v);}
function smooth(u){u=clamp(u,0,1);return u*u*(3-2*u);}
function lerp(a,b,u){return a+(b-a)*u;}
function q(a,b,c,u){return lerp(lerp(a,b,u),lerp(b,c,u),u);}

var COAT='#d9832a',COATD='#a95f1a',COATL='#eda54e',RUFF='#f3c273',CREAM='#f8e4b8';

function furBlob(c,x,y,r,col,seed,jag){
  c.fillStyle=col;c.beginPath();
  var n=14,i,a,rr,X,Y;
  for(i=0;i<=n;i++){a=i/n*Math.PI*2;rr=r*(1+(sr(seed+i)-0.5)*jag);
    X=x+Math.cos(a)*rr;Y=y+Math.sin(a)*rr;
    if(i)c.lineTo(X,Y);else c.moveTo(X,Y);}
  c.closePath();c.fill();
}
function furEll(c,x,y,rx,ry,rot,col,seed,jag){
  c.fillStyle=col;c.beginPath();
  var n=22,i,a,w,ex,ey,X,Y,cr=Math.cos(rot),srr=Math.sin(rot);
  for(i=0;i<=n;i++){a=i/n*Math.PI*2;w=1+(sr(seed+i*3)-0.5)*jag;
    ex=Math.cos(a)*rx*w;ey=Math.sin(a)*ry*w;
    X=x+ex*cr-ey*srr;Y=y+ex*srr+ey*cr;
    if(i)c.lineTo(X,Y);else c.moveTo(X,Y);}
  c.closePath();c.fill();
}
function furStrokes(c,x,y,rx,ry,seed){

}
function leg(c,sx,sy,fx,fy,bendDir,shade){

function drawDogSide(c,st,t,bonePos){

  c.save();

  var stride=lerp(13,24,run),lift=lerp(7,13,run);
  var axF=x+25,axH=x-27,fF,fF2,hH,hH2;
  if(st.mode==='jump'){
    var p=st.p,tk=Math.sin(Math.PI*p);
    fF=[axF+lerp(-14,26,smooth(p)),by+13-(1-tk)*4];
    fF2=[axF+lerp(-18,20,smooth(p)),by+15-(1-tk)*5];
    hH=[axH+lerp(16,-14,smooth(p)),by+15+(p<0.4?(0.4-p)*14:0)];

function dogState(t){
  var st={x:120,h:0,pitch:0,ph:0,run:0,dip:0,mouth:0,turn:0,mode:'idle',p:0,notice:0,crouch:0,landC:0};
  st.notice=smooth((t-0.5)/0.35)*(1-smooth((t-1.3)/0.3));
  if(t<1){st.mode='idle';st.ph=t*2.2;}
  else if(t<2.15){var u=(t-1)/1.15;st.mode='run';st.run=1;
    st.x=lerp(120,362,u);st.ph=(t-1)*14;
    if(u>0.78){var a=(u-0.78)/0.22;st.crouch=a*7;st.run=1-a*0.4;}
  }else if(t<2.55){st.mode='jump';var p=(t-2.15)/0.4;st.p=p;
    st.x=lerp(362,578,p);st.h=Math.sin(p*Math.PI)*54;
    st.pitch=lerp(-0.28,0.2,smooth(p));
  }else if(t<3.05){var u2=(t-2.55)/0.5;st.mode='run';st.run=1;
    st.x=lerp(578,642,u2);st.ph=7+(t-2.55)*13;
    st.landC=Math.sin(clamp(u2/0.45,0,1)*Math.PI)*7*(1-u2);
  }else if(t<4.45){var u3=(t-3.05)/1.4;st.mode='walk';
    st.x=lerp(642,702,smooth(u3));st.ph=13.5+(t-3.05)*7.5;st.run=0.55;
  }else if(t<6){st.mode='grab';st.x=702+smooth((t-4.45)/0.3)*2;
    st.ph=14.6+(t-4.45)*1.5;st.run=0;
    st.dip=smooth((t-4.6)/0.5)*(1-0.55*smooth((t-5.85)/0.45));
    st.mouth=smooth((t-4.95)/0.3)*(1-smooth((t-5.55)/0.35));
  }else if(t<8){st.mode='back';var u4=smooth((t-6)/2);
    st.x=lerp(704,606,u4);st.ph=-(t-6)*5.5;st.run=0.45;
    st.dip=0.18*(1-u4*0.3);
  }else if(t<10){st.mode='turn';var u5=smooth((t-8)/2);st.turn=u5;
    st.x=lerp(606,596,u5);st.ph=(t-8)*5;st.run=0.3;st.dip=0.12*(1-u5);
  }else{st.mode='front';st.x=596;}
  return st;
}
function bonePose(t,st){
  var rx=750,ry=425;
  if(t<5.3)return{x:rx,y:ry,ang:0.1,held:0};
  var u=smooth((t-5.3)/0.35);
  if(st.mode==='front')return{x:st.x,y:368,ang:0,held:1};
  if(st.mode==='turn'){
    var v=1-st.turn;
    return{x:st.x+58*v,y:lerp(368,387,v),ang:lerp(0,-0.18,v),held:1};
  }
  var dip=st.dip||0;
  var hx=st.x+42+dip*7,hy=(G-44)-25+dip*44;
  return{x:lerp(rx,hx+15,u),y:lerp(ry,hy+5,u),ang:lerp(0.1,-0.18,u),held:t>5.55?1:0};
}

function vignette(){
  var v=ctx.createRadialGradient(480,270,300,480,270,620);
  v.addColorStop(0,'rgba(0,0,0,0)');v.addColorStop(1,'rgba(20,15,5,0.22)');
  ctx.fillStyle=v;ctx.fillRect(0,0,960,540);
}
window.renderFrame=function(t){
  if(!(t>=0))t=0;
  if(t>12)t=12;
  var st=dogState(t);
  var bs=bonePose(t,st);
  drawBG();
  drawHouseInterior();
  if(!bs.held)drawBone(ctx,bs.x,bs.y,bs.ang,1);
  drawPuddle();
  drawReflection(st,t);
  dogShadow(st);
  if(st.mode==='front'){
    drawDogFront(ctx,st.x,t,1,true);
  }else if(st.mode==='turn'){
    var sx=Math.max(0.04,1-st.turn);
    ctx.save();
    ctx.translate(st.x,0);ctx.scale(sx,1);ctx.translate(-st.x,0);
    drawDogSide(ctx,st,t,null);
    ctx.restore();
    drawDogFront(ctx,st.x,t,st.turn,false);
    drawBone(ctx,bs.x,bs.y,bs.ang,1);
  }else{
    drawDogSide(ctx,st,t,(bs.held&&st.mode!=='turn')?bs:null);
  }
  drawHouseFront();
  vignette();
};
window.renderFrame(0);
})();
</script>

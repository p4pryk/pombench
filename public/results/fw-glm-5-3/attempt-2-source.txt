function drawLeg(c,sx,sy,fx,fy,l1,l2,dir,col,pc){

  fx=sx+dx/d0*d;fy=sy+dy/d0*d;

  var cc=(l1*l1+d*d-l2*l2)/(2*l1*d);
  var b=Math.acos(Math.max(-1,Math.min(1,cc)));
  var k=a+dir*b;
  var kx=sx+Math.cos(k)*l1,ky=sy+Math.sin(k)*l1;
  c.strokeStyle=col;c.lineWidth=10;c.lineCap='round';c.lineJoin='round';

}

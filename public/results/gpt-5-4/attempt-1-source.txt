const TAU=Math.PI*2;
function clamp(v,a,b){return v<a?a:v>b?b:v}
function mix(a,b,t){return a+(b-a)*t}
function smooth(t){return t<=0?0:t>=1?1:t*t*(3-2*t)}
function sstep(a,b,t){return smooth((t-a)/(b-a))}
function easeInOut(t){return t<.5?2*t*t:1-Math.pow(-2*t+2,2)/2}
function proj(p){
  return {x:90+p.x*1.02-p.z*.34,y:425-p.z*.32-p.y, d:p.z};
}
function rotY(p,a){
  const ca=Math.cos(a),sa=Math.sin(a);
  return {x:p.x*ca-p.z*sa,y:p.y,z:p.x*sa+p.z*ca};
}
function add(a,b){return{x:a.x+b.x,y:a.y+b.y,z:a.z+b.z}}
function mul(a,s){return{x:a.x*s,y:a.y*s,z:a.z*s}}
function lerp3(a,b,t){return{x:mix(a.x,b.x,t),y:mix(a.y,b.y,t),z:mix(a.z,b.z,t)}}
function furColor(k){
  const o=[226,155,68], c=[249,231,202], d=[109,67,28];
  const r=Math.round(mix(mix(d[0],o[0],.65),c[0],k));
  const g=Math.round(mix(mix(d[1],o[1],.65),c[1],k));
  const b=Math.round(mix(mix(d[2],o[2],.65),c[2],k));
  return `rgb(${r},${g},${b})`;
}

function bonePose(t,st){
  let floor={x:house.x+74,y:8,z:house.z+32,ang:.15};
  if(t<5.3)return {mode:"floor",...floor};
  if(t<6){
    const u=sstep(5.3,5.95,t);
    const m=st.mouth||{x:floor.x,y:floor.y,z:floor.z,ang:.2};
    return {mode:"lift",
      x:mix(floor.x,m.x,u),y:mix(floor.y,m.y,u),z:mix(floor.z,m.z,u),ang:mix(floor.ang,m.ang,u)};
  }
  const m=st.mouth||floor;
  return {mode:"mouth",...m};
}

function dogState(t){
  t=t<0?0:t>12?12:t;
  let pos={x:185,y:0,z:176}, yaw=.02, cycle=0, move=0, jump=0, crouch=0, reverse=0, turn=0, head=.12, front=0;
  if(t<1){
    const u=sstep(0,.9,t);
    yaw=.04; head=.16-.06*u; cycle=0;
  }else if(t<1.9){
    const u=(t-1)/.9;
    pos={x:mix(185,418,u),y:0,z:mix(176,156,u)};
    yaw=-.03; cycle=u*2.2; move=1;
  }else if(t<2.15){
    const u=(t-1.9)/.25;
    pos={x:mix(418,463,u),y:0,z:mix(156,151,u)};
    yaw=-.03; cycle=2.2+u*.45; move=.8; crouch=smooth(u); head=-.02;
  }else if(t<2.42){
    const u=(t-2.15)/.27;
    pos={x:mix(463,548,u),y:Math.sin(u*Math.PI)*52,z:mix(151,145,u)};
    yaw=-.02; jump=1; cycle=2.7; head=.04;
  }else if(t<2.62){
    const u=(t-2.42)/.20;
    pos={x:mix(548,588,u),y:Math.sin((1-u)*Math.PI)*8,z:mix(145,142,u)};
    yaw=-.01; move=.9; cycle=2.75+u*.35; crouch=.7*(1-u); head=.02;
  }else if(t<3){
    const u=(t-2.62)/.38;
    pos={x:mix(588,650,u),y:0,z:mix(142,136,u)};
    yaw=0; cycle=3.1+u*1.0; move=1;
  }else if(t<4.5){
    const u=easeInOut((t-3)/1.5);
    pos={x:mix(650,748,u),y:0,z:mix(136,133,u)};
    yaw=.03; cycle=4.1+u*1.5; move=.55; head=-.02;
  }else if(t<6){
    const u=(t-4.5)/1.5;
    pos={x:748,y:0,z:133}; yaw=.04; cycle=5.7; move=0; head=.18-.55*Math.sin(Math.min(1,u)*Math.PI); front=0;
  }else if(t<8){
    const u=(t-6)/2;
    pos={x:mix(748,650,easeInOut(u)),y:0,z:mix(133,138,easeInOut(u))};
    yaw=.04; cycle=5.7+u*2.0; move=.42; reverse=1; head=.08;
  }else if(t<10){
    const u=easeInOut((t-8)/2);
    pos={x:mix(650,620,u),y:0,z:mix(138,154,u)};
    yaw=mix(.04,-1.35,u);
    cycle=7.7+u*1.2; move=.28; turn=u; head=mix(.08,0,u); front=u;
  }else{
    pos={x:620,y:0,z:154};
    yaw=-1.35; cycle=8.9; move=0; head=0; front=1;
  }
  const breath=t<10?0:Math.sin((t-10)*2.4)*1.2;
  const tailWag=t<10?Math.sin(t*8)*.02:Math.sin((t-10)*4.2)*.08;
  const bob = move?Math.sin(cycle*TAU)*2.2*(jump?0:1):0;
  let bodyY=34 + bob - crouch*7 + breath;
  if(jump) bodyY=34+pos.y;
  const mouthBaseLocal={x:74,y:36+head*15,z:0};
  const m=add({x:pos.x,y:bodyY,z:pos.z},rotY(mouthBaseLocal,yaw));
  return {pos,yaw,cycle,move,jump,crouch,reverse,turn,head,front,bodyY,tailWag,mouth:{x:m.x,y:m.y-4,z:m.z,ang:yaw*.18}};
}
function footForLeg(st,i,anchorX,lat){
  const ph=st.cycle*TAU+[0,.5,.5,0][i]*Math.PI + (st.reverse?Math.PI:0);
  if(st.jump){
    const tuck=.55+.45*Math.sin((st.pos.y/52)*Math.PI);
    const fwd=[6,12,-4,2][i];
    return {x:anchorX+fwd*(1-tuck),y:12+16*tuck,z:lat};
  }
  const stride=(st.move*22)*(st.reverse?-1:1);
  let sx=Math.sin(ph), cy=Math.cos(ph);
  let footX=anchorX - sx*stride*.65 + (st.reverse?8:-2);
  let lift=Math.max(0,cy)*10*st.move;
  if(st.move<.1)lift=0;
  if(st.crouch>0) footX += (i<2?-6:4)*st.crouch;
  if(st.front>.1){
    footX += (i%2? -6:6)*st.front*Math.sin(st.turn*1.7);
  }
  return {x:footX,y:lift,z:lat};
}
function renderDog(st,opts={}){
  const refl=opts.reflect;
  const alpha=opts.alpha==null?1:opts.alpha;
  let bodyY=st.bodyY;
  if(refl) bodyY=-st.bodyY+2;
  const base={x:st.pos.x,y:bodyY,z:st.pos.z};
  const yaw=st.yaw;

  const shadowP=proj({x:st.pos.x+8,y:0,z:st.pos.z+3});
  if(!refl){
    x.save();
    x.globalAlpha=.18;
    x.fillStyle="#000";
    const sw=62+(1-st.front)*26, sh=16+(st.jump?8:0);
    x.beginPath(); x.ellipse(shadowP.x,shadowP.y+6,sw,sh,0,0,TAU); x.fill();
    x.restore();
  }

  const parts=[];
  function sph(local,r,k){
    let p=rotY(local,yaw); p=add(base,p);
    if(refl)p.y=-p.y+2;
    const pr=proj(p);
    parts.push({t:"s",p,pr,r:r*(1+.06*(p.z-st.pos.z)/50),k});
  }
  function leg(i,ax,lat,sideNear){
    const anchor=rotY({x:ax,y:20,z:lat},yaw);
    const a=add(base,anchor);
    const footL=footForLeg(st,i,ax,lat);
    let foot=rotY(footL,yaw); foot=add({x:st.pos.x,y:0,z:st.pos.z},foot);
    if(refl){a.y=-a.y+2;foot.y=-foot.y+2}
    const mid=lerp3(a,foot,.48);
    mid.y += (foot.y>a.y?6:12)*(refl?-1:1);
    const ap=proj(a), mp=proj(mid), fp=proj(foot);
    parts.push({t:"leg",a,ap,m,mp,f:foot,fp,near:sideNear});
    if(!refl && foot.y<2.5){
      x.save(); x.globalAlpha=.14;
      const sp=proj({x:foot.x,y:0,z:foot.z});
      x.fillStyle="#000"; x.beginPath(); x.ellipse(sp.x,sp.y+4,8,3,0,0,TAU); x.fill();
      x.restore();
    }
  }

  // body volumes
  sph({x:-14,y:27,z:0},23,.18);
  sph({x:14,y:29,z:0},25,.34);
  sph({x:35,y:36,z:0},22,.62); // ruff
  sph({x:55,y:42,z:0},18,.72); // head
  sph({x:72,y:38,z:0},10,.82); // muzzle
  sph({x:8,y:17,z:0},18,.15);
  // tail
  const tw=st.tailWag + (st.front?0.55:0);
  const tailBase=rotY({x:-30,y:40,z:4},yaw);
  const tailYaw=yaw-1.1+tw;
  for(let i=0;i<4;i++){
    let lp=rotY({x:-3-i*4,y:46+i*9,z:18+i*6},tailYaw);
    let p=add(base,add(tailBase,lp));
    if(refl)p.y=-p.y+2;
    const pr=proj(p);
    parts.push({t:"s",p,pr,r:12-i*1.5,k:.55+i*.08});
  }
  // ears
  for(let s of [-1,1]){
    let p=add(base,rotY({x:54,y:58,z:s*7},yaw));
    if(refl)p.y=-p.y+2;
    const pr=proj(p);
    parts.push({t:"ear",p,pr,s});
  }
  // legs
  leg(0,22,-8,false); leg(1,20,8,true); leg(2,-12,-8,false); leg(3,-14,8,true);

  parts.sort((a,b)=>(a.p?a.p.z:0)-(b.p?b.p.z:0));
  x.save(); x.globalAlpha=alpha;

  for(const it of parts){
    if(it.t==="leg"){
      x.strokeStyle=it.near?furColor(.18):furColor(.08);
      x.lineCap="round"; x.lineJoin="round";
      x.lineWidth=10;
      x.beginPath(); x.moveTo(it.ap.x,it.ap.y); x.lineTo(it.mp.x,it.mp.y); x.stroke();
      x.lineWidth=8;
      x.beginPath(); x.moveTo(it.mp.x,it.mp.y); x.lineTo(it.fp.x,it.fp.y); x.stroke();
      x.strokeStyle=furColor(.55);
      x.lineWidth=4;
      x.beginPath(); x.moveTo(it.fp.x-2,it.fp.y+1); x.lineTo(it.fp.x+4,it.fp.y+2); x.stroke();
    }else if(it.t==="s"){
      const p=it.pr,r=it.r,k=it.k;
      const g=x.createRadialGradient(p.x-r*.22,p.y-r*.26,r*.18,p.x,p.y,r*1.15);
      g.addColorStop(0,furColor(Math.min(1,k+.25)));
      g.addColorStop(.55,furColor(k));
      g.addColorStop(1,furColor(Math.max(0,k-.25)));
      x.fillStyle=g;
      x.beginPath(); x.ellipse(p.x,p.y,r*1.18,r*.98,0,0,TAU); x.fill();
      x.strokeStyle=`rgba(255,245,230,${.08+Math.max(0,k-.4)*.11})`;
      for(let j=0;j<18;j++){
        const a=j/18*TAU + (k*3.1);
        const rr=r*(.84+.18*((j*17)%7)/7);
        x.beginPath();
        x.moveTo(p.x+Math.cos(a)*rr*.76,p.y+Math.sin(a)*rr*.72);
        x.lineTo(p.x+Math.cos(a)*rr*1.08,p.y+Math.sin(a)*rr*1.05);
        x.stroke();
      }
    }else if(it.t==="ear"){
      x.fillStyle=furColor(.48);
      x.beginPath();
      x.moveTo(it.pr.x,it.pr.y-12);
      x.lineTo(it.pr.x+it.s*7,it.pr.y+3);
      x.lineTo(it.pr.x+it.s*1,it.pr.y+8);
      x.closePath(); x.fill();
      x.fillStyle="rgba(70,38,24,.25)";
      x.beginPath(); x.moveTo(it.pr.x,it.pr.y-7); x.lineTo(it.pr.x+it.s*4,it.pr.y+2); x.lineTo(it.pr.x+it.s*1,it.pr.y+5); x.closePath(); x.fill();
    }
  }

  // face
  const hp=proj(add(base,rotY({x:55,y:42,z:0},yaw)));
  const mp=proj(add(base,rotY({x:74,y:37,z:0},yaw)));
  if(!refl){
    x.fillStyle=furColor(.86);
    x.beginPath(); x.ellipse(mp.x,mp.y,12,8,0,0,TAU); x.fill();
    const frontal=st.front;
    const eyeSep=9+frontal*8, eyeY=-2;
    x.fillStyle="#281b14";
    for(let s of [-1,1]){
      const ex=hp.x + s*eyeSep*Math.cos(yaw+1.35)*(.65+.65*frontal) + s*frontal*5;
      const ey=hp.y + eyeY + Math.abs(s)*0;
      x.beginPath(); x.ellipse(ex,ey,2.8,3.6,0,0,TAU); x.fill();
      x.fillStyle="rgba(255,255,255,.55)";
      x.beginPath(); x.arc(ex-.6,ey-1,1,0,TAU); x.fill();
      x.fillStyle="#281b14";
    }
    x.fillStyle="#23170f";
    x.beginPath(); x.ellipse(mp.x+1,mp.y-1,3.3,2.5,0,0,TAU); x.fill();
  }
  x.restore();
}

  x.fillStyle="#2e211a";
  x.fillRect(doorFloor0.x-10,doorFloor0.y-120,120,140);
  const g=x.createRadialGradient(doorFloor0.x+20,doorFloor0.y-26,8,doorFloor0.x+28,doorFloor0.y-26,76);

  // visible dog inside
  if(t>=3.4 && t<=7.7) renderDog(st,{alpha:.96});

  if(bp.mode!=="mouth") drawBone({x:bp.x,y:bp.y,z:bp.z},bp.ang,1);

}
function drawFence(){
  x.fillStyle="#7d644e";
  for(let i=0;i<30;i++){
    const px=i*34-10, top=236+((i%4)*3);
    x.fillRect(px,top,9,76);
    x.fillRect(px+14,top+1,9,74);
  }
  x.fillRect(0,246,960,8);
}
function renderFrame(t){
  t=t<0?0:t>12?12:t;
  x.clearRect(0,0,960,540);
  drawSky();
  drawFence();
  drawYard();
  drawHouse(true);

  const st=dogState(t);

  // puddle reflection clip
  const pp=proj({x:puddle.x,y:0,z:puddle.z});
  x.save();
  x.beginPath(); x.ellipse(pp.x,pp.y,puddle.rx,puddle.rz,0,0,TAU); x.clip();
  x.globalAlpha=.23;
  renderDog(st,{reflect:true,alpha:.26});
  const lg=x.createLinearGradient(0,pp.y-puddle.rz,0,pp.y+puddle.rz);
  lg.addColorStop(0,"rgba(255,255,255,.03)");
  lg.addColorStop(1,"rgba(30,50,70,.09)");
  x.fillStyle=lg;
  x.fillRect(pp.x-puddle.rx-4,pp.y-puddle.rz-4,puddle.rx*2+8,puddle.rz*2+8);
  x.restore();

  drawPuddle();

  // outside dog before front wall
  renderDog(st,{alpha:1});

  // bone in mouth when outside
  const bp=bonePose(t,st);
  if(bp.mode==="mouth" && !(t>3.4&&t<7.7)) drawBone({x:bp.x,y:bp.y,z:bp.z},bp.ang,1);

  drawHouse(false);
  drawInteriorAndBone(t,st);

  // if bone is in mouth and visible through doorway or outside, redraw on top
  if(bp.mode==="mouth"){
    if(t>=3.4 && t<=7.7){
      x.save(); doorClipPath(); x.clip();
      drawBone({x:bp.x,y:bp.y,z:bp.z},bp.ang,1);
      x.restore();
    }else{
      drawBone({x:bp.x,y:bp.y,z:bp.z},bp.ang,1);
    }
  }

  // foreground grass
  x.strokeStyle="rgba(34,62,23,.23)";
  for(let i=0;i<120;i++){
    const px=(i*53)%960, py=430+((i*29)%105);
    x.beginPath(); x.moveTo(px,py+8); x.lineTo(px+((i%3)-1)*2,py-8); x.stroke();
  }
}
window.renderFrame=renderFrame;

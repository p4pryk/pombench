function clamp(x,a,b){return x<a?a:x>b?b:x}
function lerp(a,b,t){return a+(b-a)*t}
function smooth(t){t=clamp(t,0,1);return t*t*(3-2*t)}

function frac(x){return x-Math.floor(x)}

function hash2(x,y){return hash(x*17.13+y*91.7)}
function rotateY(p,yaw){
  const c=Math.cos(yaw), s=Math.sin(yaw);
  return {x:p.x*c-p.z*s,y:p.y,z:p.x*s+p.z*c};
}

function getDogState(t){
  if(!(t>=0)) t=0;
  if(t>12) t=12;
  const s = {
    root:{x:-2.12,y:0,z:0.1},
    yaw:0.04,
    headYaw:0,
    headPitch:0.06,
    bodyPitch:0,
    mouth:0,
    tailLift:0.9,
    phase:0,
    speed:0,
    mode:'stand',
    alert:0,
    breathe:0,
    boneInMouth:false,
    look:0
  };
  if(t<1){
    s.root.x = -2.12;
    s.root.z = 0.08;
    s.yaw = 0.03;
    s.mode='stand';
    s.alert = smooth(t);
    s.headYaw = 0.12*s.alert;
    s.headPitch = lerp(0.08,-0.02,s.alert);
    s.tailLift = 0.95;
  } else if(t<3){
    const u=(t-1)/2;
    s.root.x = easeInOut(-2.12,1.78,u);
    s.root.z = easeInOut(0.08,0.23,u);
    s.yaw = lerp(0.02,0.05,u);
    s.mode='run';
    s.speed = lerp(1.2,1.8,u);
    s.phase = u*4.2;
    s.bodyPitch = Math.sin(u*8*Math.PI)*0.02;
    s.headPitch = -0.04 + Math.sin(u*8*Math.PI+0.6)*0.01;
    s.tailLift = 1.0;
  } else if(t<4.5){
    const u=(t-3)/1.5;
    s.root.x = easeInOut(1.78,2.37,u);
    s.root.z = easeInOut(0.23,0.64,u);
    s.yaw = lerp(0.12,0.16,u);
    s.mode='enter';
    s.speed = 0.75;
    s.phase = u*2.2+0.2;
    s.headPitch = lerp(-0.02,-0.18,u);
    s.tailLift = 0.88;
  } else if(t<6){
    const u=(t-4.5)/1.5;
    const uu = u<0.35 ? u/0.35 : 1;
    s.root.x = lerp(2.37,2.5,uu);
    s.root.z = lerp(0.64,0.79,uu);
    s.yaw = 0.18;
    s.mode='grab';
    s.speed = 0.15;
    s.phase = 0.4+u*0.6;
    s.headPitch = lerp(-0.18,0.23, clamp((u-0.08)/0.35,0,1)) - clamp((u-0.45)/0.25,0,1)*0.15;
    s.mouth = Math.sin(clamp((u-0.05)/0.25,0,1)*Math.PI)*0.9 * (1-clamp((u-0.48)/0.18,0,1));
    s.tailLift = 0.78;
    s.boneInMouth = t>=5.15;
  } else if(t<8){
    const u=(t-6)/2;
    s.root.x = easeInOut(2.5,1.98,u);
    s.root.z = easeInOut(0.79,0.37,u);
    s.yaw = lerp(0.18,Math.PI+0.12,u);
    s.mode='back';
    s.speed = 0.8;
    s.phase = u*3.0;
    s.headPitch = -0.02;
    s.mouth = 0.06;
    s.tailLift = 0.84;
    s.boneInMouth = true;
  } else if(t<10){
    const u=(t-8)/2;
    s.root.x = easeInOut(1.98,1.34,u);
    s.root.z = easeInOut(0.37,0.06,u);
    s.yaw = lerp(Math.PI+0.12,-1.42,u);
    s.mode='turn';
    s.speed = 0.45*(1-u);
    s.phase = u*2.0+0.2;
    s.headYaw = lerp(0,0.18,u);
    s.headPitch = lerp(-0.02,0.02,u);
    s.tailLift = lerp(0.84,0.97,u);
    s.mouth = 0.05;
    s.boneInMouth = true;
    s.look = u;
  } else {
    const u=(t-10)/2;
    s.root.x = 1.34;
    s.root.z = 0.06;
    s.yaw = -1.42;
    s.mode='pose';
    s.phase = 0;
    s.speed = 0;
    s.headYaw = 0.18;
    s.headPitch = 0.02 + Math.sin(u*Math.PI*2)*0.01;
    s.tailLift = 0.97 + Math.sin(u*Math.PI*2+0.5)*0.02;
    s.mouth = 0.05;
    s.breathe = Math.sin(u*Math.PI*2)*0.015;
    s.boneInMouth = true;
    s.look = 1;
  }
  return s;
}

function solveLeg(anchor, foot, kneeOut, upper, lower){
  let dx=foot.x-anchor.x, dy=foot.y-anchor.y;
  let d=Math.hypot(dx,dy);
  d=clamp(d,18,upper+lower-0.001);
  const a=(upper*upper + d*d - lower*lower)/(2*upper*d);
  const h=Math.sqrt(Math.max(0,upper*upper - (a*d)*(a*d)));
  const px=anchor.x + dx*a;
  const py=anchor.y + dy*a;
  const nx=-dy/d, ny=dx/d;
  const knee={x:px+nx*h*kneeOut, y:py+ny*h*kneeOut};
  const ankle={x:(knee.x+foot.x)*0.52,y:(knee.y+foot.y)*0.52};
  return {hip:anchor,knee,ankle,foot};
}

function drawDog(state, drawInside){
  const yaw=state.yaw;
  const root=state.root;

  function wp(local){
    const r=rotateY(local,yaw);
    return project({x:root.x+r.x, y:root.y+r.y, z:root.z+r.z});
  }
  function worldOf(local){
    const r=rotateY(local,yaw);
    return {x:root.x+r.x,y:root.y+r.y,z:root.z+r.z};
  }
  function partInside(local){
    const w=worldOf(local);
    return (w.x > house.x+0.02 && w.x < house.x+house.w-0.02 && w.z > house.z+0.02 && w.z < house.z+house.d-0.02);
  }

  const p=state.phase*2*Math.PI;
  const moveDir = (state.mode==='back') ? -1 : 1;

  function footCycle(offset, front, side){
    const baseX = front ? 0.16 : -0.18;
    const spread = side*0.13;
    const stride=0.15*(state.mode==='run'?1.0:(state.mode==='enter'||state.mode==='back'?0.6:0.32));
    let cyc=Math.sin(p+offset);
    let lift=Math.max(0,Math.sin(p+offset+0.15));
    if(state.mode==='stand'||state.mode==='pose'){cyc=0; lift=0;}
    if(state.mode==='grab'){cyc*=0.15; lift*=0.08;}
    return {
      x: baseX - cyc*stride*moveDir,
      y: 0.02 + lift*(state.mode==='run'?0.09:0.05),
      z: spread + (front?0.02:-0.01)
    };
  }

  const bodyBob = state.mode==='run' ? Math.sin(p*2)*0.03 + 0.02 :
                  (state.mode==='enter'||state.mode==='back') ? Math.sin(p*2)*0.018 :
                  state.mode==='pose' ? state.breathe : 0;
  const spinePitch = state.bodyPitch;
  const chestLocal={x:0.12,y:0.43+bodyBob+state.breathe,z:0};
  const rumpLocal={x:-0.2,y:0.4+bodyBob,z:0};
  const bodyLocal={x:-0.02,y:0.41+bodyBob,z:0};
  const neckLocal={x:0.28,y:0.58+bodyBob*0.6,z:0};
  const headBase={x:0.47,y:0.63+bodyBob*0.5,z:0};
  const headYaw = state.headYaw;
  const headPitch = state.headPitch;

  function headPoint(local){
    const c=Math.cos(headPitch), s=Math.sin(headPitch);
    const yp=local.y*c-local.x*s, xp=local.y*s+local.x*c;
    let p1={x:xp,y:yp,z:local.z};
    p1=rotateY(p1,headYaw);
    p1.x += headBase.x; p1.y += headBase.y; p1.z += headBase.z;
    return wp(p1);
  }
  function headWorld(local){
    const c=Math.cos(headPitch), s=Math.sin(headPitch);
    const yp=local.y*c-local.x*s, xp=local.y*s+local.x*c;
    let p1={x:xp,y:yp,z:local.z};
    p1=rotateY(p1,headYaw);
    p1.x += headBase.x; p1.y += headBase.y; p1.z += headBase.z;
    return worldOf(p1);
  }

  const pieces=[];

  const shadowCenter=wp({x:0.02,y:0.02,z:0.01});
  const dirPt=wp({x:0.38,y:0.02,z:0.0});
  const a=Math.atan2(dirPt.y-shadowCenter.y, dirPt.x-shadowCenter.x);
  const sideScale=0.78+0.22*Math.abs(Math.sin(yaw));
  pieces.push({
    pts:[bodyLocal],
    depth:-100,
    draw(){
      drawShadowEllipse(shadowCenter.x+8, shadowCenter.y+8, 62*sideScale, 20, a, 0.18);
      drawShadowEllipse(shadowCenter.x+38, shadowCenter.y+11, 30, 12, a+0.2, 0.11);
    }
  });

  const foots = [
    footCycle(0.0,true,-1),
    footCycle(Math.PI,true,1),
    footCycle(Math.PI,false,-1),
    footCycle(0.0,false,1)
  ];
  const anchors = [
    {x:0.17,y:0.42+bodyBob,z:-0.13},
    {x:0.17,y:0.42+bodyBob,z:0.13},
    {x:-0.17,y:0.39+bodyBob,z:-0.12},
    {x:-0.17,y:0.39+bodyBob,z:0.12}
  ];

  for(let i=0;i<4;i++){
    const aLoc=anchors[i], fLoc=foots[i];
    const ap=wp(aLoc), fp=wp(fLoc);
    const near = rotateY(aLoc,yaw).z>0;
    const kneeOut = near?1:-1;
    const seg=solveLeg(ap,fp,kneeOut,32,28);
    pieces.push({
      pts:[aLoc],
      depth: rotateY(aLoc,yaw).z + (i<2?0.02:-0.02),
      draw(){ drawLeg(seg, near?'#c88544':'#9f6938', near); }
    });
  }

  const bodyC=wp(bodyLocal), chestC=wp(chestLocal), rumpC=wp(rumpLocal), neckC=wp(neckLocal);
  const bodyAxis=wp({x:0.52,y:0.43+bodyBob,z:0});
  const bodyRot=Math.atan2(bodyAxis.y-bodyC.y, bodyAxis.x-bodyC.x);

  pieces.push({
    pts:[rumpLocal],
    depth: rotateY(rumpLocal,yaw).z,
    draw(){
      drawFurBlob(rumpC, 40*(0.86+0.14*Math.abs(Math.cos(yaw))), 31, bodyRot-0.05+spinePitch, '#f0c985', '#c98543', 0.11, 15);
    }
  });
  pieces.push({
    pts:[bodyLocal],
    depth: rotateY(bodyLocal,yaw).z+0.01,
    draw(){
      drawFurBlob(bodyC, 52*(0.82+0.18*Math.abs(Math.cos(yaw))), 36, bodyRot+spinePitch, '#f3ce89', '#d08845', 0.10, 21);
      drawFurBlob(chestC, 34*(0.8+0.25*Math.abs(Math.cos(yaw))), 28, bodyRot+0.18, '#fae3bc', '#e0a455', 0.14, 24);
    }
  });
  pieces.push({
    pts:[neckLocal],
    depth: rotateY(neckLocal,yaw).z+0.08,
    draw(){
      drawFurBlob(neckC, 30*(0.82+0.32*Math.abs(Math.cos(yaw))), 26, bodyRot+0.22, '#fdeaca', '#e4b066', 0.18, 33);
    }
  });

  const tailBaseL={x:-0.3,y:0.67+bodyBob*0.5,z:0.08};
  const tailBase=wp(tailBaseL);
  const tailDirP=wp({x:-0.18,y:0.8,z:0.16});
  const tailA=Math.atan2(tailDirP.y-tailBase.y,tailDirP.x-tailBase.x)-0.45+state.tailLift*0.25;
  pieces.push({
    pts:[tailBaseL],
    depth: rotateY(tailBaseL,yaw).z+0.25,
    draw(){ drawTail(tailBase, tailA, state.tailLift, 57); }
  });

  const skull=headPoint({x:0.10,y:0.02,z:0});
  const muzzle=headPoint({x:0.28,y:0.0,z:0});
  const muzzleTop=headPoint({x:0.2,y:0.06,z:0});
  const earLP=headPoint({x:-0.02,y:0.17,z:-0.09});
  const earRP=headPoint({x:-0.02,y:0.17,z:0.09});
  const eyeLP=headPoint({x:0.12,y:0.04,z:-0.07});
  const eyeRP=headPoint({x:0.12,y:0.04,z:0.07});
  const nose=headPoint({x:0.36,y:-0.01,z:0});
  const jaw=headPoint({x:0.26,y:-0.03-state.mouth*0.08,z:0});

  pieces.push({
    pts:[headBase],
    depth: headWorld({x:0.12,y:0.02,z:0}).z+0.18,
    draw(){
      const headAxis=headPoint({x:0.28,y:0,z:0});
      const hr=Math.atan2(headAxis.y-skull.y, headAxis.x-skull.x);
      drawFurBlob(skull, 29*(0.78+0.28*Math.abs(Math.cos(yaw+headYaw))), 24, hr, '#f8ddb0', '#cf8945', 0.12, 41);
      drawFurBlob(muzzleTop, 18*(0.82+0.24*Math.abs(Math.cos(yaw+headYaw))), 12, hr, '#fff1da', '#e2b06d', 0.08, 42);

      function drawEar(p, flip){
        ctx.save();
        ctx.translate(p.x,p.y);
        ctx.rotate(hr + flip*0.18);
        ctx.fillStyle='#d2914d';
        ctx.beginPath(); ctx.moveTo(-7,10); ctx.lineTo(0,-16); ctx.lineTo(9,8); ctx.closePath(); ctx.fill();
        ctx.fillStyle='rgba(246,196,150,0.6)';
        ctx.beginPath(); ctx.moveTo(-3,6); ctx.lineTo(0,-9); ctx.lineTo(5,5); ctx.closePath(); ctx.fill();
        ctx.restore();
      }

      if (headWorld({x:0,y:0,z:-0.09}).z < headWorld({x:0,y:0,z:0.09}).z) {
        drawEar(earLP,-1);
        drawEar(earRP,1);
      } else {
        drawEar(earRP,1);
        drawEar(earLP,-1);
      }

      const faceForward = Math.abs(Math.cos(yaw+headYaw))<0.48;
      const visibleL = faceForward || headWorld({x:0,y:0,z:-0.07}).z > headWorld({x:0,y:0,z:0.07}).z;
      const visibleR = faceForward || !visibleL;
      function drawEye(p){
        ctx.fillStyle='#1d1714';
        ctx.beginPath(); ctx.arc(p.x,p.y,3.8,0,Math.PI*2); ctx.fill();
        ctx.fillStyle='rgba(255,255,255,0.9)';
        ctx.beginPath(); ctx.arc(p.x-1.3,p.y-1.2,1.2,0,Math.PI*2); ctx.fill();
      }
      if(visibleL) drawEye(eyeLP);
      if(visibleR) drawEye(eyeRP);

      ctx.strokeStyle='rgba(96,58,29,0.16)';
      ctx.beginPath(); ctx.moveTo(muzzleTop.x-8,muzzleTop.y+2); ctx.lineTo(muzzleTop.x+10,muzzleTop.y+2); ctx.stroke();

      ctx.fillStyle='#201613';
      ctx.beginPath(); ctx.arc(nose.x,nose.y,4.4,0,Math.PI*2); ctx.fill();

      if(state.mouth>0.03){
        ctx.strokeStyle='#2b1b16';
        ctx.lineWidth=2;
        ctx.beginPath();
        ctx.moveTo(muzzle.x-6,jaw.y);
        ctx.quadraticCurveTo(muzzle.x+3,jaw.y+4,muzzle.x+10,jaw.y+1);
        ctx.stroke();
        ctx.lineWidth=1;
      }

      if(state.boneInMouth){
        const bw=headPoint({x:0.30,y:-0.01,z:0});
        drawBoneAtScreen(bw.x,bw.y+3,hr+0.04,0.56);
      }
    }
  });

  const arr=[];
  for(const pc of pieces){
    const inside = pc.pts.some(partInside);
    if(inside===drawInside) arr.push(pc);
  }
  arr.sort((a,b)=>a.depth-b.depth);
  for(const p of arr) p.draw();
}

function drawForegroundPlants(){
  for(let i=0;i<18;i++){
    const x=25+i*56+hash(i)*18;
    const y=440+hash(i+9)*50;
    ctx.strokeStyle='rgba(52,92,41,0.38)';
    for(let j=0;j<7;j++){
      const h=18+hash2(i,j)*26;
      ctx.beginPath();
      ctx.moveTo(x+j*3,y);
      ctx.quadraticCurveTo(x+j*3-2+hash2(i+5,j)*10,y-h*0.5,x+j*3+hash2(i+9,j)*14-7,y-h);
      ctx.stroke();
    }
  }
}

window.renderFrame = function(timeSeconds){
  let t = timeSeconds;
  if(!(t>=0)) t=0;
  if(t>=12) t=12;

  ctx.clearRect(0,0,W,H);
  drawSkyAndYard();
  drawHouseBackAndInterior();

  const bone = getBoneWorld(t);
  if(bone.onFloor){
    const bp=project(bone.p);
    drawBoneAtScreen(bp.x,bp.y,bone.p.ang,bone.p.scale);
  }

  const dog = getDogState(t);
  drawDog(dog,true);
  drawHouseFront();
  drawDog(dog,false);

  const door=project({x:house.x+0.63,y:0,z:house.z+0.08});
  ctx.fillStyle='rgba(0,0,0,0.16)';
  ctx.beginPath();
  ctx.ellipse(door.x+6, door.y+4, 40, 13, 0.12, 0, Math.PI*2);
  ctx.fill();

  drawForegroundPlants();
};

renderFrame(0);
})();

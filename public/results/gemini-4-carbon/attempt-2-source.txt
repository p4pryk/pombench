<script>
(function() {
  "use strict";

  const canvas = document.getElementById("stage");
  const gl = canvas.getContext("webgl", {
    alpha: false,
    antialias: true,
    depth: true,
    stencil: false,
    preserveDrawingBuffer: true
  });

  if (!gl) {
    const ctx = canvas.getContext("2d");
    window.renderFrame = function() {
      if (ctx) {
        ctx.fillStyle = "#5b8266";
        ctx.fillRect(0, 0, 960, 540);
      }
    };
    window.renderFrame(0);
    return;
  }

  gl.getExtension("OES_standard_derivatives");

  // --- Math Helpers ---
  function clamp(v, a, b) { return v < a ? a : (v > b ? b : v); }
  function lerp(a, b, t) { return a + (b - a) * t; }
  function smoothstep(e0, e1, x) {
    const t = clamp((x - e0) / (e1 - e0), 0, 1);
    return t * t * (3 - 2 * t);
  }
  function smootherstep(e0, e1, x) {
    const t = clamp((x - e0) / (e1 - e0), 0, 1);
    return t * t * t * (t * (t * 6 - 15) + 10);
  }
  function vAdd(a, b) { return [a[0]+b[0], a[1]+b[1], a[2]+b[2]]; }
  function vSub(a, b) { return [a[0]-b[0], a[1]-b[1], a[2]-b[2]]; }
  function vScale(a, s) { return [a[0]*s, a[1]*s, a[2]*s]; }
  function vDot(a, b) { return a[0]*b[0] + a[1]*b[1] + a[2]*b[2]; }
  function vCross(a, b) {
    return [
      a[1]*b[2] - a[2]*b[1],
      a[2]*b[0] - a[0]*b[2],
      a[0]*b[1] - a[1]*b[0]
    ];
  }
  function vLen(a) { return Math.hypot(a[0], a[1], a[2]); }
  function vNorm(a) {
    const l = vLen(a) || 1;
    return [a[0]/l, a[1]/l, a[2]/l];
  }
  function vLerp(a, b, t) {
    return [lerp(a[0], b[0], t), lerp(a[1], b[1], t), lerp(a[2], b[2], t)];
  }
  function wrapAngle(a) {
    while (a > Math.PI) a -= Math.PI * 2;
    while (a < -Math.PI) a += Math.PI * 2;
    return a;
  }
  function lerpAngle(a, b, t) {
    return a + wrapAngle(b - a) * t;
  }

  function mat4Identity() {
    const m = new Float32Array(16);
    m[0] = m[5] = m[10] = m[15] = 1;
    return m;
  }
  function mat4Mul(a, b) {
    const out = new Float32Array(16);
    for (let c = 0; c < 4; c++) {
      for (let r = 0; r < 4; r++) {
        out[c*4 + r] =
          a[0*4 + r] * b[c*4 + 0] +
          a[1*4 + r] * b[c*4 + 1] +
          a[2*4 + r] * b[c*4 + 2] +
          a[3*4 + r] * b[c*4 + 3];
      }
    }
    return out;
  }
  function mat4Perspective(fovY, aspect, near, far) {
    const f = 1.0 / Math.tan(fovY * 0.5);
    const nf = 1.0 / (near - far);
    const out = new Float32Array(16);
    out[0] = f / aspect;
    out[5] = f;
    out[10] = (far + near) * nf;
    out[11] = -1;
    out[14] = (2 * far * near) * nf;
    return out;
  }
  function mat4Ortho(left, right, bottom, top, near, far) {
    const out = new Float32Array(16);
    const lr = 1 / (left - right);
    const bt = 1 / (bottom - top);
    const nf = 1 / (near - far);
    out[0] = -2 * lr;
    out[5] = -2 * bt;
    out[10] = 2 * nf;
    out[12] = (left + right) * lr;
    out[13] = (top + bottom) * bt;
    out[14] = (far + near) * nf;
    out[15] = 1;
    return out;
  }
  function mat4LookAt(eye, center, up) {
    const z = vNorm(vSub(eye, center));
    const x = vNorm(vCross(up, z));
    const y = vCross(z, x);
    const out = new Float32Array(16);
    out[0] = x[0]; out[4] = x[1]; out[8]  = x[2]; out[12] = -vDot(x, eye);
    out[1] = y[0]; out[5] = y[1]; out[9]  = y[2]; out[13] = -vDot(y, eye);
    out[2] = z[0]; out[6] = z[1]; out[10] = z[2]; out[14] = -vDot(z, eye);
    out[15] = 1;
    return out;
  }
  function mat4Translate(tx, ty, tz) {
    const m = mat4Identity();
    m[12] = tx; m[13] = ty; m[14] = tz;
    return m;
  }
  function mat4Scale(sx, sy, sz) {
    const m = mat4Identity();
    m[0] = sx; m[5] = sy; m[10] = sz;
    return m;
  }
  function mat4RotX(a) {
    const m = mat4Identity();
    const c = Math.cos(a), s = Math.sin(a);
    m[5] = c; m[9] = -s; m[6] = s; m[10] = c;
    return m;
  }
  function mat4RotY(a) {
    const m = mat4Identity();
    const c = Math.cos(a), s = Math.sin(a);
    m[0] = c; m[8] = s; m[2] = -s; m[10] = c;
    return m;
  }
  function mat4RotZ(a) {
    const m = mat4Identity();
    const c = Math.cos(a), s = Math.sin(a);
    m[0] = c; m[4] = -s; m[1] = s; m[5] = c;
    return m;
  }
  function mat4FromBasisPos(right, up, fwd, pos) {
    const m = new Float32Array(16);
    m[0] = right[0]; m[1] = right[1]; m[2] = right[2];
    m[4] = up[0];    m[5] = up[1];    m[6] = up[2];
    m[8] = fwd[0];   m[9] = fwd[1];   m[10]= fwd[2];
    m[12]= pos[0];   m[13]= pos[1];   m[14]= pos[2];   m[15]= 1;
    return m;
  }
  function transformPoint(m, p) {
    return [
      m[0]*p[0] + m[4]*p[1] + m[8]*p[2] + m[12],
      m[1]*p[0] + m[5]*p[1] + m[9]*p[2] + m[13],
      m[2]*p[0] + m[6]*p[1] + m[10]*p[2] + m[14]
    ];
  }

  function mulberry32(seed) {
    let a = seed >>> 0;
    return function() {
      a |= 0; a = (a + 0x6D2B79F5) | 0;
      let t = Math.imul(a ^ (a >>> 15), 1 | a);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  // --- Shaders ---
  function compileShader(type, src) {
    const s = gl.createShader(type);
    gl.shaderSource(s, src);
    gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) {
      throw new Error(gl.getShaderInfoLog(s));
    }
    return s;
  }
  function createProgram(vsSrc, fsSrc) {
    const p = gl.createProgram();
    gl.attachShader(p, compileShader(gl.VERTEX_SHADER, vsSrc));
    gl.attachShader(p, compileShader(gl.FRAGMENT_SHADER, fsSrc));
    gl.linkProgram(p);
    if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
      throw new Error(gl.getProgramInfoLog(p));
    }
    return p;
  }

  const shadowVS = `
    attribute vec3 aPos;
    uniform mat4 uLightVP;
    uniform mat4 uModel;
    varying float vDepth;
    void main() {
      vec4 cp = uLightVP * (uModel * vec4(aPos, 1.0));
      gl_Position = cp;
      vDepth = cp.z / cp.w * 0.5 + 0.5;
    }
  `;
  const shadowFS = `
    precision highp float;
    varying float vDepth;
    vec4 packDepth(float depth) {
      const vec4 bitShift = vec4(16777216.0, 65536.0, 256.0, 1.0);
      const vec4 bitMask = vec4(0.0, 1.0/256.0, 1.0/256.0, 1.0/256.0);
      vec4 res = fract(depth * bitShift);
      res -= res.xxyz * bitMask;
      return res;
    }
    void main() {
      gl_FragColor = packDepth(clamp(vDepth, 0.0, 1.0));
    }
  `;

  const mainVS = `
    attribute vec3 aPos;
    attribute vec3 aNorm;
    attribute vec3 aColor;
    attribute vec2 aUV;

    uniform mat4 uVP;
    uniform mat4 uModel;
    uniform mat4 uLightVP;
    uniform float uFlipY;
    uniform float uShellOffset;

    varying vec3 vWorldPos;
    varying vec3 vLocalPos;
    varying vec3 vWorldNorm;
    varying vec3 vColor;
    varying vec2 vUV;
    varying vec4 vLightClip;

    void main() {
      vec3 displaced = aPos + aNorm * uShellOffset;
      vec4 wp = uModel * vec4(displaced, 1.0);
      if (uFlipY > 0.5) {
        wp.y = 0.010 - wp.y;
      }
      vWorldPos = wp.xyz;
      vLocalPos = aPos;

      mat3 nMat = mat3(uModel[0].xyz, uModel[1].xyz, uModel[2].xyz);
      vec3 wn = normalize(nMat * aNorm);
      if (uFlipY > 0.5) {
        wn.y = -wn.y;
      }
      vWorldNorm = wn;
      vColor = aColor;
      vUV = aUV;
      vLightClip = uLightVP * vec4(wp.xyz, 1.0);
      gl_Position = uVP * wp;
    }
  `;

  const mainFS = `
    precision highp float;

    varying vec3 vWorldPos;
    varying vec3 vLocalPos;
    varying vec3 vWorldNorm;
    varying vec3 vColor;
    varying vec2 vUV;
    varying vec4 vLightClip;

    uniform vec3 uCamPos;
    uniform vec3 uLightDir;
    uniform sampler2D uShadowMap;
    uniform int uMatType;
    uniform float uShellLayer;
    uniform float uClipRefl;
    uniform vec4 uHouseInfo; // cx, cz, cosAng, sinAng
    uniform vec4 uPuddleData; // cx, cz, rx, rz
    uniform sampler2D uReflTex;
    uniform vec2 uScreenSize;
    uniform float uTime;

    float unpackDepth(vec4 rgba) {
      const vec4 bitShift = vec4(1.0/16777216.0, 1.0/65536.0, 1.0/256.0, 1.0);
      return dot(rgba, bitShift);
    }

    float hash12(vec2 p) {
      vec3 p3 = fract(vec3(p.xyx) * 0.1031);
      p3 += dot(p3, p3.yzx + 33.33);
      return fract((p3.x + p3.y) * p3.z);
    }
    float hash13(vec3 p3) {
      p3 = fract(p3 * 0.1031);
      p3 += dot(p3, p3.zyx + 31.32);
      return fract((p3.x + p3.y) * p3.z);
    }
    float noise2(vec2 p) {
      vec2 i = floor(p);
      vec2 f = fract(p);
      f = f * f * (3.0 - 2.0 * f);
      return mix(
        mix(hash12(i), hash12(i + vec2(1.0, 0.0)), f.x),
        mix(hash12(i + vec2(0.0, 1.0)), hash12(i + vec2(1.0, 1.0)), f.x),
        f.y
      );
    }
    float noise3(vec3 p) {
      vec3 i = floor(p);
      vec3 f = fract(p);
      f = f * f * (3.0 - 2.0 * f);
      return mix(
        mix(mix(hash13(i), hash13(i + vec3(1,0,0)), f.x),
            mix(hash13(i + vec3(0,1,0)), hash13(i + vec3(1,1,0)), f.x), f.y),
        mix(mix(hash13(i + vec3(0,0,1)), hash13(i + vec3(1,0,1)), f.x),
            mix(hash13(i + vec3(0,1,1)), hash13(i + vec3(1,1,1)), f.x), f.y),
        f.z
      );
    }

    float puddleSDF(vec2 xz) {
      vec2 d = (xz - uPuddleData.xy) / uPuddleData.zw;
      float ang = atan(d.y, d.x);
      float r = length(d);
      float wob = 0.08 * sin(ang * 3.0 + 0.7) + 0.05 * cos(ang * 5.0 - 1.2) + 0.04 * sin(ang * 2.0 + 2.1);
      return r - (1.0 + wob);
    }

    float getShadow(vec4 lightClip, vec3 N, vec3 L) {
      vec3 proj = lightClip.xyz / lightClip.w * 0.5 + 0.5;
      if (proj.x < 0.01 || proj.x > 0.99 || proj.y < 0.01 || proj.y > 0.99 || proj.z > 0.998) {
        return 1.0;
      }
      float bias = max(0.0024 * (1.0 - dot(N, L)), 0.0010);
      float s = 0.0;
      float tx = 1.0 / 1024.0;
      for (int x = -1; x <= 1; x++) {
        for (int y = -1; y <= 1; y++) {
          float d = unpackDepth(texture2D(uShadowMap, proj.xy + vec2(float(x), float(y)) * tx));
          s += (proj.z - bias > d) ? 0.0 : 1.0;
        }
      }
      return s / 9.0;
    }

    float houseOcclusion(vec3 p) {
      vec2 rel = p.xz - uHouseInfo.xy;
      float lx = rel.x * uHouseInfo.z - rel.y * uHouseInfo.w;
      float lz = rel.x * uHouseInfo.w + rel.y * uHouseInfo.z;
      float ix = smoothstep(-0.62, -0.35, lx) * (1.0 - smoothstep(0.48, 0.65, lx));
      float iz = 1.0 - smoothstep(0.38, 0.55, abs(lz));
      float iy = 1.0 - smoothstep(0.85, 1.15, p.y);
      float deep = smoothstep(-0.55, 0.45, lx);
      return clamp(1.0 - ix * iz * iy * (0.35 + 0.35 * deep), 0.28, 1.0);
    }

    void main() {
      if (uClipRefl > 0.5 && vWorldPos.y > 0.014) {
        discard;
      }

      if (uMatType == 6) {
        float h = clamp(vUV.y, 0.0, 1.0);
        vec3 col = mix(vec3(0.80, 0.87, 0.93), vec3(0.37, 0.58, 0.85), pow(h, 0.65));
        float cl = noise2(vec2(vUV.x * 9.0 + 1.3, vUV.y * 5.0)) * 0.6 +
                   noise2(vec2(vUV.x * 19.0 - 0.5, vUV.y * 11.0)) * 0.4;
        float cMask = smoothstep(0.55, 0.82, cl) * smoothstep(0.05, 0.32, h) * (1.0 - smoothstep(0.75, 0.98, h));
        col = mix(col, vec3(0.97, 0.98, 1.0), cMask * 0.52);
        vec3 V = normalize(vWorldPos - uCamPos);
        col += vec3(1.0, 0.91, 0.74) * pow(max(dot(V, uLightDir), 0.0), 10.0) * 0.22;
        gl_FragColor = vec4(col, 1.0);
        return;
      }

      vec3 N = normalize(vWorldNorm);
      vec3 V = normalize(uCamPos - vWorldPos);
      vec3 L = normalize(uLightDir);

      float shadow = getShadow(vLightClip, N, L);
      float houseAO = houseOcclusion(vWorldPos);

      vec3 baseCol = vColor;
      float specStrength = 0.08;
      float shininess = 18.0;
      float wrap = 0.25;

      if (uMatType == 1) {
        float strand = noise3(vLocalPos * 58.0) * 0.55 + noise3(vLocalPos * 118.0 + vec3(5.2, 2.7, 1.3)) * 0.45;
        if (strand < mix(0.18, 0.78, uShellLayer)) {
          discard;
        }
        baseCol *= mix(0.70, 1.08, uShellLayer) * (0.90 + 0.20 * strand);
        wrap = 0.48;
        specStrength = 0.05;
      } else if (uMatType == 0) {
        float furTex = noise3(vLocalPos * 62.0) * 0.5 + noise3(vLocalPos * 124.0) * 0.5;
        baseCol *= (0.87 + 0.23 * furTex);
        wrap = 0.42;
        specStrength = 0.06;
      } else if (uMatType == 2) {
        float pd = puddleSDF(vWorldPos.xz);
        float nLarge = noise2(vWorldPos.xz * 1.4);
        float nMed   = noise2(vWorldPos.xz * 6.5);
        float nFine  = noise2(vWorldPos.xz * 28.0);
        float blades = noise2(vec2(vWorldPos.x * 42.0, vWorldPos.z * 24.0));

        vec3 grassCol = mix(vec3(0.26, 0.47, 0.17), vec3(0.38, 0.56, 0.22), nLarge);
        grassCol = mix(grassCol, vec3(0.20, 0.38, 0.14), smoothstep(0.55, 0.8, nMed) * 0.42);
        grassCol *= (0.86 + 0.26 * blades);

        float expectedZ = mix(0.36, -0.10, clamp((vWorldPos.x + 2.2) / 3.8, 0.0, 1.0));
        float pathLine = abs(vWorldPos.z - expectedZ);
        float pathMask = 1.0 - smoothstep(0.36, 0.92, pathLine + (nMed - 0.5) * 0.26);
        pathMask *= smoothstep(-3.0, -1.9, vWorldPos.x) * (1.0 - smoothstep(1.9, 2.6, vWorldPos.x));

        vec3 dryDirt = mix(vec3(0.56, 0.46, 0.35), vec3(0.48, 0.39, 0.29), nFine);
        float peb = smoothstep(0.78, 0.86, noise2(vWorldPos.xz * 36.0));
        dryDirt = mix(dryDirt, vec3(0.65, 0.61, 0.56), peb * 0.45);
        baseCol = mix(grassCol, dryDirt, pathMask * 0.80);

        float wetRim = 1.0 - smoothstep(-0.04, 0.20, pd);
        vec3 wetMud = mix(vec3(0.23, 0.18, 0.13), vec3(0.17, 0.13, 0.09), nFine);
        baseCol = mix(baseCol, wetMud, wetRim * 0.92);
        specStrength = mix(0.03, 0.24, wetRim);
        shininess = mix(10.0, 42.0, wetRim);
      } else if (uMatType == 3) {
        float grain = noise2(vec2(vLocalPos.x * 4.0 + vLocalPos.z * 4.0, vLocalPos.y * 26.0));
        float seam = smoothstep(0.45, 0.50, abs(fract(vUV.y * 6.0) - 0.5));
        baseCol *= (0.88 + 0.20 * grain) * (1.0 - 0.25 * seam);
        specStrength = 0.10;
      } else if (uMatType == 4) {
        specStrength = 0.88;
        shininess = 96.0;
        wrap = 0.10;
      } else if (uMatType == 7) {
        float bTex = noise3(vLocalPos * 28.0);
        baseCol *= (0.95 + 0.10 * bTex);
        specStrength = 0.24;
        shininess = 32.0;
        wrap = 0.35;
      } else if (uMatType == 8) {
        float grain = noise2(vWorldPos.xz * 12.0);
        baseCol *= (0.86 + 0.20 * grain);
        specStrength = 0.03;
      } else if (uMatType == 5) {
        float pd = puddleSDF(vWorldPos.xz);
        if (pd > 0.0) {
          discard;
        }
        float edgeAlpha = smoothstep(0.0, -0.055, pd);
        float shallow = smoothstep(-0.35, 0.0, pd);

        vec2 ruv = vWorldPos.xz * 9.0;
        float r1 = sin(ruv.x * 2.1 + ruv.y * 1.7 + uTime * 1.1) * 0.0014;
        float r2 = cos(ruv.x * 1.5 - ruv.y * 2.3 - uTime * 0.85) * 0.0014;
        vec2 screenUV = gl_FragCoord.xy / uScreenSize;
        vec3 reflCol = texture2D(uReflTex, clamp(screenUV + vec2(r1, r2), vec2(0.002), vec2(0.998))).rgb;

        vec3 waterNorm = normalize(vec3(r1 * 8.0, 1.0, r2 * 8.0));
        float fresnel = 0.30 + 0.66 * pow(1.0 - clamp(dot(waterNorm, V), 0.0, 1.0), 2.8);

        vec3 bedCol = mix(vec3(0.16, 0.13, 0.09), vec3(0.24, 0.20, 0.15), noise2(vWorldPos.xz * 22.0));
        bedCol *= (0.48 + 0.52 * shadow);

        vec3 waterCol = mix(bedCol, reflCol * vec3(0.95, 0.98, 1.02), clamp(fresnel + 0.34, 0.0, 0.95));
        waterCol = mix(waterCol, bedCol * 1.12, shallow * 0.32);

        vec3 H = normalize(L + V);
        waterCol += vec3(1.0, 0.97, 0.90) * pow(max(dot(waterNorm, H), 0.0), 140.0) * shadow * 0.50;
        float rimLine = smoothstep(-0.04, -0.012, pd) * (1.0 - smoothstep(-0.012, 0.0, pd));
        waterCol += vec3(0.11, 0.13, 0.15) * rimLine;

        gl_FragColor = vec4(waterCol, edgeAlpha);
        return;
      }

      float diff = max((dot(N, L) + wrap) / (1.0 + wrap), 0.0);
      vec3 sunCol = vec3(1.04, 0.97, 0.86);
      vec3 ambient = mix(vec3(0.27, 0.25, 0.19), vec3(0.43, 0.51, 0.63), N.y * 0.5 + 0.5) * 0.68 * houseAO;

      float rim = 0.0;
      if (uMatType == 0 || uMatType == 1) {
        float fres = pow(1.0 - max(dot(N, V), 0.0), 2.6);
        rim = fres * (0.16 + 0.20 * clamp(dot(-L, V) * 0.5 + 0.5, 0.0, 1.0)) * shadow;
      }

      vec3 H = normalize(L + V);
      float spec = pow(max(dot(N, H), 0.0), shininess) * specStrength * shadow;

      vec3 lit = baseCol * (ambient + sunCol * diff * shadow * 0.88) +
                 vec3(1.0, 0.92, 0.78) * rim * baseCol +
                 vec3(1.0, 0.98, 0.95) * spec;

      float fog = smoothstep(14.0, 34.0, length(vWorldPos - uCamPos));
      lit = mix(lit, vec3(0.77, 0.85, 0.91), fog * 0.65);
      lit = lit / (1.0 + lit * 0.18);
      lit = pow(clamp(lit, 0.0, 1.0), vec3(0.92));

      gl_FragColor = vec4(lit, 1.0);
    }
  `;

  const shadowProg = createProgram(shadowVS, shadowFS);
  const mainProg = createProgram(mainVS, mainFS);

  const shadowLoc = {
    aPos: gl.getAttribLocation(shadowProg, "aPos"),
    uLightVP: gl.getUniformLocation(shadowProg, "uLightVP"),
    uModel: gl.getUniformLocation(shadowProg, "uModel")
  };

  const mainLoc = {
    aPos: gl.getAttribLocation(mainProg, "aPos"),
    aNorm: gl.getAttribLocation(mainProg, "aNorm"),
    aColor: gl.getAttribLocation(mainProg, "aColor"),
    aUV: gl.getAttribLocation(mainProg, "aUV"),
    uVP: gl.getUniformLocation(mainProg, "uVP"),
    uModel: gl.getUniformLocation(mainProg, "uModel"),
    uLightVP: gl.getUniformLocation(mainProg, "uLightVP"),
    uFlipY: gl.getUniformLocation(mainProg, "uFlipY"),
    uShellOffset: gl.getUniformLocation(mainProg, "uShellOffset"),
    uCamPos: gl.getUniformLocation(mainProg, "uCamPos"),

  function createFBO(w, h, nearest) {
    const tex = gl.createTexture();
    gl.bindTexture(gl.TEXTURE_2D, tex);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, w, h, 0, gl.RGBA, gl.UNSIGNED_BYTE, null);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, nearest ? gl.NEAREST : gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, nearest ? gl.NEAREST : gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);

    const rb = gl.createRenderbuffer();
    gl.bindRenderbuffer(gl.RENDERBUFFER, rb);
    gl.renderbufferStorage(gl.RENDERBUFFER, gl.DEPTH_COMPONENT16, w, h);

    const fbo = gl.createFramebuffer();
    gl.bindFramebuffer(gl.FRAMEBUFFER, fbo);
    gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, tex, 0);
    gl.framebufferRenderbuffer(gl.FRAMEBUFFER, gl.DEPTH_ATTACHMENT, gl.RENDERBUFFER, rb);
    gl.bindFramebuffer(gl.FRAMEBUFFER, null);
    return { fbo, tex, w, h };
  }

  // --- Mesh Builder ---
  class MeshBuilder {
    constructor() {
      this.verts = [];
      this.indices = [];
    }
    addVertex(p, n, c, uv) {
      const idx = this.verts.length / 11;
      this.verts.push(
        p[0], p[1], p[2],
        n[0], n[1], n[2],
        c[0], c[1], c[2],
        uv ? uv[0] : 0, uv ? uv[1] : 0
      );
      return idx;
    }
    addTri(i0, i1, i2) {
      this.indices.push(i0, i1, i2);
    }
    addQuad(p0, p1, p2, p3, c0, c1, c2, c3, uvX = 1, uvY = 1) {
      const n = vNorm(vCross(vSub(p1, p0), vSub(p3, p0)));
      const i0 = this.addVertex(p0, n, c0, [0, 0]);
      const i1 = this.addVertex(p1, n, c1 || c0, [uvX, 0]);
      const i2 = this.addVertex(p2, n, c2 || c0, [uvX, uvY]);
      const i3 = this.addVertex(p3, n, c3 || c0, [0, uvY]);
      this.addTri(i0, i1, i2);
      this.addTri(i0, i2, i3);
    }
    addBox(cx, cy, cz, sx, sy, sz, col, topCol) {
      const x0 = cx - sx*0.5, x1 = cx + sx*0.5;
      const y0 = cy - sy*0.5, y1 = cy + sy*0.5;
      const z0 = cz - sz*0.5, z1 = cz + sz*0.5;
      const tc = topCol || col;
      this.addQuad([x0,y0,z1],[x1,y0,z1],[x1,y1,z1],[x0,y1,z1], col, col, tc, tc, sx, sy);
      this.addQuad([x1,y0,z0],[x0,y0,z0],[x0,y1,z0],[x1,y1,z0], col, col, tc, tc, sx, sy);
      this.addQuad([x1,y0,z1],[x1,y0,z0],[x1,y1,z0],[x1,y1,z1], col, col, tc, tc, sz, sy);
      this.addQuad([x0,y0,z0],[x0,y0,z1],[x0,y1,z1],[x0,y1,z0], col, col, tc, tc, sz, sy);
      this.addQuad([x0,y1,z1],[x1,y1,z1],[x1,y1,z0],[x0,y1,z0], tc, tc, tc, tc, sx, sz);
      this.addQuad([x0,y0,z0],[x1,y0,z0],[x1,y0,z1],[x0,y0,z1], col, col, col, col, sx, sz);
    }
    addEllipsoid(cx, cy, cz, rx, ry, rz, latSegs, lonSegs, colorFn, deformFn) {
      const baseIdx = this.verts.length / 11;
      const grid = [];
      for (let i = 0; i <= latSegs; i++) {
        const v = i / latSegs;
        const th = v * Math.PI;
        const st = Math.sin(th), ct = Math.cos(th);
        for (let j = 0; j <= lonSegs; j++) {
          const u = j / lonSegs;
          const ph = u * Math.PI * 2;
          const nx = st * Math.cos(ph), ny = ct, nz = st * Math.sin(ph);
          let px = cx + nx * rx, py = cy + ny * ry, pz = cz + nz * rz;
          if (deformFn) {
            const d = deformFn(px, py, pz, nx, ny, nz, u, v);
            px = d[0]; py = d[1]; pz = d[2];
          }
          grid.push([px, py, pz, nx, ny, nz, u, v]);
        }
      }
      const stride = lonSegs + 1;
      for (let i = 0; i <= latSegs; i++) {
        for (let j = 0; j <= lonSegs; j++) {
          const cur = grid[i * stride + j];
          const jP = j > 0 ? j - 1 : lonSegs - 1;
          const jN = j < lonSegs ? j + 1 : 1;
          const iP = Math.max(0, i - 1);
          const iN = Math.min(latSegs, i + 1);
          let n = vCross(vSub(grid[iN * stride + j], grid[iP * stride + j]),
                         vSub(grid[i * stride + jN], grid[i * stride + jP]));
          if (vLen(n) < 1e-6) n = [cur[3]/rx, cur[4]/ry, cur[5]/rz];
          n = vNorm(n);
          const col = colorFn ? colorFn(cur[0], cur[1], cur[2], n[0], n[1], n[2], cur[6], cur[7]) : [1,1,1];
          this.addVertex([cur[0], cur[1], cur[2]], n, col, [cur[6], cur[7]]);
        }
      }
      for (let i = 0; i < latSegs; i++) {
        for (let j = 0; j < lonSegs; j++) {
          const a = baseIdx + i * stride + j;
          const b = a + 1, c = a + stride, d = c + 1;
          this.addTri(a, c, b);
          this.addTri(b, c, d);
        }
      }
    }
    addTube(rings, segs, colorFn) {
      const baseIdx = this.verts.length / 11;
      const pts = [];
      for (let i = 0; i < rings.length; i++) {
        const r = rings[i];
        const v = i / (rings.length - 1);
        for (let j = 0; j <= segs; j++) {
          const u = j / segs;
          const ang = u * Math.PI * 2;
          const ca = Math.cos(ang), sa = Math.sin(ang);
          const p = vAdd(r.c, vAdd(vScale(r.right, ca * r.rx), vScale(r.up, sa * r.ry)));
          const nGuess = vNorm(vAdd(vScale(r.right, ca / (r.rx || 1)), vScale(r.up, sa / (r.ry || 1))));
          pts.push({ p, n: nGuess, u, v, rCol: r.col });
        }
      }
      const stride = segs + 1;
      for (let i = 0; i < rings.length; i++) {
        for (let j = 0; j <= segs; j++) {
          const it = pts[i * stride + j];
          const jP = j > 0 ? j - 1 : segs - 1;
          const jN = j < segs ? j + 1 : 1;
          const iP = Math.max(0, i - 1);
          const iN = Math.min(rings.length - 1, i + 1);
          let n = vCross(vSub(pts[i * stride + jN].p, pts[i * stride + jP].p),
                         vSub(pts[iN * stride + j].p, pts[iP * stride + j].p));
          n = vLen(n) > 1e-6 ? vNorm(n) : it.n;
          if (vDot(n, it.n) < 0) n = vScale(n, -1);
          const col = colorFn ? colorFn(it.p, n, it.u, it.v) : (it.rCol || [1,1,1]);
          this.addVertex(it.p, n, col, [it.u, it.v]);
        }
      }
      for (let i = 0; i < rings.length - 1; i++) {
        for (let j = 0; j < segs; j++) {
          const a = baseIdx + i * stride + j;
          const b = a + 1, c = a + stride, d = c + 1;
          this.addTri(a, b, c);
          this.addTri(b, d, c);
        }
      }
    }
    build() {
      const vbo = gl.createBuffer();
      gl.bindBuffer(gl.ARRAY_BUFFER, vbo);
      gl.bufferData(gl.ARRAY_BUFFER, new Float32Array(this.verts), gl.STATIC_DRAW);
      const ibo = gl.createBuffer();
      gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, ibo);
      gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, new Uint16Array(this.indices), gl.STATIC_DRAW);
      return { vbo, ibo, count: this.indices.length };
    }
  }

  function pomTorsoMesh() {
    const mb = new MeshBuilder();
    // Compact cobby Pomeranian torso (+Z forward, +Y up)
    mb.addEllipsoid(0, 0, 0, 0.125, 0.12, 0.175, 20, 26,
      (x, y, z) => {
        const belly = clamp((-y + 0.03) * 5.0, 0, 1);
        const saddle = clamp((y - 0.02) * 5.0, 0, 1);
        const bib = clamp((z - 0.07) * 6.0, 0, 1) * clamp((-y + 0.07) * 4.0, 0, 1);
        let c = vLerp(COL_ORANGE_MID, COL_ORANGE_DEEP, saddle * 0.65);
        return vLerp(c, COL_CREAM, clamp(belly * 0.85 + bib * 0.75, 0, 1));
      },
      (x, y, z, nx, ny, nz, u, v) => {
        const fluff = 1.0 + 0.045 * Math.sin(u * Math.PI * 14) * Math.sin(v * Math.PI * 9)
                          + 0.028 * Math.cos(u * Math.PI * 22 + v * 12.0);
        let sx = 1.0, sy = 1.0;
        if (z < -0.04) { if (y < 0) sy *= 0.88; sx *= 1.04; }
        if (z > 0.05)  { sx *= 1.08; sy *= 1.06; }
        return [x * sx * fluff, y * sy * fluff, z * fluff];
      }
    );

    // Pronounced Pomeranian Lion-like Neck Ruff & Chest Frill
    mb.addEllipsoid(0, 0.045, 0.135, 0.152, 0.148, 0.115, 18, 26,
      (x, y, z) => {
        const bib = clamp((z - 0.14) * 5.5, 0, 1) * clamp((0.08 - y) * 5.5, 0, 1);
        const side = clamp((Math.abs(x) - 0.07) * 6.0, 0, 1) * clamp((y + 0.02) * 4.0, 0, 1);
        const mane = clamp((y - 0.09) * 5.5, 0, 1);
        let c = vLerp(COL_ORANGE_MID, COL_ORANGE_LIGHT, 0.45);
        c = vLerp(c, COL_ORANGE_DEEP, mane * 0.5);
        return vLerp(c, COL_CREAM_BRIGHT, clamp(bib * 0.92 + side * 0.45, 0, 1));
      },
      (x, y, z, nx, ny, nz) => {
        const ang = Math.atan2(y - 0.045, x);
        const tuft = 1.0 + 0.075 * Math.pow(Math.sin(ang * 9.0) * 0.5 + 0.5, 1.5) * (1.0 - Math.abs(nz) * 0.6)
                         + 0.04 * Math.sin(ang * 17.0 + z * 25.0);
        const dropY = (y < 0.02 && z > 0.12) ? -0.035 * (1.0 - Math.min(1.0, Math.abs(x) / 0.12)) : 0;
        return [x * tuft, (y - 0.045) * tuft + 0.045 + dropY, (z - 0.135) * tuft + 0.135];
      }
    );

    // Fluffy hindquarter pantaloons
    for (const side of [-1, 1]) {
      mb.addEllipsoid(side * 0.068, -0.02, -0.115, 0.068, 0.085, 0.075, 10, 14,
        (x, y, z) => vLerp(COL_ORANGE_LIGHT, COL_CREAM, clamp((-z - 0.12) * 8.0, 0, 1) * 0.7 + clamp(-y * 4.0, 0, 0.4)),
        (x, y, z, nx, ny, nz, u, v) => {
          const f = 1.0 + 0.05 * Math.sin(u * Math.PI * 10) * Math.sin(v * Math.PI * 7);
          return [x * f, y * f, z * f];
        }
      );
    }
    return mb.build();
  }

  function pomHeadMesh() {

    mb.addEllipsoid(0, -0.030, 0.092, 0.024, 0.008, 0.034, 6, 10, () => [0.28, 0.10, 0.11]);

    const roofDark = [0.42, 0.17, 0.14];
    const floorCol = [0.38, 0.26, 0.16];
    const innerWall= [0.34, 0.22, 0.13];

    // Low threshold sill (0.016m)
    ext.addBox(0, 0.008, 0, wLen + 0.08, 0.016, wWid + 0.08, [0.48, 0.34, 0.22], [0.54, 0.39, 0.26]);
    interior.addBox(0, 0.010, 0, wLen - th*1.8, 0.018, wWid - th*1.8, floorCol, floorCol);

    // Side walls
    for (const sz of [-1, 1]) {
      const zc = sz * (wWid * 0.5 - th * 0.5);
      ext.addBox(0, wallH * 0.5 + 0.016, zc, wLen, wallH, th, woodWarm);
      interior.addBox(0, wallH * 0.5 + 0.016, zc - sz * 0.004, wLen - th*2, wallH - 0.02, th * 0.8, innerWall);
      for (const sx of [-1, 1]) {
        ext.addBox(sx * (wLen * 0.5 - 0.025), wallH * 0.5 + 0.016, sz * (wWid * 0.5 - 0.025), 0.065, wallH + 0.01, 0.065, woodTrim);
      }
    }

    // Back wall + gable
    ext.addBox(wLen * 0.5 - th * 0.5, wallH * 0.5 + 0.016, 0, th, wallH, wWid - th, woodWarm);
    interior.addBox(wLen * 0.5 - th * 0.9, wallH * 0.5 + 0.016, 0, th * 0.5, wallH, wWid - th*2, innerWall);
    {
      const xO = wLen * 0.5, xI = wLen * 0.5 - th;
      const yB = wallH + 0.016, yT = peakH + 0.016;
      ext.addTri(
        ext.addVertex([xO, yB, wWid*0.5], [1,0,0], woodWarm, [0,0]),
        ext.addVertex([xO, yB, -wWid*0.5], [1,0,0], woodWarm, [1,0]),
        ext.addVertex([xO, yT, 0], [1,0,0], woodWarm, [0.5,1])
      );
      interior.addTri(
        interior.addVertex([xI, yB, -wWid*0.5], [-1,0,0], innerWall, [0,0]),
        interior.addVertex([xI, yB, wWid*0.5], [-1,0,0], innerWall, [1,0]),
        interior.addVertex([xI, yT, 0], [-1,0,0], innerWall, [0.5,1])
      );
    }

    // Front wall (-X) with real open arched doorway
    const doorHalfW = 0.275, jambH = 0.46, archPeakY = 0.67;
    const xFO = -wLen * 0.5, xFI = -wLen * 0.5 + th;
    const sideW = wWid * 0.5 - doorHalfW;
    for (const sz of [-1, 1]) {
      const zc = sz * (doorHalfW + sideW * 0.5);
      ext.addBox(-wLen * 0.5 + th * 0.5, wallH * 0.5 + 0.016, zc, th, wallH, sideW, woodWarm);
      interior.addBox(-wLen * 0.5 + th * 0.9, wallH * 0.5 + 0.016, zc, th * 0.4, wallH, sideW, innerWall);
      ext.addBox(-wLen * 0.5 - 0.008, jambH * 0.5 + 0.016, sz * (doorHalfW + 0.022), 0.022, jambH, 0.044, woodTrim);
    }

    const archSegs = 16;
    for (let s = 0; s < archSegs; s++) {
      const a0 = Math.PI * (s / archSegs), a1 = Math.PI * ((s + 1) / archSegs);
      const z0 = Math.cos(a0) * doorHalfW, z1 = Math.cos(a1) * doorHalfW;
      const yA0 = jambH + Math.sin(a0) * (archPeakY - jambH);
      const yA1 = jambH + Math.sin(a1) * (archPeakY - jambH);
      const rY0 = (wallH + 0.016) + (1.0 - Math.abs(z0) / (wWid * 0.5)) * (peakH - wallH);
      const rY1 = (wallH + 0.016) + (1.0 - Math.abs(z1) / (wWid * 0.5)) * (peakH - wallH);

      ext.addQuad([xFO, yA1, z1], [xFO, yA0, z0], [xFO, rY0, z0], [xFO, rY1, z1], woodWarm, woodWarm, woodWarm, woodWarm, 0.2, 0.8);
      interior.addQuad([xFI, yA0, z0], [xFI, yA1, z1], [xFI, rY1, z1], [xFI, rY0, z0], innerWall, innerWall, innerWall, innerWall, 0.2, 0.8);
      ext.addQuad([xFO, yA0, z0], [xFO, yA1, z1], [xFI, yA1, z1], [xFI, yA0, z0], woodTrim);

      const tz0o = Math.cos(a0) * (doorHalfW + 0.042), tz1o = Math.cos(a1) * (doorHalfW + 0.042);
      const ty0o = jambH + Math.sin(a0) * (archPeakY - jambH + 0.042);
      const ty1o = jambH + Math.sin(a1) * (archPeakY - jambH + 0.042);
      ext.addQuad([xFO - 0.012, yA1, z1], [xFO - 0.012, yA0, z0], [xFO - 0.012, ty0o, tz0o], [xFO - 0.012, ty1o, tz1o], woodTrim);
    }

    for (const sz of [-1, 1]) {
      const zO = sz * wWid * 0.5, zD = sz * doorHalfW;
      const yE = wallH + 0.016, yDR = yE + (1.0 - doorHalfW / (wWid * 0.5)) * (peakH - wallH);
      if (sz > 0) {
        ext.addQuad([xFO, yE, zD], [xFO, yE, zO], [xFO, yE, zO], [xFO, yDR, zD], woodWarm);
        interior.addQuad([xFI, yE, zO], [xFI, yE, zD], [xFI, yDR, zD], [xFI, yE, zO], innerWall);
      } else {
        ext.addQuad([xFO, yE, zO], [xFO, yE, zD], [xFO, yDR, zD], [xFO, yE, zO], woodWarm);
        interior.addQuad([xFI, yE, zD], [xFI, yE, zO], [xFI, yE, zO], [xFI, yDR, zD], innerWall);
      }
    }

    // Pitched shingled roof
    const rOX = 0.11, rOZ = 0.10, slopeZ = wWid * 0.5 + rOZ;
    const apexY = peakH + 0.036;
    const eaveY = apexY - (peakH - wallH) * (slopeZ / (wWid * 0.5));
    const rTh = 0.045;

    for (const sz of [-1, 1]) {
      const pTF = [-wLen*0.5 - rOX, apexY, 0], pTB = [wLen*0.5 + rOX, apexY, 0];
      const pBF = [-wLen*0.5 - rOX, eaveY, sz*slopeZ], pBB = [wLen*0.5 + rOX, eaveY, sz*slopeZ];
      const uTF = [-wLen*0.5 - rOX, apexY - rTh, 0], uTB = [wLen*0.5 + rOX, apexY - rTh, 0];
      const uBF = [-wLen*0.5 - rOX, eaveY - rTh, sz*slopeZ], uBB = [wLen*0.5 + rOX, eaveY - rTh, sz*slopeZ];
      if (sz > 0) {
        ext.addQuad(pBF, pBB, pTB, pTF, roofRed, roofRed, roofDark, roofDark, 4, 3);
        interior.addQuad(uTF, uTB, uBB, uBF, innerWall);
        ext.addQuad(uBF, uBB, pBB, pBF, woodTrim);
        ext.addQuad(uTF, uBF, pBF, pTF, woodTrim);
        ext.addQuad(uBB, uTB, pTB, pBB, woodTrim);
      } else {
        ext.addQuad(pBB, pBF, pTF, pTB, roofRed, roofRed, roofDark, roofDark, 4, 3);
        interior.addQuad(uTB, uTF, uBF, uBB, innerWall);
        ext.addQuad(uBB, uBF, pBF, pBB, woodTrim);
        ext.addQuad(uBF, uTF, pTF, pBF, woodTrim);
        ext.addQuad(uTB, uBB, pBB, pTB, woodTrim);
      }
      for (let r = 0; r < 5; r++) {
        const t0 = r / 5, t1 = (r + 0.92) / 5;
        const y0 = lerp(eaveY, apexY, t0) + 0.008, y1 = lerp(eaveY, apexY, t1) + 0.002;
        const z0 = sz * lerp(slopeZ, 0.01, t0), z1 = sz * lerp(slopeZ, 0.01, t1);
        const sc = (r % 2 === 0) ? roofRed : [0.49, 0.20, 0.17];
        const rx0 = -wLen*0.5 - rOX + 0.005, rx1 = wLen*0.5 + rOX - 0.005;
        if (sz > 0) ext.addQuad([rx0, y0, z0], [rx1, y0, z0], [rx1, y1, z1], [rx0, y1, z1], sc, sc, roofDark, roofDark, 6, 1);
        else        ext.addQuad([rx1, y0, z0], [rx0, y0, z0], [rx0, y1, z1], [rx1, y1, z1], sc, sc, roofDark, roofDark, 6, 1);
      }
    }
    ext.addBox(0, apexY + 0.012, 0, wLen + rOX * 2.02, 0.036, 0.075, woodTrim);

    return { extMesh: ext.build(), intMesh: interior.build() };
  }

      const dx = (x - PUDDLE_DATA[0]) / PUDDLE_DATA[2];
      const dz = (z - PUDDLE_DATA[1]) / PUDDLE_DATA[3];
      const ang = Math.atan2(dz, dx);
      const wob = 0.08 * Math.sin(ang * 3.0 + 0.7) + 0.05 * Math.cos(ang * 5.0 - 1.2) + 0.04 * Math.sin(ang * 2.0 + 2.1);
      return Math.hypot(dx, dz) - (1.0 + wob);
    }

    function terrainY(x, z) {
      const pd = puddleField(x, z);
      if (pd < 0.12) {
        return 0.006 - smoothstep(0.12, -0.35, pd) * 0.036;
      }
      const stageDist = Math.max(0, Math.hypot(x * 0.32, (z - 0.1) * 0.55) - 1.25);
      return smoothstep(0, 2.5, stageDist) * (0.12 * Math.sin(x * 0.35 + 0.5) * Math.cos(z * 0.4) + Math.max(0, -z - 3.2) * 0.04);
    }

    const gGrid = [];
    for (let iz = 0; iz <= gzSegs; iz++) {
      const tz = iz / gzSegs, sz = tz * 2 - 1;
      const z = lerp(-13.0, 8.0, tz) * 0.35 + (sz * Math.abs(sz) * 10.5 - 2.2) * 0.65;
      for (let ix = 0; ix <= gxSegs; ix++) {
        const tx = ix / gxSegs, sx = tx * 2 - 1;
        const x = lerp(-14.0, 14.0, tx) * 0.35 + (sx * Math.abs(sx) * 14.0) * 0.65;

  // Let's verify: if yaw = Math.PI/2 (facing +X): fwd = [1, 0, 0], right = [0, 0, -1]? Wait: if facing +X and up is +Y, right-hand thumb +X?
  // In right-handed coords: X x Y = Z => right x up = fwd => if fwd = [sin, 0, cos] and up = [0, 1, 0]:
  // right = up x fwd = [cos, 0, -sin], OR right = fwd x up = [-cos, 0, sin].
  // Wait: in dog mesh local space: +Z is forward, +Y is up, +X is dog's LEFT (or right). Since determinant of [right, up, fwd] must be +1 so triangle winding is never inverted:
  // det([[cos, 0, -sin], [0, 1, 0], [sin, 0, cos]]) = cos*cos - (-sin)*sin = +1!
  // Let's check: at yaw = 0 (facing +Z, toward viewer!):
  // fwd = [0, 0, 1], up = [0, 1, 0], col0 = [1, 0, 0] -> Identity matrix! Winding is +1!

  // So both the resting bone AND the Pomeranian's jaws gripping it are 100% visible through the open doorway!

  // sin(yaw) = HOUSE_COS, cos(yaw) = -HOUSE_SIN => yaw = Math.atan2(HOUSE_COS, -HOUSE_SIN) (~2.21 rad)
  const YAW_INTO_HOUSE = Math.atan2(HOUSE_COS, -HOUSE_SIN);
  // Yaw from POS_FINAL [0.56, 0.56] directly toward CAM_POS [0.32, 3.62]:
  const YAW_TO_CAMERA  = Math.atan2(CAM_POS[0] - POS_FINAL[0], CAM_POS[2] - POS_FINAL[1]);

  // Smooth piecewise trajectory evaluator for dog root XZ, Yaw, Height, Pitch, Roll, Gait
  function evalDogState(rawT) {
    const t = clamp(rawT, 0.0, 12.0);

    let px = POS_START[0], pz = POS_START[1];
    let bodyY = 0.215; // neutral standing torso center height
    let yaw = Math.atan2(POS_TAKEOFF[0] - POS_START[0], POS_TAKEOFF[1] - POS_START[1]);
    let pitch = 0.0; // positive pitch = nose tilts DOWN (around local X=[1,0,0]: wait! Let's check mat4RotX!)
    // In local space (+Z fwd, +Y up): mat4RotX(a) maps (0, 0, 1) -> (0, -sin(a), cos(a))!
    // Yes! Positive pitch tilts +Z (nose/forequarters) DOWN and -Z (rear) UP! Negative pitch lifts forequarters UP!
    let roll = 0.0;
    let headYaw = 0.0, headPitch = 0.0, headRoll = 0.0;
    let jawOpen = 0.03;
    let tailWag = 0.0, tailLift = 0.0;

    // 1. Planar XZ & Body Yaw across the 12-second narrative
    if (t < 1.0) {
      // 0–1s: Establish yard, puddle, house & bone. Dog notices goal!
      const notice = smoothstep(0.25, 0.78, t);
      const prepShift = smoothstep(0.72, 1.0, t);
      px = POS_START[0] + 0.03 * prepShift;
      pz = POS_START[1] - 0.005 * prepShift;
      yaw = lerpAngle(yaw - 0.18, yaw, notice);
      bodyY = 0.215 - 0.012 * Math.sin(notice * Math.PI) - 0.010 * prepShift;
      pitch = 0.03 * prepShift;
      headYaw = lerp(0.22, 0.0, notice);
      headPitch = -0.10 * notice - 0.04 * Math.sin(t * 5.0);
      headRoll = 0.08 * Math.sin(notice * Math.PI); // expressive Pomeranian head tilt noticing the bone!
      tailWag = Math.sin(t * 7.5) * (0.06 + 0.12 * notice);
    } else if (t < 1.62) {
      // 1.0–1.62s: Brisk, grounded 4-legged run from start to puddle near bank
      const u = (t - 1.0) / 0.62;
      const easeU = smoothstep(0.0, 0.18, u) * (1.0 - 0.12 * (1.0 - u)) + 0.12 * u;
      const s = clamp(u * 0.92 + 0.08 * smoothstep(0, 1, u), 0, 1);
      px = lerp(POS_START[0] + 0.03, POS_TAKEOFF[0] - 0.10, s);
      pz = lerp(POS_START[1] - 0.005, POS_TAKEOFF[1], s);
      yaw = Math.atan2(POS_LAND[0] - POS_START[0], POS_LAND[1] - POS_START[1]);
      // Subtle galloping/trotting suspension bounce & head stabilization
      const runPhase = (t - 1.0) * 4.2 * Math.PI * 2;
      bodyY = 0.216 + 0.014 * Math.abs(Math.sin(runPhase * 0.5));
      pitch = -0.04 * Math.cos(runPhase * 0.5);
      headPitch = -pitch * 0.75 - 0.06; // stabilized gaze locked toward puddle & doorway
      tailWag = Math.sin(t * 11.0) * 0.10;
      tailLift = -0.08; // coat/tail follow-through trailing slightly in run
    } else if (t < 1.80) {
      // 1.62–1.80s: Readable Jump Anticipation! Shorten final stride, lower body, load hind legs on dry near bank
      const u = (t - 1.62) / 0.18;
      const crouch = Math.sin(u * Math.PI * 0.82);
      px = lerp(POS_TAKEOFF[0] - 0.10, POS_TAKEOFF[0], smoothstep(0, 1, u));
      pz = POS_TAKEOFF[1];
      yaw = Math.atan2(POS_LAND[0] - POS_TAKEOFF[0], POS_LAND[1] - POS_TAKEOFF[1]);
      bodyY = 0.215 - 0.048 * crouch; // clear compression/loading
      pitch = 0.08 * (1.0 - u) - 0.26 * smoothstep(0.45, 1.0, u); // dip nose then coil forequarters UP for launch!
      headPitch = -0.14 - pitch * 0.5;
      tailLift = -0.14 * crouch;
    } else if (t < 2.26) {
      // 1.80–2.26s: Ballistic Arc Flight over the Shallow Puddle! All 4 paws clear water [-1.08..-0.08]
      const u = (t - 1.80) / 0.46;
      px = lerp(POS_TAKEOFF[0], POS_LAND[0], u);
      pz = lerp(POS_TAKEOFF[1], POS_LAND[1], u);
      yaw = Math.atan2(POS_LAND[0] - POS_TAKEOFF[0], POS_LAND[1] - POS_TAKEOFF[1]);
      // True parabolic CoM arc peaking +0.23m above standing height
      const arc = 4.0 * u * (1.0 - u);
      bodyY = 0.215 + 0.235 * arc;
      // Coherent spine pitch: nose-up (-0.34) at takeoff, level (0.0) at apex, nose-down (+0.32) reaching forepaws to dry bank!
      pitch = lerp(-0.34, 0.32, smoothstep(0.05, 0.95, u));
      headPitch = -pitch * 0.45 - 0.06;
      tailLift = -0.22 * Math.sin(u * Math.PI); // plumed tail streams gracefully with jump arc
      tailWag = Math.sin(t * 9.0) * 0.06;
    } else if (t < 2.48) {
      // 2.26–2.48s: Weight-Bearing Dry Landing & Joint Compression!
      const u = (t - 2.26) / 0.22;
      px = lerp(POS_LAND[0], POS_LAND[0] + 0.28, u);
      pz = lerp(POS_LAND[1], POS_PRE_DOOR[2], u * 0.35);
      const absorb = Math.sin(Math.pow(u, 0.65) * Math.PI);
      bodyY = 0.215 - 0.046 * absorb; // joints compress visibly to absorb landing impact
      pitch = lerp(0.24, -0.04, smoothstep(0.0, 0.75, u)) + 0.06 * absorb;
      const nextYaw = Math.atan2(POS_PRE_DOOR[0] - POS_LAND[0], POS_PRE_DOOR[2] - POS_LAND[1]);
      yaw = lerpAngle(Math.atan2(POS_LAND[0] - POS_TAKEOFF[0], POS_LAND[1] - POS_TAKEOFF[1]), nextYaw, smoothstep(0.2, 1.0, u));
      headPitch = 0.10 * absorb - 0.05;
      tailLift = 0.16 * absorb; // tail follow-through compression on impact
    } else if (t < 3.05) {
      // 2.48–3.05s: Supported forward run from landing spot up to doghouse approach
      const u = (t - 2.48) / 0.57;
      const s = smoothstep(0.0, 0.15, u) * 0.15 + 0.85 * u;
      px = lerp(POS_LAND[0] + 0.28, POS_PRE_DOOR[0], s);
      pz = lerp(lerp(POS_LAND[1], POS_PRE_DOOR[2], 0.35), POS_PRE_DOOR[2], s);
      const approachYaw = Math.atan2(POS_THRESHOLD[0] - POS_LAND[0], POS_THRESHOLD[2] - POS_LAND[1]);
      yaw = lerpAngle(approachYaw, YAW_INTO_HOUSE, smoothstep(0.25, 1.0, u));
      const runPhase = (t - 2.48) * 3.8 * Math.PI * 2;
      bodyY = 0.215 + 0.011 * Math.abs(Math.sin(runPhase * 0.5));
      pitch = -0.03 * Math.cos(runPhase * 0.5);
      headPitch = -0.04;
      tailWag = Math.sin(t * 11.0) * 0.10;
    } else if (t < 4.50) {
      // 3.05–4.50s: Enter through doorway head & forequarters first! Rear & curled tail remain visible!
      const u = smootherstep(3.05, 4.45, t);
      const p0 = POS_PRE_DOOR;
      const p1 = POS_GRAB_BODY;
      px = lerp(p0[0], p1[0], u);
      pz = lerp(p0[2], p1[2], u);
      yaw = YAW_INTO_HOUSE;
      const walkBob = Math.sin((t - 3.05) * 3.0 * Math.PI * 2) * 0.006 * (1.0 - smoothstep(0.8, 1.0, u));
      bodyY = 0.213 + walkBob - 0.008 * smoothstep(0.6, 1.0, u);
      pitch = 0.04 * smoothstep(0.5, 1.0, u); // slight peer downward toward bone inside
      headPitch = 0.14 * smoothstep(0.4, 1.0, u);
      tailWag = Math.sin(t * 9.5) * 0.14;
    } else if (t < 6.00) {
      // 4.50–6.00s: Lower head & forequarters, open jaws, grip bone at t = 5.30s, close jaws & lift!
      px = POS_GRAB_BODY[0];
      pz = POS_GRAB_BODY[2];
      yaw = YAW_INTO_HOUSE;

      // Reach down curve peaking around t = 5.28..5.36s
      const down = smootherstep(4.55, 5.22, t) * (1.0 - smootherstep(5.42, 5.95, t));
      // Slight forward weight shift onto forelegs during grab
      const reachFwd = 0.038 * down;
      px += Math.sin(yaw) * reachFwd;
      pz += Math.cos(yaw) * reachFwd;

      bodyY = 0.205 - 0.048 * down;
      pitch = 0.04 + 0.30 * down; // forequarters dip, hindquarters stay high!
      headPitch = 0.14 + 0.34 * down - 0.18 * smootherstep(5.45, 6.0, t);

      // Jaws open wide before grab (4.85..5.28s), clamp firmly onto bone shaft at 5.30s (jawOpen = 0.22 rad)
      if (t < 5.30) {
        jawOpen = 0.03 + 0.44 * smootherstep(4.82, 5.18, t);
      } else {
        jawOpen = lerp(0.47, 0.21, smootherstep(5.30, 5.44, t));
      }
      tailWag = Math.sin(t * 12.5) * (0.14 + 0.10 * smootherstep(5.30, 5.8, t)); // happy tail wag upon grabbing bone!
    } else if (t < 8.00) {
      // 6.00–8.00s: Back out through doorway with bone held rigidly in jaws, paws stepping in reverse!
      const u = smootherstep(6.05, 7.95, t);
      px = lerp(POS_GRAB_BODY[0], POS_BACK_OUT[0], u);
      pz = lerp(POS_GRAB_BODY[2], POS_BACK_OUT[2], u);
      yaw = lerpAngle(YAW_INTO_HOUSE, YAW_INTO_HOUSE - 0.18, smoothstep(0.65, 1.0, u));
      const revBob = Math.sin((t - 6.0) * 2.8 * Math.PI * 2) * 0.006 * Math.sin(u * Math.PI);
      bodyY = 0.213 + revBob;
      pitch = -0.04 * Math.sin(u * Math.PI); // slight proud head-up posture backing out
      headPitch = -0.08;
      jawOpen = 0.21;
      tailWag = Math.sin(t * 10.0) * 0.14;
    } else if (t < 10.00) {
      // 8.00–10.00s: Smooth 3D stepping turn to face the viewer!
      const u = smootherstep(8.00, 9.95, t);
      // Curved stepping arc from POS_BACK_OUT to POS_FINAL
      const midX = lerp(POS_BACK_OUT[0], POS_FINAL[0], 0.5) - 0.12;
      const midZ = lerp(POS_BACK_OUT[2], POS_FINAL[1], 0.5) + 0.10;
      px = (1-u)*(1-u)*POS_BACK_OUT[0] + 2*(1-u)*u*midX + u*u*POS_FINAL[0];
      pz = (1-u)*(1-u)*POS_BACK_OUT[2] + 2*(1-u)*u*midZ + u*u*POS_FINAL[1];

      // Shortest-angle turn from YAW_INTO_HOUSE - 0.18 (~2.03 rad) through +X/camera-right (1.0 -> 0.0) to YAW_TO_CAMERA (~-0.08 rad)
      yaw = lerp(YAW_INTO_HOUSE - 0.18, YAW_TO_CAMERA, u);

      const stepBob = Math.sin((t - 8.0) * 3.0 * Math.PI * 2) * 0.005 * Math.sin(u * Math.PI);
      bodyY = 0.215 + stepBob;
      roll = -0.05 * Math.sin(u * Math.PI); // natural inward bank / spine bend during turn
      headYaw = -0.18 * Math.sin(u * Math.PI); // head leads the turn naturally toward viewer!
      headPitch = lerp(-0.08, -0.14, u);
      jawOpen = 0.21;
      tailWag = Math.sin(t * 10.5) * 0.16;
    } else {
      // 10.00–12.00s: Clear front-facing proud pose holding bone, subtle breathing, ear/coat & restrained tail motion!
      const settle = smoothstep(10.0, 10.45, t);
      px = POS_FINAL[0];
      pz = POS_FINAL[1];
      yaw = YAW_TO_CAMERA;
      const breath = Math.sin((t - 10.0) * 2.6);
      bodyY = 0.214 + 0.0035 * breath;
      pitch = -0.05; // proud chest-up Pomeranian stance!
      headPitch = -0.13 + 0.012 * breath;
      headRoll = 0.045 * smoothstep(10.2, 10.8, t) * Math.cos((t - 10.2) * 1.2); // charming subtle attentive tilt
      headYaw = 0.02 * Math.sin((t - 10.0) * 1.5);
      jawOpen = 0.21;
      tailWag = Math.sin(t * 7.5) * lerp(0.14, 0.08, settle); // restrained happy plume sway
      tailLift = 0.03 * breath;
    }

    return { t, px, pz, bodyY, yaw, pitch, roll, headYaw, headPitch, headRoll, jawOpen, tailWag, tailLift };
  }

  // Exact 2-bone analytical limb IK in dog sagittal/world plane so paws plant firmly & joints bend anatomically!
  // Foreleg: elbow bends BACKWARD (-fwd), Hindleg: knee/stifle bends FORWARD (+fwd) then hock extends down to paw!
  function solveLimbIK(hipPos, pawPos, L1, L2, fwdDir, bendSign) {
    const dVec = vSub(pawPos, hipPos);
    const maxReach = (L1 + L2) * 0.992;
    const minReach = Math.abs(L1 - L2) + 0.015;
    let dist = vLen(dVec);
    let target = pawPos;
    if (dist > maxReach) {
      target = vAdd(hipPos, vScale(dVec, maxReach / dist));
      dist = maxReach;
    } else if (dist < minReach) {
      target = vAdd(hipPos, vScale(dVec, minReach / (dist || 1)));
      dist = minReach;
    }
    const dir = vScale(vSub(target, hipPos), 1.0 / dist);
    const cosA = clamp((L1*L1 + dist*dist - L2*L2) / (2.0 * L1 * dist), -1.0, 1.0);
    const sinA = Math.sqrt(Math.max(0.0, 1.0 - cosA * cosA));

    // Bend direction orthogonal to dir in sagittal plane
    let perp = vSub(vScale(fwdDir, bendSign), vScale(dir, vDot(vScale(fwdDir, bendSign), dir)));
    if (vLen(perp) < 1e-4) perp = [0, 0, bendSign];
    perp = vNorm(perp);

    const midJoint = vAdd(hipPos, vAdd(vScale(dir, L1 * cosA), vScale(perp, L1 * sinA)));
    return { midJoint, endPos: target };
  }

  // Build world matrix for a unit limb segment (0 -> -1 along local Y, local +Z forward)
  function segmentMatrix(topPos, botPos, dogFwd, dogRight) {
    const segVec = vSub(topPos, botPos); // local +Y points from bot to top
    const len = vLen(segVec) || 0.001;
    const up = vScale(segVec, 1.0 / len);
    let right = vSub(dogRight, vScale(up, vDot(dogRight, up)));
    if (vLen(right) < 1e-4) right = [1, 0, 0];
    right = vNorm(right);
    const fwd = vCross(right, up);
    const basis = mat4FromBasisPos(right, up, fwd, topPos);
    return mat4Mul(basis, mat4Scale(1, len, 1));
  }

  // Compute planted stance & swing paw targets for all 4 legs deterministically at any time t!
  // Using stride-locked world footfalls during locomotion guarantees ZERO foot skating during stance!
  const LEG_SPECS = [

    { id: 0, isFore: true,  side:  1, hipLocal: [ 0.062, -0.025,  0.105], restLocal: [ 0.066,  0.115], phase: 0.00 },
    { id: 1, isFore: true,  side: -1, hipLocal: [-0.062, -0.025,  0.105], restLocal: [-0.066,  0.115], phase: 0.50 },
    { id: 2, isFore: false, side:  1, hipLocal: [ 0.065, -0.015, -0.108], restLocal: [ 0.068, -0.112], phase: 0.50 },

  function groundHeightAt(x, z) {
    // Inside doghouse raised floor sill is y = 0.018
    const relX = x - SCENE_HOUSE_POS[0], relZ = z - SCENE_HOUSE_POS[2];
    const lx = relX * HOUSE_COS - relZ * HOUSE_SIN;
    const lz = relX * HOUSE_SIN + relZ * HOUSE_COS;
    if (lx > -0.66 && lx < 0.66 && Math.abs(lz) < 0.57) {
      return 0.018;
    }
    return 0.002;
  }

  function neutralPawWorld(state, leg) {
    const s = Math.sin(state.yaw), c = Math.cos(state.yaw);
    const rx = c, rz = -s;
    const wx = state.px + rx * leg.restLocal[0] + s * leg.restLocal[1];
    const wz = state.pz + rz * leg.restLocal[0] + c * leg.restLocal[1];
    return [wx, groundHeightAt(wx, wz), wz];
  }

  function computePawWorld(t, state, leg) {
    const s = Math.sin(state.yaw), c = Math.cos(state.yaw);
    const fwd = [s, 0, c];

    // 1) Opening stand (0.0 – 1.0s)
    if (t < 1.0) {
      const p = neutralPawWorld(evalDogState(0.0), leg);
      if (leg.id === 0 && t > 0.35) {
        // Subtle eager front-paw shift when noticing bone
        const lift = Math.sin(clamp((t - 0.35) / 0.25, 0, 1) * Math.PI);
        const prog = smoothstep(0.35, 0.60, t);
        return [p[0] + fwd[0]*0.025*prog, p[1] + 0.018*lift, p[2] + fwd[2]*0.025*prog];
      }
      return p;
    }

    // 2) Pre-jump approach run (1.0 – 1.80s) with planted stances & anticipation gather
    if (t < 1.80) {
      // Stride schedule: 3 crisp cycles between 1.0 and 1.76, planting on near dry bank for takeoff!
      if (t >= 1.64) {
        // Planted takeoff gather on dry near bank (x <= -1.14, well before puddle edge -1.08!)
        const gatherState = evalDogState(1.74);
        const base = neutralPawWorld(gatherState, leg);
        if (!leg.isFore) {
          // Hind paws coiled forward under hips to push off powerfully
          return [base[0] + fwd[0] * 0.045, base[1], base[2] + fwd[2] * 0.045];
        } else {
          // Forepaws begin lifting slightly right at end of push-off (1.73..1.80s)
          const earlyLift = smoothstep(1.72, 1.80, t) * 0.045;
          return [base[0] - fwd[0] * 0.02, base[1] + earlyLift, base[2] - fwd[2] * 0.02];
        }
      }
      const freq = 3.9; // strides/sec
      const cyc = (t - 1.0) * freq + leg.phase;
      const k = Math.floor(cyc);
      const f = cyc - k; // 0..0.58 stance (planted at fixed world anchor!), 0.58..1.0 swing!
      const stanceEnd = 0.58;
      const tAnchorA = clamp(1.0 + (k + stanceEnd * 0.5 - leg.phase) / freq, 1.0, 1.66);
      const tAnchorB = clamp(1.0 + (k + 1 + stanceEnd * 0.5 - leg.phase) / freq, 1.0, 1.72);
      const pA = neutralPawWorld(evalDogState(tAnchorA), leg);
      const pB = neutralPawWorld(evalDogState(tAnchorB), leg);
      if (f < stanceEnd) {
        return pA; // 100% locked to world ground during stance!
      } else {
        const su = (f - stanceEnd) / (1.0 - stanceEnd);
        const ease = smoothstep(0, 1, su);
        const arc = Math.sin(su * Math.PI) * 0.052;
        return [lerp(pA[0], pB[0], ease), lerp(pA[1], pB[1], ease) + arc, lerp(pA[2], pB[2], ease)];
      }
    }

    // 3) Airborne Jump Flight over the Puddle (1.80 – 2.26s)
    if (t < 2.26) {
      const u = (t - 1.80) / 0.46;
      const sYaw = Math.sin(state.yaw), cYaw = Math.cos(state.yaw);
      const rx = cYaw, rz = -sYaw;
      const tuck = Math.sin(u * Math.PI); // legs tuck neatly at mid-air, then extend for dry landing!

      let localZ, localY;
      if (leg.isFore) {
        // Forelegs fold under chest early-mid jump, then reach forward (+0.18) before landing!
        localZ = lerp(0.06, 0.19, smoothstep(0.25, 0.95, u)) - 0.05 * Math.sin(u * Math.PI);
        localY = -0.185 + 0.085 * tuck - 0.03 * smoothstep(0.65, 1.0, u);
      } else {
        // Hindlegs extend backward at pushoff (-0.21), tuck mid-flight, then swing forward for landing!
        localZ = lerp(-0.22, -0.08, smoothstep(0.15, 0.95, u));
        localY = -0.190 + 0.095 * tuck;
      }
      const wx = state.px + rx * leg.restLocal[0] + sYaw * localZ;
      const wz = state.pz + rz * leg.restLocal[0] + cYaw * localZ;
      const wy = Math.max(0.035, state.bodyY + localY); // strictly airborne above puddle water (y=0.005)!
      return [wx, wy, wz];
    }

    // 4) Dry Landing & Weight Absorption (2.26 – 2.48s): forepaws touch down first, then hind paws!
    if (t < 2.48) {
      const landForePos = neutralPawWorld(evalDogState(2.30), leg);
      const landHindPos = neutralPawWorld(evalDogState(2.38), leg);
      if (leg.isFore) {
        // Forepaws planted solidly on dry ground right at 2.26s!
        return [landForePos[0] + fwd[0] * 0.03, landForePos[1], landForePos[2] + fwd[2] * 0.03];
      } else {
        // Hind paws touch down 0.08s after forepaws (at t = 2.34s)
        if (t < 2.34) {
          const hu = (t - 2.26) / 0.08;
          return [landHindPos[0] - fwd[0]*0.04*(1-hu), landHindPos[1] + 0.045*(1-hu), landHindPos[2] - fwd[2]*0.04*(1-hu)];
        }
        return landHindPos;
      }
    }

    // 5) Post-landing run & Entry into Doghouse (2.48 – 4.50s)
    if (t < 4.50) {
      const freq = 3.2;
      const cyc = (t - 2.48) * freq + leg.phase;
      const k = Math.floor(cyc);
      const f = cyc - k;
      const stanceEnd = 0.62;
      const tA = clamp(2.48 + (k + stanceEnd * 0.5 - leg.phase) / freq, 2.48

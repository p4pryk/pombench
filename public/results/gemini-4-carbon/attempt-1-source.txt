<script>
(function() {
  "use strict";

  const canvas = document.getElementById("stage");
  const gl = canvas.getContext("webgl", {
    alpha: false,
    antialias: true,
    depth: true,
    stencil: false,
    preserveDrawingBuffer: true
  });

  if (!gl) {
    const ctx = canvas.getContext("2d");
    window.renderFrame = function() {
      if (ctx) {
        ctx.fillStyle = "#486";
        ctx.fillRect(0, 0, 960, 540);
      }
    };
    window.renderFrame(0);
    return;
  }

  gl.getExtension("OES_standard_derivatives");

  // --- Math Helpers ---
  function clamp(v, a, b) { return v < a ? a : (v > b ? b : v); }
  function lerp(a, b, t) { return a + (b - a) * t; }
  function smoothstep(e0, e1, x) {
    const t = clamp((x - e0) / (e1 - e0), 0, 1);
    return t * t * (3 - 2 * t);
  }
  function smootherstep(e0, e1, x) {
    const t = clamp((x - e0) / (e1 - e0), 0, 1);
    return t * t * t * (t * (t * 6 - 15) + 10);
  }

  function vec3(x, y, z) { return [x, y, z]; }
  function vAdd(a, b) { return [a[0]+b[0], a[1]+b[1], a[2]+b[2]]; }
  function vSub(a, b) { return [a[0]-b[0], a[1]-b[1], a[2]-b[2]]; }
  function vScale(a, s) { return [a[0]*s, a[1]*s, a[2]*s]; }
  function vDot(a, b) { return a[0]*b[0] + a[1]*b[1] + a[2]*b[2]; }
  function vCross(a, b) {
    return [
      a[1]*b[2] - a[2]*b[1],
      a[2]*b[0] - a[0]*b[2],
      a[0]*b[1] - a[1]*b[0]
    ];
  }
  function vLen(a) { return Math.hypot(a[0], a[1], a[2]); }
  function vNorm(a) {
    const l = vLen(a) || 1;
    return [a[0]/l, a[1]/l, a[2]/l];
  }
  function vLerp(a, b, t) {
    return [lerp(a[0], b[0], t), lerp(a[1], b[1], t), lerp(a[2], b[2], t)];
  }

  function mat4Identity() {
    const m = new Float32Array(16);
    m[0] = m[5] = m[10] = m[15] = 1;
    return m;
  }
  function mat4Mul(a, b) {
    const out = new Float32Array(16);
    for (let c = 0; c < 4; c++) {
      for (let r = 0; r < 4; r++) {
        out[c*4 + r] =
          a[0*4 + r] * b[c*4 + 0] +
          a[1*4 + r] * b[c*4 + 1] +
          a[2*4 + r] * b[c*4 + 2] +
          a[3*4 + r] * b[c*4 + 3];
      }
    }
    return out;
  }
  function mat4Perspective(fovY, aspect, near, far) {
    const f = 1.0 / Math.tan(fovY * 0.5);
    const nf = 1.0 / (near - far);
    const out = new Float32Array(16);
    out[0] = f / aspect;
    out[5] = f;
    out[10] = (far + near) * nf;
    out[11] = -1;
    out[14] = (2 * far * near) * nf;
    return out;
  }
  function mat4Ortho(left, right, bottom, top, near, far) {
    const out = new Float32Array(16);
    const lr = 1 / (left - right);
    const bt = 1 / (bottom - top);
    const nf = 1 / (near - far);
    out[0] = -2 * lr;
    out[5] = -2 * bt;
    out[10] = 2 * nf;
    out[12] = (left + right) * lr;
    out[13] = (top + bottom) * bt;
    out[14] = (far + near) * nf;
    out[15] = 1;
    return out;
  }
  function mat4LookAt(eye, center, up) {
    const z = vNorm(vSub(eye, center));
    const x = vNorm(vCross(up, z));
    const y = vCross(z, x);
    const out = new Float32Array(16);
    out[0] = x[0]; out[4] = x[1]; out[8]  = x[2]; out[12] = -vDot(x, eye);
    out[1] = y[0]; out[5] = y[1]; out[9]  = y[2]; out[13] = -vDot(y, eye);
    out[2] = z[0]; out[6] = z[1]; out[10] = z[2]; out[14] = -vDot(z, eye);
    out[3] = 0;    out[7] = 0;    out[11] = 0;    out[15] = 1;
    return out;
  }
  function mat4Translate(tx, ty, tz) {
    const m = mat4Identity();
    m[12] = tx; m[13] = ty; m[14] = tz;
    return m;
  }
  function mat4Scale(sx, sy, sz) {
    const m = mat4Identity();
    m[0] = sx; m[5] = sy; m[10] = sz;
    return m;
  }
  function mat4RotX(a) {
    const m = mat4Identity();
    const c = Math.cos(a), s = Math.sin(a);
    m[5] = c; m[9] = -s;
    m[6] = s; m[10] = c;
    return m;
  }
  function mat4RotY(a) {
    const m = mat4Identity();
    const c = Math.cos(a), s = Math.sin(a);
    m[0] = c;  m[8] = s;
    m[2] = -s; m[10] = c;
    return m;
  }
  function mat4RotZ(a) {
    const m = mat4Identity();
    const c = Math.cos(a), s = Math.sin(a);
    m[0] = c; m[4] = -s;
    m[1] = s; m[5] = c;
    return m;
  }
  function mat4FromBasisPos(right, up, fwd, pos) {
    const m = new Float32Array(16);
    m[0] = right[0]; m[1] = right[1]; m[2] = right[2]; m[3] = 0;
    m[4] = up[0];    m[5] = up[1];    m[6] = up[2];    m[7] = 0;
    m[8] = fwd[0];   m[9] = fwd[1];   m[10]= fwd[2];   m[11]= 0;
    m[12]= pos[0];   m[13]= pos[1];   m[14]= pos[2];   m[15]= 1;
    return m;
  }
  function transformPoint(m, p) {
    return [
      m[0]*p[0] + m[4]*p[1] + m[8]*p[2] + m[12],
      m[1]*p[0] + m[5]*p[1] + m[9]*p[2] + m[13],
      m[2]*p[0] + m[6]*p[1] + m[10]*p[2] + m[14]
    ];
  }
  function transformDir(m, d) {
    return [
      m[0]*d[0] + m[4]*d[1] + m[8]*d[2],
      m[1]*d[0] + m[5]*d[1] + m[9]*d[2],
      m[2]*d[0] + m[6]*d[1] + m[10]*d[2]
    ];
  }

  // Deterministic PRNG
  function mulberry32(seed) {
    let a = seed >>> 0;
    return function() {
      a |= 0; a = (a + 0x6D2B79F5) | 0;
      let t = Math.imul(a ^ (a >>> 15), 1 | a);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  // --- Shaders ---
  function compileShader(type, src) {
    const s = gl.createShader(type);
    gl.shaderSource(s, src);
    gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) {
      throw new Error(gl.getShaderInfoLog(s));
    }
    return s;
  }
  function createProgram(vsSrc, fsSrc) {
    const p = gl.createProgram();
    gl.attachShader(p, compileShader(gl.VERTEX_SHADER, vsSrc));
    gl.attachShader(p, compileShader(gl.FRAGMENT_SHADER, fsSrc));
    gl.linkProgram(p);
    if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
      throw new Error(gl.getProgramInfoLog(p));
    }
    return p;
  }

  const shadowVS = `
    attribute vec3 aPos;
    uniform mat4 uLightVP;
    uniform mat4 uModel;
    varying float vDepth;
    void main() {
      vec4 wp = uModel * vec4(aPos, 1.0);
      vec4 cp = uLightVP * wp;
      gl_Position = cp;
      vDepth = cp.z / cp.w * 0.5 + 0.5;
    }
  `;
  const shadowFS = `
    precision highp float;
    varying float vDepth;
    vec4 packDepth(float depth) {
      const vec4 bitShift = vec4(256.0*256.0*256.0, 256.0*256.0, 256.0, 1.0);
      const vec4 bitMask = vec4(0.0, 1.0/256.0, 1.0/256.0, 1.0/256.0);
      vec4 res = fract(depth * bitShift);
      res -= res.xxyz * bitMask;
      return res;
    }
    void main() {
      gl_FragColor = packDepth(clamp(vDepth, 0.0, 1.0));
    }
  `;

  const mainVS = `
    attribute vec3 aPos;
    attribute vec3 aNorm;
    attribute vec3 aColor;
    attribute vec2 aUV;

    uniform mat4 uVP;
    uniform mat4 uModel;
    uniform mat4 uLightVP;
    uniform float uFlipY;
    uniform float uShellOffset;

    varying vec3 vWorldPos;
    varying vec3 vLocalPos;
    varying vec3 vWorldNorm;
    varying vec3 vColor;
    varying vec2 vUV;
    varying vec4 vLightClip;

    void main() {
      vec3 displaced = aPos + aNorm * uShellOffset;
      vec4 wp = uModel * vec4(displaced, 1.0);
      if (uFlipY > 0.5) {
        wp.y = -wp.y + 0.008;
      }
      vWorldPos = wp.xyz;
      vLocalPos = aPos;

      mat3 nMat = mat3(uModel[0].xyz, uModel[1].xyz, uModel[2].xyz);
      vec3 wn = normalize(nMat * aNorm);
      if (uFlipY > 0.5) {
        wn.y = -wn.y;
      }
      vWorldNorm = wn;
      vColor = aColor;
      vUV = aUV;
      vLightClip = uLightVP * vec4(wp.xyz, 1.0);
      gl_Position = uVP * wp;
    }
  `;

  const mainFS = `
    #extension GL_OES_standard_derivatives : enable
    precision highp float;

    varying vec3 vWorldPos;
    varying vec3 vLocalPos;
    varying vec3 vWorldNorm;
    varying vec3 vColor;
    varying vec2 vUV;
    varying vec4 vLightClip;

    uniform vec3 uCamPos;
    uniform vec3 uLightDir;
    uniform sampler2D uShadowMap;
    uniform int uMatType;
    // 0 = standard matte/fur solid, 1 = fur shell, 2 = ground/yard, 3 = wood, 4 = eye/glossy, 5 = puddle water, 6 = sky/backdrop, 7 = bone, 8 = dark interior
    uniform float uShellLayer;
    uniform float uClipRefl;
    uniform vec4 uHouseBox; // minX, maxX, minZ, maxZ for interior ambient occlusion
    uniform vec4 uPuddleData; // cx, cz, rx, rz
    uniform sampler2D uReflTex;
    uniform vec2 uScreenSize;
    uniform float uTime;

    float unpackDepth(vec4 rgba) {
      const vec4 bitShift = vec4(1.0/(256.0*256.0*256.0), 1.0/(256.0*256.0), 1.0/256.0, 1.0);
      return dot(rgba, bitShift);
    }

    float hash12(vec2 p) {
      vec3 p3 = fract(vec3(p.xyx) * 0.1031);
      p3 += dot(p3, p3.yzx + 33.33);
      return fract((p3.x + p3.y) * p3.z);
    }
    float hash13(vec3 p3) {
      p3 = fract(p3 * 0.1031);
      p3 += dot(p3, p3.zyx + 31.32);
      return fract((p3.x + p3.y) * p3.z);
    }

    float noise2(vec2 p) {
      vec2 i = floor(p);
      vec2 f = fract(p);
      f = f * f * (3.0 - 2.0 * f);
      float a = hash12(i);
      float b = hash12(i + vec2(1.0, 0.0));
      float c = hash12(i + vec2(0.0, 1.0));
      float d = hash12(i + vec2(1.0, 1.0));
      return mix(mix(a, b, f.x), mix(c, d, f.x), f.y);
    }

    float noise3(vec3 p) {
      vec3 i = floor(p);
      vec3 f = fract(p);
      f = f * f * (3.0 - 2.0 * f);
      float n000 = hash13(i + vec3(0,0,0));
      float n100 = hash13(i + vec3(1,0,0));
      float n010 = hash13(i + vec3(0,1,0));
      float n110 = hash13(i + vec3(1,1,0));
      float n001 = hash13(i + vec3(0,0,1));
      float n101 = hash13(i + vec3(1,0,1));
      float n011 = hash13(i + vec3(0,1,1));
      float n111 = hash13(i + vec3(1,1,1));
      return mix(
        mix(mix(n000, n100, f.x), mix(n010, n110, f.x), f.y),
        mix(mix(n001, n101, f.x), mix(n011, n111, f.x), f.y),
        f.z
      );
    }

    float puddleSDF(vec2 xz) {
      vec2 d = (xz - uPuddleData.xy) / uPuddleData.zw;
      float ang = atan(d.y, d.x);
      float r = length(d);
      float wobble = 0.09 * sin(ang * 3.0 + 0.7) + 0.06 * cos(ang * 5.0 - 1.2) + 0.04 * sin(ang * 2.0 + 2.1);
      return r - (1.0 + wobble);
    }

    float getShadow(vec4 lightClip, vec3 N, vec3 L) {
      vec3 proj = lightClip.xyz / lightClip.w;
      proj = proj * 0.5 + 0.5;
      if (proj.x < 0.01 || proj.x > 0.99 || proj.y < 0.01 || proj.y > 0.99 || proj.z > 0.999) {
        return 1.0;
      }
      float bias = max(0.0022 * (1.0 - dot(N, L)), 0.0009);
      float shadow = 0.0;
      float texel = 1.0 / 1024.0;
      for (int x = -1; x <= 1; x++) {
        for (int y = -1; y <= 1; y++) {
          float d = unpackDepth(texture2D(uShadowMap, proj.xy + vec2(float(x), float(y)) * texel));
          shadow += (proj.z - bias > d) ? 0.0 : 1.0;
        }
      }
      return shadow / 9.0;
    }

    float houseOcclusion(vec3 p) {
      // Darken inside the doghouse volume naturally
      float ix = smoothstep(uHouseBox.x - 0.1, uHouseBox.x + 0.22, p.x) *
                 (1.0 - smoothstep(uHouseBox.y - 0.22, uHouseBox.y + 0.1, p.x));
      float iz = smoothstep(uHouseBox.z - 0.1, uHouseBox.z + 0.25, p.z) *
                 (1.0 - smoothstep(uHouseBox.w - 0.18, uHouseBox.w + 0.1, p.z));
      float iy = 1.0 - smoothstep(0.95, 1.35, p.y);
      // Deeper toward back of house (smaller x since house is on right facing left/camera-left)
      float depthFactor = smoothstep(uHouseBox.x, uHouseBox.y, p.x);
      return clamp(1.0 - ix * iz * iy * (0.42 + 0.40 * depthFactor), 0.20, 1.0);
    }

    void main() {
      if (uClipRefl > 0.5 && vWorldPos.y > 0.015) {
        discard;
      }

      if (uMatType == 6) {
        // Sky dome / distant backdrop
        float h = clamp(vUV.y, 0.0, 1.0);
        vec3 horizon = vec3(0.79, 0.86, 0.92);
        vec3 zenith  = vec3(0.38, 0.58, 0.84);
        vec3 col = mix(horizon, zenith, pow(h, 0.65));
        // Soft clouds
        float cl = noise2(vec2(vUV.x * 9.0 + 1.3, vUV.y * 5.0)) * 0.6 +
                   noise2(vec2(vUV.x * 19.0 - 0.5, vUV.y * 11.0)) * 0.4;
        float cMask = smoothstep(0.54, 0.82, cl) * smoothstep(0.05, 0.35, h) * (1.0 - smoothstep(0.75, 0.98, h));
        col = mix(col, vec3(0.97, 0.97, 0.99), cMask * 0.55);
        // Sun warm glow
        vec3 V = normalize(vWorldPos - uCamPos);
        float sunDot = max(dot(V, uLightDir), 0.0);
        col += vec3(1.0, 0.90, 0.72) * pow(sunDot, 10.0) * 0.22;
        gl_FragColor = vec4(col, 1.0);
        return;
      }

      vec3 N = normalize(vWorldNorm);
      vec3 V = normalize(uCamPos - vWorldPos);
      vec3 L = normalize(uLightDir);

      float shadow = getShadow(vLightClip, N, L);
      float houseAO = houseOcclusion(vWorldPos);

      vec3 baseCol = vColor;
      float specStrength = 0.08;
      float shininess = 18.0;
      float wrap = 0.25;

      if (uMatType == 1) {
        // Volumetric Fur Shell
        float n1 = noise3(vLocalPos * 58.0);
        float n2 = noise3(vLocalPos * 115.0 + vec3(7.1, 3.3, 1.9));
        float strand = n1 * 0.55 + n2 * 0.45;
        float radiusThresh = mix(0.16, 0.78, uShellLayer);
        if (strand < radiusThresh) {
          discard;
        }
        // Darken roots, brighten tips slightly
        float rootAO = mix(0.68, 1.08, uShellLayer);
        baseCol *= rootAO * (0.90 + 0.20 * strand);
        wrap = 0.48;
        specStrength = 0.05;
      } else if (uMatType == 0) {
        // Core dog fur / matte organic
        float furTex = noise3(vLocalPos * 62.0) * 0.5 + noise3(vLocalPos * 125.0) * 0.5;
        baseCol *= (0.86 + 0.24 * furTex);
        wrap = 0.42;
        specStrength = 0.06;
      } else if (uMatType == 2) {
        // Ground yard: lush grass + worn dirt path + puddle basin rim
        float pd = puddleSDF(vWorldPos.xz);
        float nLarge = noise2(vWorldPos.xz * 1.4);
        float nMed   = noise2(vWorldPos.xz * 6.5);
        float nFine  = noise2(vWorldPos.xz * 28.0);
        float blades = noise2(vec2(vWorldPos.x * 45.0, vWorldPos.z * 22.0));

        vec3 grassA = vec3(0.27, 0.47, 0.18);
        vec3 grassB = vec3(0.38, 0.56, 0.22);
        vec3 clover = vec3(0.21, 0.39, 0.15);
        vec3 grassCol = mix(grassA, grassB, nLarge);
        grassCol = mix(grassCol, clover, smoothstep(0.55, 0.8, nMed) * 0.45);
        grassCol *= (0.86 + 0.26 * blades);

        // Dirt path leading toward doghouse doorway
        float pathLine = abs(vWorldPos.z - (-0.18 + 0.05 * sin(vWorldPos.x * 0.7)));
        float pathMask = 1.0 - smoothstep(0.42, 0.98, pathLine + (nMed - 0.5) * 0.28);
        pathMask *= smoothstep(-3.3, -2.1, vWorldPos.x) * (1.0 - smoothstep(2.4, 3.1, vWorldPos.x));

        vec3 dryDirt = mix(vec3(0.55, 0.45, 0.34), vec3(0.47, 0.38, 0.28), nFine);
        // Tiny pebbles in dirt
        float peb = smoothstep(0.78, 0.86, noise2(vWorldPos.xz * 36.0));
        dryDirt = mix(dryDirt, vec3(0.64, 0.60, 0.55), peb * 0.45);

        baseCol = mix(grassCol, dryDirt, pathMask * 0.82);

        // Wet soil bank around puddle
        float wetRim = 1.0 - smoothstep(-0.04, 0.22, pd);
        vec3 wetMud = mix(vec3(0.24, 0.19, 0.14), vec3(0.18, 0.14, 0.10), nFine);
        baseCol = mix(baseCol, wetMud, wetRim * 0.92);
        specStrength = mix(0.03, 0.25, wetRim);
        shininess = mix(10.0, 42.0, wetRim);
      } else if (uMatType == 3) {
        // Wood planks (doghouse, fence)
        float grain = noise2(vec2(vLocalPos.x * 4.0 + vLocalPos.z * 4.0, vLocalPos.y * 26.0));
        float plank = abs(fract(vUV.y * 6.0) - 0.5);
        float seam = smoothstep(0.46, 0.50, plank);
        baseCol *= (0.88 + 0.20 * grain) * (1.0 - 0.28 * seam);
        specStrength = 0.10;
        shininess = 22.0;
      } else if (uMatType == 4) {
        // Glossy eye / wet nose
        specStrength = 0.85;
        shininess = 96.0;
        wrap = 0.1;
      } else if (uMatType == 7) {
        // Bone: warm ivory / marrow tint in crevices
        float bTex = noise3(vLocalPos * 25.0);
        baseCol *= (0.94 + 0.10 * bTex);
        specStrength = 0.22;
        shininess = 32.0;
        wrap = 0.30;
      } else if (uMatType == 8) {
        // Doghouse interior floor/walls
        float grain = noise2(vWorldPos.xz * 12.0);
        baseCol *= (0.85 + 0.22 * grain);
        specStrength = 0.03;
      } else if (uMatType == 5) {
        // Puddle water surface with planar reflection!
        float pd = puddleSDF(vWorldPos.xz);
        if (pd > 0.0) {
          discard;
        }
        float edgeAlpha = smoothstep(0.0, -0.065, pd);
        float shallowFactor = smoothstep(-0.35, 0.0, pd);

        // Subtle still-water micro-ripples
        vec2 ruv = vWorldPos.xz * 8.0;
        float r1 = sin(ruv.x * 2.1 + ruv.y * 1.7 + uTime * 1.1) * 0.0018;
        float r2 = cos(ruv.x * 1.5 - ruv.y * 2.4 - uTime * 0.85) * 0.0018;
        vec2 screenUV = gl_FragCoord.xy / uScreenSize;
        vec2 reflUV = clamp(screenUV + vec2(r1, r2), vec2(0.002), vec2(0.998));
        vec3 reflCol = texture2D(uReflTex, reflUV).rgb;

        vec3 waterNorm = normalize(vec3(r1 * 8.0, 1.0, r2 * 8.0));
        float fresnel = 0.24 + 0.72 * pow(1.0 - clamp(dot(waterNorm, V), 0.0, 1.0), 3.2);

        // Muddy shallow bed beneath clear water
        float bedNoise = noise2(vWorldPos.xz * 22.0);
        vec3 bedCol = mix(vec3(0.16, 0.13, 0.09), vec3(0.24, 0.20, 0.15), bedNoise);
        bedCol *= (0.45 + 0.55 * shadow);

        vec3 waterCol = mix(bedCol, reflCol * vec3(0.93, 0.96, 1.0), clamp(fresnel + 0.32, 0.0, 0.94));
        waterCol = mix(waterCol, bedCol * 1.15, shallowFactor * 0.38);

        // Sun specular glint on water
        vec3 H = normalize(L + V);
        float spec = pow(max(dot(waterNorm, H), 0.0), 140.0) * shadow;
        waterCol += vec3(1.0, 0.97, 0.90) * spec * 0.55;

        // Subtle meniscus highlight right at water boundary
        float rimLine = smoothstep(-0.045, -0.012, pd) * (1.0 - smoothstep(-0.012, 0.0, pd));
        waterCol += vec3(0.12, 0.14, 0.16) * rimLine;

        gl_FragColor = vec4(waterCol, edgeAlpha);
        return;
      }

      // Direct + sky/ground hemisphere lighting
      float diff = max((dot(N, L) + wrap) / (1.0 + wrap), 0.0);
      vec3 sunCol = vec3(1.04, 0.97, 0.86);
      vec3 skyAmb = vec3(0.42, 0.50, 0.62);
      vec3 gndAmb = vec3(0.26, 0.24, 0.18);
      float hemi = N.y * 0.5 + 0.5;
      vec3 ambient = mix(gndAmb, skyAmb, hemi) * 0.68 * houseAO;

      // Rim / subsurface scattering on fur
      float rim = 0.0;
      if (uMatType == 0 || uMatType == 1) {
        float fres = pow(1.0 - max(dot(N, V), 0.0), 2.6);
        float backLight = clamp(dot(-L, V) * 0.5 + 0.5, 0.0, 1.0);
        rim = fres * (0.16 + 0.22 * backLight) * shadow;
      }

      vec3 H = normalize(L + V);
      float spec = pow(max(dot(N, H), 0.0), shininess) * specStrength * shadow;

      vec3 lit = baseCol * (ambient + sunCol * diff * shadow * 0.88) +
                 vec3(1.0, 0.92, 0.78) * rim * baseCol +
                 vec3(1.0, 0.98, 0.95) * spec;

      // Subtle atmospheric perspective in far background
      float dist = length(vWorldPos - uCamPos);
      float fog = smoothstep(14.0, 34.0, dist);
      lit = mix(lit, vec3(0.76, 0.84, 0.90), fog * 0.65);

      // Gentle filmic tone curve & gamma
      lit = lit / (1.0 + lit * 0.18);
      lit = pow(clamp(lit, 0.0, 1.0), vec3(0.92));

      gl_FragColor = vec4(lit, 1.0);
    }
  `;

  function createFBO(w, h) {
    const tex = gl.createTexture();
    gl.bindTexture(gl.TEXTURE_2D, tex);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, w, h, 0, gl.RGBA, gl.UNSIGNED_BYTE, null);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);

    const rb = gl.createRenderbuffer();
    gl.bindRenderbuffer(gl.RENDERBUFFER, rb);
    gl.renderbufferStorage(gl.RENDERBUFFER, gl.DEPTH_COMPONENT16, w, h);

    const fbo = gl.createFramebuffer();
    gl.bindFramebuffer(gl.FRAMEBUFFER, fbo);
    gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, tex, 0);
    gl.framebufferRenderbuffer(gl.FRAMEBUFFER, gl.DEPTH_ATTACHMENT, gl.RENDERBUFFER, rb);

    gl.bindFramebuffer(gl.FRAMEBUFFER, null);
    return { fbo, tex, w, h };
  }

  // --- Procedural Geometry Builder ---
  class MeshBuilder {
    constructor() {
      this.verts = []; // x,y,z, nx,ny,nz, r,g,b, u,v (11 floats)
      this.indices = [];
    }
    addVertex(p, n, c, uv) {
      const idx = this.verts.length / 11;
      this.verts.push(
        p[0], p[1], p[2],
        n[0], n[1], n[2],
        c[0], c[1], c[2],
        uv ? uv[0] : 0, uv ? uv[1] : 0
      );
      return idx;
    }
    addTri(i0, i1, i2) {
      this.indices.push(i0, i1, i2);
    }
    addQuad(p0, p1, p2, p3, c0, c1, c2, c3, uvScaleX = 1, uvScaleY = 1) {
      const e1 = vSub(p1, p0);
      const e2 = vSub(p3, p0);
      const n = vNorm(vCross(e1, e2));
      const i0 = this.addVertex(p0, n, c0, [0, 0]);
      const i1 = this.addVertex(p1, n, c1 || c0, [uvScaleX, 0]);
      const i2 = this.addVertex(p2, n, c2 || c0, [uvScaleX, uvScaleY]);
      const i3 = this.addVertex(p3, n, c3 || c0, [0, uvScaleY]);
      this.addTri(i0, i1, i2);
      this.addTri(i0, i2, i3);
    }
    addBox(cx, cy, cz, sx, sy, sz, col, topCol) {
      const x0 = cx - sx*0.5, x1 = cx + sx*0.5;
      const y0 = cy - sy*0.5, y1 = cy + sy*0.5;
      const z0 = cz - sz*0.5, z1 = cz + sz*0.5;
      const tc = topCol || col;
      // Front (+Z)
      this.addQuad([x0,y0,z1],[x1,y0,z1],[x1,y1,z1],[x0,y1,z1], col, col, tc, tc, sx, sy);
      // Back (-Z)
      this.addQuad([x1,y0,z0],[x0,y0,z0],[x0,y1,z0],[x1,y1,z0], col, col, tc, tc, sx, sy);
      // Right (+X)
      this.addQuad([x1,y0,z1],[x1,y0,z0],[x1,y1,z0],[x1,y1,z1], col, col, tc, tc, sz, sy);
      // Left (-X)
      this.addQuad([x0,y0,z0],[x0,y0,z1],[x0,y1,z1],[x0,y1,z0], col, col, tc, tc, sz, sy);
      // Top (+Y)
      this.addQuad([x0,y1,z1],[x1,y1,z1],[x1,y1,z0],[x0,y1,z0], tc, tc, tc, tc, sx, sz);
      // Bottom (-Y)
      this.addQuad([x0,y0,z0],[x1,y0,z0],[x1,y0,z1],[x0,y0,z1], col, col, col, col, sx, sz);
    }
    addEllipsoid(cx, cy, cz, rx, ry, rz, latSegs, lonSegs, colorFn, deformFn) {
      const baseIdx = this.verts.length / 11;
      const posGrid = [];
      for (let i = 0; i <= latSegs; i++) {
        const v = i / latSegs;
        const theta = v * Math.PI;
        const st = Math.sin(theta), ct = Math.cos(theta);
        for (let j = 0; j <= lonSegs; j++) {
          const u = j / lonSegs;
          const phi = u * Math.PI * 2;
          const sp = Math.sin(phi), cp = Math.cos(phi);
          let nx = st * cp;
          let ny = ct;
          let nz = st * sp;
          let px = cx + nx * rx;
          let py = cy + ny * ry;
          let pz = cz + nz * rz;
          if (deformFn) {
            const d = deformFn(px, py, pz, nx, ny, nz, u, v);
            px = d[0]; py = d[1]; pz = d[2];
          }
          posGrid.push([px, py, pz, nx, ny, nz, u, v]);
        }
      }
      // Compute smooth normals via finite grid neighbors
      const stride = lonSegs + 1;
      for (let i = 0; i <= latSegs; i++) {
        for (let j = 0; j <= lonSegs; j++) {
          const cur = posGrid[i * stride + j];
          const jPrev = j > 0 ? j - 1 : lonSegs - 1;
          const jNext = j < lonSegs ? j + 1 : 1;
          const iPrev = Math.max(0, i - 1);
          const iNext = Math.min(latSegs, i + 1);
          const pL = posGrid[i * stride + jPrev];
          const pR = posGrid[i * stride + jNext];
          const pU = posGrid[iPrev * stride + j];
          const pD = posGrid[iNext * stride + j];
          const tangU = vSub(pR, pL);
          const tangV = vSub(pD, pU);
          let n = vCross(tangV, tangU);
          if (vLen(n) < 1e-6) {
            n = [cur[3]/rx, cur[4]/ry, cur[5]/rz];
          }
          n = vNorm(n);
          const col = colorFn ? colorFn(cur[0], cur[1], cur[2], n[0], n[1], n[2], cur[6], cur[7]) : [1,1,1];
          this.addVertex([cur[0], cur[1], cur[2]], n, col, [cur[6], cur[7]]);
        }
      }
      for (let i = 0; i < latSegs; i++) {
        for (let j = 0; j < lonSegs; j++) {
          const a = baseIdx + i * stride + j;
          const b = a + 1;
          const c = a + stride;
          const d = c + 1;
          this.addTri(a, c, b);
          this.addTri(b, c, d);
        }
      }
    }
    addTube(rings, segs, colorFn) {
      // rings: array of { c:[x,y,z], rx, rz, ry, up, right, col }
      const baseIdx = this.verts.length / 11;
      const pts = [];
      for (let i = 0; i < rings.length; i++) {
        const r = rings[i];
        const v = i / (rings.length - 1);
        for (let j = 0; j <= segs; j++) {
          const u = j / segs;
          const ang = u * Math.PI * 2;
          const ca = Math.cos(ang), sa = Math.sin(ang);
          const offset = vAdd(vScale(r.right, ca * r.rx), vScale(r.up, sa * r.ry));
          const p = vAdd(r.c, offset);
          const nGuess = vNorm(vAdd(vScale(r.right, ca / (r.rx || 1)), vScale(r.up, sa / (r.ry || 1))));
          pts.push({ p, n: nGuess, u, v, rCol: r.col });
        }
      }
      const stride = segs + 1;
      for (let i = 0; i < rings.length; i++) {
        for (let j = 0; j <= segs; j++) {
          const item = pts[i * stride + j];
          const jPrev = j > 0 ? j - 1 : segs - 1;
          const jNext = j < segs ? j + 1 : 1;
          const iPrev = Math.max(0, i - 1);
          const iNext = Math.min(rings.length - 1, i + 1);
          const tU = vSub(pts[i * stride + jNext].p, pts[i * stride + jPrev].p);
          const tV = vSub(pts[iNext * stride + j].p, pts[iPrev * stride + j].p);
          let n = vCross(tU, tV);
          n = vLen(n) > 1e-6 ? vNorm(n) : item.n;
          if (vDot(n, item.n) < 0) n = vScale(n, -1);
          const col = colorFn ? colorFn(item.p, n, item.u, item.v) : (item.rCol || [1,1,1]);
          this.addVertex(item.p, n, col, [item.u, item.v]);
        }
      }
      for (let i = 0; i < rings.length - 1; i++) {
        for (let j = 0; j < segs; j++) {
          const a = baseIdx + i * stride + j;
          const b = a + 1;
          const c = a + stride;
          const d = c + 1;
          this.addTri(a, b, c);
          this.addTri(b, d, c);
        }
      }
    }
    build() {
      const vbo = gl.createBuffer();
      gl.bindBuffer(gl.ARRAY_BUFFER, vbo);
      gl.bufferData(gl.ARRAY_BUFFER, new Float32Array(this.verts), gl.STATIC_DRAW);

      const ibo = gl.createBuffer();
      gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, ibo);
      const useUint32 = this.verts.length / 11 > 65535;
      gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, new Uint16Array(this.indices), gl.STATIC_DRAW);
      return { vbo, ibo, count: this.indices.length };
    }
  }

  function pomTorsoMesh() {

      (x, y, z, nx, ny, nz) => {

        c = vLerp(c, COL_CREAM, clamp(belly * 0.85 + frontBib * 0.75, 0, 1));
        return c;

      (x, y, z, nx, ny, nz, u, v) => {
        // Sculpt ribcage fullness, slight tuck at waist, fluffy buttocks
        const fluff = 1.0 + 0.045 * Math.sin(u * Math.PI * 14) * Math.sin(v * Math.PI * 9)
                          + 0.03 * Math.cos(u * Math.PI * 22 + v * 12.0);
        let sx = 1.0, sy = 1.0;

          sx *= 1.04;

          // Broad chest
          sx *= 1.08;
          sy *= 1.06;
        }
        return [x * sx * fluff, y * sy * fluff, z * fluff];
      }
    );

    // Pronounced Pomeranian Lion-like Neck Ruff & Chest Frill!
    mb.addEllipsoid(0, 0.045, 0.135, 0.152, 0.148, 0.115, 20, 28,

        const ang = Math.atan2(y - 0.045, x);
        const tuft = 1.0 + 0.075 * Math.pow(Math.sin(ang * 9.0) * 0.5 + 0.5, 1.5) * (1.0 - Math.abs(nz) * 0.6)

    );

  function pomHeadMesh() {
    const mb = new MeshBuilder();
    // Rounded foxy Pomeranian cranium + cheek fluff
    mb.addEllipsoid(0, 0.015, 0.01, 0.102, 0.094, 0.095, 20, 26,

        const browSpot = Math.exp(-((Math.abs(x) - 0.034)*(Math.abs(x) - 0.034) + (y - 0.045)*(y - 0.045) + (z - 0.08)*(z - 0.08)) * 950.0);
        let c = vLerp(COL_ORANGE_MID, COL_ORANGE_DEEP, clamp((y - 0.04) * 6.0, 0, 0.55));
        c = vLerp(c, COL_CREAM_BRIGHT, clamp(cheek * 0.88 + browSpot * 0.65, 0, 1));
        return c;
      },
      (x, y, z, nx, ny, nz, u, v) => {

        let cx = x, cy = y, cz = z;

          cx *= 1.15;
        }
        const fur = 1.0 + 0.04 * Math.sin(u * Math.PI * 16) * Math.sin(v * Math.PI * 10);
        return [cx * fur, cy * fur, cz * fur];
      }
    );

        }

      });

    mb.addEllipsoid(0, -0.030, 0.092, 0.024, 0.008, 0.034, 8, 12, () => [0.28, 0.10, 0.11]);

  function buildLimbSegmentMesh(radiusTop, radiusBot, featherBack, colTop, colBot) {
    // Unit limb segment along local Y from 0 (top joint) to -1 (bottom joint)
    const mb = new MeshBuilder();
    const rings = [];
    const steps = 9;
    for (let i = 0; i <= steps; i++) {
      const t = i / steps;

        right: [1, 0, 0],
        up: [0, 0, 1],

        ry: r + fBack * 0.55,
        col: vLerp(colTop, colBot, t)
      });
    }

    rings.unshift({ c: [0, radiusTop * 0.45, 0], right: [1,0,0], up: [0,0,1], rx: radiusTop*0.5, ry: radiusTop*0.5, col: colTop });

      const backLight = clamp(-n[2] * 0.6 + 0.2, 0, 1);

    });
    return mb.build();
  }

  function pomTailPlumeMesh() {
    // Full plumed Pomeranian tail curled forward & fanned richly over the back!
    // Built in tail-base space (0,0,0 = tail root on high croup; +Z is forward over the back, +Y is up)
    const mb = new MeshBuilder();

      const plumeEnv = Math.sin(Math.pow(t, 0.65) * Math.PI);

      const ry = 0.024 + 0.072 * plumeEnv;

      // Local tangent along curve
      const dAng = 0.01;
      const ny2 = Math.sin((archAngle + dAng) * 0.72) * 0.145 + 0.03 * (t + 0.01);
      const nz2 = (1.0 - Math.cos((archAngle + dAng) * 0.85)) * 0.135 - 0.025;
      const tang = vNorm([0, ny2 - cy, nz2 - cz]);
      const right = [1, 0, 0];
      const up = vNorm(vCross(tang, right));

      rings.push({ c: [cx, cy, cz], right, up, rx, ry });
    }

    mb.addTube(rings, 20, (p, n, u, v) => {

      }
      return c;
    });
    return mb.build();
  }

<style>

<body>
<canvas id="stage" width="960" height="540"></canvas>
<script>
(() => {
  const canvas = document.getElementById('stage');
  const ctx = canvas.getContext('2d');
  // Fixed scene constants
  const W = 960, H = 540;

  // Scene objects positions (in world space for this 2.5D look)
  const groundY = 420;
  const doghouse = {
    x: 520, y: groundY - 140, w: 140, h: 140
  };
  const doorway = { x: doghouse.x + 10, y: doghouse.y + 20, w: 40, h: 120 };

  // Bone resting position (in yard)
  const boneRest = { x: 320, y: groundY - 6, w: 22, h: 6 };

  // Utility helpers
  function lerp(a,b,t){ return a+(b-a)*t; }
  function clamp(v, a, b){ return Math.max(a, Math.min(b, v)); }

  // Deterministic random-ish shading (no external randomness)
  function shade(r, g, b, a=1){
    return `rgba(${r|0},${g|0},${b|0},${a})`;
  }

  // Draw a soft rounded rect
  function roundedRect(ctx, x, y, w, h, r){
    const rr = Math.min(r, w/2, h/2);
    ctx.beginPath();
    ctx.moveTo(x+rr, y);
    ctx.arcTo(x+w, y,   x+w, y+h, rr);
    ctx.arcTo(x+w, y+h, x,   y+h, rr);
    ctx.arcTo(x,   y+h, x,   y,   rr);
    ctx.arcTo(x,   y,   x+w, y,   rr);
    ctx.closePath();
  }

  // Draw bone (single instance)
  function drawBone(ctx, bone, angle=0){
    ctx.save();
    ctx.translate(bone.x, bone.y);
    ctx.rotate(angle);
    // bone body
    ctx.fillStyle = '#deb887';
    roundedRect(ctx, -bone.w/2, -bone.h/2, bone.w, bone.h, 3);
    ctx.fill();
    // bone ends (simple circles)
    ctx.fillStyle = '#f5f0e6';
    ctx.beginPath();
    ctx.arc(-bone.w/2, 0, 3, 0, Math.PI*2);
    ctx.arc(bone.w/2, 0, 3, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  // Draw a Pomeranian-like dog (stylized but recognizable)
  // Pose parameters:
  // tCycle: 0..1 fractional through a segment
  // facing: 1 for side view facing right; -1 for mirrored (not used heavily)
  // view: 0: 3/4 left, 1: 3/4 right, 2: frontal-ish
  function drawDog(ctx, pose){
    // pose: { x, y, scale, headAngle, bodyTilt, legPhase, tailPhase, mouthOpen, boneInMouth, facing, view, insideDoorway, turn}
    const s = pose.scale;
    const x = pose.x, y = pose.y;
    ctx.save();
    ctx.translate(x, y);
    ctx.scale(s, s);

    // Simple directional shading to imply thickness
    // Body is an organic shape via ellipses and rounded connections
    // We approximate anatomy: torso, hindquarters, neck, head, ears, tail

    // Shadow on ground
    ctx.fillStyle = 'rgba(0,0,0,0.15)';
    ctx.beginPath();
    ctx.ellipse(0, 40, 60, 22, 0, 0, Math.PI*2);
    ctx.fill();

    // Torso (core mass)
    ctx.fillStyle = '#f6a042'; // warm orange-creamy
    ctx.strokeStyle = '#c97d1d';
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    ctx.ellipse(-4, -10, 90, 42, pose.bodyTilt, 0, Math.PI*2);
    ctx.fill();
    ctx.stroke();

    // Neck & head
    ctx.save();
    ctx.translate(60, -28);
    ctx.rotate(pose.headAngle || 0.2);
    // Neck
    ctx.fillStyle = '#e7a95f';
    ctx.beginPath();
    ctx.ellipse(0, 0, 22, 42, 0, 0, Math.PI*2);
    ctx.fill();
    // Head
    ctx.fillStyle = '#d9802a';
    ctx.beginPath();
    ctx.ellipse(28, -8, 28, 26, 0.2, 0, Math.PI*2);
    ctx.fill();
    // Ears
    ctx.fillStyle = '#d5802a';
    ctx.beginPath();
    ctx.moveTo(48, -24); ctx.lineTo(56, -48); ctx.lineTo(40, -34); ctx.closePath();
    ctx.fill();
    ctx.beginPath();
    ctx.moveTo(34, -24); ctx.lineTo(22, -48); ctx.lineTo(40, -34); ctx.closePath();
    ctx.fill();
    // Eyes
    ctx.fillStyle = '#1b1b1b';
    ctx.beginPath(); ctx.arc(44, -6, 3, 0, Math.PI*2); ctx.fill();
    ctx.beginPath(); ctx.arc(50, -6, 3, 0, Math.PI*2); ctx.fill();
    // Mouth area (subtle)
    ctx.strokeStyle = '#6b4a2e';
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.arc(38, 8, 6, 0, Math.PI);
    ctx.stroke();
    // Mouth grip indicator (bone currently carried)
    if (pose.boneInMouth){
      // show lower jaw gently closed
      ctx.strokeStyle = '#000';
      ctx.lineWidth = 0.8;
      ctx.beginPath(); ctx.moveTo(28, 0); ctx.lineTo(52, 0); ctx.stroke();
    }
    ctx.restore();

    // Torso continuation to hindquarters
    ctx.save();
    ctx.translate(-40, 6);
    ctx.rotate(-pose.bodyTilt * 0.5);
    // Back end
    ctx.fillStyle = '#e88e2b';
    ctx.beginPath();
    ctx.ellipse(-60, 0, 34, 28, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();

    // Legs (four)
    // We'll position legs with slight phase offsets to simulate gait
    const legLen = 46;
    const legSpread = 28;
    const legPhase = pose.legPhase; // 0..1
    // Front left
    drawLeg(ctx, -22, 12, legLen, legSpread, legPhase, pose);
    // Front right
    drawLeg(ctx, -2, 12, legLen, legSpread, legPhase+0.5, pose);
    // Back left
    drawLeg(ctx, -28, 52, legLen, legSpread, legPhase+0.25, pose);
    // Back right
    drawLeg(ctx,  -8, 52, legLen, legSpread, legPhase+0.75, pose);

    // Tail
    ctx.save();
    ctx.translate(-20, -6);
    ctx.rotate(pose.tailPhase);
    ctx.fillStyle = '#e88e2b';
    ctx.beginPath();
    ctx.moveTo(0,0); ctx.quadraticCurveTo(-28,-20,-20,-40); ctx.lineTo(6,-20); ctx.closePath();
    ctx.fill();
    ctx.restore();

    // Bone in mouth (if gripping)
    if (pose.boneInMouth && pose.bone){
      ctx.save();
      // Bone drawn within mouth region; position relative to head
      ctx.translate(66, -4);
      drawBone(ctx, pose.bone, 0);
      ctx.restore();
    }

    // Ground contact shadow under body
    ctx.fillStyle = 'rgba(0,0,0,0.25)';
    ctx.beginPath();
    ctx.ellipse(0, 42, 60, 14, 0, 0, Math.PI*2);
    ctx.fill();

    ctx.restore();
  }

  function drawLeg(ctx, x, y, len, spread, phase, pose){
    // Leg stiffness; 2 segments
    const k = Math.sin((phase%1)*Math.PI*2);
    const upper = clamp(lerp(18, len, 0.6 + 0.4*k), 6, len);
    const lower = clamp(lerp(14, 28, 0.5 + 0.5*k), 6, 40);
    ctx.save();
    ctx.translate(x, y);
    // thigh
    ctx.strokeStyle = '#d07e1a';
    ctx.lineWidth = 6;
    ctx.beginPath();
    ctx.moveTo(0,0); ctx.lineTo(0+spread*0.5, -upper);
    ctx.stroke();
    // leg
    ctx.lineWidth = 5;
    ctx.beginPath();
    ctx.moveTo(0, -upper);
    ctx.lineTo(0, -upper + lower);
    ctx.stroke();
    // paw
    ctx.fillStyle = '#c56b1f';
    ctx.beginPath();
    ctx.arc(0, -upper + lower + 4, 4, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  // Exposed API
  window.renderFrame = function(timeSeconds){
    // timeSeconds is a real time; we map to 0..12 with clamp
    let t = timeSeconds;
    if (t < 0) t = 0;
    if (t > 12) t = 12;

    // Clear
    ctx.clearRect(0,0,W,H);

    // Background
    // sky gradient
    const g = ctx.createLinearGradient(0,0,0,540);
    g.addColorStop(0, '#87cefa');
    g.addColorStop(0.5, '#b3e0ff');
    g.addColorStop(1, '#7ec0a6');
    ctx.fillStyle = g;
    ctx.fillRect(0,0,W,H);

    // distant hills
    ctx.fillStyle = '#9bd39a';
    ctx.beginPath();
    ctx.ellipse(-100, 430, 320, 120, 0, 0, Math.PI*2);
    ctx.fill();
    // backyard ground
    ctx.fillStyle = '#5a8a4a';
    ctx.fillRect(0, groundY, W, H-groundY);

    // Doghouse
    // simple wooden box and doorway
    ctx.fillStyle = '#8b5a2b';
    roundedRect(ctx, doghouse.x, doghouse.y, doghouse.w, doghouse.h, 8);
    ctx.fill();
    // doorway opening
    ctx.fillStyle = '#6b3a1d';
    ctx.fillRect(doorway.x, doorway.y, doorway.w, doorway.h);
    // interior shading
    ctx.fillStyle = 'rgba(0,0,0,0.25)';
    ctx.fillRect(doorway.x+6, doorway.y+6, doorway.w-12, doorway.h-12);

    // Bone (rest or carried)
    const bone = {
      x: boneRest.x,
      y: boneRest.y,
      w: boneRest.w,
      h: boneRest.h
    };

    // Action scheduling
    // We'll derive a simple state machine via time to place dog position, bone state
    // 0-1s: establish
    // 1-3s: run toward doghouse
    // 3-4.5: enter
    // 4.5-6: grab
    // 6-8: back out
    // 8-10: turn to viewer
    // 10-12: front pose

    // Deterministic easing
    function seg(a,b,d){
      return clamp((t - a) / (b-a), 0, 1) ;
    }

    // Initialize pose
    let pose = {
      x: 120,
      y: groundY - 20,
      scale: 1,
      bodyTilt: 0.15,
      headAngle: 0.25,
      legPhase: 0,
      tailPhase: 0.2,
      boneInMouth: false,
      bone: bone,
      facing: 1,
      view: 0,
      insideDoorway: false,
      turn: 0
    };

    // Segmenting movement
    if (t < 1){
      // dog notices bone: place near doorway, head tilts to bone
      pose.x = 140;
      pose.y = groundY - 20;
      pose.legPhase = 0.0;
      pose.headAngle = 0.4;
      // bone remains at rest
      bone.x = boneRest.x;
    } else if (t < 3){
      // run toward doghouse
      const s = seg(1,3,t);
      pose.x = lerp(140, doghouse.x - 40, s);
      pose.legPhase = s;
      pose.headAngle = lerp(0.4, 0.2, s);
      // bone still on ground near doorway (slightly toward dog)
      bone.x = lerp(boneRest.x, boneRest.x + 40, s);
      bone.y = boneRest.y - Math.sin(s*Math.PI)*2;
    } else if (t < 4.5){
      // enter doorway (body progressively inside)
      const s = seg(3,4.5,t);
      pose.x = lerp(doghouse.x - 40, doorway.x + doorway.w - 20, s);
      pose.y = groundY - 20;
      pose.legPhase = 1 - s;
      pose.headAngle = lerp(0.2, 0.5, s);
      pose.insideDoorway = s > 0.2;
      // adjust bone location near doorway
      bone.x = lerp(bone.x, doorway.x + doorway.w - 40, s);
      bone.y = boneRest.y;
      if (s > 0.6){
        // bone appears grabbable
      }
    } else if (t < 6){
      // grab inside doorway: mouth opens, bone moves toward mouth and disappears from ground
      const s = seg(4.5,6,t);
      pose.x = lerp(doorway.x + doorway.w - 20, doorway.x + doorway.w - 6, s);
      pose.y = groundY - 26 * (1 - 0.3*Math.sin(t*4));
      pose.legPhase = 0.9 * (1 - s);
      pose.headAngle = lerp(0.5, 0.2, s);
      pose.boneInMouth = true;
      // bone collected
      bone.x = lerp( bone.x, pose.x + 24, s);
      bone.y = lerp(bone.y, pose.y - 6, s);
      bone.w = 22; bone.h = 6;
      // disappear from yard as held
    } else if (t < 8){
      // back out through doorway with bone
      const s = seg(6,8,t);
      pose.x = lerp(doorway.x + doorway.w - 6, 420, s);
      pose.y = groundY - 20;
      pose.legPhase = 0.2 + 0.8*s;
      pose.headAngle = lerp(0.2, 0.4, s);
      pose.boneInMouth = true;
      // bone stays in mouth; move by mouth position relative to pose
      bone.x = pose.x + 22;
      bone.y = pose.y - 6;
    } else if (t < 10){
      // turn to viewer
      const s = seg(8,10,t);
      pose.x = 420; pose.y = groundY - 22;
      pose.scale = 1.0;
      pose.bodyTilt = lerp(0.25, -0.25, s);
      pose.headAngle = lerp(0.4, 0.15, s);
      pose.tailPhase = 0.2 + 0.6*s;
      pose.legPhase = 0.6 + 0.4*s;
      pose.boneInMouth = true;
      // bone moves to be visible in front
      bone.x = pose.x + 30;
      bone.y = pose.y - 10;
    } else {
      // final frontal pose (12s hold)
      const s = 1;
      pose.x = 480; pose.y = groundY - 22;
      pose.scale = 1;
      pose.bodyTilt = -0.05;
      pose.headAngle = 0.05;
      pose.legPhase = 0.5;
      pose.tailPhase = 0.4;
      pose.boneInMouth = true;
      bone.x = pose.x + 34;
      bone.y = pose.y - 6;
    }

    // Draw bone (global)
    // If carried, bone is drawn in mouth region by dog drawing; we still draw a little hint in scene
    if (!pose.boneInMouth){
      // draw resting bone
      drawBone(ctx, bone, 0);
    } else {
      // draw bone near mouth (handled inside drawDog)
    }

    // Draw dog (foreground)
    drawDog(ctx, pose);

    // Foreground subtle occlusion: doorway opening should occlude parts of dog entering
    // We'll implement a simple mask by drawing a dark rectangle over the doorway area to imply depth
    ctx.fillStyle = 'rgba(0,0,0,0.15)';
    ctx.fillRect(doorway.x + 0, doorway.y, doorway.w, doorway.h);
  };

  // Expose a constant function in case external harness calls renderFrame(0)
  window.renderFrame(0);

})();
</script>
</body>

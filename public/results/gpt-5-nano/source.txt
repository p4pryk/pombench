<canvas id="stage" width="960" height="540"></canvas>
<script>
(function(){
  const canvas = document.getElementById('stage');
  const ctx = canvas.getContext('2d');
  // constants
  const W = 960, H = 540;
  // scene layout (fixed)
  const doghouse = { x: 780, y: 260, w: 140, h: 140, door: {x: 820, y: 320, w: 40, h: 60} };
  const puddle = { x: 420, y: 420, w: 220, h: 40 }; // ellipse-ish puddle along approach
  const bonePos = { x: 520, y: 420 }; // resting bone inside doghouse area before retrieval
  // dog path domain
  // time 0..12s
  // helper deterministic functions
  function clamp(v,a,b){ return Math.max(a, Math.min(b, v)); }
  function lerp(a,b,t){ return a + (b-a)*t; }
  // draw utilities
  function drawGradientGround() {
    const g = ctx.createLinearGradient(0, 0, 0, H);
    g.addColorStop(0, '#2b5a2f');
    g.addColorStop(0.6, '#4b8b4a');
    g.addColorStop(1, '#3b6e38');
    ctx.fillStyle = g;
    ctx.fillRect(0, 0, W, H);
  }
  function drawSky() {
    // subtle sky tint
    const g = ctx.createLinearGradient(0,0,0,200);
    g.addColorStop(0, 'rgba(140,190,255,0.4)');
    g.addColorStop(1, 'rgba(180,210,255,0.2)');
    ctx.fillStyle = g;
    ctx.fillRect(0,0,W,200);
  }
  function drawPuddle(oX,oY,w,h,shade){
    ctx.save();
    // water
    ctx.fillStyle = 'rgba(60,120,180,0.75)';
    ctx.beginPath();
    ctx.ellipse(oX,oY,w,h,0,0,Math.PI*2);
    ctx.fill();
    // reflection (simple)
    ctx.globalAlpha = 0.6;
    ctx.save();
    ctx.translate(0,0);
    // reflect dog will be drawn as mirror later; here leave as subtle ripples
    ctx.restore();
    ctx.restore();
    // edge highlight
    ctx.strokeStyle = 'rgba(200,240,255,0.4)';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.ellipse(oX,oY,w,h,0,0,Math.PI*2);
    ctx.stroke();
  }
  function drawBone(b){
    // bone at given pos; small, with two cylinders
    ctx.save();
    ctx.fillStyle = '#fffaf0';
    ctx.strokeStyle = '#000';
    ctx.lineWidth = 1;
    // bone body
    ctx.translate(b.x, b.y);
    ctx.rotate(b.rot);
    ctx.beginPath();
    ctx.moveTo(-12, -4);
    ctx.lineTo(-4, -4);
    ctx.lineTo(-2, -7);
    ctx.lineTo(0, -7);
    ctx.lineTo(6, -4);
    ctx.lineTo(12, -4);
    ctx.lineTo(6, 0);
    ctx.lineTo(0, 0);
    ctx.lineTo(-2, 3);
    ctx.lineTo(-6, 3);
    ctx.lineTo(-12, -0);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();
    // highlight
    ctx.fillStyle = 'rgba(255,255,255,0.9)';
    ctx.fillRect(-6, -2, 3, 2);
    ctx.restore();
  }
  // Pomeranian-like dog drawing with articulated limbs
  function drawDog(state){
    // state: {x,y,theta, limbsPhase, headTilt, mouthOpen, inDoor, carryingBone, reflection}
    ctx.save();
    // body transform
    ctx.translate(state.x, state.y);
    ctx.rotate(state.theta);
    // fur color shading
    // torso
    ctx.fillStyle = '#f6a058'; // warm orange-cream
    ctx.strokeStyle = '#2b2b2b';
    ctx.lineWidth = 1.2;
    // main body as rounded capsule
    ctx.beginPath();
    ctx.moveTo(-40, -14);
    ctx.quadraticCurveTo(0,-40, 42,-14);
    ctx.quadraticCurveTo(60, 8, 42, 28);
    ctx.quadraticCurveTo(0, 34, -28, 24);
    ctx.quadraticCurveTo(-50, 8, -40, -14);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    // neck and head
    ctx.save();
    ctx.translate(28, -12);
    ctx.rotate(state.headTilt);
    // neck fluff
    ctx.fillStyle = '#f6ae6d';
    ctx.beginPath();
    ctx.ellipse(0,0,14,12,0,0,Math.PI*2);
    ctx.fill();
    // head
    ctx.fillStyle = '#e9a058';
    ctx.beginPath();
    ctx.arc(18, 0, 12, 0, Math.PI*2);
    ctx.fill();
    // muzzle
    ctx.fillStyle = '#d0905c';
    ctx.beginPath();
    ctx.arc(28, -2, 6, 0, Math.PI*2);
    ctx.fill();
    // ears
    ctx.fillStyle = '#e38b39';
    ctx.beginPath();
    ctx.moveTo(8,-14);
    ctx.lineTo(0,-28);
    ctx.lineTo(14,-12);
    ctx.closePath();
    ctx.fill();
    ctx.beginPath();
    ctx.moveTo(22,-14);
    ctx.lineTo(30,-28);
    ctx.lineTo(16,-12);
    ctx.closePath();
    ctx.fill();
    // eyes
    ctx.fillStyle = '#1b1b1b';
    ctx.beginPath();
    ctx.arc(10, -2, 2, 0, Math.PI*2);
    ctx.arc(22, -2, 2, 0, Math.PI*2);
    ctx.fill();
    // read ear fur
    ctx.restore();

    // tail (plumed)
    ctx.save();
    ctx.translate(-40, -14);
    ctx.rotate(-state.tailTilt);
    ctx.fillStyle = '#f9922b';
    ctx.beginPath();
    ctx.moveTo(0,0);
    ctx.quadraticCurveTo(-22,-22,-42,-6);
    ctx.quadraticCurveTo(-32,-2,-20,6);
    ctx.quadraticCurveTo(-22,2,-10,6);
    ctx.quadraticCurveTo(-2,2,0,0);
    ctx.fill();
    ctx.restore();

    // legs - four limbs with phase offset for walking
    function leg(offsetX, offsetY, phase){
      ctx.save();
      ctx.translate(offsetX, offsetY);
      // simple hinge with rotation
      const ang = Math.sin(state.limbPhase + phase) * 0.6;
      ctx.rotate(ang);
      ctx.fillStyle = '#d57a2a';
      ctx.fillRect(-3,0,6,22);
      // paw
      ctx.fillStyle = '#3a2a20';
      ctx.fillRect(-4,22,8,4);
      ctx.restore();
    }
    // left front
    leg(22, 18, 0.0);
    // right front
    leg(18, 18, Math.PI);
    // left back
    leg(-18, 20, Math.PI/2);
    // right back
    leg(-22, 20, -Math.PI/2);

    // if carrying bone, draw bone near mouth
    if (state.carryingBone){
      // bone held in mouth
      ctx.save();
      ctx.translate(28, -2);
      ctx.rotate(0.1);
      drawBone({x:0,y:0,rot:0});
      ctx.restore();
    }

    // mouth open slightly or closed
    ctx.restore();
  }

  // initial bone inside doghouse
  // animation state builder
  function buildDogState(t){
    // t in seconds
    // segment schedules
    // 0-1: dog notices bone inside doghouse; near left approach; puddle visible
    // 1-3: running toward doghouse, jump puddle
    // 3-4.5: enters doorway
    // 4.5-6: grabs bone; bone leaves its resting position
    // 6-8: backs out
    // 8-10: turns toward viewer
    // 10-12: front pose
    let s = 0;
    // base ground position
    // Dog path along x from 120 to 780
    const runX = 120 + (t/12)*(640); // across frame
    // vertical slight bob
    const bob = Math.sin(t*2)*2;
    // puddle jump
    let jumping = false;
    let jumpY = 0;
    if (t>=1 && t<=3){
      jumping = true;
      // simple parabolic arc
      const a = (t-1)/2;
      // height
      const h = 40*Math.max(0, 1 - Math.abs(a-0.5)*2);
      jumpY = -h;
    }
    // entering doghouse
    let inDoor = t>=3 && t<=4.5;
    // grabbing bone
    let carrying = false;
    if (t>=4.5 && t<=6){
      carrying = (t>=5); // after grip moment around 5s; simple rule
    } else if (t>6){
      carrying = true;
    }
    // turning to viewer
    const turnToViewer = t>=8 && t<=10;
    // final pose
    // final facing viewer
    const facingAngle = t>=10 ? 0 : null;

    // compute orientation and position
    // orientation rotates a bit during turn
    const theta = (t<8) ? 0 : (turnToViewer ? 0.25 : 0); // small turn
    // doghouse occlusion: if inDoor, draw portion inside
    const distY = bob;
    // height offset so feet align
    const baseY = 360 + distY;

    // positions
    const pos = {
      x: runX,
      y: baseY - (jumpY),
      theta: theta,
      headTilt: 0.15 * Math.sin(t*3),
      limbPhase: t * 4,
      tailTilt: 0.25 * Math.sin(t*3.2),
      carryStartX: bonePos.x, // for bone position reference
      carryingBone: carrying,
      inDoor: inDoor,
      reflection: false
    };
    // adjust for approach during jump
    if (jumping){
      pos.y += jumpY;
    }
    // subtle forward/backward to simulate approach to doorway before entering
    if (t>=3 && t<4){
      pos.x = lerp(runX, doghouse.x - 20, (t-3)/1);
    }
    // entering doorway: keep some read through
    if (t>=4.5 && t<=5.5){
      // partially inside: hold head/forequarters in doorway
      pos.x = doghouse.x - 40;
      pos.theta = -0.02;
    }
    // after grab, back out mouse (move slightly to left)
    if (t>6 && t<8){
      pos.x = lerp(doghouse.x - 40, 540, (t-6)/2);
      pos.theta = 0.08;
    }
    // final face to camera
    if (t>=10){
      pos.theta = 0.0;
      pos.headTilt = -0.05;
    }

    return pos;
  }

  // initial: draw scene
  function renderFrame(timeSeconds){
    const t = timeSeconds;
    // clear
    ctx.clearRect(0,0,W,H);
    // background
    drawSky();
    drawGradientGround();

    // puddle
    // keep puddle visible
    drawPuddle(puddle.x, puddle.y, puddle.w, puddle.h, 0.6);

    // doghouse (simple depth: shadow on ground)
    ctx.save();
    ctx.fillStyle = '#7a4f25';
    ctx.fillRect(doghouse.x - doghouse.w/2, doghouse.y - doghouse.h, doghouse.w, doghouse.h);
    // doorway
    ctx.fillStyle = '#2a1a0f';
    ctx.fillRect(doghouse.door.x, doghouse.door.y, doghouse.door.w, doghouse.door.h);
    // interior edge
    ctx.fillStyle = 'rgba(0,0,0,0.25)';
    ctx.fillRect(doghouse.door.x+10, doghouse.door.y+10, 10, 20);
    ctx.restore();

    // bone resting inside doghouse (only visible outside if inside doorway)
    // We'll render a bone near the doorway edge when not carried, as one bone exists
    // Determine if bone still inside (time<5 ) it's in doghouse resting
    ctx.save();
    let boneVisible = true;
    let boneX = bonePos.x;
    let boneY = bonePos.y;
    if (t < 5){
      // resting inside behind doorway, only partial visible near threshold
      boneX = doghouse.door.x - 6;
      boneY = doghouse.door.y + 22;
    } else {
      boneX = 520;
      boneY = 420;
      // if carried later, bone is drawn by drawDog with carryingBone true
    }
    if (t<4.5){
      drawBone({x: boneX, y: boneY, rot: 0});
    }
    ctx.restore();

    // dog state
    const dog = buildDogState(t);
    // compute dog's on-canvas position and render
    // map base: we want dog near left to mid; above functions assume absolute
    // Convert state into on-canvas coordinates anchored near left-mid
    // We'll place dog.x directly; ensure within frame
    const dogState = {
      x: dog.x,
      y: dog.y,
      theta: dog.theta,
      headTilt: dog.headTilt,
      limbPhase: dog.limbPhase,
      tailTilt: dog.tailTilt,
      carryingBone: (t>=4.5 && t<=12) ? (t>=5) : false
    };
    // If near doorway and within 3-4.5s, we allow partial occlusion visually by drawing dog partly behind door (we approximate by lower y)
    if (t>=3 && t<=4.5){
      // tilt slightly into doorway by vertical offset
      dogState.y -= 6;
    }
    // Render reflection in puddle if crossing water
    // For reflection, draw a mirrored copy after drawing dog, within water area
    // We'll render reflection for the portion of dog above water line y ~ puddle.y
    // Compute a simple reflective render by saving ctx scaleY(-1) around water line
    // We'll draw the dog again but mirrored and clipped to water
    // First, normal dog
    drawDog(dogState);

    // reflection in puddle: clip to water ellipse region
    ctx.save();
    // reflect canvas
    ctx.translate(0, 0);
    // compute reflection mask: only draw within puddle ellipse
    // Create clipping path matching puddle ellipse
    ctx.beginPath();
    ctx.ellipse(puddle.x, puddle.y, puddle.w, puddle.h, 0, 0, Math.PI*2);
    ctx.clip();
    // draw mirrored dog (same state but vertical flip around puddle surface)
    ctx.save();
    ctx.scale(1, -1);
    // mirror position: reflect y around puddle.y
    const reflectY = puddle.y*2 - dogState.y;
    ctx.translate(0, puddle.y - dogState.y); // align
    const reflectedState = Object.assign({}, dogState);
    reflectedState.y = dogState.y; // position used in local clone
    // actually render with flipped coordinates by drawing at original x but y mirrored
    // For simplicity, redraw with same x but y mirrored
    reflectedState.y = puddle.y - (dogState.y - puddle.y);
    drawnDogAt(reflectedState);
    ctx.restore();
    ctx.restore();

    // helper to draw
    function drawDog(state){
      // pass through
      ctx.save();
      ctx.translate(state.x, state.y);
      ctx.rotate(state.theta);
      // body as polygon (reuse drawDog internal approach)
      // We'll draw simplified but coherent dog shape:
      // torso
      ctx.fillStyle = '#f6a058';
      ctx.strokeStyle = '#2b2b2b';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(-34,-10);
      ctx.quadraticCurveTo(-5,-34, 40,-10);
      ctx.quadraticCurveTo(52,8,40,26);
      ctx.quadraticCurveTo(-6,28,-28,20);
      ctx.quadraticCurveTo(-50,8,-34,-10);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
      // head
      ctx.save();
      ctx.translate(28,-6);
      ctx.rotate(state.headTilt);
      ctx.fillStyle = '#e9a058';
      ctx.beginPath();
      ctx.arc(18,0,12,0,Math.PI*2);
      ctx.fill();
      ctx.fillStyle = '#d0905c';
      ctx.beginPath();
      ctx.arc(28,-2,5,0,Math.PI*2);
      ctx.fill();
      // ears
      ctx.fillStyle = '#e38b39';
      ctx.beginPath();
      ctx.moveTo(6,-12); ctx.lineTo(0,-26); ctx.lineTo(14,-14); ctx.closePath(); ctx.fill();
      ctx.beginPath();
      ctx.moveTo(22,-12); ctx.lineTo(30,-26); ctx.lineTo(16,-14); ctx.closePath(); ctx.fill();
      // eyes
      ctx.fillStyle='#1b1b1b';
      ctx.beginPath(); ctx.arc(6,-2,2,0,Math.PI*2); ctx.fill();
      ctx.beginPath(); ctx.arc(16,-2,2,0,Math.PI*2); ctx.fill();
      ctx.restore();
      // legs
      function leg(x,y,phase){
        ctx.save();
        ctx.translate(x,y);
        const a = Math.sin(state.limbPhase+phase)*0.6;
        ctx.rotate(a);
        ctx.fillStyle='#d57a2a';
        ctx.fillRect(-3,0,6,22);
        ctx.fillStyle='#3a2a20';
        ctx.fillRect(-4,22,8,4);
        ctx.restore();
      }
      leg(20,24,0);
      leg(14,24,Math.PI);
      leg(-14,26,Math.PI/2);
      leg(-18,26,-Math.PI/2);
      // tail
      ctx.save();
      ctx.translate(-34,-8);
      ctx.rotate(-state.tailTilt);
      ctx.fillStyle='#f9922b';
      ctx.beginPath();
      ctx.moveTo(0,0); ctx.quadraticCurveTo(-22,-22,-42,-6);
      ctx.quadraticCurveTo(-32,-2,-20,6);
      ctx.quadraticCurveTo(-22,2,-10,6);
      ctx.quadraticCurveTo(-2,2,0,0);
      ctx.fill();
      ctx.restore();
      // mouth with bone if carrying
      if (state.carryingBone){
        ctx.save();
        ctx.translate(30,-2);
        ctx.rotate(0.1);
        drawBone({x:0,y:0,rot:0});
        ctx.restore();
      }
      ctx.restore();
    }
    function drawnDogAt(state){
      drawDog(state);
    }
  }

  // immediate bone drawing helper outside
  function drawBoneSimple(x,y){
    drawBone({x:x,y:y,rot:0});
  }

  // drawBone wrapper with simple white bone
  function drawBone(b){
    // reuse simple bone drawing
    ctx.save();
    ctx.fillStyle = '#fffaf0';
    ctx.strokeStyle = '#000';
    ctx.lineWidth = 1;
    ctx.translate(b.x,b.y);
    ctx.rotate(b.rot || 0);
    ctx.beginPath();
    ctx.moveTo(-12,-4);
    ctx.lineTo(-4,-4);
    ctx.lineTo(-2,-7);
    ctx.lineTo(0,-7);
    ctx.lineTo(6,-4);
    ctx.lineTo(12,-4);
    ctx.lineTo(6,0);
    ctx.lineTo(0,0);
    ctx.lineTo(-2,3);
    ctx.lineTo(-6,3);
    ctx.lineTo(-12,-0);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();
    ctx.restore();
  }

  // Start
  window.renderFrame = renderFrame;
  renderFrame(0);
})();
</script>
</body>

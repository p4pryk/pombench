<style>
  html, body { margin:0; width:100%; height:100%; overflow:hidden; background:#9bc5d6; }
  canvas { display:block; width:960px; height:540px; }
</style>
</head>
<body>
<canvas id="stage" width="960" height="540"></canvas>
<script>
(() => {
  const c = document.getElementById('stage');
  const x = c.getContext('2d');
  const W = c.width, H = c.height;
  const clamp=(v,a,b)=>v<a?a:v>b?b:v;
  const lerp=(a,b,t)=>a+(b-a)*t;
  const ease=(a,b,t)=>a+(b-a)*(t*t*(3-2*t));
  const smooth=t=>t*t*(3-2*t);
  const mix=(a,b,t)=>a+(b-a)*t;

  function bg(t){
    // sky
    const g = x.createLinearGradient(0,0,0,H);
    g.addColorStop(0,'#bdd7e7');
    g.addColorStop(0.55,'#b8d3dd');
    g.addColorStop(1,'#8fb08f');
    x.fillStyle=g; x.fillRect(0,0,W,H);

    // distant trees / fence
    x.fillStyle='#88a27d';
    x.fillRect(0,260,W,40);
    x.fillStyle='#7c8f70';
    for(let i=0;i<14;i++){
      const px=30+i*70+(i%2)*12;
      x.fillRect(px,205,8,95);
      x.beginPath();
      x.moveTo(px-18,220); x.lineTo(px+4,170); x.lineTo(px+26,220); x.closePath(); x.fill();
      x.beginPath();
      x.moveTo(px-22,245); x.lineTo(px+4,190); x.lineTo(px+30,245); x.closePath(); x.fill();
    }
    // ground
    const gg = x.createLinearGradient(0,270,0,H);
    gg.addColorStop(0,'#8aa36d');
    gg.addColorStop(1,'#6f8a57');
    x.fillStyle=gg; x.fillRect(0,270,W,H-270);

    // subtle lawn texture
    x.globalAlpha=.17;
    x.strokeStyle='#4f6d43';
    for(let i=0;i<130;i++){
      const px=(i*73)%W, py=285+((i*131)%250), len=5+(i%7);
      x.beginPath(); x.moveTo(px,py); x.lineTo(px+2,py-len); x.stroke();
    }
    x.globalAlpha=1;
  }

  function drawHouse(){
    const hx=640, hy=190, hw=220, hh=200;
    // shadow
    x.fillStyle='rgba(0,0,0,.18)';
    x.beginPath(); x.ellipse(hx+hw*0.48, hy+hh*1.05, 120, 26, 0, 0, Math.PI*2); x.fill();
    // base wall
    x.fillStyle='#8a5d3a';
    x.fillRect(hx,hy,hw,hh);
    // roof
    x.fillStyle='#6f4328';
    x.beginPath();
    x.moveTo(hx-18,hy+18); x.lineTo(hx+hw*0.5, hy-72); x.lineTo(hx+hw+18,hy+18); x.closePath(); x.fill();
    x.fillStyle='#5a3721';
    x.beginPath();
    x.moveTo(hx+20,hy+8); x.lineTo(hx+hw*0.5, hy-58); x.lineTo(hx+hw-20,hy+8); x.closePath(); x.fill();
    // doorway
    const dx=hx+78, dy=hy+68, dw=82, dh=92;
    x.fillStyle='#2e231d';
    x.beginPath(); roundRect(dx,dy,dw,dh,10); x.fill();
    x.fillStyle='rgba(0,0,0,.25)';
    x.beginPath(); x.ellipse(dx+dw/2,dy+dh+8,42,10,0,0,Math.PI*2); x.fill();
    // interior floor
    x.fillStyle='#1f1814';
    x.beginPath(); x.rect(dx+6,dy+18,dw-12,dh-20); x.fill();
    // bone inside
    x.save();
    x.translate(dx+46, dy+57);
    x.rotate(-.25);
    bone(0,0,1);
    x.restore();

    // trim and planks
    x.strokeStyle='rgba(255,244,220,.18)';
    x.lineWidth=2;
    for(let i=0;i<5;i++){
      x.beginPath(); x.moveTo(hx+10,hy+35+i*30); x.lineTo(hx+hw-10,hy+35+i*30); x.stroke();
    }
    x.fillStyle='#74492d';
    x.fillRect(hx-8,hy+hh-6,hw+16,12);
  }

  function roundRect(x0,y0,w,h,r){
    x.beginPath();
    x.moveTo(x0+r,y0);
    x.arcTo(x0+w,y0,x0+w,y0+h,r);
    x.arcTo(x0+w,y0+h,x0,y0+h,r);
    x.arcTo(x0,y0+h,x0,y0,r);
    x.arcTo(x0,y0,x0+w,y0,r);
    x.closePath();
  }

  function puddle(t, dog){
    const px=355, py=430, rw=115, rh=28;
    x.save();
    x.beginPath();
    x.ellipse(px,py,rw,rh,0,0,Math.PI*2);
    x.clip();
    const g=x.createLinearGradient(px-rw,py-rh,px+rw,py+rh);
    g.addColorStop(0,'#78a7b7'); g.addColorStop(.5,'#4f8092'); g.addColorStop(1,'#86b2bf');
    x.fillStyle=g; x.fillRect(px-rw,py-rh,rw*2,rh*2);

    // reflection of dog only
    if(dog && t<3.7){
      x.globalAlpha=.38;
      drawDog(dog, true);
      x.globalAlpha=1;
    }
    // water sheen
    x.globalAlpha=.18;
    x.fillStyle='#d8f3ff';
    x.beginPath();
    x.ellipse(px-25,py-6,55,8,-.15,0,Math.PI*2); x.fill();
    x.globalAlpha=1;
    // ripple edge
    x.strokeStyle='rgba(235,250,255,.20)';
    x.lineWidth=2;
    x.beginPath(); x.ellipse(px,py,rw,rh,0,0,Math.PI*2); x.stroke();
    x.restore();
  }

  function bone(x0,y0,s){
    x.save();
    x.translate(x0,y0); x.scale(s,s);
    x.fillStyle='#e7dcc6';
    x.strokeStyle='rgba(120,100,70,.25)';
    x.lineWidth=1;
    x.beginPath();
    x.moveTo(-16,-3); x.bezierCurveTo(-25,-18,-9,-26,0,-14); x.bezierCurveTo(8,-28,24,-17,16,-3);
    x.lineTo(14,3); x.bezierCurveTo(24,18,8,28,0,16); x.bezierCurveTo(-9,28,-25,17,-14,3); x.closePath();
    x.fill(); x.stroke();
    x.restore();
  }

  function drawFurEllipse(cx,cy,rx,ry,rot,base,light,edge){
    x.save(); x.translate(cx,cy); x.rotate(rot);
    const g=x.createRadialGradient(-rx*.3,-ry*.3,ry*.2,0,0,Math.max(rx,ry));
    g.addColorStop(0,light); g.addColorStop(.65,base); g.addColorStop(1,edge);
    x.fillStyle=g;
    x.beginPath(); x.ellipse(0,0,rx,ry,0,0,Math.PI*2); x.fill();
    x.restore();
  }

  function drawDog(d, reflect=false){
    // d: {x,y,s,phase,pose}
    const t=d.t;
    let x0=d.x, y0=d.y, s=d.s;
    if(reflect){ y0 = 540 - (d.y + 10); s *= 0.98; }
    x.save();
    x.translate(x0,y0);
    x.scale(s, reflect?-1:1);
    if(reflect) x.scale(1,-1);

    const pose=d.pose;
    // shadow
    if(!reflect){
      x.fillStyle='rgba(0,0,0,.18)';
      x.beginPath(); x.ellipse(0,31,42,10,0,0,Math.PI*2); x.fill();
    }
    // legs with gait/stance
    const amp=pose.legAmp||0;
    const swing=pose.swing||0;
    const lift=pose.lift||0;
    const turn=pose.turn||0;
    const front=pose.front||0;

    function leg(xb,yb,upper,lower,ang1,ang2,frontLeg,side){
      x.save();
      x.translate(xb,yb);
      x.rotate(ang1);
      // upper leg
      x.strokeStyle=frontLeg?'#b8732f':'#9a5f26';
      x.lineWidth=8;
      x.lineCap='round';
      x.beginPath(); x.moveTo(0,0); x.lineTo(0,upper); x.stroke();
      x.translate(0,upper);
      x.rotate(ang2);
      x.beginPath(); x.moveTo(0,0); x.lineTo(0,lower); x.stroke();
      // paw
      x.strokeStyle='#6f4a23';
      x.lineWidth=10;
      x.beginPath(); x.moveTo(0,lower); x.lineTo(side?2:-2,lower+4); x.stroke();
      x.restore();
    }

    // body layers
    drawFurEllipse(-1,-8,44,26,-0.08,'#c7822d','#e7b05f','#8b4f1a'); // torso
    drawFurEllipse(-21,-19,22,18,-0.45,'#d48e35','#efc06c','#9a5f26'); // chest
    drawFurEllipse(15,-27,26,19,0.32,'#d89339','#f0c875','#975c26'); // hind
    drawFurEllipse(-7,-44,28,20,-0.1,'#d39a4c','#f0d39c','#a4692b'); // neck ruff
    // tail
    x.save();
    x.translate(36,-35);
    x.rotate(-0.95 + 0.18*Math.sin(t*2));
    x.fillStyle='#cf8d35';
    x.beginPath();
    x.moveTo(0,0);
    x.bezierCurveTo(14,-10,20,10,10,16);
    x.bezierCurveTo(6,18,-2,18,-1,8);
    x.bezierCurveTo(2,2,3,-2,0,0);
    x.closePath(); x.fill();
    x.restore();

    // legs order back to front with occlusion
    const a = pose.run || 0;
    const h = pose.hold || 0;
    const z1 = -5, z2=5;
    leg(-25,8,20,18,-0.15+0.8*a,0.2-0.9*a,false,0);
    leg(6,9,22,18,0.05-0.8*a,0.15+0.8*a,false,1);
    leg(-18,9,22,19,0.05-0.8*a+0.15,0.1+0.7*a,true,0);
    leg(18,9,22,18,-0.05+0.8*a-0.1,0.15-0.7*a,true,1);

    // head
    x.save();
    x.translate(-33,-35);
    x.rotate(turn);
    drawFurEllipse(0,0,24,18,0.05,'#e0a04a','#f2ca84','#9d6328'); // head
    drawFurEllipse(4,-8,13,10,0,'#f0d8a7','#fff0cf','#c8923f'); // muzzle
    // ears
    function ear(px,py,r){
      x.save(); x.translate(px,py); x.rotate(r);
      x.fillStyle='#a7642b'; x.beginPath();
      x.moveTo(0,0); x.lineTo(-7,-16); x.lineTo(8,-13); x.closePath(); x.fill();
      x.fillStyle='#d7b175'; x.beginPath();
      x.moveTo(0,-1); x.lineTo(-4,-12); x.lineTo(5,-10); x.closePath(); x.fill();
      x.restore();
    }
    ear(-9,-14,-0.15+0.08*Math.sin(t*4));
    ear(6,-15,0.2-0.05*Math.sin(t*4.3));
    // eyes/nose
    x.fillStyle='#2f231d';
    x.beginPath(); x.ellipse(-4,-2,2.5,3,0,0,Math.PI*2); x.fill();
    x.beginPath(); x.ellipse(5,-1,2.2,2.8,0,0,Math.PI*2); x.fill();
    x.fillStyle='#1b1410';
    x.beginPath(); x.arc(12,2,2.5,0,Math.PI*2); x.fill();
    // mouth holding bone
    if(h>0.2){
      x.strokeStyle='#503226'; x.lineWidth=2.2;
      x.beginPath(); x.moveTo(8,3); x.lineTo(18,4); x.stroke();
      x.save(); x.translate(19,5); x.rotate(-0.15);
      bone(0,0,0.55);
      x.restore();
    }
    x.restore();

    // coat tufts
    x.globalAlpha=.22;
    x.strokeStyle='#fff1c6';
    x.lineWidth=2;
    for(let i=0;i<12;i++){
      const px=-34+(i*6)%70, py=-36+((i*11)%26);
      x.beginPath(); x.moveTo(px,py); x.lineTo(px-2,py-4); x.stroke();
    }
    x.globalAlpha=1;
    x.restore();
  }

  function stateAt(t){
    // Motion keyframes
    let xDog,yDog,scale,run,turn,hold,legAmp,swing,lift,front;
    if(t<1){
      const u=t/1;
      xDog=260; yDog=392;
      scale=1.02;
      run=0.05; swing=0; lift=0;
      turn=0; hold=0;
      front=0;
    } else if(t<3){
      const u=(t-1)/2;
      xDog=lerp(260, 560, smooth(u));
      yDog=392 - 1.5*Math.sin(u*Math.PI)*Math.max(0,1-u*0.25);
      scale=1.02;
      run=0.9*Math.sin(t*9);
      swing=0;
      lift=0;
      turn=0; hold=0;
      front=0;
      if(t>2.0 && t<2.55){
        const j=(t-2.0)/0.55;
        yDog -= 34*Math.sin(j*Math.PI);
        xDog += 8*Math.sin(j*Math.PI);
      }
    } else if(t<4.5){
      const u=(t-3)/1.5;
      xDog=lerp(560, 688, smooth(u));
      yDog=lerp(388, 387, u);
      scale=1.02;
      run=0.25;
      turn=0;
      hold=0;
    } else if(t<6){
      const u=(t-4.5)/1.5;
      xDog=690; yDog=388;
      scale=1.0;
      run=0.02;
      turn=0.12;
      hold=1;
    } else if(t<8){
      const u=(t-6)/2;
      xDog=lerp(690, 645, smooth(u));
      yDog=388;
      scale=1.0;
      run=0.35;
      turn=lerp(0.12, Math.PI*0.9, smooth(u));
      hold=1;
    } else if(t<10){
      const u=(t-8)/2;
      xDog=645;
      yDog=388;
      scale=1.02;
      run=0.06;
      turn=lerp(Math.PI*0.9, Math.PI, smooth(u));
      hold=1;
    } else {
      xDog=645; yDog=388;
      scale=1.02;
      run=0.02;
      turn=Math.PI;
      hold=1;
    }
    // final pose: front-facing with bone
    let pose={run,turn,hold,swing:0,legAmp:0,lift:0,front:0};
    if(t>=8){
      pose.run=0.02;
    } else if(t>=1){
      pose.run = 0.95*Math.sin(t*9);
    }
    if(t<1.2) pose.run = 0.55*Math.sin(t*10);
    return {x:xDog,y:yDog,s:scale,pose,t};
  }

  function drawScene(t){
    bg(t);
    const d=stateAt(t);
    drawHouse();
    puddle(t,d);

    // path / dirt near house
    x.fillStyle='rgba(120,93,55,.28)';
    x.beginPath(); x.ellipse(545,430,240,28,0,0,Math.PI*2); x.fill();

    // draw dog over puddle, except reflection already inside puddle
    drawDog(d,false);

    // small porch shadow / foreground detail
    x.fillStyle='rgba(0,0,0,.15)';
    x.beginPath(); x.ellipse(700,440,80,16,0,0,Math.PI*2); x.fill();

    // subtle finishing light
    const lg=x.createLinearGradient(0,0,W,H);
    lg.addColorStop(0,'rgba(255,255,255,.08)');
    lg.addColorStop(1,'rgba(255,255,255,0)');
    x.fillStyle=lg; x.fillRect(0,0,W,H);
  }

  window.renderFrame = function(timeSeconds){
    const t = timeSeconds < 0 ? 0 : (timeSeconds >= 12 ? 12 : timeSeconds);
    x.clearRect(0,0,W,H);
    drawScene(t);
  };

  renderFrame(0);
})();
</script>
</body>
</html>

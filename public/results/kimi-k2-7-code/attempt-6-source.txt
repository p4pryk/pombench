<canvas id="stage" width="960" height="540"></canvas>
<script>
const W=960,H=540;
const c=document.getElementById('stage');
const $=c.getContext('2d');

const V=(x,y,z)=>({x,y,z});
const add=(a,b)=>V(a.x+b.x,a.y+b.y,a.z+b.z);
const sub=(a,b)=>V(a.x-b.x,a.y-b.y,a.z-b.z);
const mul=(a,s)=>V(a.x*s,a.y*s,a.z*s);
const dot=(a,b)=>a.x*b.x+a.y*b.y+a.z*b.z;
const cross=(a,b)=>V(a.y*b.z-a.z*b.y,a.z*b.x-a.x*b.z,a.x*b.y-a.y*b.x);
const len=a=>Math.sqrt(dot(a,a));
const norm=a=>{const l=len(a);return l?mul(a,1/l):a;};
const lerp=(a,b,u)=>a+(b-a)*u;
const lerpV=(a,b,u)=>V(lerp(a.x,b.x,u),lerp(a.y,b.y,u),lerp(a.z,b.z,u));
const clamp=(a,b,d)=>a<b?b:a>d?d:a;
const smooth=u=>{u=clamp(u,0,1);return u*u*(3-2*u);};

const E=V(-180,110,-260),T=V(-60,12,40);
const cF=norm(sub(T,E));
const cR=norm(cross(cF,V(0,1,0)));
const cU=norm(cross(cR,cF));
const FOCAL=620;
const SUN=norm(V(-0.45,0.85,0.2));

function project(P){
  const v=sub(P,E);
  const xx=dot(v,cR),yy=dot(v,cU),zz=dot(v,cF);
  return {x:W/2+xx*(FOCAL/zz),y:H/2-yy*(FOCAL/zz),s:FOCAL/zz,z:zz};
}

function rotYPR(y,p,r){
  const cy=Math.cos(y),sy=Math.sin(y),cp=Math.cos(p),sp=Math.sin(p),cr=Math.cos(r),sr=Math.sin(r);
  const f0=V(sy,0,cy),u0=V(0,1,0),r0=V(cy,0,-sy);
  const f1=add(mul(f0,cp),mul(u0,sp));
  const u1=sub(mul(u0,cp),mul(f0,sp));
  const r2=add(mul(r0,cr),mul(u1,sr));
  const u2=sub(mul(u1,cr),mul(r0,sr));
  return {x:r2,y:u2,z:f1};
}
function toW(p,o,R){return add(o,add(mul(R.x,p.x),add(mul(R.y,p.y),mul(R.z,p.z))));}

function hash(i){return Math.abs(Math.sin(i*12.9898+78.233)*43758.5453)%1;}
function mix(a,b,u){return [a[0]+(b[0]-a[0])*u,a[1]+(b[1]-a[1])*u,a[2]+(b[2]-a[2])*u];}
function rgb(a){return 'rgb('+Math.round(a[0])+','+Math.round(a[1])+','+Math.round(a[2])+')';}

const COL={
  body:{hl:[235,168,110],base:[214,130,58],dk:[154,90,34]},
  cream:{hl:[255,242,209],base:[242,214,160],dk:[201,169,109]},
  white:{hl:[255,255,255],base:[255,253,245],dk:[224,220,200]},
  eye:[30,20,10],
  bone:{hl:[255,255,255],base:[242,230,199],dk:[201,184,150]},
  wall:[228,224,212],wallDk:[184,180,168],roof:[185,78,72],roofL:[208,106,96],
  grass1:[125,184,106],grass2:[74,130,58],interior:[42,38,34]
};

function genSphere(n){
  const a=[],g=2.39996323;
  for(let i=1;i<=n;i++){
    const y=1-2*i/(n+1),r=Math.sqrt(1-y*y),th=i*g;
    a.push(norm(V(r*Math.cos(th),y,r*Math.sin(th))));
  }
  return a;
}
const SPH=genSphere(220);

function shade(base,n){
  const l=clamp(dot(n,SUN)*0.45+0.55,0.25,1);
  return rgb(mix(base,[255,255,255],l*0.35));
}

function drawEllipsoid(items,center,radii,R,col,count,seed){
  const pc=project(center),rmax=Math.max(radii.x,radii.y,radii.z),pr=rmax*pc.s;
  const hl=project(add(center,mul(SUN,rmax)));
  items.push({z:pc.z,f(ctx){
    const g=ctx.createRadialGradient(hl.x,hl.y,pr*0.15,pc.x,pc.y,pr);
    g.addColorStop(0,rgb(mix(col.dk,col.hl,0.55)));
    g.addColorStop(0.55,rgb(col.base));
    g.addColorStop(1,rgb(col.dk));
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(pc.x,pc.y,pr,0,Math.PI*2);ctx.fill();
    for(let i=0;i<count;i++){
      const s=SPH[i];
      const lp=V(s.x*radii.x,s.y*radii.y,s.z*radii.z);
      const n=norm(add(mul(R.x,s.x),add(mul(R.y,s.y),mul(R.z,s.z))));
      const P=add(center,add(mul(R.x,lp.x),add(mul(R.y,lp.y),mul(R.z,lp.z))));
      const L=2+2*hash(i+seed+1);
      const Q=add(P,mul(n,L));
      const a=project(P),b=project(Q);
      const l=(dot(n,SUN)+1)/2;
      const cc=mix(col.dk,mix(col.base,col.hl,l),0.55+0.35*hash(i+seed));
      ctx.strokeStyle=rgb(cc);
      ctx.lineWidth=(0.4+0.6*hash(i+seed+2))*pc.s;
      ctx.beginPath();ctx.moveTo(a.x,a.y);ctx.lineTo(b.x,b.y);ctx.stroke();
    }
  }});
}

function addCapsule(items,A,B,r,col,n,seed){
  const d=sub(B,A);
  for(let i=0;i<n;i++){
    const u=i/(n-1);
    drawEllipsoid(items,add(A,mul(d,u)),V(r,r,r),{x:V(1,0,0),y:V(0,1,0),z:V(0,0,1)},col,Math.max(8,Math.floor(n*5)),seed+i);
  }
}

function drawEye(items,c,R,side){
  const p=add(c,toW(V(side*3.5,1.5,5),V(0,0,0),R));
  items.push({z:project(p).z,f(ctx){
    const q=project(p),rr=3.5*q.s;
    const g=ctx.createRadialGradient(q.x-rr*0.4,q.y-rr*0.4,rr*0.2,q.x,q.y,rr);
    g.addColorStop(0,'#3a2a1a');g.addColorStop(1,'#000');
    ctx.fillStyle=g;ctx.beginPath();ctx.arc(q.x,q.y,rr,0,Math.PI*2);ctx.fill();
    ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(q.x+rr*0.3,q.y-rr*0.35,rr*0.28,0,Math.PI*2);ctx.fill();
  }});
}

function drawBone(items,center,R){
  const p1=add(center,toW(V(-10,0,0),V(0,0,0),R));
  const p2=add(center,toW(V(10,0,0),V(0,0,0),R));
  addCapsule(items,p1,p2,2.5,COL.bone,7,400);
  drawEllipsoid(items,p1,V(4.5,4.5,4.5),R,COL.bone,18,410);
  drawEllipsoid(items,p2,V(4.5,4.5,4.5),R,COL.bone,18,411);
}

function drawWall(ctx,pts,normal,col){
  ctx.fillStyle=shade(col,normal);
  ctx.beginPath();
  for(let i=0;i<pts.length;i++){const p=project(pts[i]);if(i===0)ctx.moveTo(p.x,p.y);else ctx.lineTo(p.x,p.y);}
  ctx.closePath();ctx.fill();
  ctx.strokeStyle='rgba(0,0,0,0.08)';ctx.lineWidth=1;ctx.stroke();
}

function houseBack(ctx){
  drawWall(ctx,[V(-150,0,50),V(10,0,50),V(10,0,170),V(-150,0,170)],V(0,1,0),COL.interior);
  drawWall(ctx,[V(-150,0,50),V(-150,0,170),V(-150,90,170),V(-150,90,50)],V(-1,0,0),COL.wall);
  drawWall(ctx,[V(10,0,170),V(10,0,50),V(10,90,50),V(10,90,170)],V(1,0,0),COL.wall);
  drawWall(ctx,[V(-150,0,170),V(10,0,170),V(10,90,170),V(-150,90,170)],V(0,0,1),COL.wallDk);
}
function houseFront(ctx){
  drawWall(ctx,[V(-150,0,50),V(-94,0,50),V(-94,90,50),V(-150,90,50)],V(0,0,-1),COL.wall);
  drawWall(ctx,[V(-46,0,50),V(10,0,50),V(10,90,50),V(-46,90,50)],V(0,0,-1),COL.wall);
  drawWall(ctx,[V(-150,64,50),V(10,64,50),V(10,90,50),V(-150,90,50)],V(0,0,-1),COL.wall);
}
function houseRoof(ctx){
  drawWall(ctx,[V(-160,90,50),V(-160,90,170),V(-70,120,190),V(-70,120,30)],norm(V(-0.55,0.45,0.1)),COL.roof);
  drawWall(ctx,[V(-150,90,50),V(10,90,50),V(-70,120,30)],norm(V(0,0.55,-0.35)),COL.roofL);
}

function solveLeg(H,F,sgn,fwd){
  let d=sub(F,H),h=len(d);
  if(h>16) h=16;
  const l1=7,l2=9;
  let ca=(l1*l1+h*h-l2*l2)/(2*l1*h);
  ca=clamp(ca,-1,1);
  const sa=Math.sqrt(1-ca*ca);
  const u=mul(d,1/h);
  let ax=cross(fwd,u);
  if(len(ax)<0.001) ax=V(1,0,0);else ax=norm(ax);
  const per=norm(cross(ax,u));
  return add(H,add(mul(u,l1*ca),mul(per,sgn*l1*sa)));
}

function dogState(t){
  if(t<0)t=0;if(t>12)t=12;
  let rx=-75,rz=20,ry=14,yaw=Math.PI,hyaw=Math.PI,hpitch=0,jaw=0,phase=0,str=0;
  if(t<1){
    rx=-220;rz=-60;ry=14;yaw=0.25;hyaw=0.55;hpitch=0.05;
  }else if(t<3){
    const u=smooth((t-1)/2);
    rx=lerp(-220,-70,u);rz=lerp(-60,50,u);yaw=lerp(0.25,0,u);hyaw=lerp(0.55,0,u);
  }else if(t<4.5){
    const u=smooth((t-3)/1.5);
    rx=-70;rz=lerp(50,75,u);ry=lerp(14,10,u);
  }else if(t<6){
    const u=smooth((t-4.5)/1.5);
    rx=-70;rz=75;ry=lerp(10,11,u);
    hpitch=0.8*Math.sin(Math.PI*u);
    if(t>4.7){const g=clamp((t-4.7)/1,0,1);jaw=0.4*Math.sin(Math.PI*g);}
  }else if(t<8){
    const u=smooth((t-6)/2);
    rx=-70;rz=lerp(75,45,u);ry=lerp(11,14,u);
  }else if(t<10){
    const u=smooth((t-8)/2);
    rx=lerp(-70,-75,u);rz=lerp(45,20,u);yaw=lerp(0,Math.PI,u);hyaw=yaw;
  }
  if(jaw<0.05 && t>=5.7)jaw=0.05;

  if(t>=1&&t<3){phase=(t-1)*Math.PI*6.4;str=12;}
  else if(t>=3&&t<4.5){phase=12.8*Math.PI+(t-3)*Math.PI*3;str=5;}
  else if(t>=6&&t<8){phase=17.3*Math.PI-(t-6)*Math.PI*2.8;str=5;}
  else if(t>=8&&t<10){phase=11.7*Math.PI+(t-8)*Math.PI*2.4;str=3;}

  const bob=str*0.12*Math.abs(Math.sin(phase));
  const pitch=str*0.018*Math.sin(phase);
  const roll=str*0.01*Math.cos(phase);
  ry+=bob;
  const R=rotYPR(yaw,pitch,roll);
  const root=V(rx,ry,rz);

  const headR=rotYPR(hyaw,hpitch,0);
  const headPos=add(root,toW(V(0,6,20),V(0,0,0),R));

  let boneC,boneR;
  if(t<5.0){boneC=V(-70,4,85);boneR=rotYPR(0,0,0);}
  else{
    const m=add(headPos,toW(V(0,-1,7),V(0,0,0),headR));
    if(t<5.3){const u=smooth((t-5)/0.3);boneC=lerpV(V(-70,4,85),m,u);}else boneC=m;
    boneR=headR;
  }

  const parts=[];
  drawEllipsoid(parts,root,V(9,10,17),R,COL.body,110,1);
  drawEllipsoid(parts,add(root,toW(V(0,2,-16),V(0,0,0),R)),V(10,11,12),R,COL.body,80,2);
  drawEllipsoid(parts,add(root,toW(V(0,4,14),V(0,0,0),R)),V(10,12,13),R,COL.body,90,3);
  drawEllipsoid(parts,add(root,toW(V(0,7,20),V(0,0,0),R)),V(7,9,7),R,COL.cream,75,4);
  drawEllipsoid(parts,headPos,V(7,8,7),headR,COL.body,70,5);
  drawEllipsoid(parts,add(headPos,toW(V(0,-1,6),V(0,0,0),headR)),V(4,3,5),headR,COL.white,25,6);
  drawEllipsoid(parts,add(headPos,toW(V(-4,8,1),V(0,0,0),headR)),V(2.5,4,2),headR,COL.body,20,7);
  drawEllipsoid(parts,add(headPos,toW(V(4,8,1),V(0,0,0),headR)),V(2.5,4,2),headR,COL.body,20,8);
  drawEye(parts,headPos,headR,-1);drawEye(parts,headPos,headR,1);

  const legs=[{sx:-8,sz:16,f:1,ph:0},{sx:8,sz:16,f:1,ph:Math.PI},{sx:-8,sz:-16,f:0,ph:Math.PI},{sx:8,sz:-16,f:0,ph:0}];
  for(let i=0;i<4;i++){
    const lg=legs[i];
    let fx=lg.sx,fz=lg.sz+str*Math.sin(phase+lg.ph);
    if(t>=8&&t<10){fx+=str*0.7*Math.cos(phase+lg.ph);fz+=str*0.7*Math.sin(phase+lg.ph);}
    const hipW=add(root,toW(V(lg.sx,0,lg.sz),V(0,0,0),R));
    const footW=add(root,toW(V(fx,0,fz),V(0,0,0),R));
    const knee=solveLeg(hipW,footW,lg.f?1:-1,R.z);
    addCapsule(parts,hipW,knee,lg.f?4.2:4.8,COL.cream,4,100+i*10);
    addCapsule(parts,knee,footW,3.2,COL.cream,4,150+i*10);
    drawEllipsoid(parts,footW,V(3.5,2,4),R,COL.white,12,200+i);
  }

  const tailBase=add(root,toW(V(0,6,-18),V(0,0,0),R));
  const tps=[];
  for(let i=0;i<=8;i++){
    const u=i/8,a=u*2.0,rr=18*u;
    const lx=(Math.sin(t*4+u*2)+Math.sin(t*2))*1.2;
    const lp=V(lx,rr*Math.sin(a),-rr*Math.cos(a));
    tps.push(add(tailBase,toW(lp,V(0,0,0),R)));
  }
  for(let i=0;i<tps.length-1;i++){
    const r=5*(1-i/8)+2;
    addCapsule(parts,tps[i],tps[i+1],r,i>4?COL.white:COL.cream,3,300+i);
  }

  drawBone(parts,boneC,boneR);

  const behind=[],front=[];
  for(const it of parts){if(it.z>50.5)behind.push(it);else front.push(it);}
  behind.sort((a,b)=>b.z-a.z);
  front.sort((a,b)=>b.z-a.z);

  return {root,behind,front};
}

function renderFrame(t){
  if(t<0)t=0;if(t>12)t=12;
  $.setTransform(1,0,0,1,0,0);
  $.clearRect(0,0,W,H);

  const g=$.createLinearGradient(0,0,0,H*0.55);
  g.addColorStop(0,'#A9D0F5');g.addColorStop(1,'#E6F3E6');
  $.fillStyle=g;$.fillRect(0,0,W,H);

  const g2=$.createLinearGradient(0,H*0.45,0,H);
  g2.addColorStop(0,rgb(COL.grass1));g2.addColorStop(1,rgb(COL.grass2));
  $.fillStyle=g2;$.fillRect(0,H*0.45,W,H*0.55);

  for(let i=0;i<350;i++){
    const sx=(hash(i)-0.5)*1200,sz=hash(i+1000)*800-200;
    const p=project(V(sx,0,sz));
    if(p.y<H*0.45)continue;
    $.strokeStyle=rgb(mix(COL.grass2,COL.grass1,hash(i+200)));
    $.lineWidth=(0.5+hash(i+300))*p.s;
    $.beginPath();$.moveTo(p.x,p.y);$.lineTo(p.x+(hash(i+400)-0.5)*8,p.y-4-8*hash(i+500));$.stroke();
  }

  const s=$,hs=[V(-150,0,50),V(10,0,50),V(10,0,170),V(-150,0,170)];
  s.save();s.fillStyle='rgba(20,25,18,0.25)';s.beginPath();
  for(let i=0;i<4;i++){const p=project(add(hs[i],mul(SUN,-40)));if(i===0)s.moveTo(p.x,p.y);else s.lineTo(p.x,p.y);}
  s.closePath();s.fill();s.restore();

  const state=dogState(t);

  houseBack($);

  const p=project(state.root);
  $.save();$.fillStyle='rgba(20,25,18,0.28)';$.translate(p.x,p.y+18*p.s);$.scale(42*p.s,10*p.s);$.beginPath();$.ellipse(0,0,1,0.3,0,0,Math.PI*2);$.fill();$.restore();

  for(const it of state.behind)it.f($);
  houseFront($);
  houseRoof($);
  for(const it of state.front)it.f($);
}

window.renderFrame=renderFrame;
renderFrame(0);
</script>

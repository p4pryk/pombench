<body>
<canvas id="stage" width="960" height="540"></canvas>
<script>
(() => {
  const canvas = document.getElementById('stage');
  const ctx = canvas.getContext('2d');

  const W = 960, H = 540;
  const groundY = 402;

  function clamp(v,a,b){ return v<a?a:v>b?b:v; }
  function lerp(a,b,t){ return a+(b-a)*t; }
  function smooth(a,b,t){
    t = clamp((t-a)/(b-a),0,1);
    return t*t*(3-2*t);
  }

  function dogState(tRaw){
    const t = clamp(tRaw,0,12);
    let x=180, yaw=0, speed=0, crouch=0, jump=0, enter=0, inside=0, mouthOpen=0, hasBone=false, turn=0, idle=0;
    if(t<1){
      x=180; speed=0; idle=1;
    }else if(t<1.8){
      const u=(t-1)/0.8; x=lerp(180,440,u); speed=325;
    }else if(t<2.0){
      const u=(t-1.8)/0.2; x=lerp(440,470,u); speed=150; crouch=8*u;
    }else if(t<2.35){
      const u=(t-2)/0.35; x=lerp(470,585,u); speed=328;
      jump = 4*32*u*(1-u);
      crouch=6*(1-u);
    }else if(t<3.0){
      const u=(t-2.35)/0.65; x=lerp(585,680,u); speed=146;
      const c = 1-smooth(2.35,2.55,t);
      crouch = 6*c;
    }else if(t<4.5){
      const u=(t-3)/1.5; x=lerp(680,725,u); speed=30; enter=u; inside=u;
    }else if(t<6.0){
      x=725; speed=0; inside=1; mouthOpen=smooth(4.6,5.2,t)*(1-smooth(5.2,5.5,t)); hasBone=t>=5.2;
    }else if(t<8.0){
      const u=(t-6)/2; x=lerp(725,640,u); speed=-42; hasBone=true;
    }else if(t<10.0){
      const u=(t-8)/2; x=lerp(640,620,u); speed=18*(1-u); hasBone=true; turn=u; yaw=lerp(0,-Math.PI*0.5,u);
    }else{
      x=620; yaw=-Math.PI*0.5; speed=0; hasBone=true; turn=1;
    }
    if(t<8) yaw=0;
    const bodyLift = jump - crouch + 1.8*Math.sin(t*2.3)*(Math.abs(speed)>5?1:0.3);
    return {t,x,yaw,speed,jump,crouch,bodyLift,inside,hasBone,mouthOpen,turn,idle};
  }

  function proj(pt, rootX, rootY, yaw){
    const cy=Math.cos(yaw), sy=Math.sin(yaw);
    const rx = pt.x*cy - pt.z*sy;
    const rz = pt.x*sy + pt.z*cy;
    return {x:rootX + rx, y:rootY - pt.y + rz*0.23, d:rz};
  }

  function ellipsePath(c,x,y,rx,ry){
    c.beginPath();
    c.ellipse(x,y,rx,ry,0,0,Math.PI*2);
  }

  function drawBoneAt(c, x,y,scale,rot,alpha){
    c.save();
    c.globalAlpha = alpha==null?1:alpha;
    c.translate(x,y);
    c.rotate(rot);
    c.scale(scale,scale);
    c.fillStyle="#e7dcc3";
    c.strokeStyle="#bfb39a";
    c.lineWidth=1.2;
    c.beginPath();
    c.moveTo(-20,0); c.lineTo(20,0);
    c.lineTo(20,5); c.lineTo(-20,5); c.closePath();
    c.fill(); c.stroke();
    for(const s of [-24,24]){
      c.beginPath(); c.arc(s,-1,6,0,Math.PI*2); c.fill(); c.stroke();
      c.beginPath(); c.arc(s,6,6,0,Math.PI*2); c.fill(); c.stroke();
    }
    c.restore();
  }

  function drawLeg(c, rootX, rootY, yaw, bodyLift, phase, speed, leg, colorMul){
    const moving = Math.abs(speed)>5;
    const dir = speed>=0?1:-1;
    const stride = moving ? clamp(Math.abs(speed)*0.03,4,11) : 1.5;
    const s = Math.sin(phase+leg.phase);
    const c0 = Math.cos(phase+leg.phase);
    let fx = leg.ax - s*stride*dir;
    let fy = 0 + (moving?Math.max(0,c0)*6:0.5);
    if(leg.jumpTuck>0){ fx = leg.ax + leg.jumpTuck*leg.tuckX; fy = 8+leg.jumpTuck*8; }

    const hip = proj({x:leg.ax,y:24+bodyLift,z:leg.az},rootX,rootY,yaw);
    const foot = proj({x:fx,y:fy,z:leg.az},rootX,rootY,yaw);

    const dx = foot.x-hip.x, dy = foot.y-hip.y;
    const dist = Math.max(8,Math.hypot(dx,dy));
    const ua=18, la=16;
    const k = clamp((ua*ua + dist*dist - la*la)/(2*ua*dist),-1,1);
    const ang = Math.atan2(dy,dx);
    const bend = Math.acos(k) + (leg.az<0?-0.5:0.5);
    const knee = {x: hip.x + Math.cos(ang+bend)*ua, y: hip.y + Math.sin(ang+bend)*ua};

    c.lineCap="round";
    c.strokeStyle = `rgb(${150*colorMul},${110*colorMul},${78*colorMul})`;
    c.lineWidth = 8;
    c.beginPath(); c.moveTo(hip.x,hip.y); c.lineTo(knee.x,knee.y); c.stroke();
    c.lineWidth = 6.5;
    c.beginPath(); c.moveTo(knee.x,knee.y); c.lineTo(foot.x,foot.y); c.stroke();

    c.fillStyle = `rgb(${194*colorMul},${154*colorMul},${112*colorMul})`;
    ellipsePath(c, foot.x, foot.y+2, 6, 3);
    c.fill();
  }

  function drawDog(c, st, reflect){
    const t=st.t;
    const rootX = st.x;
    const rootY = groundY - st.bodyLift;
    const yaw = st.yaw;

    const phase = t*(Math.abs(st.speed)>50?11:7) + (st.speed<0?Math.PI*0.6:0);

    const jumpTuck = (t>=2 && t<2.35)?Math.sin((t-2)/0.35*Math.PI):0;
    const legs = [
      {ax:15,az:-7,phase:0,tuckX: 4,jumpTuck},
      {ax:13,az: 8,phase:Math.PI,tuckX:4,jumpTuck},
      {ax:-15,az:-7,phase:Math.PI,tuckX:-2,jumpTuck},
      {ax:-17,az: 8,phase:0,tuckX:-2,jumpTuck}
    ];

    const flipY = reflect ? -1 : 1;
    const yMirror = 397.5;
    function p(pt){
      const q = proj(pt,rootX,rootY,yaw);
      if(!reflect) return q;
      return {x:q.x,y:yMirror+(yMirror-q.y)*0.92,d:q.d};
    }

    c.save();
    if(reflect){
      c.globalAlpha = 0.24;
      c.filter = "blur(0.6px)";
    }

    // shadow/contact
    if(!reflect){
      c.fillStyle="rgba(0,0,0,0.19)";
      const sw = 34 + Math.abs(st.speed)*0.03;
      ellipsePath(c, rootX + (st.speed>0?4:-2), groundY+5, sw, 10);
      c.fill();
    }

    // rear legs first
    drawLeg(c,rootX,rootY,yaw,st.bodyLift,phase,st.speed,legs[2],0.95);
    drawLeg(c,rootX,rootY,yaw,st.bodyLift,phase,st.speed,legs[3],1.0);

    // tail
    const tailBase = p({x:-20,y:45,z:0});
    c.strokeStyle = reflect ? "rgba(196,148,104,0.85)" : "#c68f59";
    c.lineWidth = 13;
    c.lineCap="round";
    c.beginPath();
    c.moveTo(tailBase.x,tailBase.y);
    const curl = st.turn>0.8?0.1:0.45;
    c.bezierCurveTo(tailBase.x-10,tailBase.y-24,tailBase.x+24,tailBase.y-36,tailBase.x+30,tailBase.y-8+curl*10);
    c.stroke();
    c.strokeStyle = reflect ? "rgba(235,206,156,0.8)" : "#efc98f";
    c.lineWidth = 7;
    c.beginPath();
    c.moveTo(tailBase.x+1,tailBase.y-3);
    c.bezierCurveTo(tailBase.x+4,tailBase.y-20,tailBase.x+22,tailBase.y-24,tailBase.x+25,tailBase.y-7);
    c.stroke();

    // body
    const b = p({x:0,y:30,z:0});
    const by = 22, bx = 30*(1-0.15*st.turn), bz = 15+9*st.turn;
    const grad = c.createRadialGradient(b.x+10,b.y-8,5,b.x,b.y,40);
    grad.addColorStop(0, reflect?"rgba(250,222,175,0.9)":"#f6d9ad");
    grad.addColorStop(0.55, reflect?"rgba(229,170,112,0.9)":"#dd9b5f");
    grad.addColorStop(1, reflect?"rgba(180,120,72,0.88)":"#b77546");
    c.fillStyle=grad;
    c.beginPath();
    c.ellipse(b.x,b.y,bx,by+st.turn*2,0,0,Math.PI*2);
    c.fill();

    // neck ruff
    const n = p({x:18,y:39,z:0});
    c.fillStyle = reflect ? "rgba(244,224,188,0.9)" : "#f1d8b2";
    c.beginPath();
    c.ellipse(n.x,n.y,16+4*st.turn,14,0,0,Math.PI*2); c.fill();

    // head
    const headYaw = yaw + st.turn*0.18;
    const h = proj({x:28,y:41,z:0},rootX,rootY,headYaw);
    const h2 = reflect ? {x:h.x,y:yMirror+(yMirror-h.y)*0.92} : h;
    const hrx = 16*(1-0.15*st.turn), hry = 13;
    const hg = c.createRadialGradient(h2.x+5,h2.y-4,2,h2.x,h2.y,18);
    hg.addColorStop(0, reflect?"rgba(251,233,198,0.92)":"#f8dfbc");
    hg.addColorStop(1, reflect?"rgba(212,152,96,0.9)":"#d59258");
    c.fillStyle=hg;
    c.beginPath(); c.ellipse(h2.x,h2.y,hrx,hry,0,0,Math.PI*2); c.fill();

    // muzzle
    const m = proj({x:40,y:36,z:0},rootX,rootY,headYaw);
    const m2 = reflect ? {x:m.x,y:yMirror+(yMirror-m.y)*0.92} : m;
    c.fillStyle = reflect ? "rgba(239,220,190,0.9)" : "#ecd6ba";
    c.beginPath(); c.ellipse(m2.x,m2.y,9,6,0,0,Math.PI*2); c.fill();

    // ears
    const eL = proj({x:24,y:55,z:-8},rootX,rootY,headYaw);
    const eR = proj({x:24,y:55,z: 8},rootX,rootY,headYaw);
    const eeL = reflect ? {x:eL.x,y:yMirror+(yMirror-eL.y)*0.92} : eL;
    const eeR = reflect ? {x:eR.x,y:yMirror+(yMirror-eR.y)*0.92} : eR;
    c.fillStyle = reflect ? "rgba(206,146,95,0.9)" : "#cf935f";
    for(const e of [eeL,eeR]){
      c.beginPath();
      c.moveTo(e.x,e.y-10); c.lineTo(e.x-7,e.y+7); c.lineTo(e.x+7,e.y+7); c.closePath(); c.fill();
    }

    // eyes/nose
    if(!reflect){
      const eyeSep = 6 + 5*st.turn;
      const ex = h2.x + 6*(1-st.turn*0.8), ey=h2.y-2;
      c.fillStyle="#1f1a18";
      c.beginPath(); c.arc(ex-eyeSep*0.4,ey,2.3,0,Math.PI*2); c.arc(ex+eyeSep,ey+0.2,2.2,0,Math.PI*2); c.fill();
      c.beginPath(); c.arc(m2.x+5,m2.y+0.8,2.3,0,Math.PI*2); c.fill();
      if(st.mouthOpen>0){
        c.strokeStyle="rgba(40,14,14,0.7)";
        c.lineWidth=2;
        c.beginPath(); c.moveTo(m2.x+2,m2.y+4); c.lineTo(m2.x+12,m2.y+4+st.mouthOpen*4); c.stroke();
      }
    }

    // fur spikes
    c.strokeStyle = reflect ? "rgba(244,214,170,0.45)" : "rgba(248,226,192,0.45)";
    c.lineWidth = 1.2;
    for(let i=0;i<28;i++){
      const a = i/28*Math.PI*2;
      const rr = 29 + 2.5*Math.sin(i*1.7+t*0.6);
      const x0 = b.x + Math.cos(a)*rr*0.9;
      const y0 = b.y + Math.sin(a)*20;
      const len = 3+2*Math.sin(i*2.1+1.7);
      c.beginPath();
      c.moveTo(x0,y0);
      c.lineTo(x0+Math.cos(a)*len,y0+Math.sin(a)*len);
      c.stroke();
    }

    // front legs
    drawLeg(c,rootX,rootY,yaw,st.bodyLift,phase,st.speed,legs[0],1.02);
    drawLeg(c,rootX,rootY,yaw,st.bodyLift,phase,st.speed,legs[1],1.06);

    // bone in mouth
    if(st.hasBone){
      const jaw = proj({x:44,y:35,z:0},rootX,rootY,headYaw);
      const j = reflect ? {x:jaw.x,y:yMirror+(yMirror-jaw.y)*0.92} : jaw;
      drawBoneAt(c,j.x+3,j.y+3,reflect?0.78:0.92,0.03+(st.turn*0.08),reflect?0.9:1);
    }

    c.restore();
  }

  function drawBackground(c,t){
    // sky
    const sg = c.createLinearGradient(0,0,0,300);
    sg.addColorStop(0,"#bcd7ef");
    sg.addColorStop(1,"#dcebd0");
    c.fillStyle=sg; c.fillRect(0,0,W,330);

    // far fence
    c.fillStyle="#8b6b4c";
    c.fillRect(0,255,W,70);
    c.strokeStyle="rgba(70,52,35,0.35)";
    for(let x=0;x<=W;x+=24){
      c.beginPath(); c.moveTo(x,255); c.lineTo(x,325); c.stroke();
    }

    // ground
    const gg = c.createLinearGradient(0,300,0,H);
    gg.addColorStop(0,"#7aaa63");
    gg.addColorStop(1,"#5f8d4f");
    c.fillStyle=gg; c.fillRect(0,325,W,H-325);

    // grass texture
    c.strokeStyle="rgba(40,84,38,0.25)";
    for(let i=0;i<260;i++){
      const x = (i*37)%W;
      const y = 334 + ((i*53)%200);
      const h = 4 + (i%5);
      c.beginPath(); c.moveTo(x,y); c.lineTo(x+((i%3)-1)*2,y-h); c.stroke();
    }

    // puddle
    const px=520, py=398, prx=68, pry=18;
    const pg = c.createRadialGradient(px,py-3,8,px,py,75);
    pg.addColorStop(0,"rgba(170,205,220,0.72)");
    pg.addColorStop(1,"rgba(110,145,170,0.85)");
    c.fillStyle=pg;
    ellipsePath(c,px,py,prx,pry);
    c.fill();
    c.strokeStyle="rgba(230,245,255,0.42)";
    c.lineWidth=1.4;
    ellipsePath(c,px,py,prx,pry);
    c.stroke();

    // doghouse rear/interior opening region
    c.fillStyle="#5f402a";
    c.fillRect(640,320,190,110);
    c.fillStyle="rgba(28,20,13,0.82)";
    c.fillRect(678,344,78,90);

    // roof
    c.fillStyle="#784d30";
    c.beginPath();
    c.moveTo(620,320); c.lineTo(735,260); c.lineTo(850,320); c.closePath(); c.fill();
    c.strokeStyle="rgba(40,20,10,0.45)";
    c.stroke();

    // doorway sill shadow
    c.fillStyle="rgba(0,0,0,0.15)";
    c.fillRect(675,432,84,7);
  }

  function drawHouseFrontMask(c){
    // front wall with door cutout (even-odd) to occlude dog correctly
    c.save();
    c.fillStyle="#946341";
    c.beginPath();
    c.rect(640,320,190,120);
    c.rect(678,344,78,90);
    c.fill("evenodd");
    c.strokeStyle="rgba(55,34,20,0.45)";
    c.strokeRect(640,320,190,120);
    c.strokeRect(678,344,78,90);
    c.restore();
  }

  function drawStaticBoneInside(c, st){
    if(st.hasBone) return;
    drawBoneAt(c,721,392,0.85,0.05,0.95);
  }

  window.renderFrame = function(timeSeconds){
    const t = clamp(timeSeconds,0,12);
    const st = dogState(t);

    ctx.clearRect(0,0,W,H);
    drawBackground(ctx,t);

    // reflection clipped to puddle
    ctx.save();
    ctx.beginPath();
    ctx.ellipse(520,398,68,18,0,0,Math.PI*2);
    ctx.clip();
    drawDog(ctx,st,true);
    ctx.restore();

    // static inside bone before grab
    drawStaticBoneInside(ctx,st);

    // dog
    drawDog(ctx,st,false);

    // front wall occlusion after dog
    drawHouseFrontMask(ctx);

    // subtle breathing in final hold
    if(t>=10){
      const b = 0.5+0.5*Math.sin((t-10)*3.2);
      ctx.fillStyle=`rgba(255,255,255,${0.04*b})`;
      ctx.fillRect(0,0,W,H);
    }
  };

  window.renderFrame(0);
})();
</script>
</body>

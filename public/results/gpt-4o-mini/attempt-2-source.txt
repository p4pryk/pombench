        body {
            margin: 0;
            overflow: hidden;
        }
    </style>

<body>
<canvas id="stage" width="960" height="540"></canvas>
<script>
    const canvas = document.getElementById('stage');
    const ctx = canvas.getContext('2d');
    
    const pomeranian = {
        x: 100,
        y: 300,
        width: 50,
        height: 30,
        tailAngle: 0,
        running: false,
        direction: 1,
        boneHeld: false,
        boneX: 0,
        boneY: 0,
        boneWidth: 20,
        boneHeight: 10,
        angle: 0
    };

    function drawBackyard() {
        // Grass
        ctx.fillStyle = "#77B300";
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        
        // Doghouse
        ctx.fillStyle = "#8B4513";
        ctx.fillRect(600, 350, 100, 50);
        ctx.fillStyle = "#5C3D25";
        ctx.beginPath();
        ctx.moveTo(600, 350);
        ctx.lineTo(650, 300);
        ctx.lineTo(700, 350);
        ctx.closePath();
        ctx.fill();
        
        // Bone
        if (!pomeranian.boneHeld) {
            ctx.fillStyle = "#FFFFFF";
            ctx.fillRect(630, 370, pomeranian.boneWidth, pomeranian.boneHeight);
        }
    }

    function drawPomeranian() {
        ctx.save();
        ctx.translate(pomeranian.x, pomeranian.y);
        ctx.rotate(pomeranian.angle);
        
        // Body
        ctx.fillStyle = "#D2691E"; // Pomeranian color
        ctx.fillRect(-pomeranian.width / 2, -pomeranian.height / 2, pomeranian.width, pomeranian.height);
        
        // Tail
        ctx.save();
        ctx.rotate(pomeranian.tailAngle);
        ctx.fillRect(-pomeranian.width / 2, -pomeranian.height / 3, 15, 5);
        ctx.restore();
        
        // Head
        ctx.fillStyle = "#D2691E";
        ctx.beginPath();
        ctx.arc(0, -pomeranian.height / 2, 15, 0, Math.PI * 2);
        ctx.fill();
        
        // Ear
        ctx.fillStyle = "#D2691E";
        ctx.beginPath();
        ctx.moveTo(-10, -18);
        ctx.lineTo(-5, -25);
        ctx.lineTo(0, -18);
        ctx.fill();
        
        ctx.beginPath();
        ctx.moveTo(10, -18);
        ctx.lineTo(5, -25);
        ctx.lineTo(0, -18);
        ctx.fill();
        
        // Eyes
        ctx.fillStyle = "black";
        ctx.beginPath();
        ctx.arc(-5, -10, 3, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(5, -10, 3, 0, Math.PI * 2);
        ctx.fill();

        ctx.restore();
    }

    function renderFrame(timeSeconds) {
        ctx.clearRect(0, 0, canvas.width, canvas.height);

        drawBackyard();
        
        if (timeSeconds < 1) {
            // Initial position in yard
        } else if (timeSeconds < 3) {
            // Run to doghouse
            pomeranian.x += 4; // Speed
            pomeranian.running = true;
            pomeranian.tailAngle += 0.1;
        } else if (timeSeconds < 4.5) {
            // Enter doghouse
            pomeranian.x = 630; // Snap to entrance
        } else if (timeSeconds < 6) {
            // Retrieve bone
            if (!pomeranian.boneHeld) {
                pomeranian.boneHeld = true;
                pomeranian.boneX = pomeranian.x; // Follow position
                pomeranian.boneY = pomeranian.y;
            }
        } else if (timeSeconds < 8) {
            // Back out with bone
            pomeranian.x -= 4; // Back out speed
        } else if (timeSeconds < 10) {
            // Turn to face viewer
            pomeranian.angle += 0.1;
            if (pomeranian.angle > Math.PI / 2) {
                pomeranian.angle = Math.PI / 2; // Limit turn
            }
        } else if (timeSeconds < 12) {
            // Final position
            pomeranian.angle = 0; // Reset angle for frontal view
        }

        drawPomeranian();
        
        // Draw bone if held
        if (pomeranian.boneHeld) {
            ctx.fillStyle = "#FFFFFF";
            ctx.fillRect(pomeranian.x - pomeranian.boneWidth / 2, pomeranian.y, pomeranian.boneWidth, pomeranian.boneHeight);
        }
    }

    renderFrame(0);
</script>
</body>

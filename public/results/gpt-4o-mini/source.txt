<canvas id="stage" width="960" height="540"></canvas>
<script>
const canvas = document.getElementById("stage");
const ctx = canvas.getContext("2d");

// Scene and dog parameters
const dog = { 
    x: 100, 
    y: 400, 
    width: 60, 
    height: 60, 
    tailAngle: 0, 
    speed: 0, 
    jumping: false, 
    carryingBone: false 
};
const puddle = { x: 300, y: 440, width: 80, height: 40 };
const bone = { x: 0, y: 0, width: 20, height: 10, insideHouse: true };
const doghouse = { x: 460, y: 370, width: 100, height: 80 };

// Animation variables
let startTime;

function drawDog() {
    ctx.fillStyle = "#D19A6A"; // Pomeranian color
    ctx.beginPath();
    ctx.arc(dog.x + dog.width / 2, dog.y + dog.height / 2, dog.width / 2, 0, Math.PI * 2);
    ctx.fill();
    
    // Draw Tail
    ctx.save();
    ctx.translate(dog.x + dog.width / 2, dog.y);
    ctx.rotate(dog.tailAngle);
    ctx.fillStyle = "#D19A6A";
    ctx.fillRect(0, -10, 30, 10);
    ctx.restore();
    
    // Draw Bone if carrying
    if (dog.carryingBone) {
        ctx.fillStyle = "#FFF";
        ctx.fillRect(dog.x + (dog.width - bone.width) / 2, dog.y + dog.height / 2, bone.width, bone.height);
    }
}

function drawPuddle() {
    ctx.fillStyle = "rgba(0, 162, 255, 0.5)";
    ctx.fillRect(puddle.x, puddle.y, puddle.width, puddle.height);
    
    // Reflection Logic
    ctx.save();
    ctx.scale(1, -1);
    ctx.translate(0, -puddle.y * 2);
    ctx.fillStyle = "#D19A6A"; // Dog Color for Reflection
    ctx.beginPath();
    ctx.arc(dog.x + dog.width / 2, -dog.y + puddle.y + dog.height / 2, dog.width / 2, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
}

function drawDoghouse() {
    ctx.fillStyle = "#8B4513"; // Brown color for doghouse
    ctx.fillRect(doghouse.x, doghouse.y, doghouse.width, doghouse.height);
    ctx.fillStyle = "#000";
    ctx.fillRect(doghouse.x + 30, doghouse.y + 40, 40, 40); // Doorway
}

function renderFrame(time) {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    if (time < 1) {
        // Initial Setup
        drawDoghouse();
        drawDog();
        drawPuddle();
    } else if (time < 3) {
        // Running towards the house
        dog.speed = 2;
        dog.x += dog.speed;
        dog.tailAngle = Math.sin(time * 5) * 0.2; // Tail wagging motion
        
        // Jump over puddle
        if (dog.x + dog.width/2 > puddle.x && !dog.jumping) {
            dog.jumping = true;
            dog.speed = 0; // pause for jump
        }
        
        if (dog.jumping) {
            dog.y -= 20; // Ascend
            if (dog.y <= 360) { // Peak of jump
                dog.jumping = false;
            }
        } else if (dog.y < 400) {
            dog.y += 20; // Descend
        }
        
        drawDoghouse();
        drawDog();
        drawPuddle();
    } else if (time < 4.5) {
        // Entering the doghouse
        dog.x = 460; // Reset to doghouse entrance after jump
        dog.y = 380; // Inside doghouse
        // Retrieve bone
        if (time > 3.5) dog.carryingBone = true; // Pickup bone
        drawDoghouse();
        drawDog();
        drawPuddle();
    } else if (time < 6) {
        // Backing out of the doghouse
        if (dog.x < 520) dog.x += 2; // Backing out
        drawDoghouse();
        drawDog();
        drawPuddle();
    } else if (time < 8) {
        // Turn toward viewer
        if (dog.x < 520) {
            dog.x = 520; // Finalize x position
            dog.tailAngle = 0; // Set tail straight
        }
        drawDoghouse();
        drawDog();
        drawPuddle();
    } else {
        // Final pose
        dog.tailAngle = Math.sin(time * 5) * 0.1; // Slight tail wag
        drawDoghouse();
        drawDog();
        drawPuddle();
    }
}

window.renderFrame = renderFrame;
requestAnimationFrame(function render(timestamp) {
    if (!startTime) startTime = timestamp / 1000;
    renderFrame((timestamp / 1000) - startTime);
    if ((timestamp / 1000) < 12) requestAnimationFrame(render);
});
</script>
</body>

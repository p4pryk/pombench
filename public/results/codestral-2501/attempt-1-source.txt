```html
<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>PomBench Animation</title>
<style>
  body { margin: 0; }
  canvas { display: block; }
</style>
</head>
<body>
<canvas id="stage" width="960" height="540"></canvas>
<script>
const canvas = document.getElementById('stage');
const ctx = canvas.getContext('2d');

const doghouse = {
  x: 700,
  y: 200,
  width: 200,
  height: 150,
  doorWidth: 50,
  doorHeight: 100
};

const bone = {
  x: 750,
  y: 250,
  width: 20,
  height: 10
};

const puddle = {
  x: 400,
  y: 300,
  width: 150,
  height: 50
};

const dog = {
  x: 100,
  y: 350,
  width: 50,
  height: 30,
  tailLength: 20,
  earWidth: 10,
  earHeight: 15,
  eyeRadius: 3,
  mouthWidth: 15,
  mouthHeight: 5
};

function drawDog(t) {
  ctx.save();
  ctx.translate(dog.x, dog.y);
  if (t < 1) {
    ctx.rotate(0);
  } else if (t < 3) {
    ctx.rotate(Math.PI / 4 * (t - 1));
  } else if (t < 4.5) {
    ctx.rotate(Math.PI / 4 * 2);
  } else if (t < 6) {
    ctx.rotate(Math.PI / 4 * 2);
  } else if (t < 8) {
    ctx.rotate(Math.PI / 4 * (8 - t));
  } else if (t < 10) {
    ctx.rotate(Math.PI / 4 * (t - 8));
  } else {
    ctx.rotate(0);
  }

  // Body
  ctx.fillStyle = '#FFA500';
  ctx.beginPath();
  ctx.ellipse(0, 0, dog.width / 2, dog.height / 2, 0, 0, 2 * Math.PI);
  ctx.fill();

  // Head
  ctx.fillStyle = '#FFA500';
  ctx.beginPath();
  ctx.ellipse(-dog.width / 2, 0, dog.width / 4, dog.height / 4, 0, 0, 2 * Math.PI);
  ctx.fill();

  // Ears
  ctx.fillStyle = '#FFA500';
  ctx.beginPath();
  ctx.moveTo(-dog.width / 2 - dog.earWidth / 2, -dog.earHeight / 2);
  ctx.lineTo(-dog.width / 2 + dog.earWidth / 2, -dog.earHeight / 2);
  ctx.lineTo(-dog.width / 2, dog.earHeight / 2);
  ctx.fill();

  ctx.beginPath();
  ctx.moveTo(-dog.width / 2 - dog.earWidth / 2, dog.earHeight / 2);
  ctx.lineTo(-dog.width / 2 + dog.earWidth / 2, dog.earHeight / 2);
  ctx.lineTo(-dog.width / 2, -dog.earHeight / 2);
  ctx.fill();

  // Eyes
  ctx.fillStyle = '#000';
  ctx.beginPath();
  ctx.arc(-dog.width / 2 - 5, -dog.height / 4, dog.eyeRadius, 0, 2 * Math.PI);
  ctx.fill();
  ctx.beginPath();
  ctx.arc(-dog.width / 2 + 5, -dog.height / 4, dog.eyeRadius, 0, 2 * Math.PI);
  ctx.fill();

  // Mouth
  ctx.fillStyle = '#000';
  ctx.beginPath();
  ctx.rect(-dog.width / 2 - dog.mouthWidth / 2, dog.height / 4, dog.mouthWidth, dog.mouthHeight);
  ctx.fill();

  // Tail
  ctx.fillStyle = '#FFA500';
  ctx.beginPath();
  ctx.moveTo(dog.width / 2, 0);
  ctx.lineTo(dog.width / 2 + dog.tailLength, -dog.tailLength / 2);
  ctx.lineTo(dog.width / 2 + dog.tailLength, dog.tailLength / 2);
  ctx.fill();

  ctx.restore();
}

function drawDoghouse() {
  ctx.fillStyle = '#8B4513';
  ctx.fillRect(doghouse.x, doghouse.y, doghouse.width, doghouse.height);
  ctx.fillStyle = '#D2691E';
  ctx.fillRect(doghouse.x + doghouse.width / 2 - doghouse.doorWidth / 2, doghouse.y, doghouse.doorWidth, doghouse.doorHeight);
}

function drawPuddle() {
  ctx.fillStyle = '#ADD8E6';
  ctx.fillRect(puddle.x, puddle.y, puddle.width, puddle.height);
}

function drawBone() {
  ctx.fillStyle = '#8B4513';
  ctx.fillRect(bone.x, bone.y, bone.width, bone.height);
}

function drawScene(t) {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  drawPuddle();
  drawDoghouse();
  drawBone();
  drawDog(t);
}

window.renderFrame = function(timeSeconds) {
  if (timeSeconds < 0) {
    drawScene(0);
  } else if (timeSeconds >= 12) {
    drawScene(12);
  } else {
    drawScene(timeSeconds);
  }
};

renderFrame(0);
</script>
</body>
</html>
```

This HTML document sets up a simple 2D animation using the `<canvas>` element and JavaScript. The `renderFrame` function is called to draw each frame based on the time in seconds. The Pomeranian dog runs across the backyard, jumps over a puddle, retrieves a bone, backs out with it, turns to face the viewer, and finally settles into a front-facing pose. The animation is deterministic and follows the specified narrative.

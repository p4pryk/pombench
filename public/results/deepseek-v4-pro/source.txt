// ──────────────── Deterministic hash / noise ────────────────
function hash(x,y){let h=Math.imul(374761393,x^y*2654435761);h=Math.imul(h^(h>>>13),1274126177);return(h^(h>>>16))/2147483648;}
function fade(t){return t*t*t*(t*(t*6-15)+10);}
function lerp(a,b,t){return a+t*(b-a);}
function grad2d(ix,iy,x,y){const h=((ix*374761393+iy*2654435761)&1023)/1023*6.283185;return Math.cos(h)*x+Math.sin(h)*y;}
function perlin2(x,y){const xi=Math.floor(x),yi=Math.floor(y);const xf=x-xi,yf=y-yi;const u=fade(xf),v=fade(yf);const n00=grad2d(xi,yi,xf,yf),n01=grad2d(xi,yi+1,xf,yf-1),n10=grad2d(xi+1,yi,xf-1,yf),n11=grad2d(xi+1,yi+1,xf-1,yf-1);return lerp(lerp(n00,n10,u),lerp(n01,n11,u),v);}

// ──────────────── S-curve / easing ────────────────
function smoothstep(a,b,t){const v=Math.max(0,Math.min(1,(t-a)/(b-a)));return v*v*(3-2*v);}
function clamp(v,l,h){return Math.max(l,Math.min(h,v));}

// ──────────────── Scene constants ────────────────
const W=960,H=540;
const GROUND_Y = 400;
const DOGHOUSE_X=700, DOGHOUSE_Y=240, DOGHOUSE_W=200, DOGHOUSE_H=180, DOOR_W=70, DOOR_H=100;
const PUDDLE_CX=390, PUDDLE_CY=GROUND_Y-2, PUDDLE_RX=90, PUDDLE_RY=26;
const BONE_REST_X=DOGHOUSE_X+20, BONE_REST_Y=GROUND_Y-12;

// ──────────────── Dog state machine ────────────────
function getDogState(t){
    if(t<0) t=0; if(t>12) t=12;
    // Key times (approximate to meet narrative)
    const T_NOTICE = 0.8;
    const T_RUN_START = 1.0;
    const T_JUMP = 2.2;       // takeoff near puddle
    const T_LAND = 2.9;       // landing after puddle
    const T_ENTER_START = 3.6;
    const T_ENTER_END = 4.4;
    const T_GRAB = 5.2;
    const T_BACK_START = 5.8;
    const T_BACK_END = 7.5;
    const T_TURN_START = 8.0;
    const T_TURN_END = 9.8;
    const T_POSE = 10.2;

    let phase='idle', progress=0;
    let dogX=150, dogY=GROUND_Y, dogAngle=0, dogScale=1;
    let boneHeld=false, boneX=BONE_REST_X, boneY=BONE_REST_Y, boneAngle=0.3;
    let headTilt=0, tailWag=0, breathing=0;

    const runPathStartX=170, runPathEndX=DOGHOUSE_X+30;
    const totalRunDist = runPathEndX - runPathStartX;

    // idle / notice
    if(t < T_NOTICE){
        phase='idle'; progress = t/T_NOTICE;
        dogX = 150; dogY = GROUND_Y; dogAngle=0;
        headTilt = Math.sin(t*4)*0.1;
    }
    // run approach
    else if(t < T_JUMP){
        phase='run'; progress = (t-T_NOTICE)/(T_JUMP-T_NOTICE);
        dogX = runPathStartX + progress * (PUDDLE_CX - PUDDLE_RX - 45 - runPathStartX);
        dogY = GROUND_Y; dogAngle=0;
        headTilt = Math.sin(t*18)*0.08;
    }
    // jump over puddle (parabolic arc)
    else if(t < T_LAND){
        phase='jump'; progress = (t-T_JUMP)/(T_LAND-T_JUMP);
        const takeoffX = PUDDLE_CX - PUDDLE_RX - 45;
        const landX = PUDDLE_CX + PUDDLE_RX + 35;
        const jumpDist = landX - takeoffX;
        dogX = takeoffX + progress * jumpDist;
        const arcHeight = 55 * Math.sin(progress*Math.PI);
        dogY = GROUND_Y - arcHeight;
        dogAngle = -0.12 * Math.sin(progress*Math.PI*2);
        headTilt = 0;
    }
    // continue run to doghouse
    else if(t < T_ENTER_START){
        phase='run2'; progress = (t-T_LAND)/(T_ENTER_START-T_LAND);
        const landX = PUDDLE_CX + PUDDLE_RX + 35;
        dogX = landX + progress * (DOGHOUSE_X+15 - landX);
        dogY = GROUND_Y; dogAngle=0;
        headTilt = Math.sin(t*18)*0.07;
    }
    // enter doghouse
    else if(t < T_ENTER_END){
        phase='enter'; progress = (t-T_ENTER_START)/(T_ENTER_END-T_ENTER_START);
        dogX = DOGHOUSE_X+15 + progress * 35;
        dogY = GROUND_Y;
        dogAngle = 0;
        headTilt = 0;
    }
    // inside, grab bone
    else if(t < T_GRAB){
        phase='grab'; progress = (t-T_ENTER_END)/(T_GRAB-T_ENTER_END);
        dogX = DOGHOUSE_X+50; dogY = GROUND_Y;
        dogAngle=0;
        boneHeld = progress>0.6;
        if(boneHeld){
            boneX = dogX-12; boneY = dogY-18; boneAngle = -0.4;
        }
        headTilt = (progress<0.5)?0.3*(1-progress/0.5):0;
    }
    // back out
    else if(t < T_BACK_END){
        phase='back'; progress = (t-T_BACK_START)/(T_BACK_END-T_BACK_START);
        dogX = DOGHOUSE_X+50 - progress * 70;
        dogY = GROUND_Y; dogAngle=0;
        boneHeld=true; boneX=dogX-12; boneY=dogY-18; boneAngle=-0.4 + progress*0.6;
        headTilt = 0.1;
    }
    // turn to face viewer
    else if(t < T_TURN_END){
        phase='turn'; progress = (t-T_TURN_START)/(T_TURN_END-T_TURN_START);
        dogX = DOGHOUSE_X-20 - progress * 30;
        dogY = GROUND_Y;
        dogAngle = progress * (-0.5*Math.PI); // turn left ~90 deg toward viewer
        boneHeld=true; boneX=dogX-12; boneY=dogY-20; boneAngle=-0.4 + dogAngle;
        headTilt = 0;
    }
    // final frontal pose
    else {
        phase='pose'; progress = (t-T_POSE)/(12-T_POSE);
        dogX = DOGHOUSE_X-50; dogY = GROUND_Y-10;
        dogAngle = -Math.PI/2; // facing camera left-ish, turned toward viewer
        boneHeld=true; boneX=dogX-10; boneY=dogY-22; boneAngle= -0.2 + dogAngle;
        breathing = Math.sin(t*3.5)*0.08;
        tailWag = Math.sin(t*5)*0.2;
        headTilt = 0;
    }

    return {phase, progress, dogX, dogY, dogAngle, boneHeld, boneX, boneY, boneAngle, headTilt, tailWag, breathing};
}

// ──────────────── Drawing helpers ────────────────
function drawCircle(x,y,r,color){
    ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2); ctx.fillStyle=color; ctx.fill();
}
function drawEllipse(x,y,rx,ry,color,rot=0){
    ctx.save(); ctx.translate(x,y); ctx.rotate(rot); ctx.beginPath(); ctx.ellipse(0,0,rx,ry,0,0,Math.PI*2); ctx.fillStyle=color; ctx.fill(); ctx.restore();
}
function drawPath(points,color,close=true){
    ctx.beginPath(); ctx.moveTo(points[0][0],points[0][1]);
    for(let i=1;i<points.length;i++) ctx.lineTo(points[i][0],points[i][1]);
    if(close) ctx.closePath(); ctx.fillStyle=color; ctx.fill();
}

// ──────────────── Pomeranian drawing (anatomically styled) ────────────────
function drawPomeranian(x,y,angle,headTilt,boneHeld,boneX,boneY,boneAngle,phase,progress,breathing,tailWag){
    ctx.save(); ctx.translate(x,y); ctx.rotate(angle);
    const s=1.0; // scale

    // fur variation using perlin
    const furNoise = (dx,dy)=> perlin2(x*0.05+dx, y*0.05+dy)*0.15;

    // body (compact, fluffy)
    const bodyLen=48, bodyH=28;
    const bodyGrad = ctx.createLinearGradient(-bodyLen/2, -bodyH/2, bodyLen/2, bodyH/2);
    bodyGrad.addColorStop(0,'#E8A24A'); bodyGrad.addColorStop(0.6,'#D4882C'); bodyGrad.addColorStop(1,'#B56E1A');
    drawEllipse(0,-4,bodyLen/2,bodyH/2,bodyGrad);

    // ruff (neck fluff)
    const ruffGrad = ctx.createRadialGradient(-15,-12,4,-15,-12,18);
    ruffGrad.addColorStop(0,'#F4C07A'); ruffGrad.addColorStop(0.7,'#E8A24A'); ruffGrad.addColorStop(1,'#D4882C');
    drawEllipse(-14,-14,16,18,ruffGrad);

    // hindquarters fluff
    drawEllipse(18,-6,14,16,bodyGrad);

    // legs (4 articulated)
    function drawLeg(ox,oy,upperLen,lowerLen,phaseShift,isFront){
        const t = performance.now()/1000;
        const walkCycle = phase==='run'||phase==='run2'? Math.sin(t*14+phaseShift)*0.35 : 0;
        const kneeAngle = walkCycle * 0.4;
        const hipX=ox, hipY=oy;
        const kneeX = hipX + Math.sin(kneeAngle)*upperLen*0.6;
        const kneeY = hipY + Math.cos(kneeAngle)*upperLen;
        const footX = kneeX + Math.sin(kneeAngle*1.2)*lowerLen*0.5;
        const footY = kneeY + lowerLen*0.9;
        ctx.beginPath(); ctx.moveTo(hipX,hipY); ctx.lineTo(kneeX,kneeY); ctx.lineTo(footX,footY);
        ctx.strokeStyle='#B56E1A'; ctx.lineWidth=5; ctx.stroke();
        drawCircle(footX,footY,4,'#8B5A2B');
    }
    drawLeg(-12,10,14,12,0,true);
    drawLeg(12,10,14,12,2.5,false);
    drawLeg(-8,12,12,10,1.2,true);
    drawLeg(16,12,12,10,3.8,false);

    // tail (plumed, curled over back)
    ctx.save(); ctx.translate(20,-12);
    const tailCurl = 1.2 + tailWag*0.3;
    ctx.beginPath(); ctx.moveTo(0,0); ctx.quadraticCurveTo(10,-15,18,-5); ctx.quadraticCurveTo(14,2,8,-8);
    ctx.fillStyle='#E8A24A'; ctx.fill(); ctx.strokeStyle='#D4882C'; ctx.lineWidth=2; ctx.stroke();
    ctx.restore();

    // head
    ctx.save(); ctx.translate(-24,-16+headTilt*3);
    drawEllipse(0,0,13,11,'#E8A24A');
    // muzzle
    drawEllipse(-8,2,7,5,'#F4C07A');
    // nose
    drawCircle(-13,1,3,'#111');
    // eyes
    drawCircle(-5,-3,2.8,'#1A1A1A'); drawCircle(-1.5,-2.5,1,'white');
    drawCircle(-10,-2.5,2.8,'#1A1A1A'); drawCircle(-6.8,-2,1,'white');
    // ears (upright triangular)
    ctx.fillStyle='#D4882C';
    ctx.beginPath(); ctx.moveTo(-2,-9); ctx.lineTo(2,-19); ctx.lineTo(6,-8); ctx.fill();
    ctx.beginPath(); ctx.moveTo(-8,-8); ctx.lineTo(-4,-18); ctx.lineTo(0,-7); ctx.fill();
    ctx.restore();

    // bone drawing (if held or in mouth)
    if(boneHeld){
        ctx.save(); ctx.translate(boneX-x, boneY-y); ctx.rotate(boneAngle);
        ctx.fillStyle='#F5F0DC'; ctx.strokeStyle='#8B7D6B'; ctx.lineWidth=1.5;
        ctx.beginPath(); ctx.ellipse(0,0,16,5,0,0,Math.PI*2); ctx.fill(); ctx.stroke();
        drawCircle(-14,0,4,'#F5F0DC'); drawCircle(14,0,4,'#F5F0DC');
        ctx.restore();
    }
    ctx.restore();
}

// ──────────────── Puddle drawing with reflection ────────────────
function drawPuddleAndReflection(dogState){
    const {dogX, dogY, dogAngle, phase, progress}=dogState;
    // water surface
    ctx.save();
    ctx.beginPath(); ctx.ellipse(PUDDLE_CX, PUDDLE_CY, PUDDLE_RX, PUDDLE_RY, 0, 0, Math.PI*2);
    ctx.fillStyle='#5A7D9C'; ctx.fill();
    ctx.strokeStyle='#2C4A5E'; ctx.lineWidth=1.5; ctx.stroke();
    ctx.clip();

    // reflection of dog (only within puddle bounds)
    const reflectY = PUDDLE_CY + (PUDDLE_CY - dogY) + 4;
    if(dogX > PUDDLE_CX-PUDDLE_RX-20 && dogX < PUDDLE_CX+PUDDLE_RX+20){
        ctx.save();
        ctx.translate(dogX, reflectY);
        ctx.scale(1,-1);
        ctx.globalAlpha=0.25;
        drawPomeranian(0,0, dogAngle, 0, false,0,0,0, phase,progress,0,0);
        ctx.restore();
    }
    ctx.restore();
    // highlight
    ctx.beginPath(); ctx.ellipse(PUDDLE_CX-20, PUDDLE_CY-6, 25,5,0,0,Math.PI*2);
    ctx.fillStyle='rgba(255,255,255,0.4)'; ctx.fill();
}

// ──────────────── Scene rendering ────────────────
function drawScene(t){
    const dog = getDogState(t);

    // sky gradient
    const skyGrad = ctx.createLinearGradient(0,0,0,H);
    skyGrad.addColorStop(0,'#87CEEB'); skyGrad.addColorStop(1,'#E0F0E8');
    ctx.fillStyle=skyGrad; ctx.fillRect(0,0,W,H);

    // ground
    const groundGrad = ctx.createLinearGradient(0,GROUND_Y,0,H);
    groundGrad.addColorStop(0,'#6B8E4E'); groundGrad.addColorStop(0.2,'#4A7023'); groundGrad.addColorStop(1,'#3B5E1A');
    ctx.fillStyle=groundGrad; ctx.fillRect(0,GROUND_Y,W,H-GROUND_Y);

    // grass texture
    for(let i=0;i<80;i++){
        const gx=(i*127+13)%W, gy=GROUND_Y+15+(i*47)%(H-GROUND_Y-30);
        ctx.fillStyle=`rgb(${80+(i%30)},${100+(i%40)},${30+(i%20)})`;
        ctx.fillRect(gx,gy,2,6+Math.sin(i)*4);
    }

    // doghouse
    ctx.fillStyle='#8B5A2B'; ctx.fillRect(DOGHOUSE_X, DOGHOUSE_Y, DOGHOUSE_W, DOGHOUSE_H);
    ctx.fillStyle='#5C3A1E'; ctx.fillRect(DOGHOUSE_X+5, DOGHOUSE_Y+5, DOGHOUSE_W-10, DOGHOUSE_H-10);
    // roof
    ctx.fillStyle='#A0522D';
    ctx.beginPath(); ctx.moveTo(DOGHOUSE_X-8, DOGHOUSE_Y); ctx.lineTo(DOGHOUSE_X+DOGHOUSE_W/2, DOGHOUSE_Y-30); ctx.lineTo(DOGHOUSE_X+DOGHOUSE_W+8, DOGHOUSE_Y); ctx.fill();
    // doorway
    ctx.fillStyle='#2B1A0E'; ctx.fillRect(DOGHOUSE_X+DOGHOUSE_W/2-DOOR_W/2, DOGHOUSE_Y+DOGHOUSE_H-DOOR_H, DOOR_W, DOOR_H);
    // interior darker
    ctx.fillStyle='#1A0E05'; ctx.fillRect(DOGHOUSE_X+DOGHOUSE_W/2-DOOR_W/2+3, DOGHOUSE_Y+DOGHOUSE_H-DOOR_H+3, DOOR_W-6, DOOR_H-6);

    // bone resting (if not held)
    if(!dog.boneHeld && t<5.2){
        ctx.save(); ctx.translate(BONE_REST_X, BONE_REST_Y); ctx.rotate(0.4);
        ctx.fillStyle='#F5F0DC'; ctx.strokeStyle='#8B7D6B'; ctx.lineWidth=1.2;
        ctx.beginPath(); ctx.ellipse(0,0,16,5,0,0,Math.PI*2); ctx.fill(); ctx.stroke();
        drawCircle(-14,0,3.5,'#F5F0DC'); drawCircle(14,0,3.5,'#F5F0DC');
        ctx.restore();
    }

    // puddle
    drawPuddleAndReflection(dog);

    // drawing dog (occlusion: if entering, clip parts inside house)
    if(dog.phase==='enter' || dog.phase==='grab'){
        // draw dog behind doorway mask
        ctx.save();
        ctx.beginPath(); ctx.rect(DOGHOUSE_X+DOGHOUSE_W/2-DOOR_W/2, DOGHOUSE_Y+DOGHOUSE_H-DOOR_H, DOOR_W, DOOR_H);
        ctx.clip();
        drawPomeranian(dog.dogX, dog.dogY, dog.dogAngle, dog.headTilt, dog.boneHeld, dog.boneX, dog.boneY, dog.boneAngle, dog.phase, dog.progress, dog.breathing, dog.tailWag);
        ctx.restore();
    } else if(dog.phase==='back'){
        // partially behind doorway
        ctx.save();
        ctx.beginPath(); ctx.rect(0,0,W,DOGHOUSE_Y+DOGHOUSE_H-DOOR_H); ctx.rect(0,DOGHOUSE_Y+DOGHOUSE_H,W,DOGHOUSE_Y+DOGHOUSE_H);
        ctx.clip();
        drawPomeranian(dog.dogX, dog.dogY, dog.dogAngle, dog.headTilt, dog.boneHeld, dog.boneX, dog.boneY, dog.boneAngle, dog.phase, dog.progress, dog.breathing, dog.tailWag);
        ctx.restore();
    } else {
        drawPomeranian(dog.dogX, dog.dogY, dog.dogAngle, dog.headTilt, dog.boneHeld, dog.boneX, dog.boneY, dog.boneAngle, dog.phase, dog.progress, dog.breathing, dog.tailWag);
    }

    // contact shadows
    ctx.fillStyle='rgba(0,0,0,0.25)'; drawEllipse(dog.dogX, GROUND_Y+2, 22,7,ctx.fillStyle);
    if(!dog.boneHeld) drawEllipse(BONE_REST_X, GROUND_Y+2, 12,4,'rgba(0,0,0,0.2)');
}

// ──────────────── Public renderFrame ────────────────
window.renderFrame = function(timeSeconds){
    ctx.clearRect(0,0,W,H);
    drawScene(timeSeconds);
};

// initial call
window.renderFrame(0);

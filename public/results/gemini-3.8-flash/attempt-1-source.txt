(function() {
  const canvas = document.getElementById('stage');
  const ctx = canvas.getContext('2d');

  // Pseudo-random deterministic noise
  function sfc32(a, b, c, d) {
    return function() {
      a >>>= 0; b >>>= 0; c >>>= 0; d >>>= 0;
      var t = (a + b | 0) + d | 0;
      d = d + 1 | 0;
      a = b ^ b >>> 9;
      b = c + (c << 3) | 0;
      c = (c << 21 | c >>> 11);
      c = c + t | 0;
      return (t >>> 0) / 4294967296;
    }
  }

  // Pre-seed static grass tufts & garden details
  const rng = sfc32(0x9E3779B9, 0x243F6A88, 0xB7E15162, 123456);
  const grassBlades = [];
  for (let i = 0; i < 240; i++) {
    grassBlades.push({
      x: 30 + rng() * 900,
      y: 280 + rng() * 240,
      h: 8 + rng() * 14,
      tilt: (rng() - 0.5) * 12,
      shade: (rng() > 0.5 ? '#4b6e2d' : '#3d5a22')
    });
  }

  // Puddle configuration
  const puddle = {
    cx: 470,
    cy: 395,
    rx: 75,
    ry: 26,
    angle: -0.04
  };

  // Doghouse configuration
  const doghouse = {
    x: 710,
    y: 240,
    w: 200,
    h: 190,
    doorX: 745,
    doorY: 300,
    doorW: 68,
    doorH: 105
  };

  // Bone initial position inside doghouse
  const boneInitial = {
    x: 778,
    y: 382,
    angle: 0.15

  // Trajectory interpolation
  function getDogState(t) {
    // Clamping
    const time = Math.max(0, Math.min(12, t));

    let x, y, z = 0;
    let yaw = 0; // angle of body in world (0 = facing right, PI = left, PI/2 = facing viewer)
    let legPhase = 0;
    let mouthOpen = 0;
    let hasBone = false;
    let spineBend = 0;
    let tailWag = Math.sin(time * 6) * 0.15;
    let isJumping = false;
    let jumpProgress = 0;
    let crouch = 0;
    let facingViewerWeight = 0;

    // Timeline phases:
    // 0 - 1.0s: Alert, look at doghouse/bone, prep run
    // 1.0 - 1.8s: Run toward puddle
    // 1.8 - 2.5s: Leap over puddle (takeoff ~1.85, apex ~2.15, land ~2.45)
    // 2.5 - 3.4s: Run & enter doghouse doorway
    // 3.4 - 4.6s: Inside doorway, reach down, mouth grab bone
    // 4.6 - 6.2s: Backing out through doorway with bone
    // 6.2 - 8.2s: Backing into open yard, slowing
    // 8.2 - 10.0s: Turning around smoothly towards the camera
    // 10.0 - 12.0s: Frontal majestic presentation, breathing, holding bone

    if (time <= 1.0) {
      // Idle / anticipation
      x = 160;
      y = 405;
      yaw = 0.05;
      crouch = Math.sin(time * Math.PI) * 2;
      tailWag = Math.sin(time * 8) * 0.12;
      hasBone = false;
    } else if (time < 1.8) {
      // Run to puddle edge
      const u = (time - 1.0) / 0.8;
      x = 160 + u * (380 - 160);
      y = 405 + Math.sin(u * Math.PI * 4) * 3;
      legPhase = u * 16;
      tailWag = Math.sin(time * 14) * 0.25;
      // Pre-jump crouch near end
      if (u > 0.7) {
        crouch = Math.sin((u - 0.7) / 0.3 * Math.PI) * 7;
      }
    } else if (time < 2.5) {
      // Leap over puddle (width from x=380 to x=560)
      isJumping = true;
      const u = (time - 1.8) / 0.7;
      jumpProgress = u;
      x = 380 + u * (560 - 380);
      // Parabolic jump arc: apex reaches 48px off ground
      const arc = 4 * u * (1 - u);
      z = arc * 52;
      y = 402 - (u * 6); // slightly uphill into yard perspective
      legPhase = 2.5 + u * 1.5; // legs tucked in air, extend on landing
      tailWag = Math.sin(u * Math.PI) * 0.3;
      spineBend = (u - 0.5) * 0.3;
      if (u > 0.85) {
        crouch = (1 - (u - 0.85)/0.15) * 6; // landing cushion compression
      }
    } else if (time < 3.4) {
      // Landed, running straight into doghouse doorway
      const u = (time - 2.5) / 0.9;
      x = 560 + u * (755 - 560);
      y = 396 - u * 12; // perspective towards doghouse floor
      legPhase = u * 14;
      tailWag = Math.sin(time * 12) * 0.2;
    } else if (time < 4.6) {
      // Inside doghouse: stoop, reach, grab bone
      x = 755;
      y = 384;
      yaw = 0.05;
      const u = (time - 3.4) / 1.2;
      if (u < 0.4) {
        crouch = (u / 0.4) * 9;
        mouthOpen = u / 0.4;
        hasBone = false;
      } else if (u < 0.7) {
        crouch = 9;
        mouthOpen = 1.0 - (u - 0.4) / 0.3; // bite down
        hasBone = (u > 0.55);
      } else {
        crouch = 9 * (1.0 - (u - 0.7) / 0.3);
        mouthOpen = 0.1;
        hasBone = true;
      }
      tailWag = Math.sin(time * 10) * 0.35;
    } else if (time < 6.2) {
      // Backing out through doorway
      const u = (time - 4.6) / 1.6;
      x = 755 - u * 95; // back up to ~660
      y = 384 + u * 14;
      yaw = 0.05;
      legPhase = -u * 12; // reverse leg motion
      hasBone = true;
      tailWag = Math.sin(time * 8) * 0.2;
    } else if (time < 8.2) {
      // Continued reverse into comfortable yard space
      const u = (time - 6.2) / 2.0;
      x = 660 - u * 140; // back to ~520
      y = 398 + u * 14;  // down to ~412
      yaw = 0.05;
      legPhase = -u * 10;
      hasBone = true;
      tailWag = Math.sin(time * 8) * 0.22;
    } else if (time < 10.0) {
      // Turn around smoothly to face viewer
      const u = (time - 8.2) / 1.8;
      // Smooth Hermite blend
      const ease = u * u * (3 - 2 * u);
      x = 520 - ease * 15;
      y = 412 + ease * 10;
      yaw = ease * (Math.PI / 2); // 0 to 90 degrees (facing straight forward)
      facingViewerWeight = ease;
      legPhase = ease * 6; // pivoting step
      hasBone = true;
      tailWag = Math.sin(time * 9) * 0.25;
    } else {
      // Final frontal presentation (10s - 12s)
      const u = (time - 10.0) / 2.0;
      x = 505;
      y = 422;
      yaw = Math.PI / 2;
      facingViewerWeight = 1.0;
      hasBone = true;
      // Gentle breathing & proud chest swell
      crouch = Math.sin(u * Math.PI * 4) * 1.5;
      tailWag = Math.sin(time * 5) * 0.12;
    }

    return {
      time, x, y, z, yaw, legPhase, mouthOpen, hasBone, spineBend,
      tailWag, isJumping, jumpProgress, crouch, facingViewerWeight
    };
  }

  // Draw background environment
  function drawEnvironment() {
    // Sky
    const skyGrad = ctx.createLinearGradient(0, 0, 0, 320);
    skyGrad.addColorStop(0, '#7ba8c9');
    skyGrad.addColorStop(0.6, '#b4cee2');
    skyGrad.addColorStop(1, '#dfeaf2');
    ctx.fillStyle = skyGrad;
    ctx.fillRect(0, 0, 960, 320);

    // Distant soft warm sunlight haze
    const sunGrad = ctx.createRadialGradient(260, 90, 20, 260, 90, 280);
    sunGrad.addColorStop(0, 'rgba(255,248,220,0.55)');
    sunGrad.addColorStop(1, 'rgba(255,248,220,0)');
    ctx.fillStyle = sunGrad;
    ctx.fillRect(0, 0, 960, 320);

    // Distant trees/hedges
    ctx.fillStyle = '#49633e';
    ctx.beginPath();
    ctx.moveTo(0, 270);
    for (let x = 0; x <= 960; x += 40) {
      const h = Math.sin(x * 0.015) * 16 + Math.cos(x * 0.03) * 10;
      ctx.lineTo(x, 245 - h);
    }
    ctx.lineTo(960, 300);
    ctx.lineTo(0, 300);
    ctx.closePath();
    ctx.fill();

    // Backyard fence
    ctx.fillStyle = '#9b8772';
    for (let bx = 0; bx < 960; bx += 28) {
      ctx.fillRect(bx, 240, 24, 70);
      ctx.beginPath();
      ctx.moveTo(bx, 240);
      ctx.lineTo(bx + 12, 226);
      ctx.lineTo(bx + 24, 240);
      ctx.closePath();
      ctx.fill();
    }
    // Horizontal fence beams
    ctx.fillStyle = '#7a6754';
    ctx.fillRect(0, 252, 960, 8);
    ctx.fillRect(0, 285, 960, 8);

    // Lawn gradient with warm sunlight & natural soil tones
    const lawnGrad = ctx.createLinearGradient(0, 290, 0, 540);
    lawnGrad.addColorStop(0, '#537533');
    lawnGrad.addColorStop(0.3, '#5c8038');
    lawnGrad.addColorStop(0.7, '#496b29');
    lawnGrad.addColorStop(1, '#344e1c');
    ctx.fillStyle = lawnGrad;
    ctx.fillRect(0, 290, 960, 250);

    // Subtle trodden pathway to the doghouse
    ctx.beginPath();
    ctx.ellipse(540, 410, 320, 55, -0.05, 0, Math.PI * 2);
    const pathGrad = ctx.createRadialGradient(550, 410, 40, 550, 410, 280);
    pathGrad.addColorStop(0, 'rgba(128, 118, 86, 0.35)');
    pathGrad.addColorStop(1, 'rgba(92, 128, 56, 0)');
    ctx.fillStyle = pathGrad;
    ctx.fill();

    // Natural static grass tufts
    for (let i = 0; i < grassBlades.length; i++) {
      const g = grassBlades[i];
      // Skip if inside puddle bounds
      const dx = (g.x - puddle.cx) / puddle.rx;
      const dy = (g.y - puddle.cy) / puddle.ry;
      if (dx * dx + dy * dy < 1.1) continue;

      ctx.strokeStyle = g.shade;
      ctx.lineWidth = 1.4;
      ctx.beginPath();
      ctx.moveTo(g.x, g.y);
      ctx.quadraticCurveTo(g.x + g.tilt * 0.5, g.y - g.h * 0.6, g.x + g.tilt, g.y - g.h);
      ctx.stroke();
    }
  }

  // Draw Doghouse structure (back interior, walls, roof)
  function drawDoghouseBase(boneState) {
    const dh = doghouse;

    // Cast shadow on lawn
    ctx.fillStyle = 'rgba(20, 28, 15, 0.45)';
    ctx.beginPath();
    ctx.ellipse(dh.x + 85, dh.y + dh.h + 8, 120, 24, 0.05, 0, Math.PI * 2);
    ctx.fill();

    // Outer wall (warm cedar planking)
    ctx.fillStyle = '#8b5329';
    ctx.fillRect(dh.x, dh.y + 35, dh.w, dh.h - 35);

    // Horizontal plank seams
    ctx.strokeStyle = '#633814';
    ctx.lineWidth = 2;
    for (let py = dh.y + 55; py < dh.y + dh.h; py += 18) {
      ctx.beginPath();
      ctx.moveTo(dh.x, py);
      ctx.lineTo(dh.x + dh.w, py);
      ctx.stroke();
    }

    // Gable triangular wall front
    ctx.fillStyle = '#9c5f30';
    ctx.beginPath();
    ctx.moveTo(dh.x - 6, dh.y + 35);
    ctx.lineTo(dh.x + dh.w / 2, dh.y - 45);
    ctx.lineTo(dh.x + dh.w + 6, dh.y + 35);
    ctx.closePath();
    ctx.fill();

    // Overhanging Roof
    ctx.fillStyle = '#422415';
    ctx.beginPath();
    ctx.moveTo(dh.x - 22, dh.y + 40);
    ctx.lineTo(dh.x + dh.w / 2, dh.y - 55);
    ctx.lineTo(dh.x + dh.w + 22, dh.y + 40);
    ctx.lineTo(dh.x + dh.w + 14, dh.y + 46);
    ctx.lineTo(dh.x + dh.w / 2, dh.y - 42);
    ctx.lineTo(dh.x - 14, dh.y + 46);
    ctx.closePath();
    ctx.fill();

    // Dark Doorway Interior with realistic depth
    ctx.save();
    ctx.beginPath();
    // Arched door opening
    ctx.moveTo(dh.doorX, dh.doorY + dh.doorH);
    ctx.lineTo(dh.doorX, dh.doorY + 28);
    ctx.arc(dh.doorX + dh.doorW / 2, dh.doorY + 28, dh.doorW / 2, Math.PI, 0, false);
    ctx.lineTo(dh.doorX + dh.doorW, dh.doorY + dh.doorH);
    ctx.closePath();
    ctx.clip();

    // Deep shadowed interior
    const doorGrad = ctx.createLinearGradient(dh.doorX, dh.doorY, dh.doorX, dh.doorY + dh.doorH);
    doorGrad.addColorStop(0, '#100b08');
    doorGrad.addColorStop(0.7, '#1b120c');
    doorGrad.addColorStop(1, '#2c1e14');
    ctx.fillStyle = doorGrad;
    ctx.fillRect(dh.doorX - 10, dh.doorY - 10, dh.doorW + 20, dh.doorH + 20);

    // Warm straw bedding on interior floor
    ctx.fillStyle = '#8f773d';
    ctx.beginPath();
    ctx.ellipse(dh.doorX + dh.doorW / 2, dh.doorY + dh.doorH - 6, 32, 14, 0, 0, Math.PI * 2);
    ctx.fill();

    // Bone resting inside if not yet retrieved
    if (!boneState.hasBone) {
      drawBone(boneInitial.x, boneInitial.y, boneInitial.angle, 1.0);
    }

    ctx.restore();

    // Doorway trim arch frame
    ctx.strokeStyle = '#5a3416';
    ctx.lineWidth = 5;
    ctx.beginPath();
    ctx.moveTo(dh.doorX, dh.doorY + dh.doorH);
    ctx.lineTo(dh.doorX, dh.doorY + 28);
    ctx.arc(dh.doorX + dh.doorW / 2, dh.doorY + 28, dh.doorW / 2, Math.PI, 0, false);
    ctx.lineTo(dh.doorX + dh.doorW, dh.doorY + dh.doorH);
    ctx.stroke();
  }

  // Draw Doghouse Front Occlusion Mask (for dog partial entry)
  function drawDoghouseThresholdForeground() {
    const dh = doghouse;
    // Left & right doorway pillars and bottom ground rim
    ctx.fillStyle = '#8b5329';
    // Left wall piece
    ctx.fillRect(dh.x, dh.y + 35, dh.doorX - dh.x, dh.h - 35);
    // Right wall piece
    ctx.fillRect(dh.doorX + dh.doorW, dh.y + 35, (dh.x + dh.w) - (dh.doorX + dh.doorW), dh.h - 35);

    // Door sill ground trim
    ctx.fillStyle = '#4a2f18';
    ctx.fillRect(dh.doorX - 4, dh.doorY + dh.doorH - 2, dh.doorW + 8, 6);
  }

  // Bone renderer (isolated, clean anatomical dog bone)
  function drawBone(bx, by, angle, scale = 1.0) {
    ctx.save();
    ctx.translate(bx, by);
    ctx.rotate(angle);
    ctx.scale(scale, scale);

    // Bone shadow
    ctx.fillStyle = 'rgba(10, 10, 10, 0.35)';
    ctx.beginPath();
    ctx.ellipse(0, 4, 18, 4, 0, 0, Math.PI * 2);
    ctx.fill();

    // Bone body gradient
    const bGrad = ctx.createLinearGradient(-18, -6, 18, 6);
    bGrad.addColorStop(0, '#f9f6ed');
    bGrad.addColorStop(0.5, '#eae0cb');
    bGrad.addColorStop(1, '#ded1b4');

    ctx.fillStyle = bGrad;
    ctx.strokeStyle = '#b6a683';
    ctx.lineWidth = 1.2;

    ctx.beginPath();
    // Central bar
    ctx.moveTo(-12, -3.5);
    ctx.lineTo(12, -3.5);
    // Right condyle lobes
    ctx.arc(15, -5, 4.5, -Math.PI * 0.4, Math.PI * 0.4);
    ctx.arc(15, 4.5, 4.5, -Math.PI * 0.4, Math.PI * 0.6);
    ctx.lineTo(12, 3.5);
    // Left condyle lobes
    ctx.lineTo(-12, 3.5);
    ctx.arc(-15, 4.5, 4.5, Math.PI * 0.4, Math.PI * 1.4);
    ctx.arc(-15, -5, 4.5, Math.PI * 0.6, -Math.PI * 0.4);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    ctx.restore();
  }

  // Pomeranian Renderer
  // Rendered with volumetric double coat, neck ruff, compact spitz proportions,
  // dark eyes, upright ears, articulated legs and curled plumed tail.
  function drawPomeranian(state, isReflection = false) {
    ctx.save();

    const {
      x, y, z, yaw, legPhase, mouthOpen, hasBone, spineBend,
      tailWag, isJumping, crouch, facingViewerWeight
    } = state;

    // World positioning
    const groundY = y - z + crouch;
    ctx.translate(x, groundY);

    if (isReflection) {
      // Invert vertically for reflection with slight perspective stretch
      ctx.scale(1, -0.85);
      ctx.globalAlpha = 0.62;
    }

    // Contact Shadow on ground (skip in reflection)
    if (!isReflection) {
      ctx.save();
      const shadowDist = Math.max(4, z);
      const shadowAlpha = Math.max(0.18, 0.5 - shadowDist * 0.007);
      const shadowScale = 1.0 + shadowDist * 0.01;
      ctx.fillStyle = `rgba(22, 28, 16, ${shadowAlpha})`;
      ctx.beginPath();
      // Rotate shadow based on yaw
      ctx.ellipse(0, z - crouch + 4, 34 * shadowScale, 16 * shadowScale, yaw * 0.3, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }

    // Interpolate front view vs side/three-quarter view
    const fv = facingViewerWeight; // 0 = profile/3-quarter, 1 = direct frontal

    // Color Palette: Rich Orange & Cream double coat
    const furDeep = '#a85116';
    const furMid = '#c96a20';
    const furLight = '#e59243';
    const furCream = '#f8d9a4';
    const furHighlight = '#faebd0';

    if (fv < 0.85) {
      // ==========================================
      // SIDE / RUNNING / JUMPING / THREE-QUARTER
      // ==========================================
      const dir = (yaw > Math.PI * 0.45) ? 0.3 : 1.0; // facing direction

      ctx.save();
      ctx.rotate(spineBend * 0.4);

      // Back far legs (rendered first, behind torso)
      drawLeg(state, -14, 12, legPhase, true, true);  // back-left hind
      drawLeg(state, 15, 12, legPhase + Math.PI, true, false); // back-left fore

      // Compact Torso & Thick Woolly Body Undercoat
      const bodyGrad = ctx.createRadialGradient(-4, 0, 8, 0, 0, 36);
      bodyGrad.addColorStop(0, furLight);
      bodyGrad.addColorStop(0.5, furMid);
      bodyGrad.addColorStop(1, furDeep);

      ctx.fillStyle = bodyGrad;
      ctx.beginPath();
      ctx.ellipse(-2, -4, 28, 20, 0.08, 0, Math.PI * 2);
      ctx.fill();

      // Plumed tail curled high over the back (signature Pomeranian trait)
      ctx.save();
      ctx.translate(-22, -12);
      ctx.rotate(-0.8 + tailWag);

      const tailGrad = ctx.createRadialGradient(8, -14, 4, 12, -16, 24);
      tailGrad.addColorStop(0, furHighlight);
      tailGrad.addColorStop(0.5, furLight);
      tailGrad.addColorStop(1, furMid);
      ctx.fillStyle = tailGrad;

      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.quadraticCurveTo(-14, -22, 6, -34);
      ctx.quadraticCurveTo(24, -36, 32, -18);
      ctx.quadraticCurveTo(28, -4, 12, -6);
      ctx.quadraticCurveTo(4, -8, 0, 0);
      ctx.closePath();
      ctx.fill();

      // Wispy fur tufts along the tail plume
      ctx.strokeStyle = furCream;
      ctx.lineWidth = 1.8;
      for (let i = 0; i < 7; i++) {
        ctx.beginPath();
        ctx.moveTo(8 + i * 3, -22 - i * 1.5);
        ctx.quadraticCurveTo(14 + i * 3, -32 - i, 20 + i * 3, -26);
        ctx.stroke();
      }
      ctx.restore();

      // Near legs (in front of body)
      drawLeg(state, -11, 14, legPhase + Math.PI, false, true); // near hind
      drawLeg(state, 18, 14, legPhase, false, false);            // near fore

      // Prominent Lion-like Neck Ruff (Pomeranian Mane)
      const ruffGrad = ctx.createRadialGradient(16, -16, 4, 16, -14, 30);
      ruffGrad.addColorStop(0, furHighlight);
      ruffGrad.addColorStop(0.6, furCream);
      ruffGrad.addColorStop(0.85, furLight);
      ruffGrad.addColorStop(1, furDeep);
      ctx.fillStyle = ruffGrad;

      ctx.beginPath();
      ctx.ellipse(16, -14, 18, 22, 0.35, 0, Math.PI * 2);
      ctx.fill();

      // Fluffy layered fringe ruff detailing
      ctx.fillStyle = furCream;
      for (let a = -1.2; a < 1.4; a += 0.4) {
        ctx.beginPath();
        const rx = 16 + Math.cos(a) * 19;
        const ry = -14 + Math.sin(a) * 22;
        ctx.arc(rx, ry, 6, 0, Math.PI * 2);
        ctx.fill();
      }

      // Head & Muzzle
      const headX = 28;
      const headY = -24;
      ctx.save();
      ctx.translate(headX, headY);

      // Head spherical fluff
      const headGrad = ctx.createRadialGradient(0, 0, 4, 0, 0, 16);
      headGrad.addColorStop(0, furCream);
      headGrad.addColorStop(0.7, furMid);
      headGrad.addColorStop(1, furDeep);
      ctx.fillStyle = headGrad;
      ctx.beginPath();
      ctx.arc(0, 0, 15, 0, Math.PI * 2);
      ctx.fill();

      // Small, erect triangular ears
      // Far ear
      ctx.fillStyle = furDeep;
      ctx.beginPath();
      ctx.moveTo(-5, -13);
      ctx.lineTo(-2, -26);
      ctx.lineTo(5, -14);
      ctx.closePath();
      ctx.fill();
      // Near ear
      ctx.fillStyle = furMid;
      ctx.beginPath();
      ctx.moveTo(2, -13);
      ctx.lineTo(8, -27);
      ctx.lineTo(14, -13);
      ctx.closePath();
      ctx.fill();
      // Inner ear pinkish warmth
      ctx.fillStyle = '#caa18b';
      ctx.beginPath();
      ctx.moveTo(4, -14);
      ctx.lineTo(8, -23);
      ctx.lineTo(12, -14);
      ctx.closePath();
      ctx.fill();

      // Short pointed spitz muzzle
      ctx.fillStyle = furCream;
      ctx.beginPath();
      ctx.moveTo(8, -2);
      ctx.lineTo(21, 2);
      ctx.lineTo(8, 7);
      ctx.closePath();
      ctx.fill();

      // Nose button (jet black)
      ctx.fillStyle = '#111';
      ctx.beginPath();
      ctx.ellipse(21, 2, 2.8, 2.2, 0, 0, Math.PI * 2);
      ctx.fill();

      // Dark expressive almond eye
      ctx.fillStyle = '#150f0a';
      ctx.beginPath();
      ctx.ellipse(7, -3, 3.2, 3.6, 0.1, 0, Math.PI * 2);
      ctx.fill();
      // Eye highlight specular dot
      ctx.fillStyle = '#ffffff';
      ctx.beginPath();
      ctx.arc(6.2, -4.5, 1.1, 0, Math.PI * 2);
      ctx.fill();

      // Mouth & Bone Grip
      if (mouthOpen > 0.05 && !hasBone) {
        // Lower jaw drops slightly
        ctx.fillStyle = '#7a2e2e';
        ctx.beginPath();
        ctx.moveTo(10, 4);
        ctx.lineTo(18, 5 + mouthOpen * 3);
        ctx.lineTo(10, 7);
        ctx.closePath();
        ctx.fill();
      }

      // Bone held crosswise firmly in jaws
      if (hasBone) {
        drawBone(18, 5, 0.2, 0.85);
      }

      ctx.restore();
      ctx.restore();

    } else {
      // ==========================================

  // Articulated leg helper with natural joints & stance ground contact
  function drawLeg(state, hipX, hipY, phase, isFar, isHind) {
    ctx.save();
    ctx.translate(hipX, hipY);

    const isJumping = state.isJumping;
    let thighAngle = 0;
    let kneeAngle = 0;

    if (isJumping) {
      // Jump pose: tuck limbs at apex, extend forepaws forward on approach to landing
      const u = state.jumpProgress;
      if (isHind) {
        thighAngle = 0.5 - u * 0.4;
        kneeAngle = -0.9 + u * 0.3;
      } else {
        thighAngle = -0.4 + u * 0.6;
        kneeAngle = 0.6 - u * 0.5;
      }
    } else {
      // Natural walking/running trot gait
      const swing = Math.sin(phase);
      thighAngle = swing * 0.45;
      kneeAngle = (swing < 0 ? -swing * 0.6 : 0.15);
    }

    const legLen1 = isHind ? 12 : 11;
    const legLen2 = isHind ? 13 : 11;

    // Joint 1: Upper leg / Thigh
    ctx.rotate(thighAngle);
    ctx.fillStyle = isFar ? '#9e4612' : '#c4661e';
    ctx.beginPath();
    ctx.roundRect(-4, 0, 8, legLen1, 4);
    ctx.fill();

    // Joint 2: Lower leg / Hock / Paw
    ctx.translate(0, legLen1);
    ctx.rotate(kneeAngle);
    ctx.fillStyle = isFar ? '#b85f1c' : '#dd8737';
    ctx.beginPath();
    ctx.roundRect(-3.2, 0, 6.4, legLen2, 3);
    ctx.fill();

    // Dainty Pomeranian Paw
    ctx.fillStyle = '#fae1bb';
    ctx.beginPath();
    ctx.ellipse(1.5, legLen2, 4.5, 3.2, 0.15, 0, Math.PI * 2);
    ctx.fill();

    ctx.restore();
  }

  // Draw Puddle & Real-time Dog Reflection
  function drawPuddle(dogState) {
    const p = puddle;

    ctx.save();
    ctx.translate(p.cx, p.cy);
    ctx.rotate(p.angle);

    // Muddy wet rim around puddle
    ctx.fillStyle = '#312718';
    ctx.beginPath();
    ctx.ellipse(0, 0, p.rx + 7, p.ry + 6, 0, 0, Math.PI * 2);
    ctx.fill();

    // Deep water bed
    ctx.fillStyle = '#1a1e16';
    ctx.beginPath();
    ctx.ellipse(0, 0, p.rx, p.ry, 0, 0, Math.PI * 2);
    ctx.fill();

    // CLIP to exact water boundary for realistic reflection masking
    ctx.save();
    ctx.beginPath();
    ctx.ellipse(0, 0, p.rx, p.ry, 0, 0, Math.PI * 2);
    ctx.clip();

    // Water sky sheen
    const skySheen = ctx.createLinearGradient(0, -p.ry, 0, p.ry);
    skySheen.addColorStop(0, '#5f7f98');
    skySheen.addColorStop(0.7, '#2f3d2f');
    skySheen.addColorStop(1, '#1e2418');
    ctx.fillStyle = skySheen;
    ctx.fillRect(-p.rx, -p.ry, p.rx * 2, p.ry * 2);

    // Dynamic Reflection of Dog inside puddle
    // Transform coordinates from world to puddle local
    ctx.save();
    ctx.rotate(-p.angle);
    ctx.translate(-p.cx, -p.cy);

    // Only render reflection if dog is within horizontal vicinity of puddle
    if (dogState.x > p.cx - 220 && dogState.x < p.cx + 220) {
      drawPomeranian(dogState, true);
    }
    ctx.restore();

    // Subtle water ripples and surface fresnel highlights
    ctx.fillStyle = 'rgba(255, 255, 255, 0.15)';
    ctx.beginPath();
    ctx.ellipse(-15, -6, p.rx * 0.65, p.ry * 0.45, 0.05, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = 'rgba(210, 235, 255, 0.22)';
    ctx.beginPath();
    ctx.ellipse(8, 4, p.rx * 0.5, 3.5, 0, 0, Math.PI * 2);
    ctx.fill();

    ctx.restore(); // end clip

    // Wet perimeter highlight line
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.18)';
    ctx.lineWidth = 1.5;
    ctx.beginPath();
    ctx.ellipse(0, 0, p.rx - 1, p.ry - 1, 0, 0, Math.PI * 2);
    ctx.stroke();

    ctx.restore();
  }

  // Render Frame Main Function
  window.renderFrame = function(timeSeconds) {
    const t = Number(timeSeconds);
    const state = getDogState(t);

    // 1. Clear viewport
    ctx.clearRect(0, 0, 960, 540);

    // 2. Draw Environment & Yard
    drawEnvironment();

    // 3. Draw Doghouse interior, back walls, bone inside (if not picked)
    drawDoghouseBase(state);

    // 4. Draw Puddle with actual real reflection of the dog inside water
    drawPuddle(state);

    // 5. Draw Pomeranian in yard or inside doghouse entrance
    // Depth order: If entering doghouse, the dog is drawn before the foreground threshold
    // so doorway correctly occludes its head/forequarters as requested.
    drawPomeranian(state, false);

    // 6. Draw Doghouse Doorway Foreground Rim for proper occlusion during retrieval
    drawDoghouseThresholdForeground();
  };

  // Initial call at t = 0
  window.renderFrame(0);
})();

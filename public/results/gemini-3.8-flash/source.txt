<script>
(function() {
  const canvas = document.getElementById('stage');
  const ctx = canvas.getContext('2d');

  // Pseudo-random deterministic hash for noise/fur fibers
  function hash(n) {
    let s = Math.sin(n) * 43758.5453123;
    return s - Math.floor(s);
  }

  // Linear interpolation and clamping
  function clamp(v, min, max) { return Math.max(min, Math.min(max, v)); }
  function lerp(a, b, t) { return a + (b - a) * t; }
  function smoothstep(a, b, t) {
    let x = clamp((t - a) / (b - a), 0, 1);
    return x * x * (3 - 2 * x);
  }

  // Doghouse spatial constants in yard coordinates
  // Canvas coordinate system: Ground plane at ~y = 380 to 450
  const DH_X = 660; // front threshold X
  const DH_Y = 370; // threshold ground Y
  const DH_W = 190;
  const DH_H = 180;
  const DOOR_X = 670;
  const DOOR_Y = 372;
  const DOOR_W = 76;
  const DOOR_H = 110;

  // Initial bone resting position inside doghouse floor
  const BONE_REST_X = 708;
  const BONE_REST_Y = 362;
  const BONE_REST_ANGLE = 0.12; // radians

  /*
    Trajectory & Kinematics Planning:
    0.0 - 1.0s: Alert at X=190, Y=385, facing right (yaw ~ 0). Looking at bone.
    1.0 - 3.2s: Run forward toward doghouse threshold. X: 190 -> 635, Y: 385 -> 372.
    3.2 - 4.5s: Step inside doghouse doorway. Head/shoulders penetrate interior to X=708.
    4.5 - 6.0s: Duck head, mouth bites bone at 5.0s, bone lifts from floor.
    6.0 - 8.0s: Backing out steps. Dog moves in reverse from inside to X=590, Y=395.
    8.0 - 10.0s: Turn body smoothly to face camera (yaw rotates from 0 to +PI/2 = 1.57).
    10.0 - 12.0s: Proud front-facing pose at X=570, Y=400, holding bone, breathing, proud micro-actions.
  */

  function getKinematics(t) {
    t = clamp(t, 0, 12);

    let x, y, yaw = 0, pitch = 0, roll = 0;
    let gait = 0; // phase cycle
    let isMoving = false;
    let headDip = 0; // for grabbing bone
    let mouthOpen = 0; // 0 to 1
    let boneGrabbed = false;
    let spineCurv = 0;
    let tailWag = Math.sin(t * 8) * 0.18;
    let breathe = Math.sin(t * 3.5) * 0.04;

    if (t < 1.0) {
      // Alert & crouch readiness
      let p = smoothstep(0.4, 0.9, t);
      x = 190;
      y = 385 + Math.sin(t * 4) * 0.5;
      yaw = 0; // facing +X (right)
      headDip = p * 0.08;
      tailWag = Math.sin(t * 6) * 0.12;
      spineCurv = p * 0.05;
    } else if (t < 3.2) {
      // Running toward doghouse
      let u = (t - 1.0) / 2.2;
      let easeU = u * u * (3 - 2 * u);
      x = lerp(190, 635, easeU);
      y = lerp(385, 372, easeU) + Math.abs(Math.sin((t - 1.0) * 16)) * -6;
      yaw = 0.04;
      gait = (t - 1.0) * 16; // rapid gallop/trot
      isMoving = true;
      tailWag = Math.sin(t * 18) * 0.25;
      headDip = Math.sin(t * 16) * 0.05;
      spineCurv = Math.sin(t * 16) * 0.08;
    } else if (t < 4.5) {
      // Entering doghouse through doorway
      let u = (t - 3.2) / 1.3;
      let easeU = u * u * (3 - 2 * u);
      x = lerp(635, 688, easeU);
      y = lerp(372, 368, easeU) + Math.abs(Math.sin((t - 3.2) * 10)) * -3;
      yaw = 0.02;
      gait = (t - 3.2) * 10;
      isMoving = true;
      tailWag = Math.sin(t * 10) * 0.15;
      headDip = lerp(0, 0.2, u);
    } else if (t < 6.0) {
      // At bone position inside: grab bone
      x = 688;
      y = 368;
      yaw = 0.02;
      gait = 0;
      isMoving = false;
      let grabPhase = (t - 4.5) / 1.5;
      if (grabPhase < 0.35) {
        // lower head, open mouth
        let sub = grabPhase / 0.35;
        headDip = lerp(0.2, 0.42, sub);
        mouthOpen = lerp(0, 1.0, sub);
        boneGrabbed = false;
      } else if (grabPhase < 0.55) {
        // mouth clamped onto bone
        let sub = (grabPhase - 0.35) / 0.2;
        headDip = 0.42;
        mouthOpen = lerp(1.0, 0.1, sub);
        boneGrabbed = true;
      } else {
        // lift bone up proudly
        let sub = (grabPhase - 0.55) / 0.45;
        headDip = lerp(0.42, 0.05, sub);
        mouthOpen = 0.08;
        boneGrabbed = true;
      }
      tailWag = Math.sin(t * 14) * 0.3; // excited tail wagging
    } else if (t < 8.0) {
      // Backing out through doorway
      let u = (t - 6.0) / 2.0;
      let easeU = u * u * (3 - 2 * u);
      x = lerp(688, 580, easeU);
      y = lerp(368, 395, easeU) + Math.abs(Math.sin((t - 6.0) * 11)) * -4;
      yaw = lerp(0.02, 0.2, u);
      gait = -(t - 6.0) * 11; // reverse stepping
      isMoving = true;
      boneGrabbed = true;
      headDip = 0.05;
      mouthOpen = 0.08;
      tailWag = Math.sin(t * 12) * 0.2;
    } else if (t < 10.0) {
      // Turn smoothly to face viewer
      let u = (t - 8.0) / 2.0;
      let easeU = u * u * (3 - 2 * u);
      x = lerp(580, 560, easeU);
      y = lerp(395, 400, easeU) + Math.abs(Math.sin((t - 8.0) * 8)) * -2;
      // Turn yaw from sideways (+X ~0.2) to camera facing (Math.PI / 2 = 1.57)
      yaw = lerp(0.2, Math.PI * 0.5, easeU);
      gait = (t - 8.0) * 8;
      isMoving = u < 0.85;
      boneGrabbed = true;
      headDip = lerp(0.05, -0.05, easeU); // lifting head proudly
      mouthOpen = 0.08;
      tailWag = Math.sin(t * 10) * 0.22;
    } else {
      // 10 - 12s: Clear, settled frontal presentation facing viewer
      let u = (t - 10.0) / 2.0;
      x = 560 + Math.sin(t * 1.5) * 0.8;
      y = 400 + Math.sin(t * 3.0) * 0.5;
      yaw = Math.PI * 0.5 + Math.sin(t * 1.8) * 0.02; // facing front
      pitch = Math.sin(t * 2.2) * 0.02;
      roll = Math.sin(t * 1.2) * 0.015;
      isMoving = false;
      boneGrabbed = true;
      headDip = -0.06 + Math.sin(t * 3.5) * 0.015; // proud chin up
      mouthOpen = 0.08;
      tailWag = Math.sin(t * 7) * 0.15;
    }

    return {
      t, x, y, yaw, pitch, roll, gait, isMoving, headDip, mouthOpen, boneGrabbed, spineCurv, tailWag, breathe
    };
  }

    }

    // Garden stone path hints in foreground
    for (let j = 0; j < 5; j++) {
      let px = 80 + j * 160 + (j % 2) * 20;

      ctx.fillStyle = 'rgba(85, 80, 70, 0.35)';

  }

  // 3. Bone Rendering (Single continuous object)
  function drawBone(bx, by, angle, scale = 1.0) {
    ctx.save();
    ctx.translate(bx, by);
    ctx.rotate(angle);
    ctx.scale(scale, scale);

    // Drop shadow
    ctx.fillStyle = 'rgba(20, 15, 10, 0.35)';
    ctx.beginPath();
    ctx.ellipse(0, 7, 24, 4, 0, 0, Math.PI * 2);
    ctx.fill();

    // Bone ivory 3D shading
    let boneGrad = ctx.createLinearGradient(0, -7, 0, 7);
    boneGrad.addColorStop(0, '#fffaee');
    boneGrad.addColorStop(0.35, '#f4ece0');
    boneGrad.addColorStop(0.8, '#d9cbaf');
    boneGrad.addColorStop(1, '#b5a484');
    ctx.fillStyle = boneGrad;
    ctx.strokeStyle = '#998667';
    ctx.lineWidth = 1.1;

    // Bone central shaft
    ctx.beginPath();
    ctx.moveTo(-18, -3.5);
    ctx.lineTo(18, -3.5);
    ctx.quadraticCurveTo(20, -0.5, 18, 3.5);
    ctx.lineTo(-18, 3.5);
    ctx.quadraticCurveTo(-20, -0.5, -18, -3.5);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    // Bone left condyles/knobs
    ctx.beginPath();
    ctx.arc(-20, -4.5, 5, 0, Math.PI * 2);
    ctx.arc(-20, 4.5, 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();

    // Bone right condyles/knobs
    ctx.beginPath();
    ctx.arc(20, -4.5, 5, 0, Math.PI * 2);
    ctx.arc(20, 4.5, 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();

    // Specular highlight line along upper crest
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.75)';
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    ctx.moveTo(-16, -2.5);
    ctx.lineTo(16, -2.5);
    ctx.stroke();

    ctx.restore();
  }

  // 4. Pomeranian Anatomical Procedural Model
  // Accurately renders coat density, neck ruff, compact spitz torso, short legs,
  // curled tail over back, fox-like head, upright triangular ears, dark nose & eyes.
  function drawPomeranian(kin) {
    const {
      x, y, yaw, headDip, mouthOpen, boneGrabbed, tailWag, gait, breathe
    } = kin;

    // View angles:
    // yaw = 0 -> facing right (profile +X)
    // yaw = PI/2 (1.57) -> facing camera (frontal +Z / down)
    const viewCos = Math.cos(yaw); // 1 = side right, 0 = frontal
    const viewSin = Math.sin(yaw); // 0 = side right, 1 = frontal

    ctx.save();
    ctx.translate(x, y);

    // Realistic ground contact shadow below paws & body
    ctx.save();
    let shadowScaleX = lerp(45, 34, viewSin);
    let shadowScaleY = lerp(16, 24, viewSin);
    let shadowGrad = ctx.createRadialGradient(0, 0, 5, 0, 0, shadowScaleX);
    shadowGrad.addColorStop(0, 'rgba(15, 20, 10, 0.65)');
    shadowGrad.addColorStop(0.5, 'rgba(25, 30, 15, 0.35)');
    shadowGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.fillStyle = shadowGrad;
    ctx.beginPath();
    ctx.ellipse(0, 0, shadowScaleX, shadowScaleY, 0, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();

    // Natural Pomeranian coat colors (rich orange sable, warm cream undercoat)
    const cDeepOrange = '#b35418';
    const cOrange = '#d97226';
    const cCream = '#f8caa1';
    const cHighlight = '#ffdfbe';
    const cDarkSable = '#5a2d12';

    // Helper for rendering directional layered fur plumes
    function drawFurCluster(cx, cy, rx, ry, baseCol, tipCol, seed, count = 18) {
      ctx.save();
      ctx.translate(cx, cy);

      let radGrad = ctx.createRadialGradient(0, 0, rx * 0.15, 0, 0, Math.max(rx, ry));
      radGrad.addColorStop(0, tipCol);
      radGrad.addColorStop(0.65, baseCol);
      radGrad.addColorStop(1, cDeepOrange);
      ctx.fillStyle = radGrad;

      ctx.beginPath();
      ctx.ellipse(0, 0, rx, ry, 0, 0, Math.PI * 2);
      ctx.fill();

      // Individual fur fiber tufts around periphery
      for (let i = 0; i < count; i++) {
        let ang = (i / count) * Math.PI * 2;
        let rLen = 3 + hash(seed + i * 4.3) * 6;
        let px = Math.cos(ang) * rx;
        let py = Math.sin(ang) * ry;
        ctx.strokeStyle = (i % 3 === 0) ? cHighlight : ((i % 2 === 0) ? baseCol : cDarkSable);
        ctx.lineWidth = 1.1;
        ctx.beginPath();
        ctx.moveTo(px, py);
        ctx.lineTo(px + Math.cos(ang) * rLen, py + Math.sin(ang) * rLen);
        ctx.stroke();
      }
      ctx.restore();
    }

    // Kinematic leg coordinates based on gait cycle
    // 4 articulated legs: Left Fore, Right Fore, Left Hind, Right Hind
    let legCycle = gait;
    let lfPhase = Math.sin(legCycle);
    let rfPhase = Math.sin(legCycle + Math.PI);
    let lhPhase = Math.sin(legCycle + Math.PI * 0.7);
    let rhPhase = Math.sin(legCycle + Math.PI * 1.7);

    // Leg drawer with upper thigh fur and delicate lower paws
    function drawLeg(offsetX, offsetY, phase, isFore, isFar) {
      ctx.save();
      let hipX = offsetX;
      let hipY = offsetY;

      let swing = phase * 14 * viewCos;
      let lift = Math.max(0, -phase) * 8 * viewCos;
      let footX = hipX + swing;
      let footY = 0 - lift;

      let kneeX = (hipX + footX) * 0.5 + (isFore ? 3 : -4) * viewCos;
      let kneeY = (hipY + footY) * 0.5 - 2;

      // Leg shading darker if far leg
      ctx.strokeStyle = isFar ? '#9e4916' : '#d26d24';
      ctx.lineWidth = 9;
      ctx.lineCap = 'round';
      ctx.lineJoin = 'round';

      // Upper leg thigh
      ctx.beginPath();
      ctx.moveTo(hipX, hipY);
      ctx.lineTo(kneeX, kneeY);
      ctx.stroke();

      // Lower leg to hock / ankle
      ctx.strokeStyle = isFar ? '#893d0f' : '#be5f1d';
      ctx.lineWidth = 6;
      ctx.beginPath();
      ctx.moveTo(kneeX, kneeY);
      ctx.lineTo(footX, footY - 3);
      ctx.stroke();

      // Dainty small compact paw
      ctx.fillStyle = '#e8ba90';
      ctx.beginPath();
      ctx.ellipse(footX, footY - 1, 4.5, 3, 0, 0, Math.PI * 2);
      ctx.fill();

      // Paw contact dark shadow
      ctx.fillStyle = 'rgba(20, 10, 5, 0.45)';
      ctx.beginPath();
      ctx.ellipse(footX, 0, 5, 2, 0, 0, Math.PI * 2);
      ctx.fill();

      ctx.restore();
    }

    // DRAW BACKGROUND (FAR) LEGS
    if (viewCos > 0.15) {
      // Side/three-quarter view: draw far limbs first
      drawLeg(-18 * viewCos - 6 * viewSin, -22, lhPhase, false, true); // Left Hind (Far)
      drawLeg(14 * viewCos - 8 * viewSin, -24, lfPhase, true, true);   // Left Fore (Far)
    }

    // 1. PLUMED TAIL (Curled over back - signature Pomeranian spitz trait)
    ctx.save();
    let tailOriginX = lerp(-26, 0, viewSin);
    let tailOriginY = -34;
    let wagX = tailWag * 16;
    let wagY = Math.abs(tailWag) * 4;

    ctx.translate(tailOriginX, tailOriginY);
    let tailGrad = ctx.createLinearGradient(-15, -25, 20, 10);
    tailGrad.addColorStop(0, cHighlight);
    tailGrad.addColorStop(0.4, cCream);
    tailGrad.addColorStop(1, cOrange);
    ctx.fillStyle = tailGrad;

    // Tail plume curves up and cascades forward over pelvis/spine
    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.bezierCurveTo(-14 + wagX, -24 + wagY, 8 + wagX, -48 + wagY, 18 + wagX * 0.5, -36);
    ctx.bezierCurveTo(24 + wagX, -22, 12, -10, 0, 0);
    ctx.fill();

    // Tail fluffy wisps
    ctx.strokeStyle = cHighlight;
    ctx.lineWidth = 1.2;
    for (let t = 0; t < 16; t++) {
      let tx = -10 + t * 2.2 + wagX;
      let ty = -24 - Math.sin(t * 0.25) * 16 + wagY;
      ctx.beginPath();
      ctx.moveTo(tx, ty);
      ctx.lineTo(tx + (hash(t * 5.7) - 0.5) * 12, ty - 8 - hash(t * 2.1) * 8);
      ctx.stroke();
    }
    ctx.restore();

    // 2. MAIN BODY / COMPACT SPITZ TORSO
    let bodyX = lerp(-4, 0, viewSin);
    let bodyY = -28 + breathe * 12;
    let bodyRx = lerp(27, 24, viewSin);
    let bodyRy = lerp(21, 23, viewSin);
    drawFurCluster(bodyX, bodyY, bodyRx, bodyRy, cOrange, cCream, 101, 24);

    // DRAW FOREGROUND (NEAR) LEGS
    if (viewCos > 0.15) {
      drawLeg(-12 * viewCos + 8 * viewSin, -20, rhPhase, false, false); // Right Hind (Near)
      drawLeg(20 * viewCos + 8 * viewSin, -22, rfPhase, true, false);   // Right Fore (Near)
    } else {
      // Frontal presentation: draw all 4 paws planted/weight balanced
      drawLeg(-15, -18, 0, true, false);  // Left front paw
      drawLeg(15, -18, 0, true, false);   // Right front paw
      drawLeg(-22, -22, 0, false, true);  // Left hind base
      drawLeg(22, -22, 0, false, true);   // Right hind base
    }

    // 3. PRONOUNCED NECK RUFF (Huge fluffy mane characteristic of Pomeranians)
    let ruffX = lerp(16, 0, viewSin);
    let ruffY = -38 + headDip * 15;
    let ruffRx = lerp(24, 28, viewSin);
    let ruffRy = lerp(22, 24, viewSin);
    drawFurCluster(ruffX, ruffY, ruffRx, ruffRy, cOrange, cHighlight, 203, 30);

    // 4. HEAD, EARS, MUZZLE & FACE
    ctx.save();
    let headX = lerp(24, 0, viewSin);
    let headY = -48 + headDip * 22;
    ctx.translate(headX, headY);

    // A. Pomeranian Upright Triangular Ears
    function drawEar(ex, ey, rot, earWidth, isFarEar) {
      ctx.save();
      ctx.translate(ex, ey);
      ctx.rotate(rot);

      // Ear outer fur
      ctx.fillStyle = isFarEar ? '#a14714' : '#c8601e';
      ctx.beginPath();
      ctx.moveTo(-earWidth, 0);
      ctx.lineTo(0, -18);
      ctx.lineTo(earWidth, 0);
      ctx.closePath();
      ctx.fill();

      // Ear inner leather pinkish-cream
      ctx.fillStyle = '#f7caa6';
      ctx.beginPath();
      ctx.moveTo(-earWidth * 0.55, -2);
      ctx.lineTo(0, -14);
      ctx.lineTo(earWidth * 0.55, -2);
      ctx.closePath();
      ctx.fill();

      // Ear tuft fur
      ctx.strokeStyle = cHighlight;
      ctx.lineWidth = 1.0;
      ctx.beginPath();
      ctx.moveTo(-earWidth * 0.3, -2);
      ctx.lineTo(-earWidth * 0.6, -8);
      ctx.stroke();

      ctx.restore();
    }

    if (viewSin < 0.6) {
      // Side ears: Left (far) and Right (near)
      drawEar(-6, -14, -0.25, 7, true);
      drawEar(6, -15, 0.15, 7, false);
    } else {
      // Frontal presentation ears
      drawEar(-15, -14, -0.2, 8, false);
      drawEar(15, -14, 0.2, 8, false);
    }

    // B. Head spherical skull fluff
    let headGrad = ctx.createRadialGradient(0, -2, 4, 0, 0, 18);
    headGrad.addColorStop(0, cHighlight);
    headGrad.addColorStop(0.5, cCream);
    headGrad.addColorStop(0.85, cOrange);
    headGrad.addColorStop(1, cDeepOrange);
    ctx.fillStyle = headGrad;
    ctx.beginPath();
    ctx.arc(0, 0, 16, 0, Math.PI * 2);
    ctx.fill();

    // Head fur fringe
    ctx.strokeStyle = cCream;
    ctx.lineWidth = 1;
    for (let hf = 0; hf < 16; hf++) {
      let ha = (hf / 16) * Math.PI * 2;
      ctx.beginPath();
      ctx.moveTo(Math.cos(ha) * 15, Math.sin(ha) * 15);
      ctx.lineTo(Math.cos(ha) * 19, Math.sin(ha) * 19);
      ctx.stroke();
    }

    // C. Pomeranian Dark Expressive Almond Eyes & Eyelids
    function drawEye(eyeX, eyeY, scaleX = 1.0) {
      ctx.save();
      ctx.translate(eyeX, eyeY);
      ctx.scale(scaleX, 1.0);

      // Dark rim / eyeliner
      ctx.fillStyle = '#1a0d07';
      ctx.beginPath();
      ctx.arc(0, 0, 3.8, 0, Math.PI * 2);
      ctx.fill();

      // Deep brown iris
      ctx.fillStyle = '#3c1d0a';
      ctx.beginPath();
      ctx.arc(0, 0, 3.0, 0, Math.PI * 2);
      ctx.fill();

      // Pupil
      ctx.fillStyle = '#0a0503';
      ctx.beginPath();
      ctx.arc(0.3, 0.2, 2.0, 0, Math.PI * 2);
      ctx.fill();

      // Moisture specular glint (sparkle)
      ctx.fillStyle = '#ffffff';
      ctx.beginPath();
      ctx.arc(-1.0, -1.0, 1.0, 0, Math.PI * 2);
      ctx.arc(1.1, 0.9, 0.5, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }

    if (viewSin < 0.6) {
      // Side view eye (near eye visible)
      drawEye(6, -3, 1.0);
    } else {
      // Front view: both almond eyes visible looking at viewer
      drawEye(-8, -2, 1.0);
      drawEye(8, -2, 1.0);
    }

    // D. Short pointed fox-like Pomeranian muzzle & jaw
    let muzzleX = lerp(12, 0, viewSin);
    let muzzleY = lerp(4, 5, viewSin);
    let muzzleW = lerp(10, 12, viewSin);
    let muzzleH = lerp(7, 8, viewSin);

    let muzGrad = ctx.createRadialGradient(muzzleX, muzzleY, 1, muzzleX, muzzleY, muzzleW);
    muzGrad.addColorStop(0, '#fde4cc');
    muzGrad.addColorStop(0.65, '#e4aa7c');
    muzGrad.addColorStop(1, '#ab561e');
    ctx.fillStyle = muzGrad;
    ctx.beginPath();
    ctx.ellipse(muzzleX, muzzleY, muzzleW, muzzleH, 0, 0, Math.PI * 2);
    ctx.fill();

    // Little black Pomeranian button nose
    ctx.fillStyle = '#150d09';
    let noseX = lerp(muzzleX + 8, muzzleX, viewSin);
    let noseY = muzzleY - 2;
    ctx.beginPath();
    ctx.ellipse(noseX, noseY, lerp(2.8, 3.6, viewSin), 2.2, 0, 0, Math.PI * 2);
    ctx.fill();

    // Tiny nose highlight
    ctx.fillStyle = 'rgba(255, 255, 255, 0.6)';
    ctx.beginPath();
    ctx.arc(noseX - 0.8, noseY - 0.7, 0.8, 0, Math.PI * 2);
    ctx.fill();

    // Mouth slit & jaw opening
    let mouthY = muzzleY + 3;
    if (mouthOpen > 0.05) {
      ctx.fillStyle = '#641b1b'; // inside mouth
      ctx.beginPath();
      ctx.ellipse(muzzleX, mouthY + mouthOpen * 3, 5, 2 + mouthOpen * 3, 0, 0, Math.PI * 2);
      ctx.fill();

      // Tiny white canine teeth tips
      ctx.fillStyle = '#ffffff';
      ctx.beginPath();
      ctx.moveTo(muzzleX - 3, mouthY);
      ctx.lineTo(muzzleX - 2, mouthY + 2.5);
      ctx.lineTo(muzzleX - 1, mouthY);
      ctx.moveTo(muzzleX + 1, mouthY);
      ctx.lineTo(muzzleX + 2, mouthY + 2.5);
      ctx.lineTo(muzzleX + 3, mouthY);
      ctx.fill();
    }

    // 5. BONE IN MOUTH (When grabbed, moves rigidly with jaws)
    if (boneGrabbed) {
      let bGripX = lerp(muzzleX + 3, muzzleX, viewSin);
      let bGripY = mouthY + 1;
      let bRot = lerp(-0.06, 0.02, viewSin);
      let bScale = lerp(0.85, 0.95, viewSin);
      drawBone(bGripX, bGripY, bRot, bScale);
    }

    ctx.restore(); // end head transform
    ctx.restore(); // end dog root transform
  }

  // --- COMPOSITE FRAME RENDERER ---
  window.renderFrame = function(timeSeconds) {
    let t = clamp(timeSeconds, 0, 12);
    const kin = getKinematics(t);

    // Clear whole canvas deterministically
    ctx.clearRect(0, 0, 960, 540);

    // 1. Draw backyard environment (sky, fence, foliage, lawn)
    drawEnvironment(t);

    // 2. Draw interior doghouse floor & back walls
    drawDoghouseBack();

    // 3. Resting bone inside doghouse (BEFORE Pomeranian grabs it)
    if (!kin.boneGrabbed) {
      drawBone(BONE_REST_X, BONE_REST_Y, BONE_REST_ANGLE, 0.85);
    }

    // 4. Pomeranian dog & doorway occlusion management:
    // If dog is deep inside the doorway (kin.x > DH_X - 10), parts inside must be occluded
    // by doghouse front wall and door frame.
    // We achieve physical occlusion by clipping or layering:
    if (kin.x > DH_X - 20 && kin.yaw < 0.6) {
      // Dog is entering/inside doorway:
      // Draw dog first inside the doorway
      drawPomeranian(kin);

      // Now draw front wall and roof overhang so doorway cut occludes body passing through!
      drawDoghouseFrontRoof();

      // Doghouse left corner outer wall occlusion
      ctx.save();
      ctx.fillStyle = '#824626';
      ctx.beginPath();
      ctx.rect(DH_X - 12, DH_Y - DH_H + 20, DOOR_X - (DH_X - 12), DH_H + 20);
      ctx.fill();
      ctx.restore();
    } else {
      // Dog has backed out / is in the yard:
      // Draw doghouse complete first, then dog in front of it!
      drawDoghouseFrontRoof();
      drawPomeranian(kin);
    }

    // Subtle atmospheric vignette & lighting warmth
    ctx.save();
    let vig = ctx.createRadialGradient(480, 270, 360, 480, 270, 540);
    vig.addColorStop(0, 'rgba(0, 0, 0, 0)');
    vig.addColorStop(1, 'rgba(20, 15, 10, 0.28)');
    ctx.fillStyle = vig;
    ctx.fillRect(0, 0, 960, 540);
    ctx.restore();
  };

  // Initial call at startup required by delivery contract
  window.renderFrame(0);
})();

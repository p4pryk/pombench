  // Draw leg - thigh, shin, paw, shading with joint circles for articulation
  // posBase: 3D point of leg root (shoulder or hip) relative to dog pelvis local 3D coords
  // phase: gait cycle phase for paw lift (0..1)
  // side: "front" or "hind"
  // isLeft: boolean
  // dog body angles and position for projection and posture

  // leg animation - angles approximated for stance/swing phases

  // We simplify as 3 segments (thigh, shin, paw) connected and bending coherently

  // Returns paw 3D position projected 2D for drawing shadow/contact

  function drawLeg(ctx, dogPos, posBase, phase, yaw, pitch, isFront, isLeft, t, jumpFlight){

    // Base projected pos 2D
    let base2D = project3D(posBase);
    let baseX = dogPos.x + base2D.x;
    let baseY = dogPos.y + base2D.y;

    // Parameters

    // Joint angles modeled as function of phase

    // For stance (paw down), knee and elbow extend progressively to absorb weight

    // For swing (paw lifted), leg folds to tuck +/- paw lift height

    // Paw vertical lift
    let lift = pawLift(phase);

    // During jump flight (t between 2.95 and 3.3 s) legs are tucked bent
    if(jumpFlight){
      lift = 22 + 5*Math.sin(t * 30 + (isFront ? 0 : 2));
    }

    // Lengths in 3D
    let ul = upperLegLen;
    let ll = lowerLegLen;
    let pl = pawLen;

    // Stance phase : extended leg, swing: folded leg
    let stance = phase < 0.55;

    // Angles: bend hip and knee (or shoulder and elbow)

    // We model simple bending 0 (full extend) to max ~60deg flexion for swing

    let maxBend = Math.PI/3;

    let bendAmt = stance ? 0 : maxBend * smoothstep(0,1,(phase-0.55)/0.45);

    // Thigh angle from vertical - body pitch is baseline + hip flexion flare in swing

    // Hip angle oscillates forward/backward with gait

    let baseHipAngle = pitch;
    if(stance){
      baseHipAngle += 0;
    } else {
      baseHipAngle += bendAmt*0.8;
    }

    // Knee angle (between thigh and shin) bending during swing

    let kneeAngle = bendAmt;

    // Paw angle (foot) usually flat when stance; slight flex during swing

    let pawAngle = stance ? 0 : -bendAmt/2;

    // Compute joint positions in 2D projection (setup in 3D then project)

    // Define local 3D vectors:

    // Thigh vector down from hip:
    // Forward is dog X axis
    // bend in plane Y-Z; for front legs fold mostly backward Y-Z plane; for hind legs backward but more extended

    // We model motion in a vertical plane

    // Hip joint position (posBase)

    // Thigh vector forward and down:

    // Define a 2D plane for leg movement in Y-Z around hip position

    // For simplification assume leg bends only in vertical plane aligned with dog yaw

    // Total rotation: dog pitch + body yaw applied to base position

    // We create 3D vector along downward direction with bend

    // Direction of bend: slightly backward in X, downward in Y

    // THIGH VECTOR:

    // Angle from vertical down is bendAmt forward

    let thighAngle = -bendAmt; // bending backward in body frame Y axis

    // Thigh vector in local coordinate y - down, z- back

    let thighVec = {x:0, y:-ul * Math.cos(bendAmt), z: ul * Math.sin(bendAmt)};

    // Knee position = hip + thighVec

    let kneePos = {x:posBase.x + thighVec.x, y:posBase.y + thighVec.y, z:posBase.z + thighVec.z};

    // Lower leg/shin vector

    let shinBend = kneeAngle; // between thigh and shin

    // Shin bends upward from knee in bent angle

    let shinVec = {x: 0, y:-ll * Math.cos(shinBend), z: -ll * Math.sin(shinBend)};

    // Paw position = knee + shinVec
    let pawPos = {x:kneePos.x + shinVec.x, y:kneePos.y + shinVec.y, z:kneePos.z + shinVec.z};

    // Apply paw lift upward in Y axis

    pawPos.y -= lift;

    // Project joints to 2D:

    let hip2D = project3D(posBase);
    let knee2D = project3D(kneePos);
    let paw2D = project3D(pawPos);

    // Scale positions to canvas

    let hipX = dogPos.x + hip2D.x;
    let hipY = dogPos.y + hip2D.y;
    let kneeX = dogPos.x + knee2D.x;
    let kneeY = dogPos.y + knee2D.y;
    let pawX = dogPos.x + paw2D.x;
    let pawY = dogPos.y + paw2D.y;

    // DRAW leg segments

    ctx.strokeStyle = shadowFur;
    ctx.lineCap = "round";

    // Thigh
    let gradThigh = ctx.createLinearGradient(hipX, hipY, kneeX, kneeY);
    gradThigh.addColorStop(0, baseFur);
    gradThigh.addColorStop(1, shadowFur);
    ctx.strokeStyle = gradThigh;
    ctx.lineWidth = pawRadius * 1.8;
    ctx.beginPath();
    ctx.moveTo(hipX, hipY);
    ctx.lineTo(kneeX, kneeY);
    ctx.stroke();

    // Shin
    let gradShin = ctx.createLinearGradient(kneeX, kneeY, pawX, pawY);
    gradShin.addColorStop(0, baseFur);
    gradShin.addColorStop(1, shadowFur);
    ctx.strokeStyle = gradShin;
    ctx.lineWidth = pawRadius * 1.5;
    ctx.beginPath();
    ctx.moveTo(kneeX, kneeY);
    ctx.lineTo(pawX, pawY);
    ctx.stroke();

    // Paw : small circle
    ctx.fillStyle = shadowFur;
    ctx.beginPath();
    ctx.ellipse(pawX, pawY, pawRadius*0.65, pawRadius*0.4, 0, 0, 2*Math.PI);
    ctx.fill();

    ctx.fillStyle = baseFur;
    ctx.beginPath();
    ctx.ellipse(pawX, pawY, pawRadius*0.55, pawRadius*0.35, 0, 0, 2*Math.PI);
    ctx.fill();

    // Paw pad details: 3 small ellipses on paw
    const pads = [
      {x: 0, y: 1.5},
      {x: -4, y: 3},
      {x: 4, y: 3}
    ];
    ctx.fillStyle = "#6b3f0f";
    for(let pad of pads){
      ctx.beginPath();
      ctx.ellipse(pawX + pad.x, pawY + pad.y, 3 * SCALE, 2 * SCALE, 0, 0, 2*Math.PI);
      ctx.fill();
    }

    // Joint circles to mark hip, knee
    ctx.fillStyle = shadowFur;
    ctx.beginPath();
    ctx.ellipse(hipX, hipY, pawRadius*0.5, pawRadius*0.4, 0, 0, 2*Math.PI);
    ctx.fill();
    ctx.beginPath();
    ctx.ellipse(kneeX, kneeY, pawRadius*0.43, pawRadius*0.36, 0, 0, 2*Math.PI);
    ctx.fill();

    return {x:pawX, y:pawY};
  }

  // Draw dog with all parts in actor pose

  // Times params input various leg phases and animation states:

  function renderDog(ctx,t){

    let dogPx = dogX(t);
    let dogPy = dogY(t);

    let yaw = dogYaw(t);
    let pitch = dogPitch(t);
    let headP = headPitch(t);
    let headY = headYaw(t);
    let jumpF = (t >= 2.95 && t < 3.3);

    // Leg phases
    const phases = {
      LF: legPhase(t,0),
      RH: legPhase(t,0),
      RF: legPhase(t,0.5),
      LH: legPhase(t,0.5),
    };

    // Positions base for legs
    const frontLeftBase = {x:shoulderX, y:0, z:-0.17*dLen};
    const frontRightBase= {x:shoulderX, y:0, z:0.17*dLen};
    const hindLeftBase = {x:0, y:0, z:-0.14*dLen};
    const hindRightBase= {x:0, y:0, z:0.14*dLen};

    // Calculate dogPos 3D base pelvis at y=ground=0 (for projection we put y=0)

    let dogPos = {x:dogPx,y:dogPy};

    // TAIL
    drawTail(ctx, dogPos, pitch, yaw, t);

    // Body & neck drawn behind legs
    drawTorso(ctx,dogPos,pitch,yaw);
    drawNeckRuff(ctx,dogPos,pitch,yaw);

    // Legs drawn behind and in front depending on dog yaw rotation angle
    // Front left & right legs
    let pawLF = drawLeg(ctx,dogPos, frontLeftBase, phases.LF, yaw, pitch, true, true, t, jumpF);
    let pawRF = drawLeg(ctx,dogPos, frontRightBase, phases.RF, yaw, pitch, true, false, t, jumpF);
    // Hind legs
    let pawLH = drawLeg(ctx,dogPos, hindLeftBase, phases.LH, yaw, pitch, false, true, t, jumpF);
    let pawRH = drawLeg(ctx,dogPos, hindRightBase, phases.RH, yaw, pitch, false, false, t, jumpF);

    // Bone logic: if after grab (t>4.6) dog carries bone in mouth
    let carryingBone = (t >= 4.5);

    // Head drawn last in front of legs
    drawHead(ctx,dogPos, pitch, yaw, headP, headY);

    // Draw bone in mouth if carrying
    if(carryingBone){
      // Bone position: attached to dog's muzzle front center (head front)
      // Approx head front projected 2D position:
      let headBase3D = {x: shoulderX + neckLength3D + headLength*0.4, y: neckHeight*0.56, z: 0};
      let headFront2D = project3D(headBase3D);
      let boneX = dogPos.x + headFront2D.x;
      let boneY = dogPos.y + headFront2D.y;

      // Bone rotated approx horizontal with slight pitch following head pitch
      drawBone(ctx, boneX, boneY, pitch + 0.1, 1, 1);
    }
  }

  // Draw dog shadow contact on ground ellipse at paws position
  // Shadow darkening depending on leg support etc

  // We'll render subtle black with blur under dog feet

  // Positions paw shadow ellipses on ground under paws

  function drawDogShadow(ctx, t){
    // Get paw positions for drawLeg calls stored here

    let dogPx = dogX(t);
    let dogPy = dogY(t);

    let yaw = dogYaw(t);
    let pitch = dogPitch(t);

    let jumpF = (t >= 2.95 && t < 3.3);

    const phases = {
      LF: legPhase(t,0),
      RH: legPhase(t,0),
      RF: legPhase(t,0.5),
      LH: legPhase(t,0.5),
    };

    const frontLeftBase = {x:shoulderX, y:0, z:-0.17*dLen};
    const frontRightBase= {x:shoulderX, y:0, z:0.17*dLen};
    const hindLeftBase = {x:0, y:0, z:-0.14*dLen};
    const hindRightBase= {x:0, y:0, z:0.14*dLen};

    let dogPos = {x:dogPx,y:dogPy};

    // Compute paw positions:

    function pawPos(posBase,phase){
      // Same as drawLeg paw position function

      // Paw lift height
      let lift = pawLift(phase);
      if(jumpF) lift = 22 + 5*Math.sin(t * 30);

      let ul = upperLegLen;
      let ll = lowerLegLen;
      let bendAmt = phase < 0.55 ? 0 : Math.PI/3 * smoothstep(0,1,(phase-0.55)/0.45);
      let thighVec = {x:0, y:-ul * Math.cos(bendAmt), z: ul * Math.sin(bendAmt)};
      let kneePos = {x:posBase.x + thighVec.x, y:posBase.y + thighVec.y, z:posBase.z + thighVec.z};
      let shinVec = {x: 0, y:-ll * Math.cos(bendAmt), z: -ll * Math.sin(bendAmt)};
      let pawPos = {x:kneePos.x + shinVec.x, y:kneePos.y + shinVec.y, z:kneePos.z + shinVec.z};
      pawPos.y -= lift;

      let paw2D = project3D(pawPos);
      return {x:dogPos.x + paw2D.x, y:dogPos.y + paw2D.y};
    }

    let shadows = [
      pawPos(frontLeftBase, phases.LF),
      pawPos(frontRightBase, phases.RF),
      pawPos(hindLeftBase, phases.LH),
      pawPos(hindRightBase, phases.RH)
    ];

    // Draw shadows ellipses on ground (y=groundY)
    ctx.fillStyle = "rgba(0,0,0,0.15)";
    for(let s of shadows){
      let shadowX = s.x;

      ctx.ellipse(shadowX, shadowY, 18*SCALE, 7*SCALE, 0, 0, 2*Math.PI);

    ctx.save();
    clipPuddle();

    // Make transform origin at puddleCenterY
    ctx.translate(0, puddleCenterY);
    ctx.scale(1, -0.65);
    ctx.translate(0, -puddleCenterY);

    ctx.globalAlpha = 0.45;

    renderDog(ctx, t);

    ctx.globalAlpha = 1;
    ctx.restore();
  }

  // Main renderFrame function exposed

  window.renderFrame = function(t){
    t = Math.max(t,0);
    if(t > T_SETTLE_END) t = T_SETTLE_END;

    // Clear
    ctx.clearRect(0,0,W,H);

    // Draw background ground and puddle
    drawGround();
    drawPuddle(t);

    // Draw dog reflection inside puddle area
    drawDogReflection(ctx,t);

    // Draw doghouse behind dog (occlusion by doorway partial)
    drawDoghouse(ctx);

    // Draw bone resting in doghouse if dog not carrying bone
    if(t < 4.5){
      drawBoneRest(ctx);
    }

    // Draw dog shadows under paws on ground
    drawDogShadow(ctx,t);

    // Draw dog in normal colors on top
    renderDog(ctx,t);
  };

  // Initial draw
  window.renderFrame(0);
})();

// Generate leg angle offsets (rotation about shoulder/hip)
function runLegAngle(t,u_phase) {
  // u_phase 0..1 full cycle, t in [1..3]
  // Angle range in radians, front legs + paw flex
  let stepCycle = (t/0.5) % 1; // running footstep cycle 2 steps/s
  let phase = (stepCycle + u_phase) % 1;
  // phase close to 0 or 1 means paw on ground (stance)
  // phase close to 0.5 means paw lifted (swing)
  // We approximate a periodic curve: smooth max
  // At stance, leg extends downward (angle ~ 0 rad)
  // At swing, leg flexes and raises paw (angle ~ -0.5 rad)
  let angle = 0;
  if(phase < 0.5){
    // stance phase, leg down
    let w = phase / 0.5;
    angle = -0.5*(1-w) + 0*w;
  } else {
    // swing phase, leg raised
    let w = (phase-0.5)/0.5;
    angle = 0*-w + (-0.5)*w;
  }
  // We damp the amplitude by 0.75 for realism
  return 0.75*angle;
}
// Some tweak: smoother paw flex during swing
function pawFlex(t,u_phase){
  let stepCycle = (t/0.5) % 1;
  let phase = (stepCycle + u_phase) % 1;
  if(phase < 0.5) return 0;
  // flex increases 0 to -0.3 during swing
  return -0.3*(phase-0.5)*2;
}

function drawTail(ctx,x,y, angle, curl, coatColor) {
  // Tail starts at (x,y) rotates by base angle and curls up in segments
  ctx.save();
  ctx.translate(x,y);
  ctx.rotate(angle);
  // Draw multiple segments with upward curve
  let seg_len = 20;
  let segs = 3;
  for(let i=0;i<segs;i++){
    ctx.save();
    let curlAngle = curl * (Math.sin(Math.PI*(i/segs)))*0.7;
    ctx.rotate(curlAngle);
    // segment ellipse shape with some taper
    let rx = seg_len * (1 - i*0.2);
    let ry = 13 * (1 - i*0.35);
    // lighter cream near tip
    let c = (i < segs-1) ? coatColor : COLORS.dogCream;
    ctx.fillStyle = c;
    ctx.shadowColor = COLORS.dogShadow;
    ctx.shadowBlur=4;
    ctx.beginPath();
    ctx.ellipse(rx/2, 0, rx/2, ry/2, 0, 0, 2*Math.PI);
    ctx.fill();
    ctx.restore();
    ctx.translate(seg_len*0.75,0);
  }
  ctx.restore();
}

  // House at 780 y=430 ground level

  let dogPosY = Y_GROUND;
  let dogBodyAngle = 0;
  let tailCurl = 0.95 * Math.sin(t*14) * 0.18; // subtle tail wag default
  let headAngle = 0;
  let eyeDir = 0; // neutral
  let boneHeld = false;
  let boneRot = 0;

  // Run gait leg angles
  let runLegAngles = {LF:0, RF:0, LR:0, RR:0};
  let runPawFlex = {LF:0, RF:0, LR:0, RR:0};

  // Bone position world - needed for grab and backout synchronization
  // Bone original pos inside doghouse floor (inside front door)
  let doorXStart = HOUSE.pos.x + HOUSE.doorOffsetX;
  let doorYFloor = HOUSE.pos.y;
  // Bone resting position in yard coords
  let bonePosXRest = doorXStart + boneRestHouseLocal.x;
  let bonePosYRest = doorYFloor - boneRestHouseLocal.y;
  // During grab and hold, bone moves with dog mouth

  // Phase stepwise

  if(t < 1){
    // Phase 0 establish scene no dog move
    dogPosX = 350;
    dogPosY = Y_GROUND;
    dogBodyAngle = 0;
    headAngle = 0;
    // Dog is idle, subtle breathing tail swings
    tailCurl = 0.15 * Math.sin(t*8);
    boneHeld = false;

  } else if(t < 3){
    // run toward doghouse
    let runT = (t-1)/2; // 0..1
    dogPosX = 350 + 410*runT;
    dogPosY = Y_GROUND - 3*Math.sin(runT*Math.PI*2);
    dogBodyAngle = 0.04 * Math.sin(runT*Math.PI*14);
    tailCurl = 0.12 + 0.3 * Math.sin(t*20);
    headAngle = 0;

    // run gait legs - trot pattern diagonals
    runLegAngles.LF = runLegAngle(t,0);
    runLegAngles.RR = runLegAngle(t,0);
    runLegAngles.RF = runLegAngle(t,0.5);
    runLegAngles.LR = runLegAngle(t,0.5);
    runPawFlex.LF = pawFlex(t,0);
    runPawFlex.RR = pawFlex(t,0);
    runPawFlex.RF = pawFlex(t,0.5);
    runPawFlex.LR = pawFlex(t,0.5);

    boneHeld = false;
  } else if(t < 4.5){
    // Enter through doorway forwards, approx 1.5s duration
    // Dog pivot approaches door threshold about x= doorXStart+40 center offset

    let enterT = (t-3)/1.5; // 0..1

    // Approach onto door step a bit + slight Y offset for step up
    dogPosX = 760 + 20*enterT;
    dogPosY = Y_GROUND - 12*enterT;

    // Body rotates slightly to face house doorway (45 deg turned)
    dogBodyAngle = 0.3 * enterT;

    // Legs: slow walking backward-like paw movement but forward
    let walkT = t - 3;
    runLegAngles.LF = 0.09 * Math.sin(walkT*13 + Math.PI*0.1);
    runLegAngles.RF = 0.1 * Math.sin(walkT*13);
    runLegAngles.LR = -0.07 * Math.sin(walkT*13);
    runLegAngles.RR = -0.1 * Math.sin(walkT*13 + Math.PI*0.15);
    runPawFlex.LF = 0;
    runPawFlex.RF = 0;
    runPawFlex.LR = 0;
    runPawFlex.RR = 0;

    headAngle = 0.1 * Math.sin(t*10);
    tailCurl = 0.4 * enterT;

    boneHeld = false;
  } else if(t < 6){
    // Grab bone in 1.5s time, from door partially occluded entered position
    let grabT = (t-4.5)/1.5; // 0..1

    // Dog mostly standing in door, small adjustments in body pose
    dogPosX = 780;
    dogPosY = Y_GROUND - 15*(0.7 - Math.abs(grabT - 0.5)*2) * 0.25;
    dogBodyAngle = 0.32 - 0.07*grabT;
    headAngle = 0.2*Math.sin(grabT*Math.PI*4);

    // Legs stable stance, slight bend
    runLegAngles.LF = -0.03;
    runLegAngles.RF = -0.015;
    runLegAngles.LR = 0.01;
    runLegAngles.RR = -0.015;
    runPawFlex.LF = 0;
    runPawFlex.RF = 0;
    runPawFlex.LR = 0;
    runPawFlex.RR = 0;

    tailCurl = 0.5 * Math.sin(t*10);

    // The bone is originally resting inside house at fixed pos
    // Grab phase: mouth lowers and opens, bone moves gradually from resting pos to mouth

    // Mouth vertical / jaw open simulate by head lift on y
    // We simulate a vertical bone moving mouth - linear lerp from house pos to mouth pos
    boneHeld = grabT > 0.33; // hold after 1/3 of phase

    if(grabT < 0.33){
      boneRot = 0;
    } else {
      boneRot = 0.2*Math.sin(6*grabT*Math.PI);
    }
  } else if(t < 8){
    // Back out with bone for 2s: reverse walking, slow movement out the door
    let backT = (t-6)/2; // 0..1
    dogPosX = 780 - 290*backT;
    dogPosY = Y_GROUND + 8*backT;
    dogBodyAngle = 0.25 * Math.sin(backT*Math.PI*4) + 0.01;

    // Legs: slow backward stepping
    let backStep = t-6;
    runLegAngles.LF = 0.1 * Math.sin(backStep*7 + Math.PI*0.1);
    runLegAngles.RF = 0.12 * Math.sin(backStep*7);
    runLegAngles.LR = -0.1 * Math.sin(backStep*7);
    runLegAngles.RR = -0.12 * Math.sin(backStep*7 + Math.PI*0.15);
    runPawFlex.LF = 0;
    runPawFlex.RF = 0;
    runPawFlex.LR = 0;
    runPawFlex.RR = 0;

    tailCurl = 0.3 + 0.2 * Math.sin(t*12);

    headAngle = 0.1*Math.sin(t*10);

    // Bone rigidly held, mouth moves with dog head
    boneHeld = true;
    boneRot = 0.1*Math.sin(t*10)*0.35;

  } else if(t < 10){
    // Turn toward viewer for 2s
    let turnT = (t-8)/2; // 0..1

    dogPosX = 490 + 20*turnT;
    dogPosY = Y_GROUND - 5*Math.sin(turnT*Math.PI*1.5);
    // Body rotates from facing right (0 rad) to facing forward (PI/2 rad)
    dogBodyAngle = 0.3 * (1 - turnT);
    headAngle = 0.6 * (turnT);
    tailCurl = 0.15 * Math.sin(t*15) + 0.4*turnT;
    // Legs steady stance, no large paw movement for pivot
    runLegAngles.LF = -0.02;
    runLegAngles.RF = -0.04;
    runLegAngles.LR = 0.02;
    runLegAngles.RR = -0.01;
    runPawFlex.LF = 0;
    runPawFlex.RF = 0;
    runPawFlex.LR = 0;
    runPawFlex.RR = 0;
    boneHeld = true;
    boneRot = 0.7*turnT;
  } else {
    // Final pose: frontal facing viewer, small breathing, tail sway, steady legs
    dogPosX = 510;
    dogPosY = Y_GROUND - 5;
    dogBodyAngle = 0;
    headAngle = 0;
    tailCurl = 0.3 * Math.sin(t*6);
    runLegAngles.LF = 0;
    runLegAngles.RF = 0;
    runLegAngles.LR = 0;
    runLegAngles.RR = 0;
    runPawFlex.LF = 0;
    runPawFlex.RF = 0;
    runPawFlex.LR = 0;
    runPawFlex.RR = 0;
    boneHeld = true;
    boneRot = 0.2 * Math.sin(t*4);
  }

  return {pos:{x:dogPosX,y:dogPosY}, bodyAngle: dogBodyAngle, tailCurl: tailCurl, headAngle: headAngle, eyeDir: eyeDir,
    runLegAngles:runLegAngles, runPawFlex:runPawFlex, boneHeld: boneHeld, boneRot: boneRot};
}

  drawDogHouse(ctx);

  // Bone is only visible from outside when in doorway line of sight

  let dogstate = animationState(t);

  // Draw bone resting on floor inside doghouse if not held
  if(t < 4.8){
    // Gatekeeper threshold range: bone visible inside door
    let doorX = HOUSE.pos.x + HOUSE.doorOffsetX;
    let doorY = HOUSE.pos.y;
    // We offset bone z for layering later
    drawBone(ctx, doorX + boneRestHouseLocal.x, doorY - boneRestHouseLocal.y, 0);

  if(t>=3 && t<=8){

    clipDoor(ctx);
    drawDog(ctx,dogstate);
    ctx.restore();
  } else {
    drawDog(ctx,dogstate);
  }

  if(dogstate.boneHeld && t >= 4.8) {

    let dogY = dogstate.pos.y;

    // Add head rotation offset as small bone tilt
    let boneRot = dogstate.boneRot;

    drawBone(ctx, dogX + bx, dogY + by, boneRot);
  }

  // Draw soft shadows of dog paws on ground (approximated)
  // Paw contact points in world coords offset from dog pos (body origin)
  // For each paw, calculate positions according to joint angles (approximate)
  // This stage omitted due to complexity and visual clutter, shadows implicit by body shadow

  // Final subtle overlays or anti-aliasing - none for speed

};

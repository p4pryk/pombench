Below is a complete HTML document that meets the requirements specified for the animation of a Pomeranian dog retrieving a bone from its doghouse and proudly displaying it to the viewer.

```html
<!DOCTYPE html>
<html>
<head>
    <meta charset="UTF-8">
    <title>Pomeranian and Bone Animation</title>
    <style>
        body, html { margin: 0; padding: 0; overflow: hidden; }
        #stage { border: 1px solid black; }
    </style>
</head>
<body>
<canvas id="stage" width="960" height="540"></canvas>
<script>
    const canvas = document.getElementById('stage');
    const ctx = canvas.getContext('2d');

    // Constants
    const PomeranianSize = 50;
    const DoghouseWidth = 100;
    const DoghouseHeight = 80;
    const BoneSize = 20;
    const FPS = 30;

    // Pomeranian properties
    let pomeranianX = 100;
    let pomeranianY = 100;
    let pomeranianAngle = 0;
    let pomeranianLegAngle = 0;
    let pomeranianHasBone = false;

    // Bone properties
    let boneX = 200;
    let boneY = 250;

    function drawPomeranian(x, y, angle) {
        ctx.save();
        ctx.translate(x, y);
        ctx.rotate(angle);

        // Body
        ctx.fillStyle = 'orange';
        ctx.beginPath();
        ctx.ellipse(0, 0, PomeranianSize/2, PomeranianSize/2, 0, 0, Math.PI * 2);
        ctx.fill();

        // Head
        ctx.fillStyle = 'orange';
        ctx.beginPath();
        ctx.ellipse(-PomeranianSize/4, -PomeranianSize/2, PomeranianSize/4, PomeranianSize/4, 0, 0, Math.PI * 2);
        ctx.fill();

        // Ears
        ctx.fillStyle = 'orange';
        ctx.beginPath();
        ctx.ellipse(-PomeranianSize/4 - PomeranianSize/8, -PomeranianSize/2 - PomeranianSize/8, PomeranianSize/8, PomeranianSize/16, 0, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.ellipse(PomeranianSize/4 - PomeranianSize/8, -PomeranianSize/2 - PomeranianSize/8, PomeranianSize/8, PomeranianSize/16, 0, 0, Math.PI * 2);
        ctx.fill();

        // Eyes
        ctx.fillStyle = 'black';
        ctx.beginPath();
        ctx.ellipse(-PomeranianSize/8, -PomeranianSize/2 - PomeranianSize/16, PomeranianSize/16, PomeranianSize/16, 0, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.ellipse(PomeranianSize/8, -PomeranianSize/2 - PomeranianSize/16, PomeranianSize/16, PomeranianSize/16, 0, 0, Math.PI * 2);
        ctx.fill();

        // Legs
        ctx.fillStyle = 'orange';
        ctx.beginPath();
        ctx.ellipse(-PomeranianSize/8, PomeranianSize/4, PomeranianSize/8, PomeranianSize/16, 0, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.ellipse(PomeranianSize/8, PomeranianSize/4, PomeranianSize/8, PomeranianSize/16, 0, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.ellipse(-PomeranianSize/8, PomeranianSize/4 + PomeranianSize/8, PomeranianSize/8, PomeranianSize/16, 0, 0, Math.PI * 2);
        ctx.fill();
        ctx.beginPath();
        ctx.ellipse(PomeranianSize/8, PomeranianSize/4 + PomeranianSize/8, PomeranianSize/8, PomeranianSize/16, 0, 0, Math.PI * 2);
        ctx.fill();

        // Tail
        ctx.beginPath();
        ctx.moveTo(0, -PomeranianSize/2);
        ctx.quadraticCurveTo(PomeranianSize/4, -PomeranianSize, 0, -PomeranianSize/2 - PomeranianSize/4);
        ctx.stroke();

        ctx.restore();
    }

    function drawBone(x, y) {
        ctx.fillStyle = 'brown';
        ctx.beginPath();
        ctx.rect(x - BoneSize/2, y - BoneSize/2, BoneSize, BoneSize);
        ctx.fill();
    }

    function drawDoghouse(x, y) {
        ctx.fillStyle = 'brown';
        ctx.beginPath();
        ctx.rect(x, y, DoghouseWidth, DoghouseHeight);
        ctx.fill();
        ctx.fillStyle = 'black';
        ctx.beginPath();
        ctx.rect(x + DoghouseWidth/2 - 10, y - 20, 20, 20);
        ctx.fill();
    }

    function renderFrame(timeSeconds) {
        ctx.clearRect(0, 0, canvas.width, canvas.height);

        // Draw doghouse
        drawDoghouse(400, 400);

        // Draw bone
        if (!pomeranianHasBone) {
            drawBone(boneX, boneY);
        }

        // Pomeranian movement
        if (timeSeconds < 1) {
            pomeranianX = 100;
            pomeranianY = 100;
        } else if (timeSeconds < 3) {
            pomeranianX = 100 + (timeSeconds - 1) * 50;
            pomeranianY = 100;
        } else if (timeSeconds < 4.5) {
            pomeranianX = 400 - 20 + (timeSeconds - 3) * 20;
            pomeranianY = 400 - DoghouseHeight + PomeranianSize/2 + (timeSeconds - 3) * 10;
        } else if (timeSeconds < 6) {
            pomeranianX = 400 - 20;
            pomeranianY = 400 - DoghouseHeight + PomeranianSize/2;
        } else if (timeSeconds < 8) {
            pomeranianX = 400 - 20 - (timeSeconds - 6) * 20;
            pomeranianY = 400 - DoghouseHeight + PomeranianSize/2;
            if (timeSeconds > 6) pomeranianHasBone = true;
        } else if (timeSeconds < 10) {
            pomeranianX = 400 - 20 - 40 + (timeSeconds - 8) * 20;
            pomeranianY = 400 - DoghouseHeight + PomeranianSize/2;
            pomeranianAngle = (timeSeconds - 8) * Math.PI / 2;
        } else {
            pomeranianX = 400 - 20 - 40;
            pomeranianY = 400 - DoghouseHeight + PomeranianSize/2;
            pomeranianAngle = Math.PI / 2;
        }

        // Draw Pomeranian
        drawPomeranian(pomeranianX, pomeranianY, pomeranianAngle);

        // Keep the final pose for t >= 12
        if (timeSeconds >= 12) {
            pomeranianX = 400 - 20 - 40;
            pomeranianY = 400 - DoghouseHeight + PomeranianSize/2;
            pomeranianAngle = Math.PI / 2;
            pomeranianHasBone = true;
            drawPomeranian(pomeranianX, pomeranianY, pomeranianAngle);
            drawDoghouse(400, 400);
            drawBone(pomeranianX - 10, pomeranianY - PomeranianSize/4);
        } else {
            if (pomeranianHasBone) {
                drawBone(pomeranianX - 10, pomeranianY - PomeranianSize/4);
            }
        }
    }

    window.renderFrame = renderFrame;
    renderFrame(0);
</script>
</body>
</html>
```

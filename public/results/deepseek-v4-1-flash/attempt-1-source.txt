Here is a complete, self-contained HTML document that draws a 12-second animation of a Pomeranian retrieving a bone from its doghouse. The scene includes a shallow puddle, a reflective water surface, and a final front-facing pose with the bone in its mouth.

  <canvas id="stage" width="960" height="540"></canvas>
  <script>
    (function(){
      const canvas = document.getElementById('stage');
      const ctx = canvas.getContext('2d');
      const W = 960, H = 540;

      // ---------- deterministic PRNG ----------
      function mulberry32(a) {
        return function() {
          a |= 0; a = a + 0x6D2B79F5 | 0;
          let t = Math.imul(a ^ a >>> 15, 1 | a);
          t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
          return ((t ^ t >>> 14) >>> 0) / 4294967296;
        };
      }

      // ---------- geometric helpers ----------
      function lerp(a,b,t){ return a+(b-a)*Math.max(0,Math.min(1,t)); }
      function clamp(v,a,b){ return Math.max(a,Math.min(b,v)); }
      function smoothstep(t){ t = clamp(t,0,1); return t*t*(3-2*t); }

      // ---------- scene constants ----------
      const GROUND_Y = 400;          // main yard surface (y-coordinate at z=0)
      const HORIZON_Y = 260;         // distant yard horizon
      const DOGHOUSE = {
        x: 690, y: 310,              // top-left of front face
        w: 160, h: 140,
        doorW: 54, doorH: 70,
        doorX: 690 + (160-54)/2,     // centered door
        roofPeak: 310 - 42
      };
      const PUDDLE = {
        cx: 430, cy: 418,
        rx: 72, ry: 24
      };
      const BONE = {
        x: 742, y: 438,              // resting inside doghouse on floor
        w: 34, h: 11,
        angle: 0.12
      };

      // ---------- key frame / state interpolation ----------
      // Phases (seconds):
      // 0-1   : idle look, establish
      // 1-3   : run to puddle, leap, land
      // 3-4.5 : enter doghouse
      // 4.5-6 : grab bone
      // 6-8   : reverse out with bone
      // 8-10  : turn to face viewer
      // 10-12 : hold front-facing pose

      function getDogState(t){
        let x, y, bodyAngle, headAngle, tailCurl, legPhase, mouthOpen, hasBone, boneAngle;
        let bodyTilt = 0;

        // defaults
        x = 260; y = GROUND_Y; bodyAngle = -0.05; headAngle = 0; tailCurl = 1.0;
        legPhase = 0; mouthOpen = 0; hasBone = false; boneAngle = BONE.angle;
        bodyTilt = 0;

        if (t < 1.0) {
          // idle: look toward house
          const p = t;
          x = 260;
          bodyAngle = lerp(-0.1, -0.15, p);
          headAngle = lerp(0.0, 0.25, Math.min(p*1.2,1));
          tailCurl = 0.9 + 0.1*Math.sin(p*4);
          legPhase = 0;
        } else if (t < 3.0) {
          // run from x=260 to x=600 (approach puddle, leap, land)
          const p = (t - 1.0) / 2.0; // 0..1
          // piecewise: run, anticipate, leap, land
          if (p < 0.35) {
            // run forward on ground
            const q = p / 0.35;
            x = lerp(260, 390, q);
            y = GROUND_Y;
            bodyAngle = -0.1 - 0.08*Math.sin(q*Math.PI*2);
            legPhase = q * 6;
            headAngle = 0.15;
          } else if (p < 0.5) {
            // anticipation: shorten stride, lower body
            const q = (p - 0.35) / 0.15;
            x = lerp(390, 410, q);
            y = GROUND_Y + 6 * Math.sin(q*Math.PI);
            bodyAngle = -0.18;
            legPhase = 2.0 + q*2;
            headAngle = 0.1;
            bodyTilt = 0.08 * Math.sin(q*Math.PI);
          } else if (p < 0.72) {
            // leap: arc over puddle, all paws off
            const q = (p - 0.5) / 0.22;
            x = lerp(410, 530, q);
            // parabolic arc: peak height ~55px above ground
            const arc = 4 * 55 * q * (1 - q);
            y = GROUND_Y - arc;
            bodyAngle = -0.2 + 0.15 * Math.sin(q*Math.PI);
            legPhase = 4.0 + q*4;
            tailCurl = 0.8 + 0.4 * Math.sin(q*Math.PI);
            headAngle = -0.1;
            bodyTilt = -0.15 * Math.sin(q*Math.PI);
          } else {
            // landing and resume
            const q = (p - 0.72) / 0.28;
            x = lerp(530, 600, q);
            // landing compression then up
            if (q < 0.2) {
              y = GROUND_Y + 8 * Math.sin((q/0.2)*Math.PI);
            } else {
              y = GROUND_Y;
            }
            bodyAngle = lerp(-0.05, -0.1, q);
            legPhase = 2.0 + q*6;
            headAngle = 0.1;
          }
        } else if (t < 4.5) {
          // enter doghouse: x from 600 to 720, y constant
          const p = (t - 3.0) / 1.5;
          const q = smoothstep(p);
          x = lerp(600, 720, q);
          y = GROUND_Y;
          bodyAngle = -0.08;
          legPhase = q * 6;
          headAngle = lerp(0.2, 0.5, Math.min(q*1.5,1));
          tailCurl = 0.7;
        } else if (t < 6.0) {
          // inside, grab bone (4.5-6s)
          const p = (t - 4.5) / 1.5;
          x = 740 + 8 * Math.sin(p*Math.PI);
          y = GROUND_Y + 4 * Math.sin(p*Math.PI); // slight head lower
          bodyAngle = 0.05;
          headAngle = 0.8 - 0.6 * smoothstep(p);
          mouthOpen = Math.sin(p * Math.PI * 0.8) * 0.9;
          hasBone = (p > 0.75);
          boneAngle = 0.1;
          tailCurl = 0.6;
        } else if (t < 8.0) {
          // reverse out (6-8s)
          const p = (t - 6.0) / 2.0;
          const q = smoothstep(p);
          x = lerp(740, 620, q);
          y = GROUND_Y;
          bodyAngle = 0.1 - 0.15 * q;
          headAngle = 0.2;
          legPhase = -q * 5;
          mouthOpen = 0.55 - 0.05*Math.sin(p*Math.PI*3);
          hasBone = true;
          boneAngle = 0.15 + 0.1*Math.sin(p*Math.PI*2);
          tailCurl = 0.9;
          bodyTilt = 0.08 * Math.sin(q*Math.PI);
        } else if (t < 10.0) {
          // turn to face viewer (8-10s)
          const p = (t - 8.0) / 2.0;
          const q = smoothstep(p);
          // pivot: x moves slightly, bodyAngle rotates from side to front
          x = lerp(620, 580, q);
          y = GROUND_Y;
          bodyAngle = lerp(-0.1, 0.0, q); // 0 = front-facing
          headAngle = lerp(0.1, 0.0, q);
          legPhase = Math.sin(q*Math.PI*2) * 2;
          mouthOpen = 0.5 - 0.2*q;
          hasBone = true;
          boneAngle = 0.05 + 0.1*(1-q);
          tailCurl = lerp(0.8, 1.0, q);
          bodyTilt = 0.1 * Math.sin(q*Math.PI);
        } else {
          // hold front-facing (10-12s) with subtle breathing
          const p = (t - 10.0) / 2.0;
          x = 580;
          y = GROUND_Y;
          bodyAngle = 0.0;
          headAngle = 0.0;
          legPhase = 0;
          mouthOpen = 0.3 + 0.05 * Math.sin(p * Math.PI * 6);
          hasBone = true;
          boneAngle = 0.05;
          tailCurl = 1.0 + 0.04 * Math.sin(p * Math.PI * 8);
          // breathing
          bodyTilt = 0.015 * Math.sin(p * Math.PI * 4);
        }

        return { x, y, bodyAngle, headAngle, tailCurl, legPhase, mouthOpen, hasBone, boneAngle, bodyTilt };
      }

      // ---------- drawing functions ----------

      function drawBackground(){
        // sky gradient
        const skyGrad = ctx.createLinearGradient(0, 0, 0, HORIZON_Y+60);
        skyGrad.addColorStop(0, '#6a8a9a');
        skyGrad.addColorStop(0.5, '#9ab0b8');
        skyGrad.addColorStop(1, '#c8d4c8');
        ctx.fillStyle = skyGrad;
        ctx.fillRect(0, 0, W, HORIZON_Y+60);

        // distant trees/foliage
        ctx.fillStyle = '#4a6a4a';
        ctx.beginPath();
        ctx.moveTo(0, HORIZON_Y+60);
        for (let i=0; i<W; i+=30) {
          ctx.lineTo(i, HORIZON_Y - 20 + Math.sin(i*0.03)*15);
        }
        ctx.lineTo(W, HORIZON_Y+60);
        ctx.closePath();
        ctx.fill();

        // yard ground
        const groundGrad = ctx.createLinearGradient(0, HORIZON_Y+40, 0, H+20);
        groundGrad.addColorStop(0, '#6d8a5a');
        groundGrad.addColorStop(0.3, '#7a9a65');
        groundGrad.addColorStop(1, '#5a7a4a');
        ctx.fillStyle = groundGrad;
        ctx.fillRect(0, HORIZON_Y+40, W, H - HORIZON_Y - 40);

        // grass texture (deterministic)
        const rng = mulberry32(42);
        ctx.fillStyle = 'rgba(60, 100, 50, 0.25)';
        for (let i=0; i<200; i++) {
          const gx = rng() * W;
          const gy = HORIZON_Y + 60 + rng() * (H - HORIZON_Y - 80);
          const gw = 2 + rng()*3;
          const gh = 3 + rng()*6;
          ctx.fillRect(gx, gy, gw, gh);
        }
      }

      function drawPuddle(){
        const {cx, cy, rx, ry} = PUDDLE;
        // puddle bed
        ctx.save();
        ctx.beginPath();
        ctx.ellipse(cx, cy, rx, ry, 0, 0, Math.PI*2);
        ctx.fillStyle = '#3a4a5a';
        ctx.fill();
        // water surface with slight gradient
        const wg = ctx.createRadialGradient(cx-10, cy-5, 5, cx, cy, rx);
        wg.addColorStop(0, 'rgba(140, 180, 200, 0.85)');
        wg.addColorStop(0.6, 'rgba(80, 130, 160, 0.75)');
        wg.addColorStop(1, 'rgba(50, 90, 120, 0.9)');
        ctx.beginPath();
        ctx.ellipse(cx, cy, rx, ry, 0, 0, Math.PI*2);
        ctx.fillStyle = wg;
        ctx.fill();
        // rim highlight
        ctx.strokeStyle = 'rgba(200, 220, 240, 0.3)';
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        ctx.ellipse(cx, cy, rx*0.95, ry*0.9, 0, 0, Math.PI*2);
        ctx.stroke();
        ctx.restore();
      }

      function drawPuddleReflection(t, dogState){
        const {cx, cy, rx, ry} = PUDDLE;
        // Only if dog is near puddle and above it
        const dx = dogState.x - cx;
        const dy = dogState.y - cy;
        const dist = Math.sqrt(dx*dx/(rx*rx) + dy*dy/(ry*ry));
        if (dist > 2.2 || dogState.y < cy) return;

        // Simple masked reflection: draw a vertically flipped, faded silhouette
        ctx.save();
        ctx.beginPath();
        ctx.ellipse(cx, cy, rx-2, ry-2, 0, 0, Math.PI*2);
        ctx.clip();

        // Draw a simplified reflected dog (mirrored below puddle line)
        const reflectY = cy + (cy - dogState.y) * 0.6;
        ctx.globalAlpha = 0.35;
        ctx.translate(dogState.x, reflectY);
        ctx.scale(1, -0.7);
        ctx.rotate(-dogState.bodyAngle);

        // body mass
        ctx.fillStyle = '#b86a2a';
        ctx.beginPath();
        ctx.ellipse(0, 0, 32, 18, 0, 0, Math.PI*2);
        ctx.fill();
        // head
        ctx.fillStyle = '#c07a30';
        ctx.beginPath();
        ctx.ellipse(28, -6, 16, 12, 0.2, 0, Math.PI*2);
        ctx.fill();
        // tail
        ctx.fillStyle = '#d08030';
        ctx.beginPath();
        ctx.ellipse(-28, -12, 14, 8, -0.4, 0, Math.PI*2);
        ctx.fill();

        ctx.restore();
      }

      function drawDoghouse(){
        const {x, y, w, h, doorW, doorH, doorX} = DOGHOUSE;
        // shadow
        ctx.fillStyle = 'rgba(0,0,0,0.2)';
        ctx.beginPath();
        ctx.ellipse(x+w/2, y+h+6, w*0.6, 12, 0, 0, Math.PI*2);
        ctx.fill();

        // back wall (darker interior)
        ctx.fillStyle = '#3a2a1a';
        ctx.fillRect(x+8, y+8, w-16, h-8);

        // interior darkness
        ctx.fillStyle = '#1a1008';
        ctx.fillRect(x+12, y+12, w-24, h-16);

        // walls
        ctx.fillStyle = '#8a6a4a';
        ctx.fillRect(x, y, w, h);

        // wall texture (wood planks)
        ctx.strokeStyle = '#6a4a2a';
        ctx.lineWidth = 1.5;
        for (let i=0; i<4; i++) {
          const px = x + 15 + i * 42;
          ctx.beginPath();
          ctx.moveTo(px, y);
          ctx.lineTo(px, y+h);
          ctx.stroke();
        }
        // horizontal grain
        ctx.strokeStyle = '#7a5a3a';
        ctx.lineWidth = 1;
        for (let i=0; i<5; i++) {
          const py = y + 20 + i * 26;
          ctx.beginPath();
          ctx.moveTo(x, py);
          ctx.lineTo(x+w, py);
          ctx.stroke();
        }

        // roof (triangular)
        ctx.fillStyle = '#6a4a2a';
        ctx.beginPath();
        ctx.moveTo(x-12, y);
        ctx.lineTo(x+w/2, y-48);
        ctx.lineTo(x+w+12, y);
        ctx.closePath();
        ctx.fill();

        // roof texture
        ctx.strokeStyle = '#4a3018';
        ctx.lineWidth = 1.2;
        for (let i=0; i<6; i++) {
          const frac = i/6;
          const rx1 = lerp(x-12, x+w/2, frac);
          const ry1 = lerp(y, y-48, frac);
          const rx2 = lerp(x+w/2, x+w+12, frac);
          const ry2 = lerp(y-48, y, frac);
          ctx.beginPath();
          ctx.moveTo(rx1, ry1);
          ctx.lineTo(rx2, ry2);
          ctx.stroke();
        }

        // door opening
        ctx.fillStyle = '#1a1008';
        ctx.beginPath();
        ctx.moveTo(doorX, y+h);
        ctx.lineTo(doorX, y+h-doorH);
        ctx.quadraticCurveTo(doorX+doorW/2, y+h-doorH-14, doorX+doorW, y+h-doorH);
        ctx.lineTo(doorX+doorW, y+h);
        ctx.closePath();
        ctx.fill();

        // door frame
        ctx.strokeStyle = '#5a3a1a';
        ctx.lineWidth = 4;
        ctx.beginPath();
        ctx.moveTo(doorX, y+h);
        ctx.lineTo(doorX, y+h-doorH);
        ctx.quadraticCurveTo(doorX+doorW/2, y+h-doorH-14, doorX+doorW, y+h-doorH);
        ctx.lineTo(doorX+doorW, y+h);
        ctx.stroke();

        // interior floor
        ctx.fillStyle = '#4a3520';
        ctx.fillRect(doorX+6, y+h-6, doorW-12, 6);
      }

      function drawBoneInHouse(){
        // bone resting inside on the floor, partially visible through doorway
        ctx.save();
        ctx.translate(BONE.x, BONE.y);
        ctx.rotate(BONE.angle);
        drawBoneShape(0, 0, BONE.w, BONE.h);
        ctx.restore();
      }

      function drawBoneShape(x, y, w, h){
        // bone: two knobs at each end, shaft in middle
        const knobR = h * 1.3;
        ctx.fillStyle = '#e8d8b0';
        ctx.strokeStyle = '#b8a070';
        ctx.lineWidth = 0.8;

        // shaft
        ctx.beginPath();
        ctx.ellipse(x, y, w*0.35, h*0.45, 0, 0, Math.PI*2);
        ctx.fill();
        ctx.stroke();

        // left knobs
        ctx.beginPath();
        ctx.arc(x - w*0.32, y - h*0.4, knobR*0.7, 0, Math.PI*2);
        ctx.fill();
        ctx.stroke();
        ctx.beginPath();
        ctx.arc(x - w*0.32, y + h*0.4, knobR*0.7, 0, Math.PI*2);
        ctx.fill();
        ctx.stroke();

        // right knobs
        ctx.beginPath();
        ctx.arc(x + w*0.32, y - h*0.4, knobR*0.7, 0, Math.PI*2);
        ctx.fill();
        ctx.stroke();
        ctx.beginPath();
        ctx.arc(x + w*0.32, y + h*0.4, knobR*0.7, 0, Math.PI*2);
        ctx.fill();
        ctx.stroke();
      }

      function drawDog(state, t){
        const {x, y, bodyAngle, headAngle, tailCurl, legPhase, mouthOpen, hasBone, boneAngle, bodyTilt} = state;

        ctx.save();
        ctx.translate(x, y);
        ctx.rotate(bodyAngle + bodyTilt);

        // ---- shadow on ground ----
        ctx.save();
        ctx.rotate(-(bodyAngle + bodyTilt));
        ctx.translate(-x, -y);
        const shadowAlpha = clamp(0.25 - (GROUND_Y - y) * 0.004, 0.08, 0.3);
        ctx.fillStyle = `rgba(20, 15, 10, ${shadowAlpha})`;
        ctx.beginPath();
        ctx.ellipse(x, GROUND_Y + 7, 42, 10, 0, 0, Math.PI*2);
        ctx.fill();
        ctx.restore();

        // ---- legs ----
        // hind legs
        drawLeg(-20, 4, legPhase + Math.PI, 'hind');
        drawLeg(-14, 4, legPhase + Math.PI*0.3, 'hind');
        // front legs
        drawLeg(18, 2, legPhase, 'front');
        drawLeg(24, 2, legPhase + Math.PI*0.5, 'front');

        // ---- body ----
        // main body (fur volume)
        const bodyGrad = ctx.createRadialGradient(-5, -8, 5, 0, 0, 40);
        bodyGrad.addColorStop(0, '#e8a050');
        bodyGrad.addColorStop(0.6, '#d08030');
        bodyGrad.addColorStop(1, '#b06828');
        ctx.fillStyle = bodyGrad;
        ctx.beginPath();
        ctx.ellipse(0, 0, 38, 22, 0, 0, Math.PI*2);
        ctx.fill();

        // chest / neck ruff (lighter)
        const chestGrad = ctx.createRadialGradient(18, -6, 4, 20, -4, 20);
        chestGrad.addColorStop(0, '#f0d8a0');
        chestGrad.addColorStop(0.7, '#e0b870');
        chestGrad.addColorStop(1, '#c89848');
        ctx.fillStyle = chestGrad;
        ctx.beginPath();
        ctx.ellipse(22, -4, 16, 18, 0.15, 0, Math.PI*2);
        ctx.fill();

        // back fur ridge
        ctx.fillStyle = '#c87830';
        ctx.beginPath();
        ctx.ellipse(-12, -10, 24, 12, -0.2, 0, Math.PI*2);
        ctx.fill();

        // ---- tail (curled over back) ----
        ctx.save();
        ctx.translate(-32, -12);
        ctx.rotate(-0.6 + tailCurl * 0.3);
        const tailGrad = ctx.createLinearGradient(0, 0, -18, -8);
        tailGrad.addColorStop(0, '#d08030');
        tailGrad.addColorStop(0.5, '#e8a050');
        tailGrad.addColorStop(1, '#f0c080');
        ctx.fillStyle = tailGrad;
        ctx.beginPath();
        ctx.ellipse(-12, -6, 16, 9, -0.3, 0, Math.PI*2);
        ctx.fill();
        // tail plume
        ctx.fillStyle = '#f0c888';
        ctx.beginPath();
        ctx.ellipse(-18, -12, 12, 7, -0.5, 0, Math.PI*2);
        ctx.fill();
        ctx.restore();

        // ---- head ----
        ctx.save();
        ctx.translate(28, -14);
        ctx.rotate(headAngle * 0.7);

        // head base
        const headGrad = ctx.createRadialGradient(2, -4, 3, 0, 0, 18);
        headGrad.addColorStop(0, '#f0c080');
        headGrad.addColorStop(0.7, '#e0a050');
        headGrad.addColorStop(1, '#c88838');
        ctx.fillStyle = headGrad;
        ctx.beginPath();
        ctx.ellipse(2, -2, 15, 14, 0.1, 0, Math.PI*2);
        ctx.fill();

        // muzzle (short, pointed)
        ctx.fillStyle = '#e8b060';
        ctx.beginPath();
        ctx.ellipse(14, 0, 9, 7, 0.1, 0, Math.PI*2);
        ctx.fill();

        // nose
        ctx.fillStyle = '#2a1a0a';
        ctx.beginPath();
        ctx.ellipse(21, -1, 3, 2.5, 0.1, 0, Math.PI*2);
        ctx.fill();

        // eyes (dark, expressive)
        ctx.fillStyle = '#1a1008';
        ctx.beginPath();
        ctx.arc(6, -7, 3.2, 0, Math.PI*2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(15, -8, 2.8, 0, Math.PI*2);
        ctx.fill();
        // eye highlights
        ctx.fillStyle = 'rgba(255,255,255,0.7)';
        ctx.beginPath();
        ctx.arc(5, -8, 1.2, 0, Math.PI*2);
        ctx.fill();
        ctx.beginPath();
        ctx.arc(14, -9, 1.1, 0, Math.PI*2);
        ctx.fill();

        // ears (small triangular, upright)
        ctx.fillStyle = '#d09040';
        // left ear
        ctx.beginPath();
        ctx.moveTo(-2, -12);
        ctx.lineTo(-6, -26);
        ctx.lineTo(6, -18);
        ctx.closePath();
        ctx.fill();
        // right ear
        ctx.beginPath();
        ctx.moveTo(10, -14);
        ctx.lineTo(12, -28);
        ctx.lineTo(20, -18);
        ctx.closePath();
        ctx.fill();

        // ear inner
        ctx.fillStyle = '#e8b080';
        ctx.beginPath();
        ctx.moveTo(0, -14);
        ctx.lineTo(-4, -23);
        ctx.lineTo(4, -17);
        ctx.closePath();
        ctx.fill();
        ctx.beginPath();
        ctx.moveTo(12, -16);
        ctx.lineTo(13, -25);
        ctx.lineTo(18, -18);
        ctx.closePath();
        ctx.fill();

        // mouth (visible when grabbing bone)
        if (mouthOpen > 0.1) {
          ctx.fillStyle = '#4a2010';
          ctx.beginPath();
          ctx.ellipse(16, 2, 6, 3 * mouthOpen, 0.1, 0, Math.PI*2);
          ctx.fill();
          // tongue hint
          if (mouthOpen > 0.4) {
            ctx.fillStyle = '#d06060';
            ctx.beginPath();
            ctx.ellipse(17, 3, 3, 1.8 * (mouthOpen-0.4)/0.6, 0.1, 0, Math.PI*2);
            ctx.fill();
          }
        }

        ctx.restore(); // head

        // ---- bone in mouth ----
        if (hasBone) {
          ctx.save();
          ctx.translate(40, -8);
          ctx.rotate(boneAngle + bodyAngle*0.3);
          drawBoneShape(0, 0, 28, 9);
          ctx.restore();
        }

        ctx.restore(); // body
      }

      function drawLeg(ox, oy, phase, type){
        // ox, oy = offset from body center
        const angle = Math.sin(phase) * 0.35;
        const lift = Math.max(0, Math.sin(phase)) * 5;
        const kneeBend = 0.3 + 0.3 * Math.abs(Math.sin(phase));

        ctx.save();
        ctx.translate(ox, oy);
        // upper leg
        ctx.save();
        ctx.rotate(angle);
        ctx.fillStyle = '#c88030';
        ctx.beginPath();
        ctx.ellipse(0, 10, 5, 12, 0, 0, Math.PI*2);
        ctx.fill();
        // lower leg
        ctx.translate(0, 18 - lift * 0.3);
        ctx.save();
        ctx.rotate(-angle * 1.2 + (type==='front'?0.1:-0.1));
        ctx.fillStyle = '#d08838';
        ctx.beginPath();
        ctx.ellipse(0, 8, 4.2, 10, 0, 0, Math.PI*2);
        ctx.fill();
        // paw
        ctx.fillStyle = '#e0a050';
        ctx.beginPath();
        ctx.ellipse(2, 16, 5.5, 3.5, 0.1, 0, Math.PI*2);
        ctx.fill();
        // paw pads
        ctx.fillStyle = '#a86828';
        ctx.beginPath();
        ctx.arc(-1, 16, 1.5, 0, Math.PI*2);
        ctx.arc(2, 17, 1.5, 0, Math.PI*2);
        ctx.arc(5, 16, 1.5, 0, Math.PI*2);
        ctx.fill();
        ctx.restore();
        ctx.restore();
        ctx.restore();
      }

      // ---------- main render ----------
      function renderFrame(timeSeconds){
        // clamp time: t<0 opening, t>=12 hold final
        let t = timeSeconds;
        if (t < 0) t = 0;
        if (t > 12) t = 12;

        // clear
        ctx.clearRect(0, 0, W, H);

        // background
        drawBackground();

        // puddle (with reflection)
        const dogState = getDogState(t);
        drawPuddle();
        drawPuddleReflection(t, dogState);

        // doghouse (with bone inside)
        drawDoghouse();
        drawBoneInHouse();

        // dog (on top, but behind doghouse walls when entering)
        // For correct occlusion: if dog is inside house (x > 700 and t > 3.5), draw dog first then house overlay
        if (dogState.x > 700 && t > 3.5 && t < 8.5) {
          // dog is inside — draw dog first, then re-draw front wall
          drawDog(dogState, t);
          // redraw front wall and roof over dog's rear
          const {x, y, w, h, doorW, doorH, doorX} = DOGHOUSE;
          ctx.fillStyle = '#8a6a4a';
          ctx.fillRect(x, y, w, h);
          ctx.strokeStyle = '#6a4a2a';
          ctx.lineWidth = 1.5;
          for (let i=0; i<4; i++) {
            const px = x + 15 + i * 42;
            ctx.beginPath();
            ctx.moveTo(px, y);
            ctx.lineTo(px, y+h);
            ctx.stroke();
          }
          ctx.strokeStyle = '#7a5a3a';
          ctx.lineWidth = 1;
          for (let i=0; i<5; i++) {
            const py = y + 20 + i * 26;
            ctx.beginPath();
            ctx.moveTo(x, py);
            ctx.lineTo(x+w, py);
            ctx.stroke();
          }
          // roof
          ctx.fillStyle = '#6a4a2a';
          ctx.beginPath();
          ctx.moveTo(x-12, y);
          ctx.lineTo(x+w/2, y-48);
          ctx.lineTo(x+w+12, y);
          ctx.closePath();
          ctx.fill();
          // redraw door opening
          ctx.fillStyle = '#1a1008';
          ctx.beginPath();
          ctx.moveTo(doorX, y+h);
          ctx.lineTo(doorX, y+h-doorH);
          ctx.quadraticCurveTo(doorX+doorW/2, y+h-doorH-14, doorX+doorW, y+h-doorH);
          ctx.lineTo(doorX+doorW, y+h);
          ctx.closePath();
          ctx.fill();
          ctx.strokeStyle = '#5a3a1a';
          ctx.lineWidth = 4;
          ctx.beginPath();
          ctx.moveTo(doorX, y+h);
          ctx.lineTo(doorX, y+h-doorH);
          ctx.quadraticCurveTo(doorX+doorW/2, y+h-doorH-14, doorX+doorW, y+h-doorH);
          ctx.lineTo(doorX+doorW, y+h);
          ctx.stroke();
        } else {
          drawDog(dogState, t);
        }

        // foreground grass tufts
        const rng = mulberry32(123);
        ctx.fillStyle = 'rgba(80, 120, 60, 0.4)';
        for (let i=0; i<40; i++) {
          const fx = rng() * W;
          const fy = GROUND_Y + 20 + rng() * 80;
          const fw = 3 + rng()*5;
          const fh = 5 + rng()*10;
          ctx.beginPath();
          ctx.moveTo(fx, fy);
          ctx.lineTo(fx + fw*0.5, fy - fh);
          ctx.lineTo(fx + fw, fy);
          ctx.fill();
        }

        // subtle vignette
        const vg = ctx.createRadialGradient(W/2, H/2, H*0.4, W/2, H/2, H*0.9);
        vg.addColorStop(0, 'rgba(0,0,0,0)');
        vg.addColorStop(1, 'rgba(0,0,0,0.25)');
        ctx.fillStyle = vg;
        ctx.fillRect(0, 0, W, H);
      }

      // expose
      window.renderFrame = renderFrame;

      // initial draw
      renderFrame(0);
    })();
  </script>

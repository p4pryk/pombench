```html
<!DOCTYPE html>

    <canvas id="stage" width="960" height="540"></canvas>
    <script>
        const canvas = document.getElementById('stage');
        const ctx = canvas.getContext('2d');

        // Scene setup
        const groundColor = '#8bc34a';
        const skyColor = '#87ceeb';
        const doghouseColor = '#8d6e63';
        const boneColor = '#f5f5f5';
        const puddleColor = '#4fc3f7';
        const shadowColor = 'rgba(0,0,0,0.2)';

        // Dog properties
        const dog = {
            color: '#ff9800',
            furColor: '#ffc107',
            eyeColor: '#333',
            noseColor: '#000',
            size: 40,
            tailAngle: 0,
            earAngle: 0,
            headAngle: 0,
            legAngles: [0, 0, 0, 0],
            spineAngle: 0,
            position: { x: 100, y: 400 },
            velocity: { x: 0, y: 0 },
            state: 'idle',
            boneHeld: false,
            jumpProgress: 0,
            turnProgress: 0,
            breathProgress: 0
        };

        // Bone properties
        const bone = {
            position: { x: 700, y: 420 },
            size: 20,
            held: false,
            heldOffset: { x: 0, y: 0 }
        };

        // Puddle properties
        const puddle = {
            position: { x: 400, y: 430 },
            width: 120,
            height: 20
        };

        // Doghouse properties
        const doghouse = {
            position: { x: 600, y: 350 },
            width: 150,
            height: 120,
            depth: 80
        };

        // Animation states
        const states = {
            IDLE: 0,
            RUNNING: 1,
            JUMPING: 2,
            ENTERING: 3,
            GRABBING: 4,
            BACKING: 5,
            TURNING: 6,
            FINAL: 7
        };

        let currentState = states.IDLE;
        let stateProgress = 0;

        // Draw ground
        function drawGround() {
            ctx.fillStyle = groundColor;
            ctx.fillRect(0, 400, canvas.width, canvas.height - 400);

            // Draw puddle
            ctx.fillStyle = puddleColor;
            ctx.beginPath();
            ctx.ellipse(puddle.position.x, puddle.position.y, puddle.width/2, puddle.height/2, 0, 0, Math.PI * 2);
            ctx.fill();

            // Draw puddle edge
            ctx.strokeStyle = '#3e8e41';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.ellipse(puddle.position.x, puddle.position.y, puddle.width/2, puddle.height/2, 0, 0, Math.PI * 2);
            ctx.stroke();
        }

        // Draw sky
        function drawSky() {
            const gradient = ctx.createLinearGradient(0, 0, 0, 400);
            gradient.addColorStop(0, skyColor);
            gradient.addColorStop(1, '#e0f7fa');
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, canvas.width, 400);
        }

        // Draw doghouse
        function drawDoghouse() {
            // Base
            ctx.fillStyle = doghouseColor;
            ctx.fillRect(doghouse.position.x, doghouse.position.y, doghouse.width, doghouse.height);

            // Roof
            ctx.beginPath();
            ctx.moveTo(doghouse.position.x - 20, doghouse.position.y);
            ctx.lineTo(doghouse.position.x + doghouse.width/2, doghouse.position.y - 40);
            ctx.lineTo(doghouse.position.x + doghouse.width + 20, doghouse.position.y);
            ctx.lineTo(doghouse.position.x + doghouse.width, doghouse.position.y);
            ctx.lineTo(doghouse.position.x, doghouse.position.y);
            ctx.closePath();
            ctx.fill();

            // Doorway
            ctx.fillStyle = '#5d4037';
            ctx.fillRect(doghouse.position.x + 20, doghouse.position.y + 20, 60, 80);

            // Depth
            ctx.fillStyle = '#6d4c41';
            ctx.beginPath();
            ctx.moveTo(doghouse.position.x + doghouse.width, doghouse.position.y);
            ctx.lineTo(doghouse.position.x + doghouse.width + doghouse.depth, doghouse.position.y - doghouse.depth);
            ctx.lineTo(doghouse.position.x + doghouse.width + doghouse.depth, doghouse.position.y - doghouse.depth + doghouse.height);
            ctx.lineTo(doghouse.position.x + doghouse.width, doghouse.position.y + doghouse.height);
            ctx.closePath();
            ctx.fill();

            // Roof depth
            ctx.beginPath();
            ctx.moveTo(doghouse.position.x + doghouse.width + 20, doghouse.position.y);
            ctx.lineTo(doghouse.position.x + doghouse.width + doghouse.depth + 20, doghouse.position.y - doghouse.depth);
            ctx.lineTo(doghouse.position.x + doghouse.width/2 + doghouse.depth, doghouse.position.y - 40 - doghouse.depth);
            ctx.lineTo(doghouse.position.x + doghouse.width/2, doghouse.position.y - 40);
            ctx.closePath();
            ctx.fill();
        }

        // Draw bone
        function drawBone() {
            if (bone.held) return;

            ctx.fillStyle = boneColor;
            ctx.beginPath();
            ctx.ellipse(bone.position.x, bone.position.y, bone.size/2, bone.size/4, 0, 0, Math.PI * 2);
            ctx.fill();

            ctx.beginPath();
            ctx.ellipse(bone.position.x - bone.size, bone.position.y, bone.size/2, bone.size/4, 0, 0, Math.PI * 2);
            ctx.fill();

            ctx.fillRect(bone.position.x - bone.size, bone.position.y - bone.size/8, bone.size, bone.size/4);
        }

        // Draw dog
        function drawDog() {
            const pos = dog.position;
            const size = dog.size;

            // Save context for transformations
            ctx.save();

            // Apply body rotation for turning
            if (currentState === states.TURNING || currentState === states.FINAL) {
                const turnAngle = dog.turnProgress * Math.PI / 2;
                ctx.translate(pos.x, pos.y);
                ctx.rotate(turnAngle);
                ctx.translate(-pos.x, -pos.y);
            }

            // Body
            ctx.fillStyle = dog.color;
            ctx.beginPath();
            ctx.ellipse(pos.x, pos.y, size, size/1.5, 0, 0, Math.PI * 2);
            ctx.fill();

            // Fur texture
            ctx.fillStyle = dog.furColor;
            for (let i = 0; i < 20; i++) {
                const angle = Math.random() * Math.PI * 2;
                const distance = Math.random() * size;
                const x = pos.x + Math.cos(angle) * distance;
                const y = pos.y + Math.sin(angle) * distance * 0.7;
                const furSize = 2 + Math.random() * 3;
                ctx.beginPath();
                ctx.ellipse(x, y, furSize, furSize/2, angle, 0, Math.PI * 2);
                ctx.fill();
            }

            // Head
            ctx.fillStyle = dog.color;
            ctx.beginPath();
            ctx.ellipse(pos.x + size * 0.8, pos.y - size * 0.3, size * 0.6, size * 0.5, 0, 0, Math.PI * 2);
            ctx.fill();

            // Muzzle
            ctx.beginPath();
            ctx.ellipse(pos.x + size * 1.2, pos.y - size * 0.2, size * 0.3, size * 0.2, 0, 0, Math.PI * 2);
            ctx.fill();

            // Ears
            ctx.fillStyle = dog.color;
            ctx.beginPath();
            ctx.ellipse(pos.x + size * 0.6, pos.y - size * 0.7, size * 0.2, size * 0.3, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.ellipse(pos.x + size * 1.0, pos.y - size * 0.7, size * 0.2, size * 0.3, 0, 0, Math.PI * 2);
            ctx.fill();

            // Eyes
            ctx.fillStyle = dog.eyeColor;
            ctx.beginPath();
            ctx.ellipse(pos.x + size * 1.0, pos.y - size * 0.35, size * 0.1, size * 0.15, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.ellipse(pos.x + size * 1.1, pos.y - size * 0.35, size * 0.1, size * 0.15, 0, 0, Math.PI * 2);
            ctx.fill();

            // Nose
            ctx.fillStyle = dog.noseColor;
            ctx.beginPath();
            ctx.ellipse(pos.x + size * 1.3, pos.y - size * 0.2, size * 0.1, size * 0.08, 0, 0, Math.PI * 2);
            ctx.fill();

            // Legs
            ctx.fillStyle = dog.color;
            const legPositions = [
                { x: pos.x - size * 0.4, y: pos.y + size * 0.3 },
                { x: pos.x - size * 0.1, y: pos.y + size * 0.3 },
                { x: pos.x + size * 0.2, y: pos.y + size * 0.3 },
                { x: pos.x + size * 0.5, y: pos.y + size * 0.3 }
            ];

            for (let i = 0; i < 4; i++) {
                const leg = legPositions[i];
                const legAngle = dog.legAngles[i];

                ctx.save();
                ctx.translate(leg.x, leg.y);
                ctx.rotate(legAngle);

                // Upper leg
                ctx.beginPath();
                ctx.ellipse(0, 0, size * 0.15, size * 0.3, 0, 0, Math.PI * 2);
                ctx.fill();

                // Lower leg
                ctx.beginPath();
                ctx.ellipse(0, size * 0.3, size * 0.1, size * 0.2, 0, 0, Math.PI * 2);
                ctx.fill();

                // Paw
                ctx.beginPath();
                ctx.ellipse(0, size * 0.5, size * 0.12, size * 0.08, 0, 0, Math.PI * 2);
                ctx.fill();

                ctx.restore();
            }

            // Tail
            ctx.fillStyle = dog.color;
            ctx.beginPath();
            ctx.ellipse(pos.x - size * 0.8, pos.y - size * 0.2, size * 0.3, size * 0.5, dog.tailAngle, 0, Math.PI * 2);
            ctx.fill();

            // Fur on tail
            ctx.fillStyle = dog.furColor;
            for (let i = 0; i < 10; i++) {
                const angle = Math.random() * Math.PI * 2;
                const distance = Math.random() * size * 0.3;
                const x = pos.x - size * 0.8 + Math.cos(angle) * distance;
                const y = pos.y - size * 0.2 + Math.sin(angle) * distance;
                const furSize = 1 + Math.random() * 2;
                ctx.beginPath();
                ctx.ellipse(x, y, furSize, furSize/2, angle, 0, Math.PI * 2);
                ctx.fill();
            }

            // Bone if held
            if (dog.boneHeld) {
                ctx.fillStyle = boneColor;
                ctx.save();
                ctx.translate(pos.x + size * 0.8, pos.y - size * 0.2);
                ctx.rotate(-Math.PI/4);

                ctx.beginPath();
                ctx.ellipse(0, 0, bone.size/2, bone.size/4, 0, 0, Math.PI * 2);
                ctx.fill();

                ctx.beginPath();
                ctx.ellipse(-bone.size, 0, bone.size/2, bone.size/4, 0, 0, Math.PI * 2);
                ctx.fill();

                ctx.fillRect(-bone.size, -bone.size/8, bone.size, bone.size/4);

                ctx.restore();
            }

            ctx.restore();
        }

        // Draw puddle reflection
        function drawPuddleReflection() {
            if (dog.position.x + dog.size < puddle.position.x - puddle.width/2 ||
                dog.position.x - dog.size > puddle.position.x + puddle.width/2) {
                return;
            }

            ctx.save();
            ctx.globalAlpha = 0.5;

            // Clip to puddle shape
            ctx.beginPath();
            ctx.ellipse(puddle.position.x, puddle.position.y, puddle.width/2, puddle.height/2, 0, 0, Math.PI * 2);
            ctx.clip();

            // Draw reflected dog
            ctx.scale(1, -1);
            ctx.translate(0, -2 * puddle.position.y);

            // Adjust position for reflection
            const reflectedPos = {
                x: dog.position.x,
                y: puddle.position.y - (dog.position.y - puddle.position.y)
            };

            // Draw dog with slight distortion
            ctx.save();
            ctx.translate(reflectedPos.x, reflectedPos.y);
            ctx.scale(1, 0.8);
            ctx.translate(-reflectedPos.x, -reflectedPos.y);

            // Simplified dog reflection
            ctx.fillStyle = dog.color;
            ctx.beginPath();
            ctx.ellipse(reflectedPos.x, reflectedPos.y, dog.size, dog.size/1.5, 0, 0, Math.PI * 2);
            ctx.fill();

            // Head
            ctx.beginPath();
            ctx.ellipse(reflectedPos.x + dog.size * 0.8, reflectedPos.y - dog.size * 0.3, dog.size * 0.6, dog.size * 0.5, 0, 0, Math.PI * 2);
            ctx.fill();

            ctx.restore();

            ctx.restore();
        }

        // Draw shadows
        function drawShadows() {
            // Dog shadow
            ctx.fillStyle = shadowColor;
            ctx.beginPath();
            ctx.ellipse(dog.position.x, 430, dog.size * 0.8, dog.size * 0.3, 0, 0, Math.PI * 2);
            ctx.fill();

            // Bone shadow
            if (!bone.held) {
                ctx.beginPath();
                ctx.ellipse(bone.position.x, 430, bone.size * 0.5, bone.size * 0.2, 0, 0, Math.PI * 2);
                ctx.fill();
            }

            // Doghouse shadow
            ctx.beginPath();
            ctx.moveTo(doghouse.position.x, doghouse.position.y + doghouse.height);
            ctx.lineTo(doghouse.position.x + doghouse.width, doghouse.position.y + doghouse.height);
            ctx.lineTo(doghouse.position.x + doghouse.width + doghouse.depth, doghouse.position.y + doghouse.height - doghouse.depth);
            ctx.lineTo(doghouse.position.x + doghouse.depth, doghouse.position.y + doghouse.height - doghouse.depth);
            ctx.closePath();
            ctx.fill();
        }

        // Update dog state
        function updateDog(time) {
            const t = Math.min(time, 12);

            // State machine
            if (t < 1) {
                currentState = states.IDLE;
                stateProgress = t;
            } else if (t < 3) {
                currentState = states.RUNNING;
                stateProgress = (t - 1) / 2;
            } else if (t < 4) {
                currentState = states.JUMPING;
                stateProgress = (t - 3);
            } else if (t < 4.5) {
                currentState = states.ENTERING;
                stateProgress = (t - 4) / 0.5;
            } else if (t < 6) {
                currentState = states.GRABBING;
                stateProgress = (t - 4.5) / 1.5;
            } else if (t < 8) {
                currentState = states.BACKING;
                stateProgress = (t - 6) / 2;
            } else if (t < 10) {
                currentState = states.TURNING;
                stateProgress = (t - 8) / 2;
            } else {
                currentState = states.FINAL;
                stateProgress = 1;
            }

            // Update dog based on state
            switch (currentState) {
                case states.IDLE:
                    dog.position.x = 100;
                    dog.position.y = 400;
                    dog.velocity.x = 0;
                    dog.velocity.y = 0;
                    dog.legAngles = [0, 0, 0, 0];
                    dog.tailAngle = Math.sin(t * 2) * 0.1;
                    dog.earAngle = Math.sin(t * 3) * 0.05;
                    dog.boneHeld = false;
                    break;

                case states.RUNNING:
                    dog.velocity.x = 8;
                    dog.position.x += dog.velocity.x;
                    dog.position.y = 400;

                    // Leg animation
                    const runCycle = stateProgress * 4;
                    dog.legAngles[0] = Math.sin(runCycle) * 0.5;
                    dog.legAngles[1] = Math.sin(runCycle + Math.PI) * 0.5;
                    dog.legAngles[2] = Math.sin(runCycle + Math.PI/2) * 0.5;
                    dog.legAngles[3] = Math.sin(runCycle + Math.PI * 1.5) * 0.5;

                    // Tail wag
                    dog.tailAngle = Math.sin(runCycle * 2) * 0.3;
                    dog.earAngle = Math.sin(runCycle * 3) * 0.1;
                    dog.boneHeld = false;
                    break;

                case states.JUMPING:
                    if (stateProgress < 0.3) {
                        // Anticipation
                        dog.position.x += 5;
                        dog.position.y = 400;
                        dog.legAngles = [0.3, 0.3, -0.3, -0.3];
                    } else if (stateProgress < 0.7) {
                        // Jumping
                        const jumpProgress = (stateProgress - 0.3) / 0.4;
                        dog.position.x += 10;
                        dog.position.y = 400 - Math.sin(jumpProgress * Math.PI) * 60;
                        dog.legAngles = [0.8, 0.8, 0.8, 0.8];
                    } else {
                        // Landing
                        const landProgress = (stateProgress - 0.7) / 0.3;
                        dog.position.x += 8;
                        dog.position.y = 400 - Math.sin((1 - landProgress) * Math.PI) * 20;
                        dog.legAngles = [-0.3, -0.3, 0.3, 0.3];
                    }
                    dog.tailAngle = 0.5;
                    dog.earAngle = 0.1;
                    break;

                case states.ENTERING:
                    dog.position.x = 550 + (600 - 550) * (1 - stateProgress);
                    dog.position.y = 400;
                    dog.legAngles = [0, 0, 0, 0];
                    dog.tailAngle = 0.2;
                    break;

                case states.GRABBING:
                    dog.position.x = 600;
                    dog.position.y = 400;
                    dog.legAngles = [0, 0, 0, 0];
                    dog.tailAngle = Math.sin(stateProgress * 5) * 0.3;
                    dog.boneHeld = stateProgress > 0.5;
                    break;

                case states.BACKING:
                    dog.position.x = 600 - 100 * stateProgress;
                    dog.position.y = 400;
                    dog.legAngles = [0, 0, 0, 0];
                    dog.tailAngle = 0.2;
                    dog.boneHeld = true;
                    break;

                case states.TURNING:
                    dog.position.x = 500;
                    dog.position.y = 400;
                    dog.turnProgress = stateProgress;
                    dog.legAngles = [0, 0, 0, 0];
                    dog.tailAngle = 0.3;
                    dog.boneHeld = true;
                    break;

                case states.FINAL:
                    dog.position.x = 500;
                    dog.position.y = 400;
                    dog.turnProgress = 1;
                    dog.legAngles = [0, 0, 0, 0];
                    dog.tailAngle = Math.sin(t * 2) * 0.1;
                    dog.earAngle = Math.sin(t * 3) * 0.05;
                    dog.boneHeld = true;
                    break;
            }

            // Breathing animation
            dog.breathProgress = Math.sin(t * 2) * 0.02;
            dog.position.y += dog.breathProgress * 5;
        }

        // Main render function
        window.renderFrame = function(timeSeconds) {
            // Clear canvas
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Draw scene
            drawSky();
            drawGround();
            drawDoghouse();
            drawBone();
            drawShadows();
            drawDog();
            drawPuddleReflection();

            // Update dog state
            updateDog(timeSeconds);
        };

        // Initial render
        renderFrame(0);
    </script>
</body>
</html>
```

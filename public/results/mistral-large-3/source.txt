```html
<!DOCTYPE html>
<html>

    <style>

<body>
    <canvas id="stage" width="960" height="540"></canvas>
    <script>
        const canvas = document.getElementById('stage');
        const ctx = canvas.getContext('2d');

        // Scene setup
        const yardColor = '#a8c686';
        const doghouseColor = '#8B4513';
        const doghouseShadowColor = '#654321';
        const boneColor = '#f5e7d3';
        const furColor = '#ff9966';
        const furShadowColor = '#e68a55';
        const eyeColor = '#333333';
        const noseColor = '#222222';

        // Dog parameters
        const dog = {
            size: 0.15,
            headSize: 0.05,
            bodyLength: 0.12,
            legLength: 0.08,
            tailLength: 0.06,
            earSize: 0.025,
            neckRuffSize: 0.04,
            position: { x: 0.2, y: 0.7, z: 0 },
            rotation: { x: 0, y: 0, z: 0 },
            headRotation: { x: 0, y: 0, z: 0 },
            legPhase: 0,
            tailWag: 0,
            breathing: 0,
            hasBone: false,
            bonePosition: { x: 0, y: 0, z: 0 },
            boneRotation: { x: 0, y: 0, z: 0 }
        };

        // Doghouse parameters
        const doghouse = {
            position: { x: 0.7, y: 0.65, z: 0 },
            width: 0.25,
            height: 0.2,
            depth: 0.2,
            entranceWidth: 0.12,
            entranceHeight: 0.15
        };

        // Bone parameters
        const bone = {
            position: { x: 0.75, y: 0.7, z: 0.05 },
            length: 0.08,
            width: 0.02,
            height: 0.015
        };

        // Camera parameters
        const camera = {
            position: { x: 0.5, y: 0.5, z: 1.5 },
            lookAt: { x: 0.5, y: 0.5, z: 0 },
            fov: Math.PI / 3
        };

        // Lighting
        const light = {
            position: { x: 0.3, y: 0.2, z: 1 },
            ambient: 0.3,
            diffuse: 0.7
        };

        // Animation state
        let animationState = {
            time: 0,
            running: false,
            entering: false,
            backingOut: false,
            turning: false,
            finalPose: false
        };

        // 3D projection
        function project(x, y, z) {
            const scale = canvas.width / 2;
            const aspect = canvas.width / canvas.height;

            // Simple perspective projection
            const distance = 2;
            const factor = distance / (distance + z);

            return {
                x: (x - camera.position.x) * factor * scale * aspect + canvas.width / 2,
                y: (y - camera.position.y) * factor * scale + canvas.height / 2,
                z: z
            };
        }

        // Draw yard
        function drawYard() {
            ctx.fillStyle = yardColor;
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Draw some grass texture
            ctx.fillStyle = 'rgba(0, 0, 0, 0.1)';
            for (let i = 0; i < 200; i++) {
                const x = Math.random() * canvas.width;
                const y = Math.random() * canvas.height;
                const size = Math.random() * 2 + 1;
                ctx.fillRect(x, y, size, size);
            }
        }

        // Draw doghouse
        function drawDoghouse() {
            const pos = project(doghouse.position.x, doghouse.position.y, doghouse.position.z);
            const width = doghouse.width * canvas.width;
            const height = doghouse.height * canvas.height;
            const depth = doghouse.depth * canvas.width;

            // Base
            ctx.fillStyle = doghouseColor;
            ctx.beginPath();
            ctx.moveTo(pos.x - width/2, pos.y);
            ctx.lineTo(pos.x - width/2, pos.y - height);
            ctx.lineTo(pos.x + width/2, pos.y - height);
            ctx.lineTo(pos.x + width/2, pos.y);
            ctx.closePath();
            ctx.fill();

            // Roof
            ctx.fillStyle = doghouseShadowColor;
            ctx.beginPath();
            ctx.moveTo(pos.x - width/2, pos.y - height);
            ctx.lineTo(pos.x, pos.y - height - depth/2);
            ctx.lineTo(pos.x + width/2, pos.y - height);
            ctx.closePath();
            ctx.fill();

            // Entrance
            ctx.fillStyle = '#5D4037';
            const entrancePos = project(
                doghouse.position.x - doghouse.width/2 + doghouse.entranceWidth/2,
                doghouse.position.y - doghouse.entranceHeight/2,
                doghouse.position.z
            );
            const entranceWidth = doghouse.entranceWidth * canvas.width;
            const entranceHeight = doghouse.entranceHeight * canvas.height;
            ctx.fillRect(
                entrancePos.x - entranceWidth/2,
                entrancePos.y - entranceHeight/2,
                entranceWidth,
                entranceHeight
            );

            // Shadow
            ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
            ctx.beginPath();
            ctx.moveTo(pos.x - width/2, pos.y);
            ctx.lineTo(pos.x - width/2 - depth/2, pos.y + depth/2);
            ctx.lineTo(pos.x + width/2 - depth/2, pos.y + depth/2);
            ctx.lineTo(pos.x + width/2, pos.y);
            ctx.closePath();
            ctx.fill();
        }

        // Draw bone
        function drawBone() {
            if (dog.hasBone) {
                // Bone is in dog's mouth
                return;
            }

            const pos = project(bone.position.x, bone.position.y, bone.position.z);
            const length = bone.length * canvas.width;
            const width = bone.width * canvas.width;
            const height = bone.height * canvas.height;

            ctx.fillStyle = boneColor;
            ctx.beginPath();
            // Main shaft
            ctx.moveTo(pos.x - length/2, pos.y);
            ctx.lineTo(pos.x + length/2, pos.y);
            ctx.lineTo(pos.x + length/2, pos.y - height);
            ctx.lineTo(pos.x - length/2, pos.y - height);
            ctx.closePath();
            ctx.fill();

            // Ends
            ctx.beginPath();
            ctx.ellipse(pos.x - length/2, pos.y - height/2, width, height/2, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.ellipse(pos.x + length/2, pos.y - height/2, width, height/2, 0, 0, Math.PI * 2);
            ctx.fill();

            // Shadow
            ctx.fillStyle = 'rgba(0, 0, 0, 0.2)';
            ctx.beginPath();
            ctx.ellipse(pos.x, pos.y + height/2, length/2, height/4, 0, 0, Math.PI * 2);
            ctx.fill();
        }

        // Draw dog
        function drawDog() {
            const time = animationState.time;
            const pos = project(dog.position.x, dog.position.y, dog.position.z);

            // Calculate animation parameters
            let legPhase = dog.legPhase;
            let tailWag = dog.tailWag;
            let breathing = dog.breathing;
            let headRotationY = dog.headRotation.y;
            let bodyRotationY = dog.rotation.y;

            if (time < 1) {
                // Initial pose
                legPhase = 0;
                tailWag = Math.sin(time * 2) * 0.1;
                breathing = Math.sin(time * 3) * 0.01;
            } else if (time < 3) {
                // Running to doghouse
                const runTime = (time - 1) * 2;
                legPhase = runTime % 1;
                tailWag = Math.sin(runTime * 10) * 0.2;
                breathing = Math.sin(runTime * 5) * 0.02;
                dog.position.x = 0.2 + (0.7 - 0.2) * (runTime / 2);
                if (dog.position.x > 0.65) dog.position.x = 0.65;
            } else if (time < 4.5) {
                // Entering doghouse
                const enterTime = (time - 3) * 2;
                legPhase = 0.5;
                tailWag = Math.sin(enterTime * 5) * 0.1;
                breathing = Math.sin(enterTime * 3) * 0.01;
                dog.position.x = 0.65 + (0.75 - 0.65) * (enterTime / 1.5);
                dog.position.z = 0 + 0.05 * (enterTime / 1.5);
                if (dog.position.x > 0.75) dog.position.x = 0.75;
            } else if (time < 6) {
                // Grabbing bone
                const grabTime = (time - 4.5) * 2;
                legPhase = 0.5;
                tailWag = 0;
                breathing = Math.sin(grabTime * 3) * 0.01;
                headRotationY = -0.3 * grabTime;
                if (grabTime > 0.5) {
                    dog.hasBone = true;
                    dog.bonePosition = {
                        x: dog.position.x + 0.03,
                        y: dog.position.y - 0.02,
                        z: dog.position.z + 0.02
                    };
                    dog.boneRotation = { x: 0, y: 0.5, z: 0 };
                }
            } else if (time < 8) {
                // Backing out
                const backTime = (time - 6) * 2;
                legPhase = 0.5 - backTime * 0.5;
                tailWag = Math.sin(backTime * 5) * 0.1;
                breathing = Math.sin(backTime * 3) * 0.01;
                dog.position.x = 0.75 - (0.75 - 0.65) * (backTime / 2);
                dog.position.z = 0.05 - 0.05 * (backTime / 2);
                if (dog.position.x < 0.65) dog.position.x = 0.65;
                if (dog.position.z < 0) dog.position.z = 0;
            } else if (time < 10) {
                // Turning to face camera
                const turnTime = (time - 8) * 2;
                legPhase = 0;
                tailWag = Math.sin(turnTime * 3) * 0.1;
                breathing = Math.sin(turnTime * 3) * 0.01;
                bodyRotationY = turnTime * Math.PI / 2;
                headRotationY = 0;
            } else {
                // Final pose
                legPhase = 0;
                tailWag = Math.sin(time * 2) * 0.05;
                breathing = Math.sin(time * 3) * 0.01;
                bodyRotationY = Math.PI / 2;
                headRotationY = 0;
            }

            // Save context
            ctx.save();
            ctx.translate(pos.x, pos.y);

            // Body rotation
            ctx.rotate(bodyRotationY);

            // Draw body
            ctx.fillStyle = furColor;
            const bodyWidth = dog.bodyLength * canvas.width * 0.8;
            const bodyHeight = dog.bodyLength * canvas.width * 0.5;

            // Main body
            ctx.beginPath();
            ctx.ellipse(0, 0, bodyWidth/2, bodyHeight/2, 0, 0, Math.PI * 2);
            ctx.fill();

            // Neck ruff
            ctx.fillStyle = furShadowColor;
            ctx.beginPath();
            ctx.ellipse(-bodyWidth/4, 0, dog.neckRuffSize * canvas.width, bodyHeight/3, 0, 0, Math.PI * 2);
            ctx.fill();

            // Head
            const headSize = dog.headSize * canvas.width;
            ctx.fillStyle = furColor;
            ctx.beginPath();
            ctx.ellipse(-bodyWidth/2 - headSize/2, 0, headSize/2, headSize/1.5, 0, 0, Math.PI * 2);
            ctx.fill();

            // Head rotation
            ctx.save();
            ctx.translate(-bodyWidth/2 - headSize/2, 0);
            ctx.rotate(headRotationY);

            // Muzzle
            ctx.fillStyle = furColor;
            ctx.beginPath();
            ctx.ellipse(-headSize/2, 0, headSize/3, headSize/4, 0, 0, Math.PI * 2);
            ctx.fill();

            // Nose
            ctx.fillStyle = noseColor;
            ctx.beginPath();
            ctx.ellipse(-headSize/2 - headSize/6, 0, headSize/8, headSize/6, 0, 0, Math.PI * 2);
            ctx.fill();

            // Eyes
            ctx.fillStyle = eyeColor;
            ctx.beginPath();
            ctx.ellipse(-headSize/2, -headSize/6, headSize/10, headSize/8, 0, 0, Math.PI * 2);
            ctx.fill();
            ctx.beginPath();
            ctx.ellipse(-headSize/2, headSize/6, headSize/10, headSize/8, 0, 0, Math.PI * 2);
            ctx.fill();

            // Ears
            ctx.fillStyle = furColor;
            const earSize = dog.earSize * canvas.width;
            ctx.beginPath();
            ctx.moveTo(-headSize/2 + earSize/2, -headSize/3);
            ctx.lineTo(-headSize/2 + earSize, -headSize/3 - earSize);
            ctx.lineTo(-headSize/2, -headSize/3 - earSize);
            ctx.closePath();
            ctx.fill();

            ctx.beginPath();
            ctx.moveTo(-headSize/2 + earSize/2, headSize/3);
            ctx.lineTo(-headSize/2 + earSize, headSize/3 + earSize);
            ctx.lineTo(-headSize/2, headSize/3 + earSize);
            ctx.closePath();
            ctx.fill();

            ctx.restore();

            // Legs
            const legLength = dog.legLength * canvas.width;
            const legWidth = legLength * 0.2;

            // Front left leg
            ctx.fillStyle = furColor;
            const frontLeftX = -bodyWidth/4;
            const frontLeftY = bodyHeight/4;
            const frontLeftPhase = legPhase < 0.5 ? legPhase * 2 : (1 - legPhase) * 2;
            ctx.beginPath();
            ctx.moveTo(frontLeftX, frontLeftY);
            ctx.lineTo(frontLeftX - legWidth/2, frontLeftY + legLength * (0.8 + frontLeftPhase * 0.2));
            ctx.lineTo(frontLeftX + legWidth/2, frontLeftY + legLength * (0.8 + frontLeftPhase * 0.2));
            ctx.closePath();
            ctx.fill();

            // Front right leg
            const frontRightX = -bodyWidth/4;
            const frontRightY = -bodyHeight/4;
            const frontRightPhase = legPhase < 0.5 ? (1 - legPhase) * 2 : (legPhase - 0.5) * 2;
            ctx.beginPath();
            ctx.moveTo(frontRightX, frontRightY);
            ctx.lineTo(frontRightX - legWidth/2, frontRightY + legLength * (0.8 + frontRightPhase * 0.2));
            ctx.lineTo(frontRightX + legWidth/2, frontRightY + legLength * (0.8 + frontRightPhase * 0.2));
            ctx.closePath();
            ctx.fill();

            // Back left leg
            const backLeftX = bodyWidth/4;
            const backLeftY = bodyHeight/4;
            const backLeftPhase = legPhase;
            ctx.beginPath();
            ctx.moveTo(backLeftX, backLeftY);
            ctx.lineTo(backLeftX - legWidth/2, backLeftY + legLength * (0.8 + backLeftPhase * 0.2));
            ctx.lineTo(backLeftX + legWidth/2, backLeftY + legLength * (0.8 + backLeftPhase * 0.2));
            ctx.closePath();
            ctx.fill();

            // Back right leg
            const backRightX = bodyWidth/4;
            const backRightY = -bodyHeight/4;
            const backRightPhase = 1 - legPhase;
            ctx.beginPath();
            ctx.moveTo(backRightX, backRightY);
            ctx.lineTo(backRightX - legWidth/2, backRightY + legLength * (0.8 + backRightPhase * 0.2));
            ctx.lineTo(backRightX + legWidth/2, backRightY + legLength * (0.8 + backRightPhase * 0.2));
            ctx.closePath();
            ctx.fill();

            // Tail
            const tailLength = dog.tailLength * canvas.width;
            ctx.fillStyle = furColor;
            ctx.beginPath();
            ctx.moveTo(bodyWidth/2, 0);
            ctx.quadraticCurveTo(
                bodyWidth/2 + tailLength/2,
                -tailLength/2 + tailWag * tailLength,
                bodyWidth/2 + tailLength,
                0 + tailWag * tailLength
            );
            ctx.quadraticCurveTo(
                bodyWidth/2 + tailLength/2,
                tailLength/2 + tailWag * tailLength,
                bodyWidth/2,
                0
            );
            ctx.closePath();
            ctx.fill();

            // Bone in mouth
            if (dog.hasBone) {
                const bonePos = project(
                    dog.bonePosition.x,
                    dog.bonePosition.y,
                    dog.bonePosition.z
                );

                ctx.save();
                ctx.translate(bonePos.x - pos.x, bonePos.y - pos.y);
                ctx.rotate(dog.boneRotation.y);

                const boneLength = bone.length * canvas.width * 0.8;
                const boneWidth = bone.width * canvas.width;
                const boneHeight = bone.height * canvas.height;

                ctx.fillStyle = boneColor;
                ctx.beginPath();
                // Main shaft
                ctx.moveTo(-boneLength/2, 0);
                ctx.lineTo(boneLength/2, 0);
                ctx.lineTo(boneLength/2, -boneHeight);
                ctx.lineTo(-boneLength/2, -boneHeight);
                ctx.closePath();
                ctx.fill();

                // Ends
                ctx.beginPath();
                ctx.ellipse(-boneLength/2, -boneHeight/2, boneWidth, boneHeight/2, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.beginPath();
                ctx.ellipse(boneLength/2, -boneHeight/2, boneWidth, boneHeight/2, 0, 0, Math.PI * 2);
                ctx.fill();

                ctx.restore();
            }

            // Restore context
            ctx.restore();

            // Shadow
            ctx.fillStyle = 'rgba(0, 0, 0, 0.15)';
            ctx.beginPath();
            ctx.ellipse(pos.x, pos.y + bodyHeight/2, bodyWidth/2, bodyHeight/4, 0, 0, Math.PI * 2);
            ctx.fill();
        }

        // Main render function
        window.renderFrame = function(timeSeconds) {
            animationState.time = timeSeconds;

            // Clear canvas
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Draw scene
            drawYard();
            drawDoghouse();
            if (!dog.hasBone) {
                drawBone();
            }
            drawDog();
        };

        // Initial render
        renderFrame(0);
    </script>
</body>
</html>
```

3. A bone that's properly grabbed and carried in the dog's mouth

6. All the required animation phases in sequence

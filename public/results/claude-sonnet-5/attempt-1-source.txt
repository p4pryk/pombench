<script>
const cv = document.getElementById('stage');
const ctx = cv.getContext('2d');

const GY = 400; // ground line y
const PUD = {x:370, y:400, rx:62, ry:15};
const HOUSE = {x:650, wallX:682, hallW:230, roofH:95, h:170, holeX0:672,holeX1:758,holeY0:238,holeY1:398};

function hash(i){ return (Math.sin(i*12.9898)*43758.5453)%1; }
function frac(x){ return x - Math.floor(x); }
function nrand(i){ return frac(Math.abs(hash(i)))*2-1; }

function clamp(v,a,b){ return Math.max(a,Math.min(b,v)); }
function lerp(a,b,t){ return a+(b-a)*t; }
function smooth(t){ return t*t*(3-2*t); }
function easeInOut(t){ t=clamp(t,0,1); return t*t*(3-2*t); }

// ---------- Background ----------
function drawBackdrop(ctx){
  let g = ctx.createLinearGradient(0,0,0,GY);
  g.addColorStop(0,'#bcd9ec');
  g.addColorStop(0.7,'#dce9d8');
  g.addColorStop(1,'#e4ecd9');
  ctx.fillStyle = g;
  ctx.fillRect(0,0,960,GY);

  // distant fence
  ctx.fillStyle = '#9c8468';
  for(let i=0;i<24;i++){
    let x = i*42-10;
    ctx.fillRect(x, 230, 26, 70);
  }
  ctx.fillStyle = 'rgba(0,0,0,0.08)';
  ctx.fillRect(0,230,960,70);

  // yard ground
  let gg = ctx.createLinearGradient(0,GY,0,540);
  gg.addColorStop(0,'#8a9f5c');
  gg.addColorStop(1,'#6f8248');
  ctx.fillStyle = gg;
  ctx.fillRect(0,GY,960,140);

  // grass texture (deterministic)
  ctx.strokeStyle = 'rgba(40,60,20,0.25)';
  ctx.lineWidth = 1;
  for(let i=0;i<260;i++){
    let x = frac(hash(i*3.1))*960;
    let y = GY+10+frac(hash(i*7.7))*125;
    let l = 4+frac(hash(i*1.3))*6;
    ctx.beginPath();
    ctx.moveTo(x,y);
    ctx.lineTo(x+nrand(i)*2, y-l);
    ctx.stroke();
  }
}

// ---------- Puddle ----------
function drawPuddleBase(ctx){
  ctx.save();
  ctx.beginPath();
  ctx.ellipse(PUD.x,PUD.y,PUD.rx,PUD.ry,0,0,Math.PI*2);
  let g = ctx.createLinearGradient(PUD.x,PUD.y-PUD.ry,PUD.x,PUD.y+PUD.ry);
  g.addColorStop(0,'#aebfc9');
  g.addColorStop(1,'#6c8694');
  ctx.fillStyle = g;
  ctx.fill();
  ctx.strokeStyle = 'rgba(255,255,255,0.4)';
  ctx.lineWidth=1.5;
  ctx.stroke();
  ctx.restore();
}
function puddleRipples(ctx,t){
  ctx.save();
  ctx.beginPath();
  ctx.ellipse(PUD.x,PUD.y,PUD.rx,PUD.ry,0,0,Math.PI*2);
  ctx.clip();
  ctx.strokeStyle='rgba(255,255,255,0.25)';
  for(let i=0;i<3;i++){
    let ph = (t*0.3+i*0.33)%1;
    ctx.beginPath();
    ctx.ellipse(PUD.x, PUD.y, PUD.rx*0.3+ph*PUD.rx*0.7, PUD.ry*0.2+ph*PUD.ry*0.7,0,0,Math.PI*2);
    ctx.globalAlpha = 0.3*(1-ph);
    ctx.stroke();
  }
  ctx.restore();
}

// ---------- Doghouse ----------
function drawHouseInterior(ctx){
  ctx.save();
  ctx.fillStyle = '#1c1510';
  ctx.beginPath();
  ctx.rect(HOUSE.holeX0-4, HOUSE.holeY0-4, HOUSE.holeX1-HOUSE.holeX0+8, HOUSE.holeY1-HOUSE.holeY0+10);
  ctx.fill();
  ctx.restore();
}
function drawBone(ctx,x,y,ang,scale){
  ctx.save();
  ctx.translate(x,y);
  ctx.rotate(ang);
  ctx.scale(scale,scale);
  ctx.fillStyle = '#eee3c5';
  ctx.strokeStyle = '#b7a978';
  ctx.lineWidth=1;
  ctx.beginPath();
  ctx.roundRect ? ctx.roundRect(-14,-3,28,6,3) : ctx.rect(-14,-3,28,6);
  ctx.fill(); ctx.stroke();
  for(const sx of [-14,14]){
    for(const sy of [-4,4]){
      ctx.beginPath();
      ctx.arc(sx,sy,4.2,0,Math.PI*2);
      ctx.fill();
      ctx.stroke();
    }
  }
  ctx.restore();
}
function drawHouseFront(ctx){
  ctx.save();
  // walls
  ctx.fillStyle = '#9a4b32';
  ctx.beginPath();
  ctx.rect(HOUSE.x, GY-HOUSE.h, HOUSE.hallW, HOUSE.h);
  // doorway hole (evenodd cut)
  ctx.rect(HOUSE.holeX0, HOUSE.holeY0, HOUSE.holeX1-HOUSE.holeX0, HOUSE.holeY1-HOUSE.holeY0);
  ctx.fill('evenodd');
  // wood planks
  ctx.strokeStyle='rgba(0,0,0,0.12)';
  for(let i=0;i<8;i++){
    ctx.beginPath();
    ctx.moveTo(HOUSE.x, GY-HOUSE.h+i*22);
    ctx.lineTo(HOUSE.x+HOUSE.hallW, GY-HOUSE.h+i*22);
    ctx.stroke();
  }
  // roof
  ctx.fillStyle = '#6b3524';
  ctx.beginPath();
  ctx.moveTo(HOUSE.x-18, GY-HOUSE.h);
  ctx.lineTo(HOUSE.x+HOUSE.hallW/2, GY-HOUSE.h-HOUSE.roofH);
  ctx.lineTo(HOUSE.x+HOUSE.hallW+18, GY-HOUSE.h);
  ctx.closePath();
  ctx.fill();
  ctx.fillStyle='rgba(0,0,0,0.18)';
  ctx.beginPath();
  ctx.moveTo(HOUSE.x+HOUSE.hallW/2, GY-HOUSE.h-HOUSE.roofH);
  ctx.lineTo(HOUSE.x+HOUSE.hallW+18, GY-HOUSE.h);
  ctx.lineTo(HOUSE.x+HOUSE.hallW-10, GY-HOUSE.h);
  ctx.closePath();
  ctx.fill();
  // doorway frame shadow
  ctx.fillStyle='rgba(0,0,0,0.35)';
  ctx.fillRect(HOUSE.holeX0-4,HOUSE.holeY0-4,HOUSE.holeX1-HOUSE.holeX0+8,6);
  ctx.restore();
}

// ---------- Dog state ----------
function computeState(t){
  const s = {};
  s.t = t;
  s.boneHeld = false;
  s.mouthOpen = 0;
  s.headPitch = 0;
  s.turn = 0; // 0 side(right-facing) -> 1 front
  s.legPhase = 0;
  s.crouch = 0;
  s.jump = 0;
  s.tailWag = 0.15*Math.sin(t*2.2);
  s.breathe = Math.sin(t*2.1)*0.015;

  if(t < 1){
    s.x = 90 + Math.sin(t*3)*2;
    s.y = GY;
    s.legPhase = Math.sin(t*3)*0.15;
    s.boneWorld = {x:740,y:355,ang:0.15};
  } else if(t < 3){
    let p = (t-1)/2;
    s.x = lerp(90, 610, p);
    s.legPhase = t*11;
    s.y = GY;
    if(t>=1.65 && t<1.9){
      let cp = (t-1.65)/0.25;
      s.crouch = smooth(cp);
    } else if(t>=1.9 && t<2.42){
      let jp = (t-1.9)/0.52;
      s.jump = Math.sin(Math.PI*jp);
      s.crouch = (1-jp)*0.3;
    }
    s.y = GY - s.jump*46;
    s.boneWorld = {x:740,y:355,ang:0.15};
  } else if(t < 4.5){
    let p = easeInOut((t-3)/1.5);
    s.x = lerp(610, 760, p);
    s.y = GY;
    s.legPhase = t*8;
    s.enter = p;
    s.boneWorld = {x:740,y:355,ang:0.15};
  } else if(t < 6){
    let p = (t-4.5)/1.5;
    s.x = 760; s.y = GY; s.enter = 1;
    s.headPitch = Math.sin(Math.min(p*1.6,1)*Math.PI*0.5)*0.55;
    s.mouthOpen = clamp(Math.sin(p*Math.PI),0,1);
    if(p>0.55){
      s.boneHeld = true;
    }
    s.boneWorld = {x:740,y:355,ang:0.15};
    s.legPhase = t*2;
  } else if(t < 8){
    let p = easeInOut((t-6)/2);
    s.x = lerp(760, 610, p);
    s.y = GY; s.enter = 1-p;
    s.boneHeld = true;
    s.headPitch = 0.15;
    s.legPhase = t*8+4;
  } else if(t < 10){
    let p = easeInOut((t-8)/2);
    s.x = lerp(610, 585, p);
    s.y = GY;
    s.boneHeld = true;
    s.turn = p;
    s.legPhase = 0;
  } else {
    s.x = 585; s.y = GY;
    s.boneHeld = true;
    s.turn = 1;
    s.legPhase = 0;
  }
  return s;
}

// ---------- Leg drawing helper ----------
function drawLeg(ctx, hipX, hipY, phase, L1, L2, liftAmp, dir){
  // phase controls swing; produces foot position & draws 2-seg leg
  let swing = Math.sin(phase)*0.55;
  let lift = Math.max(0, Math.sin(phase))*liftAmp;
  let a1 = 1.25 + swing*dir;
  let kneeX = hipX + Math.sin(a1)*L1;
  let kneeY = hipY + Math.cos(a1)*L1;
  let a2 = -0.25 - swing*0.8*dir - lift*0.03;
  let footX = kneeX + Math.sin(a1+a2)*L2;
  let footY = kneeY + Math.cos(a1+a2)*L2 - lift;
  ctx.lineWidth = 6;
  ctx.strokeStyle = '#c87b3d';
  ctx.lineCap='round';
  ctx.beginPath();
  ctx.moveTo(hipX,hipY);
  ctx.lineTo(kneeX,kneeY);
  ctx.lineTo(footX,footY);
  ctx.stroke();
  ctx.fillStyle='#8a5425';
  ctx.beginPath();
  ctx.ellipse(footX,footY,4.5,3,0,0,Math.PI*2);
  ctx.fill();
  return {footX,footY};
}

// ---------- Side view dog ----------
function drawDogSide(ctx, s, alpha){
  ctx.save();
  ctx.globalAlpha = alpha;
  let cx = s.x, cy = s.y;
  let bob = s.jump? -s.jump*2 : Math.abs(Math.sin(s.legPhase))*2.2*(s.enter===undefined?1:0.3);
  let crouchY = s.crouch*8;
  let by = cy - 34 + crouchY + bob*0 ;
  ctx.translate(cx, by + s.breathe*20);

  // contact shadow
  ctx.save();
  ctx.globalAlpha *= 0.4;
  ctx.fillStyle='#000';
  ctx.beginPath();
  ctx.ellipse(0, 34-crouchY+ (s.jump?22:0), 30-(s.jump?18:0), 6,0,0,Math.PI*2);
  ctx.fill();
  ctx.restore();

  // legs (back pair then front pair for depth)
  let legY = 20;
  let ph = s.legPhase;
  drawLeg(ctx, -16, legY, ph+Math.PI, 15,15, 10,1);
  drawLeg(ctx, 16, legY, ph, 15,15, 10,1);
  drawLeg(ctx, -14, legY, ph+Math.PI+0.3, 15,15, 9,1);
  drawLeg(ctx, 18, legY, ph+0.3, 15,15, 9,1);

  // tail
  ctx.save();
  ctx.translate(-30, -4);
  ctx.rotate(-0.9 + s.tailWag*0.3);
  let tg = ctx.createRadialGradient(0,0,2,0,0,22);
  tg.addColorStop(0,'#f0a94e');
  tg.addColorStop(1,'#d88226');
  ctx.fillStyle = tg;
  ctx.beginPath();
  ctx.ellipse(0,-10,15,18,0.3,0,Math.PI*2);
  ctx.fill();
  ctx.restore();

  // torso
  let tg2 = ctx.createLinearGradient(0,-26,0,20);
  tg2.addColorStop(0,'#f4b05a');
  tg2.addColorStop(1,'#c97a30');
  ctx.fillStyle = tg2;
  ctx.beginPath();
  ctx.ellipse(0,-2,30,20,0,0,Math.PI*2);
  ctx.fill();

  // ruff
  ctx.fillStyle = '#f6c178';
  ctx.beginPath();
  ctx.ellipse(16,-10,16,16,0,0,Math.PI*2);
  ctx.fill();

  // neck/head group
  ctx.save();
  ctx.translate(26,-18);
  ctx.rotate(-s.headPitch);
  // head
  ctx.fillStyle = '#f0a94e';
  ctx.beginPath();
  ctx.ellipse(6,-2,13,11,0,0,Math.PI*2);
  ctx.fill();
  // muzzle
  ctx.fillStyle = '#f3b162';
  ctx.beginPath();
  ctx.ellipse(18,2,8,6,0,0,Math.PI*2);
  ctx.fill();
  ctx.fillStyle='#2a1810';
  ctx.beginPath();
  ctx.ellipse(25,1,2.6,2,0,0,Math.PI*2);
  ctx.fill();
  // mouth open
  if(s.mouthOpen>0.02){
    ctx.fillStyle='#5b1c14';
    ctx.beginPath();
    ctx.ellipse(18, 5+s.mouthOpen*3, 6, 2+s.mouthOpen*3,0,0,Math.PI*2);
    ctx.fill();
  }
  // eye
  ctx.fillStyle='#211208';
  ctx.beginPath();
  ctx.ellipse(8,-5,2.2,2.6,0,0,Math.PI*2);
  ctx.fill();
  // ears
  ctx.fillStyle='#e8984a';
  ctx.beginPath();
  ctx.moveTo(-4,-10); ctx.lineTo(0,-24); ctx.lineTo(6,-11); ctx.closePath(); ctx.fill();
  ctx.beginPath();
  ctx.moveTo(6,-11); ctx.lineTo(13,-25); ctx.lineTo(15,-10); ctx.closePath(); ctx.fill();

  // bone in mouth
  if(s.boneHeld){
    drawBone(ctx, 22, 4, 0.1, 0.9);
  }
  ctx.restore();

  // fur strokes on torso for texture
  ctx.strokeStyle='rgba(120,70,20,0.25)';
  ctx.lineWidth=1;
  for(let i=0;i<22;i++){
    let ax = nrand(i)*28, ay=-2+nrand(i*2)*16;
    ctx.beginPath();
    ctx.moveTo(ax,ay);
    ctx.lineTo(ax+2,ay+5);
    ctx.stroke();
  }

  ctx.restore();

  // bone resting in house (not held) drawn separately by caller
}

function drawDogFront(ctx, s, alpha){
  ctx.save();
  ctx.globalAlpha = alpha;
  ctx.translate(s.x, s.y-34 + s.breathe*20);
  // shadow
  ctx.save();
  ctx.globalAlpha*=0.4;
  ctx.fillStyle='#000';
  ctx.beginPath();
  ctx.ellipse(0,36,28,6,0,0,Math.PI*2);
  ctx.fill();
  ctx.restore();

  // back legs faint
  ctx.fillStyle='#c87b3d';
  ctx.fillRect(-20,10,7,22);
  ctx.fillRect(13,10,7,22);
  // front legs
  let sway = Math.sin(s.t*2.1)*1.2;
  ctx.fillStyle='#d8893f';
  ctx.beginPath();
  ctx.ellipse(-14+sway,22,5,16,0,0,Math.PI*2); ctx.fill();
  ctx.beginPath();
  ctx.ellipse(14-sway,22,5,16,0,0,Math.PI*2); ctx.fill();
  ctx.fillStyle='#8a5425';
  ctx.beginPath(); ctx.ellipse(-14+sway,36,6,4,0,0,Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.ellipse(14-sway,36,6,4,0,0,Math.PI*2); ctx.fill();

  // tail peeking behind
  ctx.fillStyle='#e8984a';
  ctx.beginPath();
  ctx.ellipse(-26,-14,12,16,0.3,0,Math.PI*2); ctx.fill();
  ctx.beginPath();
  ctx.ellipse(26,-14,12,16,-0.3,0,Math.PI*2); ctx.fill();

  // torso
  let g = ctx.createRadialGradient(0,-6,6,0,0,34);
  g.addColorStop(0,'#f6ba66');
  g.addColorStop(1,'#cf8536');
  ctx.fillStyle=g;
  ctx.beginPath();
  ctx.ellipse(0,-4,26,24,0,0,Math.PI*2);
  ctx.fill();

  // ruff
  ctx.fillStyle='#f8c67e';
  ctx.beginPath();
  ctx.ellipse(0,-16,24,16,0,0,Math.PI*2);
  ctx.fill();

  // head
  ctx.save();
  ctx.translate(0,-34);
  ctx.fillStyle='#f3af5a';
  ctx.beginPath();
  ctx.ellipse(0,0,15,13,0,0,Math.PI*2);
  ctx.fill();
  // muzzle
  ctx.fillStyle='#f6ba66';
  ctx.beginPath();
  ctx.ellipse(0,9,8,6,0,0,Math.PI*2);
  ctx.fill();
  ctx.fillStyle='#2a1810';
  ctx.beginPath(); ctx.ellipse(0,10,2.6,2,0,0,Math.PI*2); ctx.fill();
  // eyes
  ctx.fillStyle='#211208';
  ctx.beginPath(); ctx.ellipse(-6,-2,2.4,2.8,0,0,Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.ellipse(6,-2,2.4,2.8,0,0,Math.PI*2); ctx.fill();
  // ears
  ctx.fillStyle='#e8984a';
  ctx.beginPath();
  ctx.moveTo(-13,-6); ctx.lineTo(-18,-26); ctx.lineTo(-3,-12); ctx.closePath(); ctx.fill();
  ctx.beginPath();
  ctx.moveTo(13,-6); ctx.lineTo(18,-26); ctx.lineTo(3,-12); ctx.closePath(); ctx.fill();

  // bone
  if(s.boneHeld){
    drawBone(ctx, 0, 11, Math.PI/2, 0.85);
  }
  ctx.restore();

  ctx.restore();
}

// bone resting
function drawBoneResting(ctx, s){
  if(!s.boneHeld){
    drawBone(ctx, s.boneWorld.x, s.boneWorld.y, s.boneWorld.ang, 1.0);
  }
}

// reflection
function drawReflection(ctx, s){
  if(s.x < PUD.x-PUD.rx-40 || s.x > PUD.x+PUD.rx+40) return;
  ctx.save();
  ctx.beginPath();
  ctx.ellipse(PUD.x,PUD.y,PUD.rx,PUD.ry,0,0,Math.PI*2);
  ctx.clip();
  ctx.translate(0, 2*GY);
  ctx.scale(1,-1);
  ctx.globalAlpha = 0.33;
  drawDogSide(ctx, s, 1);
  ctx.restore();
}

function renderFrame(t){
  t = Math.max(0, Math.min(12, t));
  ctx.clearRect(0,0,960,540);
  drawBackdrop(ctx);
  drawPuddleBase(ctx);
  puddleRipples(ctx,t);

  const s = computeState(t);

  drawReflection(ctx, s);

  drawHouseInterior(ctx);
  drawBoneResting(ctx, s);

  // draw dog (blend side/front for turn)
  if(s.turn <= 0.001){
    drawDogSide(ctx, s, 1);
  } else if(s.turn >= 0.999){
    drawDogFront(ctx, s, 1);
  } else {
    let sq = Math.cos(s.turn*Math.PI/2);
    let sq2 = Math.sin(s.turn*Math.PI/2);
    ctx.save();
    ctx.translate(s.x,0);
    ctx.scale(Math.max(sq,0.02),1);
    ctx.translate(-s.x,0);
    drawDogSide(ctx, s, Math.max(sq,0));
    ctx.restore();
    ctx.save();
    ctx.translate(s.x,0);
    ctx.scale(Math.max(sq2,0.02),1);
    ctx.translate(-s.x,0);
    drawDogFront(ctx, s, Math.max(sq2,0));
    ctx.restore();
  }

  drawHouseFront(ctx);
}

window.renderFrame = renderFrame;
renderFrame(0);
</script>

<canvas id="stage" width="960" height="540"></canvas>
<script>
(function(){
const cv=document.getElementById('stage');
const ctx=cv.getContext('2d');
const W=960,H=540;
const GROUND=430;
const HOUSE={x:610,y:GROUND-150,w:190,h:150};
const DOOR={x:HOUSE.x+50,y:HOUSE.y+62,w:90,h:88};
const BONE_REST={x:HOUSE.x+95,y:DOOR.y+DOOR.h-12};

function clamp(v,a,b){return v<a?a:v>b?b:v}
function lerp(a,b,t){return a+(b-a)*t}
function ease(t){t=clamp(t,0,1);return t*t*(3-2*t)}
function seg(t,a,b){return clamp((t-a)/(b-a),0,1)}
function rnd(n){const s=Math.sin(n*127.1+311.7)*43758.5453;return s-Math.floor(s)}

// Dog state as function of time
function dogState(t){
  // phases
  let s={x:90,y:0,face:1,bodyTurn:0,headDown:0,mouthBone:0,inside:0,tail:0,
          legPhase:0,legAmp:0,breath:0,bonePicked:0,turn:0,ear:0,
          frontal:0,headYaw:0};
  const T=clamp(t,0,12);
  // run to house (1-3s): x 90 -> 470
  let run=ease(seg(T,1,3.2));
  // enter house 3.2-4.5
  let enter=ease(seg(T,3.2,4.5));
  // grab 4.5-6
  let grab=ease(seg(T,4.5,6));
  // back out 6-8
  let back=ease(seg(T,6,8));
  // turn 8-10
  let turn=ease(seg(T,8,10));
  // settle 10-12
  s.x=lerp(90,500,run);
  if(T>=3.2){s.x=lerp(500,575,enter);}
  if(T>=6 && T<8){s.x=lerp(575,500,back);}
  if(T>=8){s.x=500;}
  s.inside = (T>=3.2&&T<6)?enter*(1-(T>=4.5?0:0))*(1-0):0;
  if(T>=3.2 && T<4.5) s.inside=enter;
  if(T>=4.5 && T<6) s.inside=1;
  if(T>=6 && T<8) s.inside=1-back;
  s.headDown=(T>=4.5&&T<6)?grab:(T>=6&&T<8?1-back:0);
  s.mouthBone=(T>=5.2)?1:0;
  s.bonePicked=(T>=5.2)?1:0;
  s.legAmp=(T>=1&&T<3.2)?1:((T>=3.2&&T<4.5)?0.6:((T>=6&&T<8)?0.5:0));
  s.legPhase=T*(T<3.2?12:T<8?6:0);
  s.turn=turn;
  s.frontal=ease(seg(T,8,10));
  s.breath=Math.sin(T*2.2)*0.5+0.5;
  s.tail=Math.sin(T*6)*0.5+0.5;
  return s;
}

function drawBackground(t){
  // sky gradient
  const g=ctx.createLinearGradient(0,0,0,GROUND);
  g.addColorStop(0,'#9cc8e8');g.addColorStop(1,'#e6eedb');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
  // distant fence
  ctx.fillStyle='#b9a58a';
  for(let i=0;i<22;i++){ctx.fillRect(i*46,GROUND-120,14,120);}
  ctx.fillStyle='#a08a6e';
  ctx.fillRect(0,GROUND-96,W,8);ctx.fillRect(0,GROUND-50,W,8);
  // hedge
  for(let i=0;i<40;i++){
    const r=rnd(i)*20+20;
    ctx.fillStyle='rgba(60,110,60,'+(0.8)+')';
    ctx.beginPath();ctx.arc(i*26,GROUND-70+rnd(i+3)*6,r,0,Math.PI*2);ctx.fill();
  }
  // lawn
  const lg=ctx.createLinearGradient(0,GROUND,0,H);
  lg.addColorStop(0,'#6fa352');lg.addColorStop(1,'#4c7a38');
  ctx.fillStyle=lg;ctx.fillRect(0,GROUND,W,H-GROUND);
  // grass strokes
  ctx.strokeStyle='rgba(40,80,30,0.35)';ctx.lineWidth=1;
  for(let i=0;i<260;i++){
    const x=rnd(i*3)*W, y=GROUND+rnd(i*5+1)*(H-GROUND);
    ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x+rnd(i)*3-1.5,y-4-rnd(i+9)*4);ctx.stroke();
  }
}

function drawHouse(){
  const h=HOUSE;
  // shadow
  ctx.fillStyle='rgba(0,0,0,0.22)';
  ctx.beginPath();ctx.ellipse(h.x+h.w/2+10,GROUND+4,h.w*0.6,9,0,0,Math.PI*2);ctx.fill();
  // body walls
  const wg=ctx.createLinearGradient(h.x,0,h.x+h.w,0);
  wg.addColorStop(0,'#c26a3c');wg.addColorStop(1,'#a2532b');
  ctx.fillStyle=wg;
  ctx.fillRect(h.x,h.y+18,h.w,h.h-18);
  // plank lines
  ctx.strokeStyle='rgba(60,25,10,0.35)';ctx.lineWidth=1;
  for(let x=h.x+12;x<h.x+h.w;x+=20){ctx.beginPath();ctx.moveTo(x,h.y+18);ctx.lineTo(x,h.y+h.h);ctx.stroke();}
  // roof
  ctx.fillStyle='#5a3a2a';
  ctx.beginPath();
  ctx.moveTo(h.x-14,h.y+22);ctx.lineTo(h.x+h.w/2,h.y-8);ctx.lineTo(h.x+h.w+14,h.y+22);ctx.closePath();ctx.fill();
  ctx.fillStyle='#704c38';
  ctx.beginPath();
  ctx.moveTo(h.x-14,h.y+22);ctx.lineTo(h.x+h.w/2,h.y-8);ctx.lineTo(h.x+h.w/2,h.y+8);ctx.lineTo(h.x-6,h.y+22);ctx.closePath();ctx.fill();
  // door interior (dark)
  const d=DOOR;
  ctx.fillStyle='#1d120c';
  ctx.fillRect(d.x,d.y,d.w,d.h);
  ctx.fillStyle='#2e1c12';
  ctx.fillRect(d.x+4,d.y+4,d.w-8,d.h-8);
  // door frame
  ctx.strokeStyle='#e7dccb';ctx.lineWidth=5;
  ctx.strokeRect(d.x-2,d.y-2,d.w+4,d.h+4);
  // roof trim
  ctx.fillStyle='#e7dccb';ctx.fillRect(h.x-16,h.y+18,h.w+32,6);
}

function drawBone(t,s){
  // resting before pick; carried after
  let bx,by,ang,vis=true;
  const T=clamp(t,0,12);
  if(T<5.2){
    // resting just inside doorway floor, slightly visible near threshold
    bx=BONE_REST.x-4;by=BONE_REST.y-2;ang=-0.15;
    // draw only if not occluded: inside house, partially visible from door
    // draw inside dark door area using dimmed color
    drawBoneShape(bx,by,ang,0.7);
    return;
  }
  // carried: attached to mouth of dog
  const mx=s.x+ (s.face||1)*0; // placeholder
  // computed from dog head position in dogPose
  const hp=headPoint(t,s);
  bx=hp.x+18; by=hp.y+2; ang=0.1;
  if(T>=10){ // presented at front
    bx=hp.x+4;by=hp.y+14;ang=0;
  }
  drawBoneShape(bx,by,ang,1);
}

function drawBoneShape(x,y,a,alpha){
  ctx.save();
  ctx.translate(x,y);ctx.rotate(a);
  ctx.globalAlpha=alpha;
  ctx.fillStyle='#f3ead6';
  ctx.strokeStyle='#b9a77f';ctx.lineWidth=1.2;
  ctx.beginPath();
  ctx.moveTo(-20,-3);ctx.lineTo(16,-3);
  ctx.arc(16,-7,4,0,Math.PI*2);
  ctx.arc(18,2,4,0,Math.PI*2);
  ctx.lineTo(-18,3);
  ctx.arc(-20,5,4,0,Math.PI*2);
  ctx.arc(-22,-1,4,0,Math.PI*2);
  ctx.closePath();ctx.fill();ctx.stroke();
  ctx.restore();
}

function headPoint(t,s){
  // approximate head centre in world coords for dog at state s
  const T=clamp(t,0,12);
  const base=dogBase(t,s);
  if(T>=8){ // frontal: head top-centre
    return {x:base.cx,y:base.gy-52};
  }
  return {x:base.cx+ (s.face||1)*22, y:base.gy-30+s.headDown*18};
}

function dogBase(t,s){
  // root positions
  const T=clamp(t,0,12);
  const cx=s.x;
  const gy=GROUND+2;
  return {cx,gy};
}

function drawDog(t){
  const s=dogState(t);
  const T=clamp(t,0,12);
  const base=dogBase(t,s);
  const cx=base.cx, gy=base.gy;
  // occlusion: when inside house, clip everything above door threshold line
  ctx.save();
  if(s.inside>0.01 && T<6){
    // clip to outside of house except door
    ctx.beginPath();
    ctx.rect(0,0,W,H);
    // remove house body region
    ctx.rect(HOUSE.x,HOUSE.y,HOUSE.w,HOUSE.h);
    ctx.clip('evenodd');
  }
  // partial turn: stretch horizontally
  const frontal=s.frontal;
  const sx=lerp(1,0.78,frontal);
  ctx.translate(cx,gy);
  ctx.scale(sx,1);
  const flip=1;
  ctx.translate(0,0);

  const bob=Math.abs(Math.sin(s.legPhase*0.5))*s.legAmp*2.5;
  const bodyY=-38-bob-(s.headDown*0)-s.breath*0.6;

  // shadow
  ctx.save();
  ctx.scale(1/sx,1);
  ctx.fillStyle='rgba(0,0,0,0.25)';
  ctx.beginPath();ctx.ellipse(0,2,32,6,0,0,Math.PI*2);ctx.fill();
  ctx.restore();

  // tail (curled over back)
  ctx.strokeStyle='#e0822f';ctx.lineCap='round';
  ctx.lineWidth=9;
  const tw=s.tail*4*(1-frontal);
  ctx.beginPath();
  ctx.moveTo(-26,bodyY-2);
  ctx.bezierCurveTo(-42,bodyY-18+tw,-44,bodyY-40,-26,bodyY-42);
  ctx.stroke();
  ctx.strokeStyle='#f6dfb5';ctx.lineWidth=4;
  ctx.beginPath();
  ctx.moveTo(-26,bodyY-2);
  ctx.bezierCurveTo(-40,bodyY-18+tw,-42,bodyY-38,-26,bodyY-40);
  ctx.stroke();

  // legs (back legs)
  drawLeg(-14,bodyY+10,s,false,0.5);
  drawLeg(-22,bodyY+10,s,false,0.25);
  // body
  ctx.fillStyle='#e8842f';
  ctx.beginPath();
  ctx.ellipse(0,bodyY,26,15+frontal*0,0,0,Math.PI*2);ctx.fill();
  // fur lighter underside
  ctx.fillStyle='#f6d7a5';
  ctx.beginPath();ctx.ellipse(2,bodyY+7,22,7,0,0,Math.PI*2);ctx.fill();
  // neck ruff
  ctx.fillStyle='#f3dcb0';
  ctx.beginPath();
  ctx.ellipse(14,bodyY-6,14,14,0,0,Math.PI*2);ctx.fill();
  // front legs
  drawLeg(14,bodyY+10,s,true,0);
  drawLeg(22,bodyY+10,s,true,0.5);

  // head
  const hy=bodyY-18+s.headDown*16;
  const hx=18+s.headDown*4;
  ctx.fillStyle='#e8842f';
  ctx.beginPath();ctx.ellipse(hx,hy,15,13,0,0,Math.PI*2);ctx.fill();
  // ears (small triangular)
  ctx.fillStyle='#e8842f';
  ctx.beginPath();
  ctx.moveTo(hx-8,hy-8);ctx.lineTo(hx-9,hy-22+s.ear*0);ctx.lineTo(hx-1,hy-10);ctx.closePath();ctx.fill();
  ctx.beginPath();
  ctx.moveTo(hx+4,hy-10);ctx.lineTo(hx+8,hy-22);ctx.lineTo(hx+11,hy-8);ctx.closePath();ctx.fill();
  // muzzle
  ctx.fillStyle='#f6dfb5';
  ctx.beginPath();
  ctx.moveTo(hx+4,hy-2);ctx.lineTo(hx+22,hy+1);ctx.lineTo(hx+4,hy+7);ctx.closePath();ctx.fill();
  // eye
  ctx.fillStyle='#1a120c';
  ctx.beginPath();ctx.arc(hx+4,hy-3,2,0,Math.PI*2);ctx.fill();
  // nose
  ctx.fillStyle='#111';
  ctx.beginPath();ctx.arc(hx+21,hy+0,1.6,0,Math.PI*2);ctx.fill();

  // bone in mouth
  if(s.mouthBone>0.5 && T<10){
    // handled by drawBone
  }
  ctx.restore();
  // redraw bone above dog if held
  if(T>=5.2){
    ctx.save();
    if(s.inside>0.01 && T<6){}
    ctx.restore();
  }
}

function drawLeg(x,y,s,front,ph){
  const T=s.legPhase;
  const swing=s.legAmp*Math.sin(T+ph*6.28)*6;
  ctx.strokeStyle='#e8842f';ctx.lineWidth=6;ctx.lineCap='round';
  ctx.beginPath();
  ctx.moveTo(x,y);
  ctx.lineTo(x+swing*0.4,y+10);
  ctx.lineTo(x+swing,y+20);
  ctx.stroke();
  ctx.fillStyle='#f6dfb5';
  ctx.beginPath();ctx.ellipse(x+swing+1,y+21,4,2.4,0,0,Math.PI*2);ctx.fill();
}

window.renderFrame=function(timeSeconds){
  let t=timeSeconds;
  if(t<0)t=0;
  if(t>12)t=12;
  ctx.setTransform(1,0,0,1,0,0);
  ctx.clearRect(0,0,W,H);
  drawBackground(t);
  drawHouse();
  // bone at rest, under dog when inside
  const s=dogState(t);
  if(t<5.2) drawBone(t,s);
  drawDog(t);
  if(t>=5.2) drawBone(t,s);
};

renderFrame(0);
})();
</script>

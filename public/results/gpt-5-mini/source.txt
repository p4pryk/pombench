const canvas = document.getElementById('stage');
const ctx = canvas.getContext('2d');

const W = canvas.width, H = canvas.height;

// Scene layout constants (fixed)
const yard = {
  groundY: 380,          // horizon-ish ground plane y
  houseX: 660, houseY: 220, houseW: 220, houseH: 200,

const puddle = {

// Bone resting position inside doghouse interior coordinates (relative)
const bone = {
  x: yard.houseX + 48, y: yard.houseY + yard.houseH - 48, w: 36, h: 12

// Dog scale and initial positions (approach from left)
const dogParams = {
  initialX: 120, initialY: yard.groundY - 6,

  height: 80, length: 98

// Animation timing (seconds)
function clamp(v,a,b){return Math.max(a,Math.min(b,v));}
function smoothstep(a,b,t){ if(t<=a) return 0; if(t>=b) return 1; t=(t-a)/(b-a); return t*t*(3-2*t); }

const T_TOTAL = 12;

// Precompute key time ranges per spec
const T_notice = [0,1];
const T_run = [1,3];
const T_enter = [3,4.5];
const T_grab = [4.5,6];
const T_backout = [6,8];
const T_turn = [8,10];
const T_settle = [10,12];

// Simple deterministic pseudo-random (for subtle fur pattern)
function pseudo(seed){
  let s = seed|0;
  return function(){ s = (s*1664525 + 1013904223) | 0; return ((s >>> 0) / 4294967295); }
}
const rnd = pseudo(12345);

// Utility drawing helpers
function drawRoundedRect(ctx, x,y,w,h,r){

  ctx.moveTo(x+r,y);

  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
  ctx.fill();
}

// Dog procedural model
function makeDogState(t){
  // position along approach path (x)
  // Define phases:
  // run: 1->3: move from initialX to door threshold
  // enter: 3->4.5: move forward into doorway (x increases into house slightly)
  // grab: 4.5->6: little pause inside (head lower)
  // backout: 6->8: move backwards to outside threshold
  // turn: 8->10: rotate toward viewer and step to face finalFrontX
  // settle: 10->12: maintain final frontal pose
  // We'll produce x,y,bodyAngle,heading, leg phases, tail/ruff movement, mouthOpen and bone attached boolean
  let state = {
    x: dogParams.initialX,
    y: dogParams.initialY,
    bodyAngle: 0,
    heading: 0, // 0 = left->right
    scale: 1,
    mouthOpen: 0,
    boneHeld: false,
    // leg angles for 4 legs: frontLeft, frontRight, hindLeft, hindRight
    legs: [0,0,0,0],
    tailCurl: 0,
    earTilt: 0,
    coatShift: 0,
    inAir: false
  };

  // helper for interpolation
  function lerp(a,b,u){return a + (b-a)*u;}
  // path x target positions
  const runStartX = dogParams.initialX;
  const runEndX = yard.houseX + yard.doorW*0.5 - 18; // just outside door
  // During enter, x increases slightly into doorway interior
  const enterEndX = yard.houseX + yard.doorW*0.5 + 12;
  const backoutEndX = runEndX; // come back to just outside door
  const turnEndX = dogParams.finalFrontX;

  // compute progress in segments
  const p_run = smoothstep(T_run[0], T_run[1], t);
  const p_enter = smoothstep(T_enter[0], T_enter[1], t);
  const p_grab = smoothstep(T_grab[0], T_grab[1], t);
  const p_backout = smoothstep(T_backout[0], T_backout[1], t);
  const p_turn = smoothstep(T_turn[0], T_turn[1], t);
  const p_settle = smoothstep(T_settle[0], T_settle[1], t);

  if (t < T_run[0]) {
    state.x = runStartX;
  } else if (t < T_run[1]) {
    state.x = lerp(runStartX, runEndX, p_run);
  } else if (t < T_enter[1]) {
    state.x = lerp(runEndX, enterEndX, p_enter);
  } else if (t < T_grab[1]) {
    // remain with slight shift
    state.x = lerp(enterEndX, enterEndX + 2, p_grab);
  } else if (t < T_backout[1]) {
    // back out from interior to outside
    state.x = lerp(enterEndX+2, backoutEndX, p_backout);
  } else if (t < T_turn[1]) {
    state.x = lerp(backoutEndX, turnEndX, p_turn);
  } else {
    state.x = turnEndX;
  }

  // vertical position: ground normally, but during jump have arc
  // Jump scheduled around run end: anticipate near t ~ 2.0s (run in 1-3). We'll place jump center at 2.0.
  const jumpCenter = 2.0;
  const jumpDuration = 0.8; // total time dog is airborne ~ centered
  const jumpStart = jumpCenter - jumpDuration*0.5;
  const jumpEnd = jumpCenter + jumpDuration*0.5;
  let inAir = (t >= jumpStart && t <= jumpEnd);
  state.inAir = inAir;
  if (inAir) {
    // normalized 0..1 through jump
    const u = (t - jumpStart) / (jumpEnd - jumpStart);
    // arc height relative
    const height = 56; // peak height
    const yoff = 4 + height * Math.sin(Math.PI * u); // smooth arc
    state.y = dogParams.initialY - yoff;
    // body tilt during flight: slight nose-up then nose-down
    state.bodyAngle = Math.PI*0.06 * Math.cos(Math.PI*(u)); // small rotation
  } else {
    state.y = dogParams.initialY;
    state.bodyAngle = 0;
  }

  // If during enter/grab: head lowering and mouth open
  if (t >= T_enter[0] && t < T_grab[0]) {
    // entering, head low
    state.mouthOpen = 0;
  }
  if (t >= T_grab[0] && t < T_grab[1]) {
    state.mouthOpen = smoothstep(T_grab[0], T_grab[1], t);
    if (t >= T_grab[0]+0.15) state.boneHeld = true; // bone removed from interior at start of grab
  }
  if (t >= T_backout[0]) {
    state.mouthOpen = 0.0;
    // keep bone held once grabbed
    if (t >= T_grab[0]+0.15) state.boneHeld = true;
  }

  // Tail and ears behaviour:
  state.tailCurl = 0.72 + 0.06*Math.sin(t*4 + 0.3); // curled over back baseline
  state.earTilt = 0.08*Math.cos(t*3);
  state.coatShift = 1.0 + 0.03*Math.cos(t*6);

  // Leg gait: during running/backing produce phases; during enter/inside adjust; during turn different stepping
  // We'll model gait cycle frequency; ensure all paws leave ground during jump (set to lifted)
  function legPhaseForTime(t, forward){
    // forward gait frequency
    const freq = forward ? 2.6 : 2.0;
    const phase = (t * freq) % 1;
    // produce angle for protraction/retraction
    // returns angle and groundContact boolean based on phase windows
    const angle = Math.sin(phase*Math.PI*2) * 0.9;
    const contact = (phase < 0.45); // foot on ground ~ first half cycle
    return {angle, contact, phase};
  }

  // assign legs
  if (inAir) {
    // tuck legs slightly
    state.legs = [-0.9, -0.8, 0.4, 0.45]; // front left, front right, hind left, hind right (angles)
  } else if (t >= T_backout[0] && t < T_turn[0]) {
    // backing gait: reverse direction
    const lf = legPhaseForTime(t, false);
    const rf = legPhaseForTime(t + 0.5/freqHack(), false);
    const lh = legPhaseForTime(t + 0.25/freqHack(), false);
    const rh = legPhaseForTime(t + 0.75/freqHack(), false);
    state.legs = [lf.angle, rf.angle, lh.angle, rh.angle].map(a=>a*0.9);
  } else if (t >= T_turn[0] && t < T_turn[1]) {
    // turning: staggered steps, slower
    const turnPhase = (t - T_turn[0])/(T_turn[1]-T_turn[0]);
    // legs move to reposition feet for turn
    state.legs = [
      -0.2 - 0.6*Math.sin(turnPhase*Math.PI*2),
      -0.1 - 0.6*Math.cos(turnPhase*Math.PI*2),
      0.2 + 0.4*Math.cos(turnPhase*Math.PI*2),
      0.15 + 0.4*Math.sin(turnPhase*Math.PI*2)
    ];
  } else {
    // forward running gait in approach and entering
    const freq = 2.6;
    function lph(off){ const phase = ((t*freq)+off) % 1; return Math.sin(phase*Math.PI*2) * 1.05; }
    state.legs = [ lph(0), lph(0.5), lph(0.25), lph(0.75) ];
    // if just entering (3-4.5) reduce amplitude
    if (t >= T_enter[0] && t < T_enter[1]) {
      const reduce = 1 - smoothstep(T_enter[0], T_enter[1], t)*0.6;
      state.legs = state.legs.map(a=>a*reduce);
    }
  }

  // For realistic paw contacts during landing: if just after jump end, make front legs compress
  if (t > jumpEnd && t < jumpEnd + 0.25) {
    const c = (t - jumpEnd) / 0.25;
    state.legs[0] -= 0.6*(1-c); state.legs[1] -= 0.6*(1-c);
    // body compress
    state.bodyAngle += 0.02*(1-c);
  }

  // head and neck orientation tailored to phases:
  // default head faces forward (0). During grab lowers and opens mouth; during turn rotates to viewer
  if (t >= T_turn[0] && t < T_turn[1]) {
    // rotate to face viewer: body rotates negative to show frontal
    const u = p_turn;
    state.bodyAngle += lerp(0, -Math.PI*0.65, u); // rotates so that dog turns more toward camera
  } else if (inAir) {
    state.bodyAngle += -0.06;
  }

  // subtle bobbing of coat/ruff driven by speed
  const speed = Math.abs( (state.x - dogParams.initialX) / (runEndX - dogParams.initialX + 0.0001) );
  state.coatShift += 0.06*Math.sin(t*8)*speed;

  return state;
  // small helper for back gait phasing freq hack
  function freqHack(){ return 2.0; }
}

// Draw the dog as layered shapes using state
function drawDog(ctx, state, options){
  // options: reflectY boolean, alpha multiplier, clip? used for reflection
  const reflect = options && options.reflect;
  const alpha = (options && options.alpha!=null)? options.alpha : 1;
  ctx.save();
  ctx.globalAlpha = alpha;
  // compute transform: place dog center at state.x,state.y; apply bodyAngle and scale
  const sx = state.x, sy = state.y;
  ctx.translate(sx, sy);
  ctx.rotate(state.bodyAngle);
  // body scale
  const scale = 1.0;
  ctx.scale(scale, scale);

  // Basic skeleton proportions
  const bodyLen = 84;
  const bodyH = 46;
  const neckLen = 22;
  const headW = 36, headH = 30;
  const tailOffset = -bodyLen*0.2;

  // If reflected, flip vertically
  if (reflect) {
    ctx.scale(1, -1);
  }

  // Draw shadow contact under dog (simple ellipse) - but for reflection skip
  if (!reflect) {
    ctx.save();
    ctx.rotate(-state.bodyAngle);
    ctx.fillStyle = 'rgba(0,0,0,0.12)';
    ctx.beginPath();
    ctx.ellipse(0+10, bodyH+22, 44, 12, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  // Fur base color for Pomeranian (orange-cream)
  const baseA = '#ee9b45';
  const baseB = '#f6d9b2';

  // Draw hindquarters (behind)
  ctx.save();
  ctx.translate(-6, 8); // move origin to roughly center
  // hind body ellipse
  const grad = ctx.createLinearGradient(-20,-10,40,40);
  grad.addColorStop(0,baseB); grad.addColorStop(1,baseA);
  ctx.fillStyle = grad;
  ctx.beginPath();
  ctx.ellipse(0, 6, bodyLen*0.5, bodyH*0.6, -0.05, 0, Math.PI*2);
  ctx.fill();

  // tail plume
  ctx.save();
  ctx.translate(bodyLen*0.5 - 6, -16);
  ctx.rotate(-0.6 + 0.12*Math.sin(2 + state.tailCurl*3));
  // plume shape
  ctx.fillStyle = '#f7e0c1';
  ctx.beginPath();
  ctx.moveTo(0,0);
  ctx.bezierCurveTo(40,-6,44,-40,12,-46);
  ctx.bezierCurveTo(28,-34,8,-10,0,0);
  ctx.closePath();
  ctx.fill();
  ctx.restore();

  // hind leg shapes (two legs)
  drawLeg(ctx, 14, 18, state.legs[2], true);
  drawLeg(ctx, -6, 18, state.legs[3], true);

  ctx.restore();

  // chest and neck
  ctx.save();
  ctx.translate(-6, -2);
  // chest ruff
  ctx.fillStyle = '#fbeddc';
  ctx.beginPath();
  ctx.ellipse(-6, -6, 34, 28, -0.15, 0, Math.PI*2);
  ctx.fill();

  // torso top to connect
  ctx.fillStyle = baseA;
  ctx.beginPath();
  ctx.ellipse(16, 2, bodyLen*0.46, bodyH*0.6, 0.08, 0, Math.PI*2);
  ctx.fill();

  // front legs
  drawLeg(ctx, 34, 18, state.legs[0], false);
  drawLeg(ctx, 12, 18, state.legs[1], false);

  // head & muzzle
  ctx.save();
  ctx.translate(46, -10); // neck position
  // neck rotation subtle: when turning, head more independent
  const headTilt = (state.bodyAngle< -0.2) ? -0.3*(Math.abs(state.bodyAngle)/1.5) : Math.sin(Date.now()*0.002)*0.02;
  ctx.rotate(headTilt);
  // neck ruff
  ctx.fillStyle = '#f9e6cf';
  ctx.beginPath();
  ctx.ellipse(-6, 6, 26, 22, -0.2, 0, Math.PI*2);
  ctx.fill();

  // head main
  ctx.save();
  // if mouth open, tilt jaw down a bit
  const mouthOpen = state.mouthOpen;
  ctx.translate(0, -4);
  ctx.fillStyle = '#f1c28a';
  ctx.beginPath();
  ctx.ellipse(0, 0, headW*0.9, headH*0.9, 0, 0, Math.PI*2);
  ctx.fill();

  // muzzle
  ctx.fillStyle = '#f6e2c8';
  ctx.beginPath();
  ctx.ellipse(10, 6, 18, 12, -0.12, 0, Math.PI*2);
  ctx.fill();

  // mouth / bone interaction area (draw lower jaw as shape if mouthOpen)
  ctx.save();
  if (mouthOpen > 0.001) {
    ctx.translate(8, 10);
    ctx.rotate(0.05);
    ctx.fillStyle = '#3b2516';
    ctx.beginPath();
    ctx.ellipse(0, 2, 10 + 2*mouthOpen, 5 + 2*mouthOpen, 0, 0, Math.PI*2);
    ctx.fill();
  } else {
    // closed mouth crease
    ctx.strokeStyle = 'rgba(60,40,30,0.85)';
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    ctx.moveTo(-2, 8);
    ctx.quadraticCurveTo(8,12,18,10);
    ctx.stroke();
  }
  ctx.restore();

  // eyes
  ctx.fillStyle = '#24170f';
  ctx.beginPath();
  ctx.ellipse(-4, -4, 6, 6, 0, 0, Math.PI*2);
  ctx.fill();
  ctx.fillStyle = '#000';
  ctx.beginPath();
  ctx.ellipse(-3, -4, 2.2, 2.6, 0, 0, Math.PI*2);
  ctx.fill();
  // small highlight
  ctx.fillStyle = 'rgba(255,255,255,0.85)';
  ctx.beginPath();
  ctx.ellipse(-1.5, -5.5, 1.0, 0.6, 0, 0, Math.PI*2);
  ctx.fill();

  // ears
  ctx.fillStyle = '#f1c28a';
  ctx.beginPath();
  ctx.moveTo(-16, -20);
  ctx.lineTo(-8, -6);
  ctx.lineTo(-26, -8);
  ctx.closePath();
  ctx.fill();
  ctx.beginPath();
  ctx.moveTo(2, -22);
  ctx.lineTo(12, -6);
  ctx.lineTo(-4, -8);
  ctx.closePath();
  ctx.fill();

  // nose
  ctx.fillStyle = '#2a1b14';
  ctx.beginPath();
  ctx.ellipse(18, 2, 4.4, 3.2, 0, 0, Math.PI*2);
  ctx.fill();

  ctx.restore(); // head

  ctx.restore(); // chest

  // If bone is being held, draw bone relative to head/mouth position
  if (!options || !options.skipBone) {
    // compute bone world coordinates from state.x + offsets rotated by bodyAngle and head transform
    const bonePos = computeBoneWorld(state);
    // draw bone (single object)
    ctx.save();
    ctx.globalAlpha = alpha;
    // If reflection, bone should reflect too
    if (options && options.reflect) {
      // reflected draw handled by same code (mirror)
    }
    ctx.translate(bonePos.x - state.x, bonePos.y - state.y);
    // apply body rotation in drawing space: we've already rotated context by state.bodyAngle, so bone local align works
    ctx.rotate(bonePos.rot + (reflect? Math.PI:0));
    // bone shape
    ctx.fillStyle = '#e9e0c8';
    ctx.strokeStyle = '#bfae86';
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(-bone.w/2, 0);
    ctx.lineTo(bone.w/2, 0);
    ctx.fill();
    // ends
    ctx.beginPath();
    ctx.ellipse(bone.w/2, 0, 8, 6, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.beginPath();
    ctx.ellipse(-bone.w/2, 0, 8, 6, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.stroke();
    ctx.restore();
  }

  ctx.restore();

  // small helper draws legs relative to origin
  function drawLeg(ctx, ox, oy, angle, isHind){
    ctx.save();
    ctx.translate(ox, oy);
    // upper leg
    ctx.rotate(angle*0.7);
    ctx.fillStyle = '#f2d0a0';
    ctx.beginPath();
    ctx.ellipse(0, 8, 9, 12, 0, 0, Math.PI*2);
    ctx.fill();
    // lower leg
    ctx.translate(0, 16);
    ctx.rotate(angle*0.8);
    ctx.fillStyle = '#f0cfa0';
    ctx.beginPath();
    ctx.ellipse(0, 8, 7, 10, 0, 0, Math.PI*2);
    ctx.fill();
    // paw contact
    ctx.fillStyle = '#e6c7a0';
    ctx.beginPath();
    ctx.ellipse(0, 18, 9, 5, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  // Compute bone position in world coordinates based on head/mouth placement
  function computeBoneWorld(state){
    // approximate head world position transform
    const sx = state.x, sy = state.y;
    const bodyAngle = state.bodyAngle;
    // head pivot relative to dog origin used earlier translate(46,-10)
    const hx = 46, hy = -10;
    // head rotation headTilt computed earlier roughly - use mouthOpen
    const headTilt = (state.bodyAngle< -0.2) ? -0.3*(Math.abs(state.bodyAngle)/1.5) : 0;
    // mouth local offset
    const mx = 18 + state.mouthOpen*6;
    const my = 6 + state.mouthOpen*2;
    // compute transform
    const cosB = Math.cos(bodyAngle + headTilt), sinB = Math.sin(bodyAngle + headTilt);
    const wx = sx + (hx * cosB - hy * sinB);
    const wy = sy + (hx * sinB + hy * cosB);
    // bone rotation relative
    const rot = bodyAngle + headTilt - 0.15;
    // bone position offset
    const bx = wx + (mx * Math.cos(bodyAngle + headTilt));
    const by = wy + (mx * Math.sin(bodyAngle + headTilt));
    return {x: bx, y: by, rot: rot};
  }
}

// Render reflection of dog into puddle: draw clipped mirrored dog with alpha and slight blur-like effect
function drawDogReflection(ctx, state){
  // Clip to puddle ellipse
  ctx.save();
  ctx.beginPath();
  ctx.ellipse(puddle.cx, puddle.cy, puddle.rx, puddle.ry, 0, 0, Math.PI*2);
  ctx.clip();
  // The reflection is vertically flipped relative to puddle.cy
  ctx.save();
  ctx.translate(0, puddle.cy*2);
  ctx.scale(1, -1);
  // Translate coordinate so that normal drawing uses world coords: renderDog expects translation by state.x,state.y
  // But to simulate faint blurred reflection, draw with lower alpha, vertical offset proportional to y-distance from puddle
  const distance = Math.max(0, puddle.cy - state.y);
  // Slight fade and blur emulate
  drawDog(ctx, state, {reflect:true, alpha:0.36});
  ctx.restore();
  // overlay subtle water distortion lines
  ctx.globalAlpha = 0.12;
  ctx.fillStyle = '#ffffff';
  for(let i=0;i<6;i++){
    ctx.beginPath();
    const y = puddle.cy - puddle.ry + (i/6)*puddle.ry*2 + (Math.sin(i*1.6 + state.x*0.01)*6);
    ctx.ellipse(puddle.cx + Math.cos(i)*8, y, puddle.rx*0.86 - i*6, puddle.ry*0.06, 0, 0, Math.PI*2);
    ctx.fill();
  }
  ctx.globalAlpha = 1.0;
  ctx.restore();
}

// Draw bone resting inside doghouse (only until grabbed)
function drawBoneInHouse(ctx, t){
  ctx.save();
  // bone inside interior coordinates: bone.x, bone.y
  // If bone has been grabbed (state.boneHeld), don't draw in interior after that moment
  if (t >= T_grab[0]+0.15) { ctx.restore(); return; }
  // interior transform: door opening is at dx,dy
  const dx = yard.houseX + (yard.houseW - yard.doorW)/2;
  const dy = yard.houseY + yard.houseH - yard.doorH;
  // place bone a little inside
  ctx.translate(dx + bone.x - yard.houseX, dy + bone.y - yard.houseY - 6);
  ctx.rotate(-0.05);
  ctx.fillStyle = '#e9e0c8';
  ctx.strokeStyle = '#bfae86';
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(-bone.w/2, 0);
  ctx.lineTo(bone.w/2, 0);
  ctx.fill();
  // ends
  ctx.beginPath(); ctx.ellipse(bone.w/2, 0, 8, 6, 0, 0, Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.ellipse(-bone.w/2, 0, 8, 6, 0, 0, Math.PI*2); ctx.fill();
  ctx.stroke();
  ctx.restore();
}

// Main renderFrame as required (synchronous)
window.renderFrame = function(timeSeconds){
  let t = Number(timeSeconds);
  if (isNaN(t)) t = 0;
  if (t < 0) t = 0;
  if (t > T_TOTAL) t = T_TOTAL;
  // clear
  ctx.clearRect(0,0,W,H);

  // background scene
  drawYard(ctx);
  drawHouse(ctx);

  // compute dog state
  const state = makeDogState(t);

  // Draw reflection in puddle: reflection should follow dog during approach and flight.
  // We must draw reflection BEFORE dog and also ensure masking so reflection only inside puddle
  drawDogReflection(ctx, state);

  // Draw bone in house interior (if not yet removed)
  drawBoneInHouse(ctx, t);

  // Draw occlusion: the dog may enter door: we need to draw dog with correct overlap relative to doorway.
  // Approach: compute whether dog's body is partly inside door threshold; if so, draw back half first, then house doorway occludes mid, then draw front parts.

  // Determine if dog's torso crosses door threshold (x coordinate relative)
  const doorCenterX = yard.houseX + yard.doorW*0.5;
  // We'll render sequence: hindquarters, house interior occlusion, front parts
  // To approximate, draw hind portion with clip if behind doorway.
  // For simplicity: if state.x > doorCenterX - 6, some of body is inside.

  // Draw hindquarter/back half
  ctx.save();
  // clip region for hind one: left of a local split
  if (state.x < doorCenterX + 10) {
    // fully outside: draw full dog normally
    drawDog(ctx, state, {alpha:1});
    ctx.restore();
  } else {
    // split draw:
    // 1) draw hind portion only: create clip rectangle covering rear half of dog local coords
    // We'll clip world to a polygon to draw rear where (dog-local x) < 6
    // Simpler: mask by drawing dog into offscreen canvas? Avoid offscreen. Instead approximate by drawing a darkened rear first then let doorway occlude front.
    // Draw entire dog at slightly desaturated to represent rear then later draw front normally but masked by door.
    // To achieve correct occlusion we will:
    // - Draw hind one with translation but mask to left side of dog bounding box via globalCompositeOperation
    ctx.restore();
    // Strategy simplified: draw full dog; then draw doorway (already drawn) will occlude front; then draw front exposed (the parts that should be in front of door) — but doorway must appear in front of dog when dog is inside. So draw dog fully, then redraw doorway (again) on top to occlude; then draw only front-most bits that are outside (like tail/rear) appropriately. Simpler: draw dog fully, then draw doorway overlay to occlude interior parts, then draw front of dog remaining on top (like head/forelegs) to show outside. We'll implement: draw full dog, then draw door overlay, then draw front-facing pieces (head and forelegs) if outside.
    // Actually easier: draw full dog; draw door overlay (frame) so interior bits are occluded; then if dog is mostly outside draw nothing; if dog is emerging, draw front bits on top.
    // So proceed:
    drawDog(ctx, state, {alpha:1});
    // redraw door frame overlay to occlude interior parts correctly
    // door overlay: darker inner lip and threshold
    ctx.save();
    const dx = yard.houseX + (yard.houseW - yard.doorW)/2;
    const dy = yard.houseY + yard.houseH - yard.doorH;
    // draw frame strokes
    ctx.fillStyle = 'rgba(60,40,30,0.98)';
    drawRoundedRect(ctx, dx, dy, yard.doorW, yard.doorH, 8);
    // inside visible (cut hole) we draw interior darker as before
    ctx.fillStyle = '#22170f';
    ctx.fillRect(dx+8, dy+10, yard.doorW-16, yard.doorH-20);
    // threshold highlight again
    ctx.fillStyle = 'rgba(255,255,255,0.06)';
    ctx.fillRect(dx+6, dy+yard.doorH-12, yard.doorW-12, 6);
    ctx.restore();

    // Now draw head and forelegs on top if they are outside (state.x < doorCenterX + 30)
    if (state.x < doorCenterX + 18) {
      // dog mostly outside: no need
    } else {
      // draw only head/fore area: approximate by drawing the dog again clipped to front region
      ctx.save();
      // clip to rectangle right of doorway so that parts outside are visible
      ctx.beginPath();
      ctx.rect(0,0, W, H);
      // allow drawing only where x < door left? Actually draw parts to the left of door to show them outside - we want head/fore visible outside: clip to area left of door inner edge
      const doorLeftEdge = yard.houseX + (yard.houseW - yard.doorW)/2 + 2;
      ctx.rect(0,0,doorLeftEdge, H);
      ctx.clip('nonzero');
      // draw dog again (this will show portions outside)
      drawDog(ctx, state, {alpha:1});
      ctx.restore();
    }
    ctx.restore();
  }

  // If dog was not going through, ensure doorway overlay drawn to keep consistent
  if (!(state.x >= doorCenterX - 40 && state.x <= doorCenterX + 80)) {
    // ensure door drawn (already drawn above by drawHouse), skip
  }

  // Finally draw contact shadows / paw shadows for stance and landing
  drawContactShadows(ctx, state);

  // ensure final frontal pose subtle breathing when t>=10: hold slight motion
  // but main visuals done
};

// Contact shadows
function drawContactShadows(ctx, state){
  ctx.save();
  ctx.globalAlpha = 0.28;
  ctx.fillStyle = 'rgba(0,0,0,0.28)';
  // approximate paw positions relative to dog body
  const footOffsets = [
    {x: 34, y: 36}, // front left
    {x: 12, y: 36}, // front right
    {x: 14, y: 38}, // hind left
    {x: -6, y: 38}  // hind right
  ];
  for(let i=0;i<4;i++){
    const off = footOffsets[i];
    // transform local to world with rotation
    const bx = state.x + (off.x*Math.cos(state.bodyAngle) - off.y*Math.sin(state.bodyAngle));
    const by = state.y + (off.x*Math.sin(state.bodyAngle) + off.y*Math.cos(state.bodyAngle)) + 18;
    ctx.beginPath();
    ctx.ellipse(bx, by, 8, 4, 0, 0, Math.PI*2);
    ctx.fill();
  }
  // extra landing shadow if recently landed
  ctx.restore();
}

// initialize by rendering t=0
window.renderFrame(0);
})();

(() => {
const canvas = document.getElementById('stage');
const ctx = canvas.getContext('2d', { alpha: false });
const W = canvas.width, H = canvas.height;

// Scene constants
const DURATION = 12;
const yard = {
  groundY: 420,
  backFenceY: 160,
};

// Dog model base parameters
const dogBase = {
  // world coordinates: x increases right, y increases down, z is depth toward camera (smaller z = further back)
  startX: 160, startZ: 0.32, startYGround: yard.groundY,
  houseX: 560, houseZ: 0.20, houseY: yard.groundY,
  scaleBase: 1.0
};

// Bone model
const boneModel = {
  length: 42, radius: 8, color: '#efe7d0'
};

// Doghouse geometry
const house = {
  x: 520, y: yard.groundY - 120, w: 220, h: 140, doorW: 90, doorH: 100, color: '#7b4f26', trim: '#5a3618', floorDepth: 18
};

// Seeded deterministic pseudo-random for fur stroke placement
function rnd(seed) {
  let s = seed|0;
  return function() {
    s = (s * 1664525 + 1013904223) | 0;
    return ((s >>> 0) / 4294967295);
  }
}

// Utilities
function clamp(v,a,b){ return Math.max(a,Math.min(b,v)); }
function lerp(a,b,t){ return a + (b-a)*t; }
function smoothstep(t){ return t*t*(3-2*t); }
function easeInOut(t){ return t<0.5? 2*t*t : -1 + (4-2*t)*t; }
function deg(d){ return d*Math.PI/180; }

// Projection: simple 2.5D - objects scale and y shift with z
function project(worldX, groundY, z, sizeScale=1) {
  // z range roughly 0 (back) to 1 (front)
  const zClamp = clamp(z, 0, 1);
  const perspectiveScale = lerp(0.6, 1.05, zClamp); // further back smaller
  const screenX = worldX;
  const screenY = groundY - (1 - perspectiveScale) * 60 * sizeScale;
  return { x: screenX, y: screenY, s: perspectiveScale * sizeScale };
}

// Draw grounded soft shadow under footprint/body
function drawContactShadow(x,y,s,alpha=0.28, rx=32, ry=12) {
  ctx.save();
  ctx.translate(x, y+6);
  ctx.scale(s, 1);
  const g = ctx.createLinearGradient(-rx,0,rx,0);
  g.addColorStop(0, `rgba(0,0,0,0)`);
  g.addColorStop(0.3, `rgba(0,0,0,${alpha*0.85})`);
  g.addColorStop(1, `rgba(0,0,0,0)`);
  ctx.fillStyle = g;
  ctx.beginPath();
  ctx.ellipse(0,0,rx,ry,0,0,Math.PI*2);
  ctx.fill();
  ctx.restore();
}

// Draw the doghouse with a dark interior; we will apply occlusion by drawing a front frame
function drawHouse() {
  // back wall
  ctx.fillStyle = '#795033';
  roundRect(ctx, house.x, house.y, house.w, house.h, 8);
  ctx.fill();

  // roof
  ctx.fillStyle = '#5a3a22';
  ctx.beginPath();
  ctx.moveTo(house.x - 20, house.y);
  ctx.lineTo(house.x + house.w/2, house.y - 46);
  ctx.lineTo(house.x + house.w + 20, house.y);
  ctx.closePath();
  ctx.fill();

  // entrance cutout: darker interior
  const doorX = house.x + (house.w - house.doorW)/2;
  const doorY = house.y + house.h - house.doorH - 10;
  ctx.fillStyle = '#24180f';
  roundRect(ctx, doorX, doorY, house.doorW, house.doorH, 8);
  ctx.fill();

  // inner floor slight lighter
  ctx.fillStyle = '#2b1d12';
  ctx.fillRect(doorX, doorY + house.doorH - house.floorDepth, house.doorW, house.floorDepth);

  // front trim (to occlude when dog is inside)
  ctx.fillStyle = house.trim;
  roundRect(ctx, doorX-6, doorY-6, house.doorW+12, house.doorH+12, 10);
  ctx.fill();
}

// Helper for rounded rect
function roundRect(ctx,x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}

// Single bone draw: oriented by angle, positioned in world coordinates relative to dog mouth
function drawBone(screen, angle, color, scale=1) {
  ctx.save();
  ctx.translate(screen.x, screen.y);
  ctx.rotate(angle);
  ctx.scale(scale, scale);
  // central bar
  ctx.fillStyle = color;
  roundedRect(ctx, -boneModel.length/2, -6, boneModel.length, 12, 6);
  ctx.fill();
  // ends - little knobs
  ctx.beginPath();
  ctx.ellipse(-boneModel.length/2 - 10, 0, 12, 10, 0, 0, Math.PI*2);
  ctx.fill();
  ctx.beginPath();
  ctx.ellipse(boneModel.length/2 + 10, 0, 12, 10, 0, 0, Math.PI*2);
  ctx.fill();

  // subtle shading
  const g = ctx.createLinearGradient(-boneModel.length/2, -8, boneModel.length/2, 8);
  g.addColorStop(0, 'rgba(255,255,255,0.07)');
  g.addColorStop(1, 'rgba(0,0,0,0.06)');
  ctx.fillStyle = g;
  roundedRect(ctx, -boneModel.length/2, -6, boneModel.length, 12, 6);
  ctx.fill();

  ctx.restore();
}

function roundedRect(ctx,x,y,w,h,r){ ctx.beginPath(); ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath(); }

// Dog drawing: hierarchical parts. We compute world positions of joints then paint layered fur strokes and shapes.
function drawDog(state) {
  // state contains world position (x,z), orientation ang (radians) facing right=0 toward house
  // body geometry base
  const seed = 12345; // deterministic
  const rndGen = rnd(seed);

  const worldX = state.x;
  const z = state.z;
  const proj = project(worldX, dogBase.startYGround, z, state.scale);
  const baseS = proj.s;
  // Spine endpoints: chest and pelvis in world coords (x offsets)
  const spineLen = 86 * baseS;
  const spineAngle = state.bodyAngle; // rad, positive rotates clockwise (down)
  // chest point is slightly forward of worldX
  const chest = { x: worldX + Math.cos(spineAngle) * 14 * baseS, y: proj.y + Math.sin(spineAngle) * 6 * baseS };
  const pelvis = { x: chest.x - Math.cos(spineAngle) * spineLen, y: chest.y - Math.sin(spineAngle) * spineLen };

  // head pivot sits forward of chest
  const headPivot = { x: chest.x + Math.cos(spineAngle) * 32 * baseS, y: chest.y + Math.sin(spineAngle) * 12 * baseS };

  // compute legs: four legs positions relative to chest/pelvis
  function legIK(hipX, hipY, phase, reach, stance) {
    // simplified two-segment leg using phase for foot target anterior/posterior
    const stepForward = Math.cos(phase) * reach * baseS;
    const footX = hipX + stepForward;
    const footY = dogBase.startYGround - (1 - proj.s)*40; // ground Y projected slightly
    // height lift
    const lift = Math.max(0, Math.sin(phase)) * 18 * baseS * stance;
    return { hipX, hipY, footX, footY - lift };
  }

  // legs hip positions
  const hipOffsetForward = 18 * baseS;
  const hipOffsetBack = 58 * baseS;
  const hipL = { x: chest.x - Math.cos(spineAngle) * 6 * baseS - Math.sin(spineAngle)*14*baseS, y: chest.y - Math.sin(spineAngle)*6*baseS + Math.cos(spineAngle)*14*baseS };
  const hipR = { x: hipL.x, y: hipL.y + 0.1 }; // small lateral offset ignored
  const hipBack = { x: pelvis.x + Math.cos(spineAngle)*12*baseS, y: pelvis.y + Math.sin(spineAngle)*12*baseS };

  // leg cycles: based on gaitPhase
  const gp = state.gaitPhase;
  const lf = legIK(chest.x, chest.y + 14*baseS, gp + Math.PI*1.0, 28, 1.0); // front-left
  const rf = legIK(chest.x, chest.y + 14*baseS, gp + Math.PI*0.0, 28, 1.0); // front-right
  const lb = legIK(pelvis.x, pelvis.y + 14*baseS, gp + Math.PI*0.0, 30, 1.0); // back-left
  const rb = legIK(pelvis.x, pelvis.y + 14*baseS, gp + Math.PI*1.0, 30, 1.0); // back-right

  // head pose compute
  const headAngle = state.headAngle;
  const headScale = 0.9 * baseS;
  const headCenter = { x: headPivot.x + Math.cos(headAngle) * 18 * headScale, y: headPivot.y + Math.sin(headAngle) * 6 * headScale };

  // tail - attached to pelvis, curls over back
  const tailBase = { x: pelvis.x - Math.cos(spineAngle)*8*baseS, y: pelvis.y - Math.sin(spineAngle)*8*baseS };
  const tailCurl = state.tailCurl; // angle relative to body
  const tailLength = 46 * baseS;

  // Draw contact shadows first to ground dog
  // Use a single big shadow beneath belly and smaller for feet in contact
  drawContactShadow(proj.x, proj.y + 28 * baseS, 0.9 + state.z*0.3, 0.26, 56*baseS, 18*baseS);
  // foot small shadows placed at foot positions
  [lf, rf, lb, rb].forEach(f => {
    drawContactShadow(f.footX, f.footY + 6, 0.9 + state.z*0.3, 0.22, 18*baseS, 8*baseS);
  });

  // BODY LAYERS: draw rear fur first, then limbs, then torso, head, facial details, bone overlays, then front trim
  // To simulate depth as dog moves through doorway, callers will control global composite order.

  // Rear fur (pelvis region)
  ctx.save();
  ctx.translate(0,0);
  // fur color palette
  const coatA = '#f1b16b';
  const coatB = '#f8e3c5';
  const ruff = '#ffd99f';
  // draw main torso ellipse
  ctx.fillStyle = coatA;
  ctx.beginPath();
  ctx.ellipse(pelvis.x, pelvis.y, 44*baseS, 30*baseS, spineAngle, 0, Math.PI*2);
  ctx.fill();

  // draw layered directional fur strokes around torso
  ctx.lineCap = 'round';
  for (let i=0;i<60;i++) {
    const t = i/59;
    const ang = spineAngle - Math.PI/2 + (t-0.5)*0.6;
    const len = lerp(12,28, 1-Math.abs(t-0.5)*2) * baseS;
    const ox = pelvis.x + Math.cos(spineAngle+ (t-0.5)*0.8) * 36*baseS * (0.6 + t*0.5);
    const oy = pelvis.y + Math.sin(spineAngle+ (t-0.5)*0.8) * 22*baseS * (0.6 + t*0.5);
    ctx.strokeStyle = (i%6===0) ? coatB : coatA;
    ctx.lineWidth = lerp(1.2, 3.2, Math.random());
    ctx.beginPath();
    ctx.moveTo(ox, oy);
    ctx.lineTo(ox + Math.cos(ang)*len, oy + Math.sin(ang)*len);
    ctx.stroke();
  }
  ctx.restore();

  // Legs: draw back pair then front pair as they are behind/forward depending on x depth
  function drawLeg(leg, isFront) {
    ctx.save();
    ctx.strokeStyle = '#b86f2f';
    ctx.fillStyle = '#b86f2f';
    ctx.lineWidth = 6 * baseS;
    // upper leg (thigh) from hip to knee approximate
    const hipX = isFront ? chest.x : pelvis.x;
    const hipY = isFront ? chest.y + 14*baseS : pelvis.y + 14*baseS;
    const kneeX = lerp(hipX, leg.footX, 0.46);
    const kneeY = lerp(hipY, leg.footY, 0.46);
    ctx.beginPath();
    ctx.lineCap = 'round';
    ctx.moveTo(hipX, hipY);
    ctx.lineTo(kneeX, kneeY);
    ctx.stroke();
    // lower leg
    ctx.lineWidth = 5 * baseS;
    ctx.beginPath();
    ctx.moveTo(kneeX, kneeY);
    ctx.lineTo(leg.footX, leg.footY);
    ctx.stroke();
    // paw
    ctx.fillStyle = '#3b2a1b';
    ctx.beginPath();
    ctx.ellipse(leg.footX, leg.footY + 4*baseS, 8*baseS, 6*baseS, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  // Decide draw order by x to allow simple occlusion when inside doorway - caller handles major occlusion.
  // Draw back legs:
  drawLeg(lb, false);
  drawLeg(rb, false);

  // Torso mid: belly / ruff
  ctx.save();
  // main belly
  ctx.fillStyle = coatB;
  ctx.beginPath();
  ctx.ellipse((pelvis.x + chest.x)/2 - Math.cos(spineAngle)*6*baseS, (pelvis.y + chest.y)/2 - Math.sin(spineAngle)*6*baseS, 54*baseS, 34*baseS, spineAngle, 0, Math.PI*2);
  ctx.fill();

  // neck ruff near chest
  const ruffX = chest.x + Math.cos(spineAngle)*8*baseS;
  const ruffY = chest.y + Math.sin(spineAngle)*6*baseS;
  ctx.fillStyle = ruff;
  for (let i=0;i<22;i++) {
    const a = spineAngle - 0.8 + i*(1.6/21);
    const rx = ruffX + Math.cos(a) * lerp(10,26, Math.abs(i-11)/11) * baseS;
    const ry = ruffY + Math.sin(a) * lerp(6,22, Math.abs(i-11)/11) * baseS;
    ctx.beginPath();
    ctx.ellipse(rx, ry, lerp(6,20, Math.abs(i-11)/11)*baseS, lerp(4,14, Math.abs(i-11)/11)*baseS, a+Math.PI/6, 0, Math.PI*2);
    ctx.fill();
  }
  ctx.restore();

  // Front legs
  drawLeg(lf, true);
  drawLeg(rf, true);

  // Tail: curling plume over back
  ctx.save();
  ctx.translate(tailBase.x, tailBase.y);
  ctx.rotate(tailCurl);
  // tail strokes
  const tailS = tailLength;
  for (let i=0;i<16;i++){
    const t = i/15;
    ctx.beginPath();
    ctx.strokeStyle = (i%3===0)? coatB : coatA;
    ctx.lineWidth = lerp(6, 18, 1-t) * baseS;
    ctx.moveTo(Math.cos(0)*t*10*baseS, Math.sin(0)*t*4*baseS);
    ctx.lineTo(Math.cos(0)*(t*tailS + 8) , Math.sin(0)*(t*8));
    ctx.stroke();
  }
  ctx.restore();

  // Head & face
  ctx.save();
  ctx.translate(headCenter.x, headCenter.y);
  ctx.rotate(headAngle);
  const headW = 46 * headScale;
  const headH = 40 * headScale;
  // main head volume
  ctx.fillStyle = coatA;
  ctx.beginPath();
  ctx.ellipse(0, 0, headW, headH, 0, 0, Math.PI*2);
  ctx.fill();
  // muzzle
  ctx.fillStyle = '#e7c397';
  ctx.beginPath();
  ctx.ellipse(headW*0.18, headH*0.06, 18*headScale, 14*headScale, 0, 0, Math.PI*2);
  ctx.fill();

  // ears (upright triangles) - left & right
  ctx.fillStyle = '#d07a31';
  ctx.beginPath();
  ctx.moveTo(-18*headScale, -18*headScale);
  ctx.lineTo(-32*headScale, -42*headScale);
  ctx.lineTo(-6*headScale, -30*headScale);
  ctx.closePath();
  ctx.fill();
  ctx.beginPath();
  ctx.moveTo(18*headScale, -18*headScale);
  ctx.lineTo(6*headScale, -42*headScale);
  ctx.lineTo(36*headScale, -30*headScale);
  ctx.closePath();
  ctx.fill();

  // eyes
  const eyeY = -2*headScale;
  ctx.fillStyle = '#2a1f18';
  ctx.beginPath();
  ctx.ellipse(-8*headScale, eyeY, 5*headScale, 6*headScale, 0, 0, Math.PI*2);
  ctx.fill();
  ctx.beginPath();
  ctx.ellipse(10*headScale, eyeY, 5*headScale, 6*headScale, 0, 0, Math.PI*2);
  ctx.fill();

  // small glossy highlight
  ctx.fillStyle = 'rgba(255,255,255,0.24)';
  ctx.beginPath();
  ctx.ellipse(-6*headScale, eyeY-2*headScale, 2*headScale, 3*headScale, 0, 0, Math.PI*2);
  ctx.fill();

  // nose
  ctx.fillStyle = '#1e1410';
  ctx.beginPath();
  ctx.ellipse(18*headScale, 6*headScale, 5*headScale, 4*headScale, 0, 0, Math.PI*2);
  ctx.fill();

  // mouth region: draw a slight gap if biting
  const bite = state.biting; // 0..1 for open mouth
  // lower jaw
  ctx.save();
  ctx.translate(8*headScale, 12*headScale);
  ctx.rotate( bite * 0.08 );
  ctx.fillStyle = '#b16a39';
  ctx.beginPath();
  ctx.ellipse(0, 6*headScale, 16*headScale, 8*headScale, 0, 0, Math.PI);
  ctx.fill();
  ctx.restore();

  ctx.restore(); // head local

  // If bone is held by mouth, caller will draw bone subsequent to head so it overlays correctly.

  // slight breathing movement: subtle canvas translation for coat fluff near chest
  ctx.save();
  ctx.globalAlpha = 0.7;
  for (let i=0;i<6;i++){
    ctx.fillStyle = i%2? coatB : coatA;
    const off = (Math.sin(state.time*2 + i)*1.8) * baseS;
    ctx.beginPath();
    ctx.ellipse(chest.x - Math.cos(spineAngle)*8*baseS + off, chest.y - Math.sin(spineAngle)*6*baseS + i*2*baseS, 18*baseS - i*1.8*baseS, 10*baseS - i*1.2*baseS, spineAngle, 0, Math.PI*2);
    ctx.fill();
  }
  ctx.restore();

  // expose some diagnostic points if needed (disabled)
  // returning mouth world pos for bone placement
  const mouthWorld = {
    x: headCenter.x + Math.cos(headAngle)*22*headScale,
    y: headCenter.y + Math.sin(headAngle)*8*headScale
  };

  return { mouthWorld, headAngle, headScale };
}

// Compute dog state over time
function dogStateAt(t) {
  // clamp
  const tc = clamp(t, 0, DURATION);
  // timeline phases
  // 0-1 establish: dog at start noticing bone -> slight head turn
  // 1-3 run toward house
  // 3-4.5 enter through doorway
  // 4.5-6 bite (head lowers opens mouth)
  // 6-8 back out
  // 8-10 turn toward viewer
  // 10-12 settle front pose

  // compute path X and Z as functions
  // start at dogBase.startX, approach house x near house.x - some margin
  const startX = dogBase.startX;
  const approachX = house.x + house.w/2 - 22; // house doorway center - 22
  const insideX = approachX + 10; // slightly inside
  const exitX = approachX - 6;

  let x = startX;
  let z = dogBase.startZ;
  let bodyAngle = 0; // small pitch
  let headAngle = 0;
  let tailCurl = -0.9; // upwards over back (negative to curl back)
  let gaitPhase = 0;
  let scale = 1.0;
  let biting = 0;
  let time = tc;

  if (tc < 1.0) {
    // noticing: slight attention toward house
    const p = smoothstep(tc/1.0);
    x = startX;
    z = lerp(dogBase.startZ, dogBase.startZ, p);
    bodyAngle = lerp(0.02, 0.0, p);
    headAngle = lerp(0.12, 0.06, p) ; // turning slightly
    gaitPhase = 0;
    tailCurl = -0.75;
  } else if (tc < 3.0) {
    const p = (tc - 1.0)/2.0;
    const sp = easeInOut(p);
    x = lerp(startX, approachX - 8, sp);
    // run: lower body angle, faster gait
    bodyAngle = lerp(0.06, 0.00, sp);
    headAngle = lerp(0.06, -0.02, sp);
    tailCurl = lerp(-0.7, -1.0, sp);
    // gait phase drives leg cycles; faster
    gaitPhase = (tc - 1.0) * 10.0; // speed up
    z = lerp(dogBase.startZ, dogBase.startZ + 0.04, sp);
    scale = lerp(1.0, 1.02, sp);
  } else if (tc < 4.5) {
    // entering 3-4.5: position crosses door threshold; z becomes smaller (goes back into house)
    const p = (tc - 3.0)/1.5;
    const sp = smoothstep(p);
    x = lerp(approachX - 8, insideX, sp);
    z = lerp(dogBase.startZ + 0.04, house.doorW? 0.16:0.18, sp);
    bodyAngle = lerp(0.0, -0.06, sp); // pitch slightly when entering
    headAngle = lerp(-0.02, 0.12, sp); // head moving inside to look down at bone
    tailCurl = lerp(-1.0, -0.5, sp);
    gaitPhase = (tc - 3.0)*6;
    scale = lerp(1.02, 0.98, sp);
  } else if (tc < 4.5 + 0.0001) {
    // guard exact boundary
  } else if (tc < 6.0) {
    // 4.5-6 biting: mostly stationary at inner threshold, head moves down, mouth opens
    const p = clamp((tc - 4.5) / 1.5, 0, 1);
    x = lerp(insideX, insideX, p);
    z = lerp(0.175, 0.165, p);
    bodyAngle = -0.06;
    headAngle = lerp(0.12, 0.28, p);
    tailCurl = -0.45;
    gaitPhase = (tc - 4.5) * 1.8;
    biting = smoothstep(p);
    scale = 0.98;
  } else if (tc < 8.0) {
    // 6-8 backing out with bone
    const p = clamp((tc - 6.0) / 2.0, 0, 1);
    const sp = smoothstep(p);
    x = lerp(insideX, exitX, sp);
    z = lerp(0.165, dogBase.startZ + 0.02, sp);
    bodyAngle = lerp(-0.04, 0.02, sp);
    headAngle = lerp(0.28, 0.06, sp);
    tailCurl = lerp(-0.4, -0.95, sp);
    gaitPhase = (tc - 6.0) * 6.0; // backward stepping rhythm
    biting = 1.0;
    scale = lerp(0.98, 1.01, sp);
  } else if (tc < 10.0) {
    // 8-10 turning to face viewer
    const p = clamp((tc - 8.0)/2.0, 0, 1);
    const sp = smoothstep(p);
    // move slightly left to center frame as it turns
    x = lerp(exitX, 420, sp);
    z = lerp(dogBase.startZ + 0.02, 0.42, sp); // come forward for front-facing
    // body rotates: initially facing right (0 rad), finally facing viewer (pi/2)
    // We'll represent orientation by bodyAngle (pitch) and heading by a heading variable
    bodyAngle = lerp(0.02, 0.0, sp);
    // To simulate body rotation we will shift headAngle and tailCurl; actual frontal body drawn by dog renderer via mouth/head positions
    headAngle = lerp(0.06, 0.0, sp);
    tailCurl = lerp(-0.95, -1.4, sp);
    gaitPhase = (tc - 8.0) * 3.5;
    biting = 1.0;
    scale = lerp(1.01, 1.06, sp);
  } else {
    // 10-12 final settle facing viewer: hold pose
    const p = clamp((tc - 10.0) / 2.0, 0, 1);
    x = lerp(420, 420, p);
    z = lerp(0.42, 0.44, p);
    bodyAngle = 0.0;
    headAngle = 0.0;
    tailCurl = -1.45;
    gaitPhase = (tc - 10.0)*1.2;
    biting = 1.0;
    scale = 1.08;
  }

  // For turning 8-10 we also need to rotate facing direction from right to frontal.
  // Represent heading progress as headingT 0..1 where 0=side view facing right, 1=front.
  let headingT;
  if (tc < 8.0) headingT = 0;
  else headingT = clamp((tc - 8.0)/2.0, 0, 1);
  // We'll map this to a composite where the renderer uses headAngle and also adjusts x offsets for frontal look.
  // Compose final state
  const state = {
    x, z, bodyAngle, headAngle: headAngle - headingT*0.18, tailCurl: tailCurl + headingT* -0.35,
    gaitPhase: gaitPhase, scale: scale, biting: biting, time: time, headingT: headingT
  };
  return state;
}

// Main renderFrame synchronous function as required.
window.renderFrame = function(timeSeconds) {
  // deterministic clamp
  const t = (typeof timeSeconds === 'number') ? timeSeconds : 0;
  const tc = (t < 0) ? 0 : (t > DURATION ? DURATION : t);

  // Clear background: sky and yard
  ctx.fillStyle = '#88bcd6';
  ctx.fillRect(0,0,W,H);

  // distant fence and plants
  ctx.fillStyle = '#5b8a3a';
  ctx.fillRect(0, yard.backFenceY, W, 40);
  // ground
  const grd = ctx.createLinearGradient(0, yard.groundY, 0, H);
  grd.addColorStop(0, '#7bb86b');
  grd.addColorStop(1, '#5f8c4e');
  ctx.fillStyle = grd;
  ctx.fillRect(0, yard.groundY, W, H - yard.groundY);

  // small path toward doghouse
  ctx.save();
  ctx.fillStyle = '#d7cfae';
  ctx.beginPath();
  ctx.ellipse(house.x + house.w/2, yard.groundY + 22, 120, 32, 0, 0, Math.PI*2);
  ctx.fill();
  ctx.restore();

  // draw house back and interior
  drawHouse();

  // Bone position initial resting location: inside house near floor center. We'll move it once bitten.
  // Determine bone world position based on timeline: before grab bone stays on floor; after grabbed it will follow mouth.
  // Initial bone coords:
  const doorCenterX = house.x + house.w/2;
  const doorFloorY = house.y + house.h - 12;
  const boneStartWorld = { x: doorCenterX + 6, y: doorFloorY - 8, z: 0.165 };
  // Dog state
  const dstate = dogStateAt(tc);

  // Draw scene elements in depth: choose ordering:
  // 1) items behind dog in deeper z (house interior behind)
  // 2) dog parts behind doorway trim when entering: to achieve correct occlusion when dog is inside, we draw dog, then draw the house front trim which we've already drawn as part of drawHouse. To manage occlusion properly we should overlay front trim after drawing dog, but earlier drawHouse already drew trim. To ensure correct occlusion across timeline, we will redraw the front trim last to occlude appropriately.
  // So we will: redraw house interior (done), then draw dog, then bone if inside mouth (after), then draw house front trim overlay.

  // Decide if bone has been picked up (time >= 4.5 bite event begins at 4.5; bone leaves original resting once bitten sometime 4.5-6)
  // We'll move bone to mouth starting at tGrab = 4.6 -> by 4.8 it's attached rigidly. Use small offset.
  const tGrabStart = 4.5;
  const tGrabEnd = 4.9;
  let boneWorldPos = { ...boneStartWorld };
  let boneHeld = false;
  if (tc < tGrabStart) {
    boneHeld = false;
    boneWorldPos = boneStartWorld;
  } else if (tc < tGrabEnd) {
    // transitioning: bone moves as dog grabs it, we lerp position toward mouth
    const p = smoothstep((tc - tGrabStart)/(tGrabEnd - tGrabStart));
    // compute approximate mouth world pos at this subtime by evaluating dog state slightly earlier
    const interimState = dogStateAt(clamp(tc, 0, DURATION));
    // get mouth pos by calling dog drawing helper but not drawing (we call drawDog later to get mouth).
    const mouthInfo = (function(){ return null; })();
    // we need mouthWorld - easiest: approximate as head pivot forward of chest
    const approxState = dstate;
    // approximate mouth world:
    const mouthApproxX = approxState.x + Math.cos(approxState.headAngle) * 26 * approxState.scale;
    const mouthApproxY = dogBase.startYGround - (1 - approxState.z)*60 + Math.sin(approxState.headAngle)*16*approxState.scale;
    boneWorldPos.x = lerp(boneStartWorld.x, mouthApproxX, p);
    boneWorldPos.y = lerp(boneStartWorld.y, mouthApproxY, p);
    boneWorldPos.z = lerp(boneStartWorld.z, approxState.z, p);
    boneHeld = p > 0.6;
  } else {
    boneHeld = true;
    // bone follows mouth exactly
    // compute mouth world from dog state
    // To get accurate mouth pos we will call drawDog but we need mouthWorld returned. Instead, call a helper that computes mouth without drawing: replicate logic.
    const st = dstate;
    const proj = project(st.x, dogBase.startYGround, st.z, st.scale);
    const spineLen = 86 * proj.s;
    const spineAngle = st.bodyAngle;
    const chest = { x: st.x + Math.cos(spineAngle) * 14 * proj.s, y: proj.y + Math.sin(spineAngle) * 6 * proj.s };
    const headPivot = { x: chest.x + Math.cos(spineAngle) * 32 * proj.s, y: chest.y + Math.sin(spineAngle) * 12 * proj.s };
    const headAngle = st.headAngle;
    const headScale = 0.9 * proj.s;
    const mouthX = headPivot.x + Math.cos(headAngle) * 22 * headScale;
    const mouthY = headPivot.y + Math.sin(headAngle) * 8 * headScale;
    boneWorldPos = { x: mouthX, y: mouthY, z: st.z };
  }

  // Draw the dog onto canvas with partial occlusion: to simulate going into doorway, if dog's x is past door center we should draw torso then re-draw front trim to occlude parts.
  // We'll calculate depth ordering: if dog is inside house (z <= 0.18) we will allow front trim to occlude lower torso.
  // Draw dog into an offscreen ordering by splitting some parts: rear parts, then house trim, then front parts.
  // To emulate that we will:
  // - Draw rear parts (pelvis, tail, back legs, rear fur)
  // - Draw house trim overlay if needed
  // - Draw front parts (belly, chest, front legs, head)
  // However drawDog currently draws everything. So instead replicate simplified ordering: draw back elements manually here.

  // To keep consistent look, we will call drawDog to draw everything except head, then draw bone, then head to ensure bone overlays mouth correctly.
  // But drawDog returned mouthWorld used earlier only when bone held. To keep simple: we'll call drawDog once to draw body, get mouth pos, then draw bone anchored to mouth if held. Then draw house front trim again to occlude overlap when inside.

  // Draw dog entire (body + head)
  const mouthInfo = drawDog({ x: dstate.x, z: dstate.z, bodyAngle: dstate.bodyAngle, headAngle: dstate.headAngle, tailCurl: dstate.tailCurl, gaitPhase: dstate.gaitPhase, scale: dstate.scale, biting: dstate.biting, time: dstate.time });

  // Draw bone either resting on floor (if not held) or at mouth position (held)
  if (!boneHeld) {
    // project boneStartWorld to screen
    const boneProj = project(boneWorldPos.x, boneWorldPos.y, boneWorldPos.z, 1.0);
    // angle oriented flat
    drawBone(boneProj, deg(-10), boneModel.color, 1.0);
  } else {
    // bone attached to mouth: get mouth pos and orientation aligned with head angle
    const boneScreen = project(boneWorldPos.x, boneWorldPos.y, boneWorldPos.z, 1.0);
    // bone angle: align with head angle
    const angle = mouthInfo.headAngle + 0.12;
    drawBone(boneScreen, angle, boneModel.color, 1.0);
  }

  // Now re-draw house front trim to occlude parts when dog inside (ensure consistent occlusion)
  // Front trim rectangle over doorway:
  const doorX = house.x + (house.w - house.doorW)/2;
  const doorY = house.y + house.h - house.doorH - 10;
  ctx.fillStyle = house.trim;
  roundRect(ctx, doorX-6, doorY-6, house.doorW+12, house.doorH+12, 10);
  ctx.fill();

  // Small contact shadows for dog's paws and final stance slightly updated for t>=12 clamp
  // If final pose hold, add tiny breathing / ear twitch: draw subtle highlight on coat
  if (tc >= 10) {
    // gentle breathing: tiny vertical bob of chest and tail shimmer already in drawDog
    // draw highlight over ruff
    ctx.save();
    ctx.globalAlpha = 0.12;
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.ellipse(dstate.x + 8, dogBase.startYGround - 160 * dstate.scale, 40 * dstate.scale, 12 * dstate.scale, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  // For t < 1 show subtle attention: small ear perk up (already in drawDog head)
  // End of frame - deterministic output
};

window.renderFrame(0);

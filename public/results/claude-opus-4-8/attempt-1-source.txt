<script>
const cv=document.getElementById('stage');
const ctx=cv.getContext('2d');
const W=960,H=540,groundY=430;

function lerp(a,b,u){return a+(b-a)*u;}
function clamp(v,a,b){return v<a?a:v>b?b:v;}
function smooth(x,a,b){let u=clamp((x-a)/(b-a),0,1);return u*u*(3-2*u);}
function rnd(seed){let x=Math.sin(seed*12.9898)*43758.5453;return x-Math.floor(x);}

// ---------- TIMELINE ----------
function getState(t){
  let s={cx:130,lift:0,crouch:0,tilt:0,legPhase:(t*2.2)%1,mode:'idle',
         headYaw:0,headPitch:0,mouth:0,bone:false,frontal:0,facing:1,breathe:0};
  if(t<0)t=0;
  if(t<1){
    s.cx=130;s.mode='idle';s.headYaw=smooth(t,0.3,0.9)*0.35;s.breathe=Math.sin(t*2)*0.5;
  } else if(t<1.6){
    let u=(t-1)/0.6;s.cx=lerp(130,382,u);s.mode='run';s.legPhase=(t*3.1)%1;s.headYaw=0.1;
  } else if(t<1.85){
    let u=(t-1.6)/0.25;s.cx=lerp(382,432,u);s.mode='load';s.crouch=smooth(u,0,1);s.tilt=-0.06;
  } else if(t<2.45){
    let u=(t-1.85)/0.6;s.cx=lerp(432,596,u);s.mode='air';
    s.lift=Math.sin(u*Math.PI)*98;s.tilt=lerp(-0.14,0.14,u);
  } else if(t<2.72){
    let u=(t-2.45)/0.27;s.cx=lerp(596,626,u);s.mode='land';s.crouch=Math.sin(u*Math.PI)*0.85;s.tilt=0.06*(1-u);
  } else if(t<3.1){
    let u=(t-2.72)/0.38;s.cx=lerp(626,702,u);s.mode='run';s.legPhase=(t*3)%1;
  } else if(t<4.5){
    let u=(t-3.1)/1.4;s.cx=lerp(702,770,u);s.mode='walk';s.legPhase=(t*1.8)%1;
  } else if(t<6){
    let u=(t-4.5)/1.5;s.cx=770;s.mode='stand';
    let down=smooth(u,0,0.4),up=smooth(u,0.62,1);
    s.headPitch=0.95*down-0.95*up;
    s.mouth=(u>0.25&&u<0.7)?1:0;
    s.bone=u>0.56;
  } else if(t<8){
    let u=(t-6)/2;s.cx=lerp(770,688,u);s.mode='back';s.bone=true;s.legPhase=(t*1.8)%1;s.headPitch=0.05;
  } else if(t<10){
    let u=(t-8)/2;s.cx=lerp(688,662,u);s.frontal=smooth(u,0,1);s.bone=true;s.mode='turn';
    s.headYaw=lerp(0.1,0,u);
  } else {
    let u=(t-10);s.cx=662;s.frontal=1;s.bone=true;s.mode='front';
    s.breathe=Math.sin(u*1.8)*0.5;s.headYaw=Math.sin(u*0.7)*0.04;
  }
  return s;
}

// ---------- LEG IK ----------
function ikKnee(hx,hy,fx,fy,a,b,bend){
  let dx=fx-hx,dy=fy-hy,d=Math.hypot(dx,dy);
  d=clamp(d,0.01,a+b-0.01);
  let base=Math.atan2(dy,dx);
  let ca=clamp((a*a+d*d-b*b)/(2*a*d),-1,1);
  let A=Math.acos(ca);
  let ka=base+bend*A;
  return[hx+a*Math.cos(ka),hy+a*Math.sin(ka)];
}
function drawLeg(hx,hy,fx,fy,bend,wTop,wBot,col,colD){
  let[kx,ky]=ikKnee(hx,hy,fx,fy,27,29,bend);
  ctx.lineCap='round';
  ctx.strokeStyle=col;
  ctx.lineWidth=wTop;ctx.beginPath();ctx.moveTo(hx,hy);ctx.lineTo(kx,ky);ctx.stroke();
  ctx.lineWidth=wBot;ctx.beginPath();ctx.moveTo(kx,ky);ctx.lineTo(fx,fy);ctx.stroke();
  // paw
  ctx.fillStyle=colD;ctx.beginPath();ctx.ellipse(fx,fy,wBot*0.65,wBot*0.5,0,0,7);ctx.fill();
}

// ---------- FUR BLOB ----------
function furBlob(cx,cy,rx,ry,base,light,seed){
  let g=ctx.createRadialGradient(cx-rx*0.3,cy-ry*0.4,ry*0.2,cx,cy,Math.max(rx,ry));
  g.addColorStop(0,light);g.addColorStop(0.6,base);g.addColorStop(1,base);
  ctx.fillStyle=g;ctx.beginPath();ctx.ellipse(cx,cy,rx,ry,0,0,7);ctx.fill();
  // tufts
  let n=42;
  for(let i=0;i<n;i++){
    let a=i/n*6.283;
    let r1=rnd(seed+i)*0.5+0.6;
    let ex=cx+Math.cos(a)*rx*r1, ey=cy+Math.sin(a)*ry*r1;
    let len=4+rnd(seed+i*1.7)*6;
    let ox=ex+Math.cos(a)*len, oy=ey+Math.sin(a)*len;
    ctx.strokeStyle=(i%3==0)?light:base;
    ctx.lineWidth=1.6;ctx.lineCap='round';
    ctx.beginPath();ctx.moveTo(ex,ey);ctx.lineTo(ox,oy);ctx.stroke();
  }
}

// ---------- DOG SIDE ----------
function drawDogSide(cx,s,alpha,sx){
  const base='#d4873a',light='#f1c988',dark='#a05d1f',cream='#f5e3c0';
  ctx.save();
  ctx.globalAlpha=alpha;
  ctx.translate(cx,0);
  ctx.scale(sx,1);
  ctx.rotate(s.tilt*0.0);
  let bob=s.mode==='run'?Math.sin(s.legPhase*12.566)*2:0;
  let bodyY=groundY-48-s.lift-s.crouch*9+bob+s.breathe;
  ctx.translate(0,0);
  ctx.rotate(s.tilt);

  // feet targets
  let fnH=[22,bodyY-6],frH=[-24,bodyY-2];
  let feet={};
  function gaitFoot(hip,ph,stride,lift){
    let p=(s.legPhase+ph)%1;
    let fx=hip[0]+Math.cos(p*6.283)*stride;
    let lf=Math.max(0,Math.sin(p*6.283));
    let fy=groundY-s.crouch*9-lf*lift;
    return[fx,fy];
  }
  if(s.mode==='run'){
    feet.fn=gaitFoot(fnH,0,16,18);feet.ff=gaitFoot(fnH,0.12,16,15);
    feet.rn=gaitFoot(frH,0.5,17,18);feet.rf=gaitFoot(frH,0.62,17,15);
  } else if(s.mode==='air'){
    let u=clamp((s.tilt+0.14)/0.28,0,1);
    feet.fn=[28,bodyY+6];feet.ff=[33,bodyY+2];
    feet.rn=[-30,bodyY+8];feet.rf=[-35,bodyY+4];
  } else if(s.mode==='load'){
    feet.fn=[20,groundY];feet.ff=[26,groundY];feet.rn=[-28,groundY+ -2];feet.rf=[-34,groundY];
  } else if(s.mode==='land'){
    feet.fn=[24,groundY];feet.ff=[30,groundY];feet.rn=[-26,groundY];feet.rf=[-32,groundY];
  } else if(s.mode==='back'||s.mode==='walk'){
    let dir=s.mode==='back'?-1:1;
    feet.fn=gaitFoot(fnH,0,10*1,10);feet.ff=gaitFoot(fnH,0.3,10,8);
    feet.rn=gaitFoot(frH,0.5,10,10);feet.rf=gaitFoot(frH,0.8,10,8);
  } else {
    feet.fn=[20,groundY];feet.ff=[26,groundY];feet.rn=[-26,groundY];feet.rf=[-32,groundY];
  }

  // far legs (behind body)
  drawLeg(fnH[0],fnH[1],feet.ff[0],feet.ff[1],1,9,7,dark,'#8a4f18');
  drawLeg(frH[0],frH[1],feet.rf[0],feet.rf[1],-1,10,7,dark,'#8a4f18');

  // tail plume (behind, curled over back)
  ctx.save();
  let tw=Math.sin(s.legPhase*6.283)* (s.mode==='run'?6:2) + s.breathe*2;
  furBlob(-38,bodyY-30+tw*0.3, 24,20, base,light, 71);
  furBlob(-30,bodyY-40+tw*0.4, 20,16, light,cream, 73);
  ctx.restore();

  // body
  furBlob(-2,bodyY-2, 38,30, base,light, 11);
  // chest/ruff
  furBlob(24,bodyY+2, 24,26, base,light, 23);

  // near legs (front of body)
  drawLeg(frH[0],frH[1],feet.rn[0],feet.rn[1],-1,11,8,base,dark);
  drawLeg(fnH[0],fnH[1],feet.fn[0],feet.fn[1],1,10,8,base,dark);

  // neck ruff
  furBlob(30,bodyY-14, 20,20, light,cream, 31);

  // head
  ctx.save();
  let hx=46,hy=bodyY-24+s.headPitch*34;
  ctx.translate(hx,hy);
  ctx.rotate(s.headPitch*0.5 - s.headYaw*0.2);
  // ears
  ctx.fillStyle=dark;
  ctx.beginPath();ctx.moveTo(-4,-14);ctx.lineTo(2,-30);ctx.lineTo(10,-14);ctx.closePath();ctx.fill();
  ctx.fillStyle=base;
  ctx.beginPath();ctx.moveTo(6,-13);ctx.lineTo(13,-28);ctx.lineTo(20,-11);ctx.closePath();ctx.fill();
  // head fluff
  furBlob(4,-2, 20,18, base,light, 51);
  // muzzle
  ctx.fillStyle=cream;
  ctx.beginPath();ctx.ellipse(18,2,12,9,0.1,0,7);ctx.fill();
  // mouth open
  if(s.mouth>0){ctx.fillStyle='#5a2a1a';ctx.beginPath();ctx.ellipse(24,6,6,4*s.mouth,0,0,7);ctx.fill();}
  // nose
  ctx.fillStyle='#241712';ctx.beginPath();ctx.ellipse(28,0,3.2,3,0,0,7);ctx.fill();
  // eye
  ctx.fillStyle='#1a120c';ctx.beginPath();ctx.ellipse(8,-3,2.8,3.2,0,0,7);ctx.fill();
  ctx.fillStyle='#fff';ctx.beginPath();ctx.ellipse(9,-4,0.9,0.9,0,0,7);ctx.fill();
  // bone in mouth
  if(s.bone){drawBone(30,4,0.1,0.8);}
  ctx.restore();

  ctx.restore();
}

// ---------- DOG FRONT ----------
function drawDogFront(cx,s,alpha){
  const base='#d4873a',light='#f1c988',dark='#a05d1f',cream='#f5e3c0';
  ctx.save();ctx.globalAlpha=alpha;ctx.translate(cx,0);
  let bodyY=groundY-44+s.breathe;
  let br=1+s.breathe*0.01;
  // tail peek
  furBlob(26,bodyY-18,16,14,light,cream,77);
  // front legs
  ctx.strokeStyle=base;ctx.lineCap='round';
  ctx.lineWidth=12;
  ctx.beginPath();ctx.moveTo(-12,bodyY+6);ctx.lineTo(-13,groundY);ctx.stroke();
  ctx.beginPath();ctx.moveTo(12,bodyY+6);ctx.lineTo(13,groundY);ctx.stroke();
  ctx.fillStyle=dark;
  ctx.beginPath();ctx.ellipse(-13,groundY,7,5,0,0,7);ctx.fill();
  ctx.beginPath();ctx.ellipse(13,groundY,7,5,0,0,7);ctx.fill();
  // body fluffy
  furBlob(0,bodyY,40*br,34,base,light,91);
  // chest ruff
  furBlob(0,bodyY+6,30,24,light,cream,93);
  // head
  let hx=0,hy=bodyY-28;
  // ears
  ctx.fillStyle=dark;
  ctx.beginPath();ctx.moveTo(hx-18,hy-6);ctx.lineTo(hx-24,hy-26);ctx.lineTo(hx-6,hy-12);ctx.closePath();ctx.fill();
  ctx.beginPath();ctx.moveTo(hx+18,hy-6);ctx.lineTo(hx+24,hy-26);ctx.lineTo(hx+6,hy-12);ctx.closePath();ctx.fill();
  ctx.fillStyle=base;
  ctx.beginPath();ctx.moveTo(hx-14,hy-6);ctx.lineTo(hx-19,hy-22);ctx.lineTo(hx-5,hy-11);ctx.closePath();ctx.fill();
  ctx.beginPath();ctx.moveTo(hx+14,hy-6);ctx.lineTo(hx+19,hy-22);ctx.lineTo(hx+5,hy-11);ctx.closePath();ctx.fill();
  // head fluff
  furBlob(hx,hy,22,20,base,light,95);
  // muzzle
  ctx.fillStyle=cream;ctx.beginPath();ctx.ellipse(hx,hy+8,11,9,0,0,7);ctx.fill();
  // nose
  ctx.fillStyle='#241712';ctx.beginPath();ctx.ellipse(hx,hy+6,3.4,2.8,0,0,7);ctx.fill();
  // eyes
  ctx.fillStyle='#1a120c';
  ctx.beginPath();ctx.ellipse(hx-8,hy-1,3,3.6,0,0,7);ctx.fill();
  ctx.beginPath();ctx.ellipse(hx+8,hy-1,3,3.6,0,0,7);ctx.fill();
  ctx.fillStyle='#fff';
  ctx.beginPath();ctx.ellipse(hx-7,hy-2.5,1,1,0,0,7);ctx.fill();
  ctx.beginPath();ctx.ellipse(hx+9,hy-2.5,1,1,0,0,7);ctx.fill();
  // bone horizontal across muzzle
  if(s.bone){drawBone(hx,hy+11,0,1.0);}
  ctx.restore();
}

function drawBone(x,y,rot,sc){
  ctx.save();ctx.translate(x,y);ctx.rotate(rot);ctx.scale(sc,sc);
  ctx.fillStyle='#efe6cf';ctx.strokeStyle='#cbbd9a';ctx.lineWidth=1.2;
  ctx.beginPath();ctx.roundRect(-13,-3,26,6,3);ctx.fill();ctx.stroke();
  for(let s1=-1;s1<=1;s1+=2){
    ctx.beginPath();ctx.ellipse(13*s1,-4,4,4,0,0,7);ctx.fill();ctx.stroke();
    ctx.beginPath();ctx.ellipse(13*s1,4,4,4,0,0,7);ctx.fill();ctx.stroke();
  }
  ctx.restore();
}

// ---------- SCENE ----------
function drawBackground(){
  // sky
  let g=ctx.createLinearGradient(0,0,0,groundY);
  g.addColorStop(0,'#bfe4f2');g.addColorStop(1,'#e8f3e0');
  ctx.fillStyle=g;ctx.fillRect(0,0,W,groundY);
  // distant fence
  ctx.fillStyle='#9fb98a';ctx.fillRect(0,groundY-70,W,18);
  ctx.fillStyle='#8aa677';
  for(let x=0;x<W;x+=22){ctx.fillRect(x,groundY-90,12,38);}
  // grass
  let gg=ctx.createLinearGradient(0,groundY,0,H);
  gg.addColorStop(0,'#6fae52');gg.addColorStop(1,'#4f8c3a');
  ctx.fillStyle=gg;ctx.fillRect(0,groundY,W,H-groundY);
  // grass texture
  for(let i=0;i<400;i++){
    let x=rnd(i)*W,y=groundY+rnd(i*1.3)*(H-groundY);
    ctx.strokeStyle=rnd(i*2.1)>0.5?'#5c9a44':'#79b85c';
    ctx.lineWidth=1;ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x+rnd(i*3)*2-1,y-3-rnd(i*4)*3);ctx.stroke();
  }
}

const puddle={cx:500,cy:groundY+22,rx:78,ry:16};
function drawPuddle(){
  ctx.save();
  ctx.fillStyle='rgba(60,90,110,0.5)';
  ctx.beginPath();ctx.ellipse(puddle.cx,puddle.cy,puddle.rx,puddle.ry,0,0,7);ctx.fill();
  let g=ctx.createLinearGradient(0,puddle.cy-puddle.ry,0,puddle.cy+puddle.ry);
  g.addColorStop(0,'#aecfe0');g.addColorStop(0.5,'#7ba6bf');g.addColorStop(1,'#5f8aa6');
  ctx.fillStyle=g;
  ctx.beginPath();ctx.ellipse(puddle.cx,puddle.cy,puddle.rx-4,puddle.ry-3,0,0,7);ctx.fill();
  ctx.restore();
}
function drawReflection(s){
  // only when dog over/near puddle
  if(s.cx<puddle.cx-180||s.cx>puddle.cx+180)return;
  ctx.save();
  ctx.beginPath();ctx.ellipse(puddle.cx,puddle.cy,puddle.rx-4,puddle.ry-3,0,0,7);ctx.clip();
  let axis=puddle.cy-6;
  ctx.translate(0,2*axis);ctx.scale(1,-1);
  ctx.globalAlpha=0.33;
  // bluish tint via composite
  drawDogSide(s.cx,s,0.33,1);
  ctx.restore();
}

function drawHouse_back(){
  // interior dark seen through doorway
  ctx.fillStyle='#2a211a';
  ctx.fillRect(695,groundY-96,90,96);
  // floor shading
  ctx.fillStyle='#3a2d22';
  ctx.beginPath();ctx.moveTo(695,groundY);ctx.lineTo(785,groundY);ctx.lineTo(778,groundY-14);ctx.lineTo(702,groundY-14);ctx.closePath();ctx.fill();
}
function drawBone_rest(){
  drawBone(766,groundY-8,-0.05,0.85);
}
function drawHouse_front(){
  // walls around doorway
  ctx.fillStyle='#b06a3e';
  ctx.fillRect(672,groundY-112,24,112); // left wall
  ctx.fillRect(785,groundY-112,58,112); // right wall
  ctx.fillRect(672,groundY-132,171,22); // top band
  // plank lines
  ctx.strokeStyle='#8f5330';ctx.lineWidth=1.5;
  for(let y=groundY-108;y<groundY;y+=14){
    ctx.beginPath();ctx.moveTo(785,y);ctx.lineTo(843,y);ctx.stroke();
  }
  // doorway frame shadow
  ctx.strokeStyle='#20160f';ctx.lineWidth=3;
  ctx.strokeRect(695,groundY-96,90,96);
  // roof
  ctx.fillStyle='#7a4428';
  ctx.beginPath();ctx.moveTo(656,groundY-128);ctx.lineTo(757,groundY-172);ctx.lineTo(859,groundY-128);ctx.closePath();ctx.fill();
  ctx.fillStyle='#8a4f30';
  ctx.beginPath();ctx.moveTo(662,groundY-130);ctx.lineTo(757,groundY-168);ctx.lineTo(852,groundY-130);ctx.lineTo(843,groundY-124);ctx.lineTo(757,groundY-160);ctx.lineTo(672,groundY-124);ctx.closePath();ctx.fill();
}

function contactShadow(cx,w,a){
  ctx.save();ctx.globalAlpha=a;ctx.fillStyle='rgba(20,30,15,0.5)';
  ctx.beginPath();ctx.ellipse(cx,groundY+6,w,8,0,0,7);ctx.fill();ctx.restore();
}

// ---------- MAIN ----------
function renderFrame(t){
  if(t<0)t=0;if(t>12)t=12;
  let s=getState(t);
  ctx.clearRect(0,0,W,H);
  drawBackground();
  drawPuddle();
  drawReflection(s);

  drawHouse_back();
  if(!s.bone) drawBone_rest();

  // dog
  let shW=34*(1-s.lift/200);
  if(s.mode!=='air') contactShadow(s.cx, s.frontal>0.5?36:34, 1-s.frontal*0.1);
  else contactShadow(s.cx, 20, 0.3);

  if(s.frontal<0.95){
    let a=1-smooth(s.frontal,0.35,0.9);
    if(a>0.01) drawDogSide(s.cx,s,a,1-0.7*s.frontal);
  }
  if(s.frontal>0.05){
    let a=smooth(s.frontal,0.35,0.9);
    drawDogFront(s.cx,s,a);
  }

  drawHouse_front();
}
window.renderFrame=renderFrame;
renderFrame(0);
</script>

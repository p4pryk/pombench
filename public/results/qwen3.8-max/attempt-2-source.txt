<canvas id="stage" width="960" height="540"></canvas>
<script>
var cv=document.getElementById('stage'),cx=cv.getContext('2d');
function renderFrame(t){
if(t<0)t=0;if(t>12)t=12;
cx.clearRect(0,0,960,540);
drawScene(t);
}
window.renderFrame=renderFrame;

function lerp(a,b,f){return a+(b-a)*f;}
function clamp(v,a,b){return Math.max(a,Math.min(b,v));}
function smoothstep(a,b,v){var x=clamp((v-a)/(b-a),0,1);return x*x*(3-2*x);}

function drawScene(t){
// Sky
var sky=cx.createLinearGradient(0,0,0,300);
sky.addColorStop(0,'#87CEEB');sky.addColorStop(1,'#B0E0E6');
cx.fillStyle=sky;cx.fillRect(0,0,960,300);
// Ground
var gnd=cx.createLinearGradient(0,280,0,540);
gnd.addColorStop(0,'#5a9e3a');gnd.addColorStop(1,'#3d7a28');
cx.fillStyle=gnd;cx.fillRect(0,280,960,260);
// Fence
cx.fillStyle='#8B6914';
for(var i=0;i<12;i++){cx.fillRect(i*85+10,200,12,100);}
cx.fillRect(0,220,960,10);cx.fillRect(0,260,960,10);
// Doghouse
drawDoghouse(t);
// Puddle
drawPuddle(t);
// Bone position
var bonePos=getBonePos(t);
// Dog
var dp=getDogPose(t);
// Draw bone if not held
if(!dp.hasBone)drawBone(bonePos.x,bonePos.y,bonePos.rot,1);
// Draw dog
drawDog(dp,t);
// Draw bone in mouth
if(dp.hasBone)drawBone(dp.boneX,dp.boneY,dp.boneRot,1);
}

function getDogPose(t){
var p={x:150,y:380,scale:1,facing:1,legPhase:0,bodyPitch:0,headPitch:0,
tailWag:0,hasBone:false,boneX:0,boneY:0,boneRot:0,mouthOpen:0,
crouch:0,airborne:false,turnAngle:0,frontness:0,earPerk:0.5,breath:0};
p.breath=Math.sin(t*4)*0.01;
p.earPerk=0.5+0.3*Math.sin(t*2.5);

if(t<1){
p.x=150;p.y=380;p.facing=1;p.legPhase=0;
p.headPitch=-0.1*smoothstep(0.3,0.8,t);
p.earPerk=0.5+0.4*smoothstep(0.3,0.8,t);
}else if(t<3){
var rt=(t-1)/2;
var runX=lerp(150,520,rt);
// Puddle at x=420..500, jump from ~390 to ~510
var jumpStart=0.55,jumpEnd=0.75;
if(rt<jumpStart){
p.x=lerp(150,390,rt/jumpStart);
p.legPhase=rt*12;
p.crouch=0;
}else if(rt<jumpStart+0.08){
// anticipation
var js=(rt-jumpStart)/0.08;
p.x=lerp(390,400,js);
p.crouch=js;
p.legPhase=jumpStart*12;
p.bodyPitch=-0.05*js;
}else if(rt<jumpEnd){
var jt=(rt-jumpStart-0.08)/(jumpEnd-jumpStart-0.08);
p.x=lerp(400,530,jt);
p.y=380-Math.sin(jt*Math.PI)*45;
p.airborne=true;
p.legPhase=jumpStart*12;
p.bodyPitch=-0.1+0.2*jt;
p.crouch=0;
}else{
var ls=(rt-jumpEnd)/(1-jumpEnd);
p.x=lerp(530,620,ls);
p.y=380;
p.legPhase=rt*12;
p.crouch=Math.max(0,0.3-ls*0.6);
p.bodyPitch=0.05*(1-ls);
}
p.facing=1;p.turnAngle=0;p.frontness=0;
}else if(t<4.5){
var et=(t-3)/1.5;
p.x=lerp(620,720,et);
p.y=380;
p.facing=1;p.legPhase=et*4;
p.turnAngle=0;p.frontness=0;
p.headPitch=0.1*smoothstep(0.5,1,et);
}else if(t<6){
var gt=(t-4.5)/1.5;
p.x=720;p.y=380;
p.facing=1;p.legPhase=0;
p.headPitch=0.25*smoothstep(0,0.4,gt);
p.mouthOpen=smoothstep(0.3,0.6,gt)*(1-smoothstep(0.7,1,gt));
p.hasBone=gt>0.6;
if(p.hasBone){p.boneX=p.x+15;p.boneY=p.y-15;p.boneRot=0.3;}
p.turnAngle=0;p.frontness=0;
}else if(t<8){
var bt=(t-6)/2;
p.x=lerp(720,580,bt);
p.y=380;
p.facing=-1;p.legPhase=bt*6;
p.hasBone=true;
p.boneX=p.x-15;p.boneY=p.y-20;p.boneRot=-0.3;
p.turnAngle=0;p.frontness=0;
p.headPitch=-0.05;
}else if(t<10){
var tt=(t-8)/2;
p.x=lerp(580,520,tt);
p.y=380;
p.turnAngle=lerp(Math.PI,0,smoothstep(0,1,tt));
p.frontness=smoothstep(0.2,0.9,tt);
p.facing=1;p.legPhase=tt*2;
p.hasBone=true;
p.boneX=p.x;p.boneY=p.y-25;p.boneRot=0;
p.headPitch=0;
}else{
p.x=520;p.y=380;
p.turnAngle=0;p.frontness=1;
p.facing=1;p.legPhase=0;
p.hasBone=true;
p.boneX=p.x;p.boneY=p.y-28+Math.sin(t*3)*1;
p.boneRot=0;
p.headPitch=-0.05;
p.earPerk=0.7+0.1*Math.sin(t*2);
p.tailWag=Math.sin(t*3)*0.15;
}
return p;
}

function getBonePos(t){
if(t<4.5)return{x:740,y:395,rot:0.2};
if(t<6){var gt=(t-4.5)/1.5;if(gt<0.6)return{x:740,y:395,rot:0.2};}
return{x:740,y:395,rot:0.2};
}

function drawDoghouse(t){
cx.save();
cx.translate(700,320);
// walls
cx.fillStyle='#A0522D';
cx.fillRect(-60,-80,120,100);
// roof
cx.fillStyle='#8B0000';
cx.beginPath();cx.moveTo(-75,-80);cx.lineTo(0,-120);cx.lineTo(75,-80);cx.closePath();cx.fill();
// doorway
cx.fillStyle='#1a0a00';
cx.beginPath();cx.arc(0,-10,35,Math.PI,0);cx.lineTo(35,20);cx.lineTo(-35,20);cx.closePath();cx.fill();
cx.restore();
}

function drawPuddle(t){
cx.save();
cx.translate(460,420);
cx.fillStyle='rgba(100,150,200,0.5)';
cx.beginPath();cx.ellipse(0,0,50,18,0,0,Math.PI*2);cx.fill();
cx.strokeStyle='rgba(60,100,150,0.4)';cx.lineWidth=2;
cx.beginPath();cx.ellipse(0,0,50,18,0,0,Math.PI*2);cx.stroke();
cx.restore();
}

function drawBone(x,y,rot,s){
cx.save();cx.translate(x,y);cx.rotate(rot);cx.scale(s,s);
cx.fillStyle='#F5F0DC';
cx.fillRect(-12,-3,24,6);
cx.beginPath();cx.arc(-12,-3,4,0,Math.PI*2);cx.fill();
cx.beginPath();cx.arc(-12,3,4,0,Math.PI*2);cx.fill();
cx.beginPath();cx.arc(12,-3,4,0,Math.PI*2);cx.fill();
cx.beginPath();cx.arc(12,3,4,0,Math.PI*2);cx.fill();
cx.restore();
}

function drawDog(p,t){
cx.save();
cx.translate(p.x,p.y);
var sc=0.9;
cx.scale(sc*p.facing,sc);
if(p.frontness>0.5){
drawDogFront(p,t);
}else{
drawDogSide(p,t);
}
cx.restore();
// Shadow
cx.save();
cx.fillStyle='rgba(0,0,0,0.2)';
cx.beginPath();
cx.ellipse(p.x,p.y+22,30*sc,8,0,0,Math.PI*2);
cx.fill();
cx.restore();
}

function drawDogSide(p,t){
var crouch=p.crouch*5;
var bodyY=-20+crouch+Math.sin(p.breath*10)*1;
// Legs
var legColor='#D2691E';
var phase=p.legPhase;
cx.fillStyle=legColor;
for(var i=0;i<4;i++){
var lx=(i<2)?-12:12;
var swing=p.airborne?0.3:Math.sin(phase+i*1.57)*0.4;
cx.save();
cx.translate(lx,bodyY+15);
cx.rotate(swing*0.5);
cx.fillRect(-3,0,6,18);
cx.restore();
}
// Body
var grad=cx.createRadialGradient(0,bodyY,5,0,bodyY,25);
grad.addColorStop(0,'#F4A460');grad.addColorStop(1,'#D2691E');
cx.fillStyle=grad;
cx.beginPath();cx.ellipse(0,bodyY,22,16,0,0,Math.PI*2);cx.fill();
// Ruff
cx.fillStyle='#FFDAB9';
cx.beginPath();cx.ellipse(14,bodyY-5,10,12,0,0,Math.PI*2);cx.fill();
// Head
var hx=20,hy=bodyY-12+p.headPitch*20;
cx.fillStyle='#F4A460';
cx.beginPath();cx.ellipse(hx,hy,10,9,0,0,Math.PI*2);cx.fill();
// Muzzle
cx.fillStyle='#FFDAB9';
cx.beginPath();cx.ellipse(hx+8,hy+2,5,4,0,0,Math.PI*2);cx.fill();
// Nose
cx.fillStyle='#222';
cx.beginPath();cx.arc(hx+12,hy+1,2,0,Math.PI*2);cx.fill();
// Eye
cx.fillStyle='#1a0a00';
cx.beginPath();cx.arc(hx+4,hy-2,2.5,0,Math.PI*2);cx.fill();
// Ears
cx.fillStyle='#D2691E';
cx.beginPath();cx.moveTo(hx-4,hy-8);cx.lineTo(hx-1,hy-16*p.earPerk-2);cx.lineTo(hx+3,hy-8);cx.closePath();cx.fill();
cx.beginPath();cx.moveTo(hx+2,hy-8);cx.lineTo(hx+5,hy-16*p.earPerk-2);cx.lineTo(hx+9,hy-8);cx.closePath();cx.fill();
// Tail
var tailA=-0.8+Math.sin(t*5)*0.2+p.tailWag;
cx.save();
cx.translate(-18,bodyY-8);
cx.rotate(tailA);
cx.fillStyle='#F4A460';
cx.beginPath();cx.ellipse(0,-8,6,12,0,0,Math.PI*2);cx.fill();
cx.fillStyle='#FFDAB9';
cx.beginPath();cx.ellipse(0,-12,4,6,0,0,Math.PI*2);cx.fill();
cx.restore();
// Fur texture
cx.strokeStyle='rgba(180,100,40,0.3)';
cx.lineWidth=1;
for(var i=0;i<8;i++){
var fx=-15+i*4,fy=bodyY-10+((i*7)%10);
cx.beginPath();cx.moveTo(fx,fy);cx.lineTo(fx+3,fy+4);cx.stroke();
}
}

function drawDogFront(p,t){
var bodyY=-20+Math.sin(p.breath*10)*1;
// Body (front view - wider)
var grad=cx.createRadialGradient(0,bodyY,5,0,bodyY,22);
grad.addColorStop(0,'#F4A460');grad.addColorStop(1,'#D2691E');
cx.fillStyle=grad;
cx.beginPath();cx.ellipse(0,bodyY,20,18,0,0,Math.PI*2);cx.fill();
// Ruff (prominent)
cx.fillStyle='#FFDAB9';
cx.beginPath();cx.ellipse(0,bodyY-8,14,10,0,0,Math.PI*2);cx.fill();
// Front legs
cx.fillStyle='#D2691E';
cx.fillRect(-10,bodyY+12,6,18);
cx.fillRect(4,bodyY+12,6,18);
// Head
var hy=bodyY-22+p.headPitch*15;
cx.fillStyle='#F4A460';
cx.beginPath();cx.ellipse(0,hy,11,10,0,0,Math.PI*2);cx.fill();
// Face markings
cx.fillStyle='#FFDAB9';
cx.beginPath();cx.ellipse(0,hy+3,6,5,0,0,Math.PI*2);cx.fill();
// Eyes
cx.fillStyle='#1a0a00';
cx.beginPath();cx.arc(-4,hy-2,2.5,0,Math.PI*2);cx.fill();
cx.beginPath();cx.arc(4,hy-2,2.5,0,Math.PI*2);cx.fill();
// eye highlights
cx.fillStyle='#fff';
cx.beginPath();cx.arc(-3.5,hy-2.5,0.8,0,Math.PI*2);cx.fill();
cx.beginPath();cx.arc(4.5,hy-2.5,0.8,0,Math.PI*2);cx.fill();
// Nose
cx.fillStyle='#222';
cx.beginPath();cx.ellipse(0,hy+3,3,2.5,0,0,Math.PI*2);cx.fill();
// Mouth / bone area
if(p.hasBone){
cx.fillStyle='#333';
cx.beginPath();cx.ellipse(0,hy+7,4,2,0,0,Math.PI);cx.fill();
}
// Ears
cx.fillStyle='#D2691E';
var ep=p.earPerk;
cx.beginPath();cx.moveTo(-8,hy-7);cx.lineTo(-12,hy-16*ep-4);cx.lineTo(-3,hy-8);cx.closePath();cx.fill();
cx.beginPath();cx.moveTo(8,hy-7);cx.lineTo(12,hy-16*ep-4);cx.lineTo(3,hy-8);cx.closePath();cx.fill();
// Inner ears
cx.fillStyle='#FFDAB9';
cx.beginPath();cx.moveTo(-7,hy-7);cx.lineTo(-10,hy-13*ep-2);cx.lineTo(-4,hy-8);cx.closePath();cx.fill();
cx.beginPath();cx.moveTo(7,hy-7);cx.lineTo(10,hy-13*ep-2);cx.lineTo(4,hy-8);cx.closePath();cx.fill();
// Tail visible behind (slight)
cx.fillStyle='#F4A460';
cx.beginPath();cx.ellipse(14,bodyY-10,5,10,0.3,0,Math.PI*2);cx.fill();
// Fur texture
cx.strokeStyle='rgba(180,100,40,0.25)';
cx.lineWidth=1;
for(var i=0;i<6;i++){
var fx=-8+i*3.2,fy=bodyY-5+((i*5)%8);
cx.beginPath();cx.moveTo(fx,fy);cx.lineTo(fx+2,fy+3);cx.stroke();
}
}

renderFrame(0);
</script>

function worldToScreen(wx,wy,wz){

var rlen=Math.sqrt(rx*rx+ry*ry+rz*rz);rx/=rlen;ry/=rlen;rz/=rlen;
var ux=ry*lz-rz*ly,uy=rz*lx-rx*lz,uz=rx*ly-ry*lx;
var vx=dx*rx+dy*ry+dz*rz;

if(vz<=0.01)vz=0.01;
var s=canvas.height*WORLD.fov/vz;

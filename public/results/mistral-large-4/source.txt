function solvePose(t){
  var P={x:0,y:0,scale:1,bodyRot:0,legs:[0,0,0,0],head:{x:30,y:-22,rot:0,jaw:0},tail:{curl:1,swish:0},bone:false,boneRot:0.35,phase:'idle'};
  var doorCx=DOOR_X+DOOR_W/2;
  function trot(ph,speed){
    var a=ph*Math.PI*2;
    var fr=Math.sin(a)*0.9*speed;
    var fl=Math.sin(a+Math.PI)*0.9*speed;
    var hl=Math.sin(a+Math.PI)*0.8*speed;
    var hr=Math.sin(a)*0.8*speed;
    return [fl,fr,hl,hr];
  }
  if(t<1){
    var tt=t;
    P.x=lerp(120,150,tt); P.y=groundY(P.x)+8;
    P.scale=1.0;
    P.legs=[0.1,0.1,0.1,0.1];
    P.head.x=30+Math.sin(tt*2)*2; P.head.y=-22;
    P.head.rot=-0.15-0.25*Math.sin(tt*1.5);
    P.tail.curl=1; P.tail.swish=Math.sin(tt*3)*0.3;
    P.phase='idle';
  } else if(t<3){
    var u=(t-1)/2;
    var jumpStart=0.35, jumpEnd=0.72;
    var x;
    if(u<jumpStart){ x=lerp(150,330,sstep(0,jumpStart,u)); }
    else if(u<jumpEnd){
      var ju=(u-jumpStart)/(jumpEnd-jumpStart);
      x=lerp(330,540,ju);
    } else { x=lerp(540,600,sstep(jumpEnd,1,u)); }
    P.x=x; P.y=groundY(x)+8;
    P.scale=1.0;
    var speed=1;
    var ph=u*6.0;
    P.legs=trot(ph,speed);
    if(u>jumpStart-0.08 && u<jumpEnd+0.06){
      var jw=0;
      if(u<jumpEnd) jw=sstep(jumpStart-0.08,jumpEnd,u); else jw=1-sstep(jumpEnd,jumpEnd+0.06,u);
      var arc=Math.sin(jw*Math.PI);
      P.y-=arc*46;
      var tuck=arc;
      P.legs=[ -0.9-0.5*tuck, 0.9+0.4*tuck, 0.8+0.5*tuck, -0.8-0.4*tuck ];
      P.bodyRot=-0.18*Math.sin(jw*Math.PI*1.2);
      if(u<jumpStart){ var an=sstep(jumpStart-0.12,jumpStart,u); P.y+=an*6; P.bodyRot+=an*0.12; }
      if(u>jumpEnd){ var lc=sstep(jumpEnd,jumpEnd+0.05,u); P.legs=[0.3,0.3,-0.4,-0.4]; P.bodyRot+=lc*0.1; }
    }
    P.head.x=30; P.head.y=-22-2*Math.sin(ph*Math.PI*2);
    P.head.rot=-0.05;
    P.tail.curl=1+0.15*Math.sin(ph*Math.PI*2);
    P.tail.swish=Math.sin(ph*2)*0.5;
    P.phase='run';
  } else if(t<4.5){
    var u=(t-3)/1.5;
    P.x=lerp(600,doorCx-6,sstep(0,0.85,u));
    P.y=groundY(P.x)+8;
    P.scale=1.0;
    var ph=u*4;
    P.legs=trot(ph,0.6);
    P.head.x=34; P.head.y=-24;
    P.head.rot=-0.05;
    P.tail.curl=1; P.tail.swish=Math.sin(t*4)*0.3;
    P.phase='enter';
    P.enter=sstep(0,1,u);
  } else if(t<6){
    var u=(t-4.5)/1.5;
    P.x=doorCx-6; P.y=groundY(P.x)+8;
    P.scale=1.0;
    P.legs=[0.1,0.1,0.1,0.1];
    var low=sstep(0.2,0.7,u);
    P.head.x=lerp(34,26,low);
    P.head.y=lerp(-24,-6,low);
    P.head.rot=lerp(-0.05,0.55,low);
    P.head.jaw=sstep(0.5,0.9,u)*0.9;
    P.tail.curl=1; P.tail.swish=Math.sin(t*3)*0.2;
    P.phase='grab';
    P.bone=sstep(0.75,1,u)>0;
    P.boneRot=0.35;
    P.enter=1;
  } else if(t<8){
    var u=(t-6)/2;
    P.x=lerp(doorCx-6,560,sstep(0,0.9,u));
    P.y=groundY(P.x)+8;
    P.scale=1.0;
    var ph=u*3;
    P.legs=[Math.sin(ph*Math.PI*2)*0.6,Math.sin(ph*Math.PI*2+Math.PI)*0.6,Math.sin(ph*Math.PI*2+Math.PI)*0.5,Math.sin(ph*Math.PI*2)*0.5];
    P.head.x=32; P.head.y=-22;
    P.head.rot=-0.05;
    P.head.jaw=0.55;
    P.tail.curl=1; P.tail.swish=Math.sin(t*3)*0.4;
    P.phase='back';
    P.bone=true; P.boneRot=0.35;
    P.enter=1-sstep(0,0.5,u);
  } else if(t<10){
    var u=(t-8)/2;
    P.x=lerp(560,470,sstep(0,0.8,u));
    P.y=groundY(P.x)+8;
    P.scale=1.0;
    var turn=sstep(0.15,0.95,u);
    P.turn=turn;
    P.legs=[0.15,0.15,0.15,0.15];
    P.head.x=lerp(32,0,turn);
    P.head.y=lerp(-22,-26,turn);
    P.head.rot=lerp(-0.05,-0.0,turn);
    P.head.jaw=0.55;
    P.tail.curl=1; P.tail.swish=Math.sin(t*2)*0.3;
    P.phase='turn';
    P.bone=true; P.boneRot=lerp(0.35,-0.2,turn);
  } else {
    var u=clamp((t-10)/2,0,1);
    P.x=470; P.y=groundY(P.x)+8;
    P.scale=1.05;
    P.turn=1;
    P.legs=[0.12,0.12,0.12,0.12];
    P.head.x=0; P.head.y=-26+Math.sin(t*2.2)*0.8;
    P.head.rot=0;
    P.head.jaw=0.55+Math.sin(t*2.2)*0.03;
    P.tail.curl=1+Math.sin(t*1.3)*0.06;
    P.tail.swish=Math.sin(t*1.3)*0.25;
    P.phase='front';
    P.bone=true; P.boneRot=-0.2;
    P.breath=Math.sin(t*2.2)*0.02;
  }
  return P;
}

function renderFrame(t){
  if(t<0)t=0; if(t>=12)t=12;
  ctx.clearRect(0,0,W,H);
  drawSky();
  drawGround();
  ctx.strokeStyle='rgba(120,100,70,0.5)';ctx.lineWidth=3;
  for(var x=0;x<W;x+=60){ctx.beginPath();ctx.moveTo(x,HORIZ-30);ctx.lineTo(x,HORIZ-6);ctx.stroke();}
  ctx.strokeStyle='rgba(120,100,70,0.4)';ctx.lineWidth=2;
  ctx.beginPath();ctx.moveTo(0,HORIZ-24);ctx.lineTo(W,HORIZ-24);ctx.stroke();
  ctx.beginPath();ctx.moveTo(0,HORIZ-12);ctx.lineTo(W,HORIZ-12);ctx.stroke();

  drawPuddle();

  var P=solvePose(t);

  var nearPuddle = Math.abs(P.x-PUDDLE_CX)<PUDDLE_RX+80 && P.y<PUDDLE_CY+10;
  if(nearPuddle && P.phase!=='enter' && P.phase!=='grab'){
    drawDogReflection(P);
    ctx.save();
    ctx.beginPath();ctx.ellipse(PUDDLE_CX,PUDDLE_CY,PUDDLE_RX,PUDDLE_RY,0,0,Math.PI*2);
    ctx.strokeStyle='rgba(200,230,240,0.4)';ctx.lineWidth=1;ctx.stroke();
    ctx.restore();
  }

  drawDoghouse();

  var grabbed = t>=5.25;
  if(!grabbed){ drawBoneRest(); }

  drawDogShadow(P);

  var turnAmt = P.turn||0;
  if(turnAmt<0.98){
    var P2=Object.assign({},P);
    P2.scaleX=1-turnAmt*0.35;
    drawDogScaledFull(P, P2.scaleX||1, 1);
  } else {
    drawDogFront(P);
  }

  if(P.phase==='enter'||P.phase==='grab'||(P.phase==='back'&&P.enter>0)){
    ctx.fillStyle='rgb(30,18,12)';
    ctx.fillRect(DOOR_X,DOOR_Y,DOOR_W,DOOR_H);
    var ig=ctx.createLinearGradient(0,DOOR_Y,0,DOOR_Y+DOOR_H);
    ig.addColorStop(0,'rgb(60,40,26)');ig.addColorStop(1,'rgb(20,12,8)');
    ctx.fillStyle=ig;ctx.fillRect(DOOR_X,DOOR_Y,DOOR_W,DOOR_H);
    if(P.phase==='grab'||P.phase==='back'){
      ctx.save();
      ctx.beginPath();ctx.rect(DOOR_X,DOOR_Y,DOOR_W,DOOR_H);ctx.clip();
      drawDogHeadInside(P);
      ctx.restore();
    }
    ctx.strokeStyle=rgb(HOUSE,0.6);ctx.lineWidth=5;
    ctx.strokeRect(DOOR_X-2.5,DOOR_Y-2.5,DOOR_W+5,DOOR_H+5);
    ctx.fillStyle=rgb(HOUSE,0.7);
    ctx.fillRect(DOOR_X-4,DOOR_Y+DOOR_H-4,DOOR_W+8,6);
    if(!grabbed){ drawBoneRest(); }
  }

  ctx.strokeStyle='rgba(50,90,40,0.5)';ctx.lineWidth=2;
  for(var i=0;i<40;i++){
    var x=rnd()*W, y=H-4-rnd()*30;
    ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x+(rnd()-0.5)*6,y-8-rnd()*10);ctx.stroke();
  }
}

window.renderFrame=renderFrame;
renderFrame(0);

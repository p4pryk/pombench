<canvas id="stage" width="960" height="540"></canvas>
<script>
const canvas = document.getElementById('stage');
const ctx = canvas.getContext('2d');

// Scene parameters
const sceneWidth = 960;
const sceneHeight = 540;

// Timing
const totalDuration = 12; // seconds
// Function to get normalized time for each segment
function getTime(t) {
    if (t < 0) return 0;
    if (t > totalDuration) return totalDuration;
    return t;
}

// Utility functions
function lerp(a, b, t) {
    return a + (b - a) * t;
}
function lerpPoint(p1, p2, t) {
    return {x: lerp(p1.x,p2.x,t), y: lerp(p1.y,p2.y,t)};
}
function rotatePoint(p, angle, origin) {
    const cosA = Math.cos(angle);
    const sinA = Math.sin(angle);
    const dx = p.x - origin.x;
    const dy = p.y - origin.y;
    return {
        x: origin.x + dx * cosA - dy * sinA,
        y: origin.y + dx * sinA + dy * cosA
    };
}
// Animate dog
function getDogPose(t) {
    // Determine phase based on t
    if (t < 1) {
        // scene setting, dog spotted bone and puddle
        return {phase: 'approach'};
    }
    if (t <= 3) {
        // running towards doghouse with anticipation
        const progress = (t - 1)/(2);
        return {phase: 'run', progress: progress};
    }
    if (t <= 4.5) {
        // entering doghouse
        const progress = (t - 3)/(1.5);
        return {phase: 'enter', progress: progress};
    }
    if (t <= 6) {
        // grabbing bone
        const progress = (t - 4.5)/1.5;
        return {phase: 'grab', progress: progress};
    }
    if (t <=8) {
        // backing out with bone
        const progress = (t - 6)/2;
        return {phase: 'backout', progress: progress};
    }
    if (t <=10) {
        // turn to face viewer
        const progress = (t - 8)/2;
        return {phase: 'turn', progress: progress};
    }
    // final pose
    return {phase: 'final'};
}

// Puddle parameters
const puddle = {
    x: 300,
    y: 420,
    radius: 60,
    width: 20
    // shape can be made irregular but for simplicity, circles
};

// Ground
function drawGround() {
    ctx.fillStyle = '#4a6b3a';
    ctx.fillRect(0, sceneHeight - 100, sceneWidth, 100);
}

// Water puddle
function drawPuddle() {
    ctx.save();
    ctx.globalAlpha = 0.3;
    ctx.fillStyle = '#6699cc';
    ctx.beginPath();
    ctx.arc(puddle.x, puddle.y, puddle.radius, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
}

// Reflection inside puddle
function drawDogReflection(pose) {
    // Draw a mirrored dog in the puddle
    ctx.save();
    ctx.translate(puddle.x, puddle.y);
    ctx.scale(1, -1);
    ctx.translate(-puddle.x, -puddle.y);
    drawDogBody(pose, true);
    ctx.restore();
}

// Draw dog
function drawDogBody(pose, reflection=false) {
    // For shape, basic silhouette, enrich with fur and features
    // approximate proportions of Pomeranian
    const bodyCenterX = sceneWidth/2;
    const bodyCenterY = 420;
    // position depends on phase
    let baseX = sceneWidth/2;
    let baseY = 420;
    let bodyOffsetX = 0;
    let bodyOffsetY = 0;
    let headOffsetX = 0;
    let headOffsetY = 0;
    let tailOffsetX = 0;
    let tailOffsetY = 0;
    let headTurn = 0;
    let tailCurl = Math.PI/4;
    let headGimbal = 0;
    let bodyDirection = 0;
    let bodyScale = 1;
    // set pose based on phase
    switch (pose.phase) {
        case 'approach':
            // waiting, static pose
            break;
        case 'run':
            // moving forward
            bodyOffsetX = -lerp(0,100, pose.progress);
            break;
        case 'enter':
            bodyOffsetX = -100; // into doghouse
            break;
        case 'grab':
            bodyOffsetX = -100;
            // perhaps slightly lean head down
            headGimbal = Math.sin(pose.progress * Math.PI)*0.1;
            break;
        case 'backout':
            bodyOffsetX = -100 + lerp(0, 20, pose.progress);
            break;
        case 'turn':
            // rotate towards camera
            headTurn = Math.PI/4 * (1 - pose.progress); // rotate head
            bodyDirection = Math.PI/4 * (1 - pose.progress);
            break;
        case 'final':
            bodyOffsetX = 0;
            headTurn = 0;
            bodyDirection = 0;
            break;
    }
    // run pose
    const direction = pose.phase === 'turn' ? bodyDirection : 0;
    // body shape
    ctx.save();
    ctx.translate(baseX + bodyOffsetX, baseY);
    ctx.rotate(direction);
    // draw body
    ctx.fillStyle = '#e68a34'; // orange coat
    ctx.beginPath();
    ctx.ellipse(0,0, 30, 20, 0, 0, Math.PI*2);
    ctx.fill();

    // fur volume: add some strokes for textures (simulate volume)
    ctx.strokeStyle = '#d55f24';
    ctx.lineWidth=2;
    ctx.stroke();

    // head
    ctx.save();
    ctx.translate(20, -10);
    ctx.rotate(headGimbal);
    // head shape
    ctx.fillStyle= '#e68a34';
    ctx.beginPath();
    ctx.ellipse(0,0,12,10,0,0,Math.PI*2);
    ctx.fill();
    ctx.stroke();

    // ears
    ctx.fillStyle='#c1542c';
    ctx.beginPath();
    ctx.moveTo(4,-5);
    ctx.lineTo(0,-12);
    ctx.lineTo(8,-7);
    ctx.closePath();
    ctx.fill();
    ctx.beginPath();
    ctx.moveTo(-4,-5);
    ctx.lineTo(0,-12);
    ctx.lineTo(-8,-7);
    ctx.closePath();
    ctx.fill();

    // eyes
    ctx.fillStyle='black';
    ctx.beginPath();
    ctx.arc(4,-2,1.5,0,Math.PI*2);
    ctx.arc(-4,-2,1.5,0,Math.PI*2);
    ctx.fill();

    // muzzle
    ctx.fillStyle='#d97b5b';
    ctx.beginPath();
    ctx.ellipse(8,2,3,2,0,0,Math.PI*2);
    ctx.fill();

    ctx.restore();

    // tail
    ctx.save();
    ctx.translate(-10, -8);
    ctx.rotate(tailCurl);
    ctx.fillStyle='#e68a34';
    ctx.beginPath();
    ctx.ellipse(0,0,4,8,0,0,Math.PI*2);
    ctx.fill();

    ctx.restore();

    // paws
    ctx.strokeStyle='#c1542c';
    ctx.lineWidth=2;

    // front left
    ctx.beginPath();
    ctx.moveTo(-15,10);
    ctx.lineTo(-15,14);
    ctx.stroke();
    // front right
    ctx.beginPath();
    ctx.moveTo(15,10);
    ctx.lineTo(15,14);
    ctx.stroke();
    // back left
    ctx.beginPath();
    ctx.moveTo(-8,22);
    ctx.lineTo(-8,26);
    ctx.stroke();
    // back right
    ctx.beginPath();
    ctx.moveTo(8,22);
    ctx.lineTo(8,26);
    ctx.stroke();

    // if reflection, flip Y
}

// Draw scene
function drawScene(time) {
    ctx.clearRect(0,0,sceneWidth,sceneHeight);
    drawGround();
    drawPuddle();

    // determine pose
    const t = getTime(time);
    const pose = getDogPose(t);

    // draw reflection inside puddle
    if (pose.phase==='approach' || pose.phase==='run'|| pose.phase==='enter'||pose.phase==='grab'||pose.phase==='backout') {
        drawDogReflection(pose);
    }

    // Draw dog
    drawDogBody(pose);

    // Draw doghouse
    ctx.save();
    ctx.fillStyle = '#774b3e';
    ctx.fillRect(650, 370, 80, 70);
    ctx.fillStyle='#4a2f1a';
    ctx.fillRect(648, 370, 84, 70);
    ctx.fillStyle='#222';
    ctx.fillRect(675, 410, 10, 30); // step
    ctx.restore();

    // inside door: draw entrance
    ctx.save();
    ctx.fillStyle='#774b3e';
    ctx.fillRect(698, 410, 14, 30);
    ctx.restore();

    // Bone resting in front of doghouse,
    // within reach during approach/grab
    if (t>=0 && t<=6){
        // animate bone close to door
        ctx.save();
        ctx.fillStyle='white';
        const boneX = 700;
        const boneY = 445;
        ctx.beginPath();
        ctx.arc(boneX, boneY,4,0,Math.PI*2);
        ctx.fill();
        ctx.restore();
        // show bone in mouth when grabbed
        if (t>=4.5 && t<=8){
            // draw bone in mouth during grab/backout
            ctx.save();
            ctx.fillStyle='white';
            ctx.translate(0,0);
            // draw the bone in the dog's mouth
            // approximate position inside dog's head
            ctx.translate(sceneWidth/2 + (pose.phase==='backout' ? 0 : 0)+ (pose.phase==='grab' ? 0 : 0), 410);
            ctx.beginPath();
            ctx.arc(0,0,3,0,Math.PI*2);
            ctx.fill();
            ctx.restore();
        }
    }

    // Bone in mouth during final pose
    if (t>=8 && t<=10){
        ctx.save();
        ctx.fillStyle='white';
        ctx.translate(sceneWidth/2 + (0), 410);
        ctx.beginPath();
        ctx.arc(0,0,3,0,Math.PI*2);
        ctx.fill();
        ctx.restore();
    }
}

// Expose renderFrame
window.renderFrame = function(timeSeconds) {
    drawScene(timeSeconds);
}

// optional: automatic animation for testing

</script>

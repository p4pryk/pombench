<script>
const canvas=document.getElementById("stage");
const ctx=canvas.getContext("2d");
const TAU=Math.PI*2;
const clamp=(v,a=0,b=1)=>Math.max(a,Math.min(b,v));
const mix=(a,b,t)=>a+(b-a)*t;
const smooth=t=>{t=clamp(t);return t*t*(3-2*t)};
const ease=t=>{t=clamp(t);return t<.5?2*t*t:1-Math.pow(-2*t+2,2)/2};

function rr(x,y,w,h,r){
  ctx.beginPath();
  ctx.moveTo(x+r,y);ctx.lineTo(x+w-r,y);ctx.quadraticCurveTo(x+w,y,x+w,y+r);
  ctx.lineTo(x+w,y+h-r);ctx.quadraticCurveTo(x+w,y+h,x+w-r,y+h);
  ctx.lineTo(x+r,y+h);ctx.quadraticCurveTo(x,y+h,x,y+h-r);
  ctx.lineTo(x,y+r);ctx.quadraticCurveTo(x,y,x+r,y);ctx.closePath();
}
function ellipse(x,y,rx,ry,fill){
  ctx.beginPath();ctx.ellipse(x,y,rx,ry,0,0,TAU);ctx.fillStyle=fill;ctx.fill();
}
function line(x1,y1,x2,y2,w,c){
  ctx.beginPath();ctx.moveTo(x1,y1);ctx.lineTo(x2,y2);
  ctx.lineWidth=w;ctx.lineCap="round";ctx.strokeStyle=c;ctx.stroke();
}
function pathFill(points,fill){
  ctx.beginPath();ctx.moveTo(points[0][0],points[0][1]);
  for(let i=1;i<points.length;i++)ctx.lineTo(points[i][0],points[i][1]);
  ctx.closePath();ctx.fillStyle=fill;ctx.fill();
}
function scene(){
  let g=ctx.createLinearGradient(0,0,0,540);
  g.addColorStop(0,"#9fc5d0");g.addColorStop(.43,"#d9d5ae");
  g.addColorStop(.44,"#72925e");g.addColorStop(1,"#395336");
  ctx.fillStyle=g;ctx.fillRect(0,0,960,540);

  // distant trees and fence
  ctx.fillStyle="#56734e";
  for(let i=0;i<13;i++){
    let x=i*82-25, h=55+18*Math.sin(i*2.7);
    ellipse(x,170-h*.25,70,h,"#56734e");
  }
  ctx.fillStyle="#b8a47a";ctx.fillRect(0,212,960,82);
  ctx.fillStyle="rgba(63,53,35,.22)";
  for(let x=0;x<960;x+=72){ctx.fillRect(x,212,5,82);ctx.fillRect(x,237,72,5)}
  ctx.fillStyle="#496a42";ctx.fillRect(0,289,960,125);

  // grass texture
  ctx.lineWidth=1;
  for(let i=0;i<390;i++){
    let x=(i*73)%960, y=301+((i*47)%205);
    let h=2+(i%5), c=i%3===0?"#89a263":i%3===1?"#54743e":"#35552f";
    ctx.strokeStyle=c;ctx.beginPath();ctx.moveTo(x,y);ctx.lineTo(x+(i%3)-1,y-h);ctx.stroke();
  }

  // path
  let pg=ctx.createLinearGradient(0,325,0,480);
  pg.addColorStop(0,"#998c65");pg.addColorStop(1,"#746a4e");
  ctx.fillStyle=pg;ctx.beginPath();
  ctx.moveTo(82,393);ctx.bezierCurveTo(260,359,515,365,732,361);
  ctx.lineTo(854,455);ctx.bezierCurveTo(530,462,250,457,60,474);ctx.closePath();ctx.fill();
  ctx.globalAlpha=.23;
  for(let i=0;i<160;i++){
    let x=65+(i*101)%780,y=380+(i*37)%75;
    ellipse(x,y,1+(i%3),.7,"#302d22");
  }
  ctx.globalAlpha=1;
}
function puddlePath(){
  ctx.beginPath();
  ctx.moveTo(333,411);ctx.bezierCurveTo(348,385,405,378,456,385);
  ctx.bezierCurveTo(493,390,517,405,505,421);
  ctx.bezierCurveTo(487,438,411,441,361,430);
  ctx.bezierCurveTo(339,426,326,419,333,411);ctx.closePath();
}
function drawPuddleBase(){
  ctx.save();puddlePath();ctx.clip();
  let w=ctx.createLinearGradient(0,382,0,442);
  w.addColorStop(0,"#718b86");w.addColorStop(.5,"#456d72");w.addColorStop(1,"#314f50");
  ctx.fillStyle=w;ctx.fillRect(320,375,205,75);
  ctx.globalAlpha=.25;ctx.fillStyle="#d5e8df";ctx.fillRect(330,390,180,8);
  ctx.restore();
  ctx.strokeStyle="rgba(31,43,34,.55)";ctx.lineWidth=2;puddlePath();ctx.stroke();
}
function dogState(t){
  let s={x:150,y:411,q:0,phase:0,jump:0,compress:0,lower:0,held:0,inside:0};
  if(t<1){
    s.x=150+10*smooth(t);s.phase=0;s.q=0;
  }else if(t<1.62){
    let u=(t-1)/.62;s.x=mix(160,316,ease(u));s.phase=u*2.3;
  }else if(t<1.78){
    let u=(t-1.62)/.16;s.x=mix(316,334,u);s.phase=2.3+.2*u;s.compress=Math.sin(u*Math.PI)*.9;
  }else if(t<2.48){
    let u=(t-1.78)/.70;s.x=mix(334,514,u);
    s.y=411-92*4*u*(1-u);s.jump=Math.sin(Math.PI*u);s.phase=2.5+u;
  }else if(t<2.73){
    let u=(t-2.48)/.25;s.x=mix(514,544,u);s.phase=3.5+u*.25;s.compress=Math.sin(Math.PI*u);
  }else if(t<3.35){
    let u=(t-2.73)/.62;s.x=mix(544,676,ease(u));s.phase=3.8+u*2;
  }else if(t<4.5){
    let u=(t-3.35)/1.15;s.x=mix(676,756,smooth(u));s.phase=5.8+u*1.6;s.inside=smooth(u);
  }else if(t<6){
    let u=(t-4.5)/1.5;s.x=756;s.lower=Math.sin(clamp(u*1.3)*Math.PI*.52);
    s.held=smooth((u-.38)/.24);s.inside=1;s.phase=7.4;
  }else if(t<8){
    let u=(t-6)/2;s.x=mix(756,660,smooth(u));s.lower=mix(.95,0,smooth(u*.65));
    s.held=1;s.inside=1-smooth((u-.15)/.75);s.phase=7.4-u*3.3;
  }else if(t<10){
    let u=(t-8)/2;s.x=mix(660,616,smooth(u));s.q=smooth(u);s.held=1;s.phase=4.1+u*2.1;
  }else{
    s.x=616;s.q=1;s.held=1;s.phase=6.2;
  }
  return s;
}
function pawPoint(baseX,baseY,phase,back,jump,compress,reverse){
  let p=phase*TAU+(back?Math.PI:0);
  let stride=reverse?-16:20;
  let swing=Math.sin(p);
  let lift=Math.max(0,-Math.cos(p))*11;
  if(jump>.02)return {x:baseX+(back?-7:5),y:baseY-22*jump+(back?8:-2)*jump};
  return {x:baseX+stride*swing*(1-compress*.5),y:baseY-lift*(1-compress*.4)};
}
function drawLeg(x,y,hipX,hipY,foot,far){
  let kneeX=mix(hipX,foot.x,.47)+(far?-5:5);
  let kneeY=mix(hipY,foot.y,.48)+7;
  ctx.strokeStyle=far?"#9a572e":"#bd7440";
  ctx.lineWidth=far?13:15;ctx.lineCap="round";ctx.lineJoin="round";
  ctx.beginPath();ctx.moveTo(hipX,hipY);ctx.lineTo(kneeX,kneeY);ctx.lineTo(foot.x,foot.y-3);ctx.stroke();
  line(foot.x-5,foot.y,foot.x+8,foot.y,far?8:10,far?"#75422b":"#9c5b35");
  ctx.strokeStyle=far?"rgba(238,183,119,.35)":"rgba(255,205,139,.45)";
  ctx.lineWidth=2;ctx.beginPath();ctx.moveTo(hipX-2,hipY);ctx.lineTo(kneeX-2,kneeY);ctx.stroke();
}
function furEllipse(x,y,rx,ry,c1,c2,rot=0){
  ctx.save();ctx.translate(x,y);ctx.rotate(rot);
  let g=ctx.createRadialGradient(-rx*.3,-ry*.35,2,0,0,Math.max(rx,ry));
  g.addColorStop(0,c2);g.addColorStop(.7,c1);g.addColorStop(1,"#8b4b29");
  ctx.fillStyle=g;ctx.beginPath();ctx.ellipse(0,0,rx,ry,0,0,TAU);ctx.fill();
  ctx.restore();
}
function boneShape(x,y,a,scale=1,alpha=1){
  ctx.save();ctx.translate(x,y);ctx.rotate(a);ctx.scale(scale,scale);ctx.globalAlpha=alpha;
  ctx.shadowColor="rgba(45,30,16,.35)";ctx.shadowBlur=4;ctx.shadowOffsetY=2;
  ctx.fillStyle="#e7d3a2";ctx.strokeStyle="#9c8057";ctx.lineWidth=1.5;
  ctx.beginPath();
  ctx.moveTo(-27,-5);ctx.bezierCurveTo(-34,-13,-43,-8,-40,-1);
  ctx.bezierCurveTo(-47,4,-39,12,-31,6);ctx.lineTo(29,6);
  ctx.bezierCurveTo(36,13,45,7,40,1);ctx.bezierCurveTo(46,-7,36,-13,29,-6);
  ctx.closePath();ctx.fill();ctx.stroke();ctx.restore();
}
function drawDog(s,t,reflection=false){
  const x=s.x,y=s.y+(s.compress?7*s.compress:0);
  const q=s.q, sc=.78;
  ctx.save();ctx.translate(x,y);ctx.scale(sc,sc);
  if(reflection)ctx.globalAlpha=.30;

  let breathe=t>=10?Math.sin((t-10)*4)*1.2:0;
  let bodyW=mix(67,48,q), bodyH=mix(39,47,q);
  let headX=mix(49,0,q), headY=-80+s.lower*25+breathe*.2;
  let gait=s.phase;
  let rev=(t>=6&&t<8);

  // far legs
  let rearBase=mix(-43,-22,q), frontBase=mix(34,22,q);
  let f1=pawPoint(frontBase,0,gait+.5,false,s.jump,s.compress,rev);
  let r1=pawPoint(rearBase,0,gait,true,s.jump,s.compress,rev);
  drawLeg(0,0,frontBase,-48,f1,true);drawLeg(0,0,rearBase,-43,r1,true);

  // plume tail, behind body
  ctx.save();
  ctx.translate(mix(-59,0,q),-68);
  ctx.rotate(mix(-.35,.05,q)+.08*Math.sin(t*3));
  ctx.strokeStyle="#a95e32";ctx.lineWidth=31;ctx.lineCap="round";
  ctx.beginPath();ctx.moveTo(0,13);ctx.bezierCurveTo(-30,-6,-26,-44,5,-45);
  ctx.bezierCurveTo(27,-46,27,-24,13,-15);ctx.stroke();
  ctx.strokeStyle="#e2a264";ctx.lineWidth=19;ctx.beginPath();ctx.moveTo(-2,9);
  ctx.bezierCurveTo(-22,-9,-18,-34,5,-36);ctx.bezierCurveTo(18,-37,20,-27,12,-19);ctx.stroke();
  ctx.restore();

  furEllipse(0,-59+breathe,bodyW,bodyH,"#a95e31","#df9556");
  furEllipse(mix(28,0,q),-66,36+9*q,43,"#b66736","#efad6b");
  // cream chest and ruff
  furEllipse(mix(34,0,q),-57,22+14*q,35,"#d39a63","#f4d4a1");

  // near legs
  let f2=pawPoint(frontBase,0,gait,false,s.jump,s.compress,rev);
  let r2=pawPoint(rearBase,0,gait+.5,true,s.jump,s.compress,rev);
  drawLeg(0,0,rearBase,-42,r2,false);drawLeg(0,0,frontBase,-47,f2,false);

  // toes
  [f2,r2].forEach(p=>{
    ctx.strokeStyle="rgba(64,38,27,.45)";ctx.lineWidth=1;
    for(let k=-1;k<=1;k++)line(p.x+k*3,p.y-3,p.x+k*3+1,p.y+1,1,"rgba(62,39,28,.5)");
  });

  // head volume and ears
  let hx=headX,hy=headY;
  let earSpread=mix(17,24,q);
  pathFill([[hx-earSpread,hy-22],[hx-mix(10,16,q),hy-60],[hx+1,hy-29]],"#9a502d");
  pathFill([[hx+earSpread,hy-22],[hx+mix(11,16,q),hy-60],[hx-1,hy-29]],"#a95b32");
  pathFill([[hx-earSpread+4,hy-28],[hx-mix(10,16,q),hy-51],[hx-2,hy-30]],"#d39469");
  pathFill([[hx+earSpread-4,hy-28],[hx+mix(10,16,q),hy-51],[hx+2,hy-30]],"#d39469");

  furEllipse(hx,hy-14,mix(31,36,q),mix(29,34,q),"#b96738","#eaa260");
  // cheek fluff
  furEllipse(hx-mix(13,23,q),hy-5,mix(16,18,q),22,"#c87d48","#f0bd82");
  furEllipse(hx+mix(10,23,q),hy-5,mix(16,18,q),22,"#b86a3b","#e9a565");

  // eyes
  let eyeY=hy-17;
  if(q<.62){
    ellipse(hx+9,eyeY,4.5,5.2,"#211b16");
    ellipse(hx+10.5,eyeY-1.5,1.2,1.4,"#f4eee0");
    if(q>.18){
      ellipse(hx-mix(2,14,q),eyeY,3.8,4.8,`rgba(31,25,20,${q})`);
    }
  }else{
    ellipse(hx-13,eyeY,4.6,5.3,"#211b16");ellipse(hx+13,eyeY,4.6,5.3,"#211b16");
    ellipse(hx-11.5,eyeY-1.5,1.2,1.4,"#fff7df");ellipse(hx+14.5,eyeY-1.5,1.2,1.4,"#fff7df");
  }

  // muzzle
  let muzzleX=mix(hx+24,hx,q), muzzleY=hy+3;
  furEllipse(muzzleX,muzzleY,mix(20,20,q),mix(13,15,q),"#dbad79","#f1d2a6");
  ellipse(mix(hx+40,hx,q),hy-2,mix(6,7,q),mix(5,5.5,q),"#251c18");

  if(s.held>0){
    let bx=mix(hx+39,hx,q),by=hy+9;
    boneShape(bx,by,mix(.08,0,q),1,s.held);
    // lower jaw over the centre grip
    furEllipse(mix(hx+29,hx,q),hy+10,mix(12,13,q),7,"#c88854","#eac092");
    line(mix(hx+31,hx-8,q),hy+8,mix(hx+42,hx+8,q),hy+9,2,"#4a2d22");
  }else{
    line(mix(hx+31,hx-7,q),hy+9,mix(hx+39,hx+7,q),hy+9,1.7,"#5f3326");
  }

  // deterministic guard hairs
  ctx.strokeStyle="rgba(247,205,150,.55)";ctx.lineWidth=1;
  for(let i=0;i<18;i++){
    let a=-1.9+i*.21, px=Math.cos(a)*bodyW*.86,py=-59+Math.sin(a)*bodyH*.72;
    ctx.beginPath();ctx.moveTo(px,py);ctx.lineTo(px+Math.cos(a)*6,py+Math.sin(a)*6);ctx.stroke();
  }
  ctx.restore();
}
function drawHouseBack(){
  // contact shadow
  ellipse(779,421,154,24,"rgba(25,30,19,.32)");
  // house body
  let g=ctx.createLinearGradient(650,230,900,430);
  g.addColorStop(0,"#9d5a33");g.addColorStop(1,"#65351f");
  ctx.fillStyle=g;ctx.fillRect(654,226,242,188);
  // side wall
  pathFill([[896,226],[935,261],[935,414],[896,414]],"#553321");
  // roof
  pathFill([[623,238],[758,134],[929,238]],"#503429");
  pathFill([[758,134],[940,226],[929,245],[758,164],[620,250],[610,237]],"#6d4532");
  for(let i=0;i<9;i++)line(640+i*35,226-i*12,785+i*19,154+i*9,3,"rgba(31,23,21,.24)");
  // doorway recess

  ctx.moveTo(699,414);ctx.lineTo(699,292);ctx.quadraticCurveTo(699,252,756,252);

  dg.addColorStop(0,"#4a3b2d");dg.addColorStop(1,"#171512");

  ctx.fillStyle="#754325";rr(685,241,151,18,4);ctx.fill();
}
function houseForeground(){
  // doorway jambs and threshold, masks portions crossing solid walls
  let g=ctx.createLinearGradient(0,0,30,0);g.addColorStop(0,"#522c1c");g.addColorStop(1,"#9b5931");
  ctx.fillStyle=g;ctx.fillRect(680,273,20,141);ctx.fillRect(824,273,20,141);
  ctx.fillStyle="#4c2b1b";rr(693,402,139,14,3);ctx.fill();
  ctx.strokeStyle="#bd7748";ctx.lineWidth=4;
  ctx.beginPath();ctx.moveTo(698,405);ctx.lineTo(698,294);ctx.quadraticCurveTo(698,251,758,251);
  ctx.quadraticCurveTo(825,251,825,296);ctx.lineTo(825,405);ctx.stroke();
}
function shadowForDog(s,t){
  let airborne=clamp((411-s.y)/95);
  ctx.save();ctx.globalAlpha=.32*(1-airborne*.55);
  ellipse(s.x,416,55*(1-airborne*.35),11*(1-airborne*.5),"#20261c");
  ctx.restore();
}
function drawBoneRest(s){
  if(s.held>=1)return;
  let u=s.held;
  let mouthX=s.x+(.78*(49+39));
  let mouthY=s.y+.78*(-80+s.lower*25+9);
  let x=mix(809,mouthX,u),y=mix(393,mouthY,u);
  boneShape(x,y,mix(-.12,.08,u),.82,1);
}
function renderFrame(timeSeconds){
  let t=clamp(Number(timeSeconds)||0,0,12);
  ctx.clearRect(0,0,960,540);
  scene();
  drawPuddleBase();
  drawHouseBack();
  let s=dogState(t);

  // reflection of this same dog, clipped strictly to water
  ctx.save();puddlePath();ctx.clip();
  ctx.translate(0,826);ctx.scale(1,-.42);
  drawDog(s,t,true);
  ctx.restore();
  // water sheen over reflected image
  ctx.save();puddlePath();ctx.clip();ctx.globalAlpha=.18;
  ctx.fillStyle="#b9d5d1";ctx.fillRect(330,395,180,15);
  ctx.strokeStyle="#d4e6df";ctx.lineWidth=1;
  for(let i=0;i<5;i++){ctx.beginPath();ctx.ellipse(402+i*11,414+i%2*5,40-i*4,4,0,0,TAU);ctx.stroke()}
  ctx.restore();

  shadowForDog(s,t);

  // resting bone is behind dog but remains visible through doorway
  if(t<5.75)drawBoneRest(s);

  drawDog(s,t,false);

  // doorway correctly occludes entering/backing animal
  if((t>=3.55&&t<8.15))houseForeground();
  else {
    // normal architectural trim, kept behind an outside dog
    ctx.save();
    if(t<3.55){houseForeground()}
    ctx.restore();
  }

  // Repaint dog in front only while fully outside before entry;
  // at these times trim belongs behind its visible silhouette.
  if(t<3.55 && s.x>625){
    drawDog(s,t,false);
  }

  // foreground blades
  ctx.strokeStyle="#29432a";ctx.lineWidth=2;
  for(let i=0;i<54;i++){
    let x=(i*181)%960,y=468+(i*29)%66;
    ctx.beginPath();ctx.moveTo(x,y);ctx.quadraticCurveTo(x+4,y-10,x+2,y-17-(i%8));ctx.stroke();
  }

  // subtle vignette
  let vg=ctx.createRadialGradient(480,270,190,480,270,570);
  vg.addColorStop(.55,"rgba(0,0,0,0)");vg.addColorStop(1,"rgba(12,18,14,.28)");
  ctx.fillStyle=vg;ctx.fillRect(0,0,960,540);
}
window.renderFrame=renderFrame;
renderFrame(0);

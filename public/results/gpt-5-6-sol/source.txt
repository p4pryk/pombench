const canvas=document.getElementById("stage");

const PI=Math.PI, TAU=PI*2;
const clamp=(v,a=0,b=1)=>Math.max(a,Math.min(b,v));
const mix=(a,b,t)=>a+(b-a)*t;
const smooth=t=>{t=clamp(t);return t*t*(3-2*t)};
const ease=t=>{t=clamp(t);return t<.5?4*t*t*t:1-Math.pow(-2*t+2,3)/2};
const fract=x=>x-Math.floor(x);
const hash=n=>fract(Math.sin(n*127.1+31.7)*43758.5453123);

function ellipse(x,y,rx,ry,fill,rot=0){
  ctx.save();ctx.translate(x,y);ctx.rotate(rot);ctx.beginPath();
  ctx.ellipse(0,0,rx,ry,0,0,TAU);ctx.fillStyle=fill;ctx.fill();ctx.restore();
}
function line(x1,y1,x2,y2,w,color){
  ctx.beginPath();ctx.moveTo(x1,y1);ctx.lineTo(x2,y2);
  ctx.lineWidth=w;ctx.strokeStyle=color;ctx.stroke();
}
function grad(x0,y0,x1,y1,stops){
  const g=ctx.createLinearGradient(x0,y0,x1,y1);
  stops.forEach(s=>g.addColorStop(s[0],s[1]));return g;
}
function radial(x,y,r0,r1,stops){
  const g=ctx.createRadialGradient(x,y,r0,x,y,r1);
  stops.forEach(s=>g.addColorStop(s[0],s[1]));return g;
}
function path(points,fill,stroke,w=1){
  ctx.beginPath();ctx.moveTo(points[0][0],points[0][1]);
  for(let i=1;i<points.length;i++)ctx.lineTo(points[i][0],points[i][1]);
  ctx.closePath();if(fill){ctx.fillStyle=fill;ctx.fill()}
  if(stroke){ctx.lineWidth=w;ctx.strokeStyle=stroke;ctx.stroke()}
}
function roundedRect(x,y,w,h,r,fill){
  ctx.beginPath();ctx.roundRect(x,y,w,h,r);ctx.fillStyle=fill;ctx.fill();
}

  legGeometry(x+frontSep*(1-q)-19*q,bodyY+19,true,phase,activity,q,scale);

  const bodyGrad=radial(x-15,bodyY-19,4,83,[
    [0,dark?"#b66d3b":"#e9a15a"],
    [.43,dark?"#98502b":"#c97035"],
    [1,"#733c24"]

    radial(x-14,bodyY-18,2,50,[[0,"#efb470"],[.48,"#d77e3f"],[1,"#8c4928"]]));

    const px=x+Math.cos(a)*rx;
    const py=bodyY+Math.sin(a)*bodyH*.76;
    const len=3+hash(i+44)*8;

      i%3?"rgba(244,166,92,.31)":"rgba(91,44,24,.27)");

  const headY=bodyY-mix(24,35,q)+state.headDrop;
  const headW=mix(31,39,q), headH=mix(31,37,q);

    ctx.save();ctx.translate(ex,ey);ctx.rotate(flip*mix(.05,.12,q));

function getState(t){
  t=clamp(t,0,12);
  let s={
    x:182,y:421,q:0,phase:0,activity:0,bob:0,headDrop:0,
    bone:false,mouthOpen:0,tail:0,inside:0,scale:1
  };
  if(t<1){
    const n=smooth(t);
    s.x=182;s.headDrop=-3*Math.sin(n*PI);
    s.q=.05;s.tail=2*Math.sin(t*4);
    s.earL=-1+.14*Math.sin(t*6);
  }else if(t<3){
    const u=ease((t-1)/2);
    s.x=mix(182,588,u);s.activity=1;
    s.phase=(t-1)*TAU*2.25;
    s.bob=-3*Math.abs(Math.sin(s.phase));
    s.tail=5*Math.sin(s.phase*.5);
  }else if(t<4.5){
    const u=smooth((t-3)/1.5);
    s.x=mix(588,696,u);
    s.activity=mix(.75,.2,u);
    s.phase=(t-3)*TAU*1.45;
    s.bob=-2*Math.abs(Math.sin(s.phase));
    s.inside=smooth((s.x-630)/65);
    s.tail=3*Math.sin(s.phase);
  }else if(t<6){
    const u=(t-4.5)/1.5;
    s.x=696;s.inside=1;s.activity=0;
    s.headDrop=26*smooth(clamp(u/0.48));
    s.mouthOpen=Math.sin(clamp((u-.25)/.45)*PI);
    if(u>.58){s.bone=true;s.mouthOpen=0}
    if(u>.72)s.headDrop=mix(26,7,smooth((u-.72)/.28));
  }else if(t<8){
    const u=smooth((t-6)/2);
    s.x=mix(696,584,u);s.inside=1-u;
    s.activity=.72;s.phase=(t-6)*TAU*1.25;
    s.bob=-1.7*Math.abs(Math.sin(s.phase));
    s.headDrop=mix(7,0,u);s.bone=true;s.tail=3*Math.sin(s.phase+.5);
  }else if(t<10){
    const u=smooth((t-8)/2);
    s.x=mix(584,548,u);s.q=u;s.bone=true;
    s.activity=mix(.58,.12,u);
    s.phase=(t-8)*TAU*.9;
    s.bob=-1.4*Math.abs(Math.sin(s.phase))*(1-u);
    s.tail=3*Math.sin((t-8)*3)*(1-u);
  }else{
    s.x=548;s.q=1;s.bone=true;s.activity=0;
    const u=t-10;
    s.breathe=1.1*Math.sin(u*PI*1.1);
    s.bob=.45*Math.sin(u*PI*1.1);
    s.tail=1.5*Math.sin(u*2.3);
    s.earL=-1+.06*Math.sin(u*2.7);
    s.earR=1+.05*Math.sin(u*2.2+.8);
  }
  return s;
}

window.renderFrame=function(timeSeconds){
  const t=clamp(Number.isFinite(timeSeconds)?timeSeconds:0,0,12);
  ctx.clearRect(0,0,960,540);
  ctx.lineJoin="round";ctx.lineCap="round";
  drawBackground();
  drawHouseBack();

  const grabbed=t>=5.37;
  if(!grabbed) boneShape(752,405,-.08,.83);

  const s=getState(t);
  drawDog(s);
  drawHouseFront();

  if(t<1){
    const a=.12+.10*Math.sin(t*PI);
    ctx.fillStyle="rgba(255,238,182,"+a+")";
    ctx.beginPath();ctx.ellipse(752,405,43,13,0,0,TAU);ctx.fill();
  }

  const vignette=radial(480,270,260,610,[
    [0,"rgba(0,0,0,0)"],[.72,"rgba(22,30,18,.03)"],[1,"rgba(9,15,10,.30)"]
  ]);
  ctx.fillStyle=vignette;ctx.fillRect(0,0,960,540);

  ctx.fillStyle="rgba(255,247,218,.035)";ctx.fillRect(0,0,960,540);
};
window.renderFrame(0);

function clampT(t){return t<0?0:t>12?12:t;}

function proj(f,r,u,S){
  const yaw=S.yaw*1.5707963;
  const c=Math.cos(yaw), s=Math.sin(yaw);
  const sx=S.x + f*c + r*s;
  const sy=S.gy - u + f*s*0.20 - r*c*0.06;
  const near=f*s - r*c;
  return {sx:sx,sy:sy,near:near,c:c,s:s};
}

function furEll(g,sx,sy,rx,ry,rot,light,mid,dark,seed,tufts){
  rx=Math.max(2,rx); ry=Math.max(2,ry);
  g.save();
  g.translate(sx,sy);
  g.rotate(rot||0);
  const grd=g.createRadialGradient(-rx*0.32,-ry*0.38,Math.min(rx,ry)*0.15,0,0,Math.max(rx,ry));
  grd.addColorStop(0,light);
  grd.addColorStop(0.55,mid);
  grd.addColorStop(1,dark);
  g.fillStyle=grd;
  g.beginPath();
  g.ellipse(0,0,rx,ry,0,0,6.2832);
  g.fill();
  g.lineWidth=1;
  const n=tufts||12;
  for(let i=0;i<n;i++){
    const a=(i/n)*6.2832+seed;
    const j=hsh(seed*100+i*17);
    const rr=0.82+j*0.18;
    const len=1.08+j*0.22;
    g.strokeStyle=j>0.45?light:mid;
    g.globalAlpha=0.85;
    g.beginPath();

    g.lineTo(Math.cos(a)*rx*len, Math.sin(a)*ry*len);
    g.stroke();
  }
  g.restore();
  g.globalAlpha=1;
}

function solveKnee(hipF,hipU,footF,footU,L1,L2,bend){
  let df=footF-hipF, du=footU-hipU;
  let dist=Math.hypot(df,du);
  const maxL=L1+L2-0.2, minL=Math.abs(L1-L2)+0.2;
  if(dist<0.001){dist=0.001; du=-1;}
  if(dist>maxL){df*=maxL/dist; du*=maxL/dist; dist=maxL;}
  if(dist<minL){df*=minL/dist; du*=minL/dist; dist=minL;}
  const ang=Math.atan2(du,df);
  const cosA=clamp((L1*L1+dist*dist-L2*L2)/(2*L1*dist),-1,1);
  const off=Math.acos(cosA)*bend;
  return {f:hipF+Math.cos(ang+off)*L1, u:hipU+Math.sin(ang+off)*L1};
}

function legPose(S){
  const specs=[
    {id:0,f:18,r:-8.5,phase:0.0,fore:1},
    {id:1,f:18,r:8.5,phase:0.5,fore:1},
    {id:2,f:-16,r:-8,phase:0.5,fore:0},
    {id:3,f:-16,r:8,phase:0.0,fore:0}
  ];
  const sign=S.mode==='back'?-1:1;
  for(let i=0;i<specs.length;i++){
    const L=specs[i];
    let footF=L.f, footU=0, footR=L.r;
    if(S.mode==='run'||S.mode==='back'){
      let u=(S.t*S.freq+L.phase)%1; if(u<0)u+=1;
      const duty=0.52;
      if(u<duty){
        footF=L.f+sign*S.stride*(0.5-u/duty);
        footU=0;
      }else{
        const s=(u-duty)/(1-duty);
        footF=L.f+sign*S.stride*(-0.5+s);
        footU=Math.sin(s*Math.PI)*S.liftH;
      }
    }else if(S.mode==='crouch'){
      const k=S.crouchN||0;
      footU=0;
      footF=L.f+(L.fore?-4*k:-14*k);
    }else if(S.mode==='jump'){
      const u=S.jumpU;
      const tuck=Math.sin(u*Math.PI);
      if(L.fore){
        footF=L.f+lerp(12,2,Math.min(1,u*1.3))+(u>0.62?lerp(0,12,(u-0.62)/0.38):0);
        footU=9+tuck*15;
        if(u>0.8){footU=lerp(12,0.8,(u-0.8)/0.2); footF=L.f+16;}
      }else{
        footF=L.f-lerp(16,6,Math.min(1,u*1.2));
        footU=8+tuck*13;
        if(u>0.86){footU=lerp(10,0.4,(u-0.86)/0.14); footF=L.f-2;}
      }
      if(u<0.96) footU=Math.max(footU,7);
    }else if(S.mode==='land'){
      let u=(S.t*S.freq+L.phase)%1; if(u<0)u+=1;
      footF=L.f+(L.fore?6:-2)+sign*6*(0.5-u);
      footU=(u>0.6)?Math.sin((u-0.6)/0.4*Math.PI)*4:0;
    }else if(S.mode==='turn'){
      const u=(S.t*1.15+L.phase)%1;
      const step=u<0.22?Math.sin(u/0.22*Math.PI):0;
      footU=step*9;
      footF=L.f+(L.fore?1:-1)*(S.yaw*4);
      footR=L.r+Math.sin(S.t*2.2+L.phase*6)*1.4;
    }else if(S.mode==='pose'){
      footU=0.4;
      footF=L.f+(L.fore?1.5:0);
      footR=L.r;
    }else if(S.mode==='grab'){
      footU=0;
      footF=L.f+(L.fore?2:-1);
    }else{
      const u=(S.t*0.5+L.phase)%1;
      footU=u>0.82?Math.sin((u-0.82)/0.18*Math.PI)*2.4:0;
      footF=L.f;
    }
    L.footF=footF; L.footU=footU; L.footR=footR;
  }
  return specs;
}

function headPoint(f,r,u,S,neck){
  const pitch=S.headPitch;
  let df=f-neck.f, du=u-neck.u;
  const c=Math.cos(pitch), s=Math.sin(pitch);
  const f2=neck.f+df*c+du*s;
  const u2=neck.u-df*s+du*c;
  const hy=S.headYaw;
  const cy=Math.cos(hy), sy=Math.sin(hy);
  const df2=f2-neck.f;
  return {f:neck.f+df2*cy-r*sy, r:r*cy+df2*sy, u:u2};
}

    g.beginPath();
    g.ellipse(p.sx,p.sy,2.15,2.5*(1-shut*0.92),0,0,6.2832);
    g.fill();
    if(shut<0.7){
      g.fillStyle='#fff';
      g.fillRect(p.sx-1.1,p.sy-1.5,1.1,1.1);
    }
  });
  addH(eyeR,1.05,function(){
    const p=proj(eyeR.f,eyeR.r,eyeR.u+jump,S);
    const shut=S.blink;
    g.fillStyle='#24160f';
    g.beginPath();
    g.ellipse(p.sx,p.sy,2.05,2.4*(1-shut*0.92),0,0,6.2832);
    g.fill();
    if(shut<0.7){
      g.fillStyle='#fff';
      g.fillRect(p.sx-0.9,p.sy-1.4,1.0,1.0);
    }
  });
  addH(nose,1.3,function(){
    const p=proj(nose.f,nose.r,nose.u+jump,S);
    g.fillStyle='#1c120e';
    g.beginPath();
    g.ellipse(p.sx,p.sy,2.6+S.yaw*0.8,2.0,0,0,6.2832);
    g.fill();
    g.fillStyle='rgba(255,255,255,0.45)';
    g.fillRect(p.sx-1.3,p.sy-1.1,1.1,0.8);
  });

  // carried bone
  if(S.grip>0){
    const rest={x:778,y:456};
    const mp=proj(muzC.f+2,muzC.r,muzC.u-1+jump,S);
    const bx=lerp(rest.x,mp.sx,S.grip);
    const by=lerp(rest.y,mp.sy+2,S.grip);
    blobs.push({n:mp.near+1.5,fn:function(){
      drawBone(g,bx,by,S.yaw,1,1);
    }});
  }

  blobs.sort(function(a,b){return a.n-b.n;});
  if(reflect){
    g.fillStyle='rgba(70,110,130,0.18)';
  }
  for(let i=0;i<blobs.length;i++) blobs[i].fn();

  S._legs=legs;
  return {nose:proj(nose.f,nose.r,nose.u+jump,S), muz:proj(muzC.f,muzC.r,muzC.u+jump,S)};
}

function drawYard(){
  const sky=ctx.createLinearGradient(0,0,0,320);
  sky.addColorStop(0,'#9ec8e4');
  sky.addColorStop(0.55,'#d5ead4');
  sky.addColorStop(1,'#e7f0cf');
  ctx.fillStyle=sky;
  ctx.fillRect(0,0,960,540);

  // far trees
  for(let i=0;i<9;i++){
    const x=40+i*110+hsh(i+3)*20;
    const hgt=70+hsh(i+9)*50;
    ctx.fillStyle=i%2?'#6e9a52':'#5d8744';
    ctx.beginPath();
    ctx.ellipse(x,300-hgt*0.2,28,hgt*0.45,0,0,6.2832);
    ctx.fill();
    ctx.fillStyle='#6a4a30';
    ctx.fillRect(x-4,300,8,18);
  }

  const grass=ctx.createLinearGradient(0,300,0,540);
  grass.addColorStop(0,'#87b85a');
  grass.addColorStop(0.45,'#6ea348');
  grass.addColorStop(1,'#5b8c3c');
  ctx.fillStyle=grass;
  ctx.fillRect(0,300,960,240);

  // worn path
  ctx.fillStyle='rgba(120,92,52,0.28)';
  ctx.beginPath();
  ctx.ellipse(420,478,280,28,0,0,6.2832);
  ctx.fill();
  ctx.fillStyle='rgba(96,74,40,0.18)';
  ctx.beginPath();
  ctx.ellipse(430,486,180,14,-0.05,0,6.2832);
  ctx.fill();

  // grass blades
  for(let i=0;i<70;i++){
    const x=hsh(i*5+1)*960;
    const y=330+hsh(i*3+2)*190;
    if(x>640 && x<930 && y>360 && y<500) continue;
    ctx.strokeStyle=hsh(i)>0.5?'#d7e7a4':'#4f7a34';
    ctx.lineWidth=1;
    ctx.beginPath();
    ctx.moveTo(x,y);
    ctx.lineTo(x+2+(hsh(i+4)-0.5)*4,y-6-hsh(i+8)*8);
    ctx.stroke();
  }

  // fence
  ctx.fillStyle='#cbb48a';
  for(let i=0;i<14;i++){
    const x=20+i*48;
    ctx.fillRect(x,286,6,28);
  }
  ctx.fillRect(16,292,660,4);
  ctx.fillRect(16,304,660,3);
}

function drawReflection(S){
  const p=puddleGeom();
  ctx.save();
  ctx.beginPath();
  ctx.ellipse(p.x,p.y,p.rx-2,p.ry-2,0,0,6.2832);
  ctx.clip();
  ctx.translate(p.x,p.y);
  ctx.scale(1,-0.40);
  ctx.translate(-p.x,-p.y);
  ctx.translate(0,-8);
  ctx.globalAlpha=0.55;
  drawDog(ctx,S,true);
  ctx.restore();
}

function drawShadow(S){
  const elev=S.jump||0;
  ctx.save();
  ctx.fillStyle='rgba(30,40,20,'+(0.30-Math.min(0.18,elev*0.0022))+')';
  ctx.beginPath();
  ctx.ellipse(S.x+6+elev*0.05,S.gy+5,36+S.yaw*6-elev*0.08,9,0,0,6.2832);
  ctx.fill();
  if(S._legs){
    for(let i=0;i<S._legs.length;i++){
      const fp=S._legs[i]._fp;
      if(!fp) continue;
      if((S._legs[i].footU||0)<2 && elev<8){
        ctx.beginPath();
        ctx.ellipse(fp.sx,S.gy+3,5,2.2,0,0,6.2832);
        ctx.fill();
      }
    }
  }
  ctx.restore();
}

function drawLandingDust(S){
  if(S.t<2.76||S.t>3.15) return;
  const u=(S.t-2.76)/0.39;
  ctx.save();
  ctx.globalAlpha=(1-u)*0.45;
  for(let i=0;i<6;i++){
    const x=560+ (hsh(i+20)-0.5)*40 + u*18*(i-3);
    const y=470 - u*16 - hsh(i+2)*10;
    ctx.fillStyle='#d9c7a0';
    ctx.beginPath();
    ctx.ellipse(x,y,3+u*4,2+u*1.5,0,0,6.2832);
    ctx.fill();
  }
  ctx.restore();
}

window.renderFrame=function(timeSeconds){
  const S=getState(timeSeconds);
  ctx.clearRect(0,0,960,540);
  ctx.globalAlpha=1;
  drawYard();
  drawHouse('back');
  if(S.grip<=0){
    drawBone(ctx,778,456,0,1,1);
  }
  drawPuddle();
  drawReflection(S);
  drawPuddleRim();
  drawShadow(S);
  drawDog(ctx,S,false);
  drawLandingDust(S);
  drawHouse('front');
};

renderFrame(0);
})();

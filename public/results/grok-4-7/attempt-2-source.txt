function clamp(v,a,b){return v<a?a:v>b?b:v;}
function lerp(a,b,u){return a+(b-a)*u;}
function smooth(u){u=clamp(u,0,1);return u*u*(3-2*u);}
function hash(n){const s=Math.sin(n*127.1+311.7)*43758.5453;return s-Math.floor(s);}
function pose(time){
  const t=clamp(time,0,12);
  let x=198,z=100,gait="idle",crouch=0,mouth=0,grip=0,headPitch=0.15,look=0.55,alert=0;
  if(t<1){
    alert=smooth(t); look=lerp(0.7,0.05,alert); headPitch=lerp(0.2,0.02,alert);
  }else if(t<3){
    const u=smooth((t-1)/2);
    x=lerp(198,505,u); z=lerp(100,96,u); gait="run"; headPitch=0.04; look=0.02;
  }else if(t<4.5){
    const u=smooth((t-3)/1.5);
    x=lerp(505,655,u); z=lerp(96,94,u); gait="walk"; headPitch=0.08; look=0;
  }else if(t<6){
    const u=smooth((t-4.5)/1.5);
    x=lerp(655,678,u); z=94; gait="grab";
    crouch=smooth(clamp((t-4.5)/0.55,0,1))*(1-smooth(clamp((t-5.55)/0.45,0,1))*0.35);
    headPitch=lerp(0.15,1.05,smooth(clamp((t-4.5)/0.7,0,1)));
    mouth=Math.sin(clamp((t-4.7)/1.0,0,1)*Math.PI)*0.62;
    grip=smooth(clamp((t-5.25)/0.55,0,1));
    look=0.02;
  }else if(t<8){
    const u=smooth((t-6)/2);
    x=lerp(678,392,u); z=lerp(94,108,u); gait="back";
    headPitch=0.22; mouth=0.05; grip=1; look=0;
  }else if(t<10){
    const u=smooth((t-8)/2);
    x=lerp(392,448,u); z=lerp(108,150,u); gait="turn";
    headPitch=lerp(0.2,0.04,u); mouth=0.04; grip=1; look=lerp(0,-0.05,u);
  }else{
    const b=Math.sin((t-10)*2.5);
    x=448; z=150; gait="stand"; headPitch=0.03+b*0.015; mouth=0.03; grip=1; look=-0.02;
  }
  let yaw=0.02;
  if(t>=8&&t<10) yaw=lerp(0.02,1.5708,smooth((t-8)/2));
  else if(t>=10) yaw=1.5708;
  const sc=lerp(1.15,1.42,clamp((z-94)/70,0,1));
  const bob=gait==="run"?Math.abs(Math.sin(t*10.5))*3.4:gait==="walk"?Math.abs(Math.sin(t*7))*1.7:gait==="back"?Math.abs(Math.sin(t*6.2))*1.5:gait==="turn"?Math.abs(Math.sin(t*5))*1.2:gait==="stand"?Math.sin(t*2.5)*0.6:0;
  return {t,x,z,yaw,gait,crouch,mouth,grip,headPitch,look,alert,sc,bob};
}
function groundY(z){return 318+(z-100)*1.05;}
function project(px,pz,yaw,fx,rz,up,sc){
  const c=Math.cos(yaw), s=Math.sin(yaw);
  const wx=px+(fx*c-rz*s)*sc;
  const wz=pz+(fx*s+rz*c)*sc*0.42;
  const wy=up*sc;
  return {wx,wz,sx:wx,sy:groundY(wz)-wy,d:wz+up*0.02};
}

function ell(sx,sy,rx,ry,rot,c0,c1){
  ctx.save(); ctx.translate(sx,sy); ctx.rotate(rot);
  const g=ctx.createRadialGradient(-rx*0.35,-ry*0.4,2,0,0,Math.max(rx,ry));
  g.addColorStop(0,c0); g.addColorStop(1,c1);
  ctx.fillStyle=g;
  ctx.beginPath(); ctx.ellipse(0,0,Math.max(0.5,rx),Math.max(0.5,ry),0,0,6.283); ctx.fill();

}
function drawDog(p){
  const S=p.sc, yaw=p.yaw;
  const bodyUp=28-p.crouch*10+p.bob*0.35;
  const freq=p.gait==="run"?3.15:p.gait==="walk"?2.05:p.gait==="back"?1.75:p.gait==="turn"?1.35:0;
  const reach=p.gait==="run"?18:p.gait==="walk"?12:p.gait==="back"?11:p.gait==="turn"?6:0;
  const dir=p.gait==="back"?-1:1;
  const mounts=[[20,-8],[20,8],[-14,-8],[-14,8]];
  const offs=[0,0.5,0.5,0];
  const parts=[];
  function add(d,fn){parts.push({d,fn});}
  for(let i=0;i<4;i++){
    let along=i<2?3:-2, lift=0;
    if(freq){
      const ph=((p.t*freq+offs[i])%1+1)%1;
      if(ph<0.58){along=dir*reach*(1-2*(ph/0.58)); lift=0;}
      else {const u=(ph-0.58)/0.42; along=dir*(-reach+2*reach*u); lift=Math.sin(u*Math.PI)*(p.gait==="run"?11:6);}
    }
    if(p.gait==="grab"){along=i<2?5:-3; lift=0;}
    const hip=project(p.x,p.z,yaw,mounts[i][0],mounts[i][1],bodyUp-6,S);
    const paw=project(p.x,p.z,yaw,mounts[i][0]+along,mounts[i][1]*1.05,lift+3,S);
    const knee=project(p.x,p.z,yaw,mounts[i][0]+along*0.45,mounts[i][1],bodyUp-14+lift*0.2,S);
    add(hip.d-1,()=>{

      ell(paw.sx,paw.sy,5.2*S,3.2*S,0,"#f0c49a","#c47a45");

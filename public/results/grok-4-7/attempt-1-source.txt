function clamp(v,a,b){return v<a?a:v>b?b:v;}
function lerp(a,b,u){return a+(b-a)*u;}
function smooth(u){u=clamp(u,0,1);return u*u*(3-2*u);}
function hash(n){const s=Math.sin(n*127.1+311.7)*43758.5453;return s-Math.floor(s);}
function yawOf(t){
  t=clamp(t,0,12);
  if(t<8)return 0.04;
  if(t<10)return lerp(0.04,1.5708,smooth((t-8)/2));
  return 1.5708;
}
function poseOf(t){
  t=clamp(t,0,12);
  let x=210,z=80,gait="idle",spd=0,crouch=0,hp=0,mouth=0,grip=0,headY=0.55,alert=0;
  if(t<1){
    x=210;z=80;gait="idle";hp=-0.08;headY=lerp(0.62,0.08,smooth(t));alert=smooth(t);
  }else if(t<3){
    const u=smooth((t-1)/2);
    x=lerp(210,468,u);z=lerp(80,78,u);gait="run";spd=1;hp=0.02;headY=0.02;
  }else if(t<4.5){
    const u=smooth((t-3)/1.5);
    x=lerp(468,628,u);z=lerp(78,74,u);gait="walk";spd=0.72;hp=0.08;headY=0;
  }else if(t<6){
    const u=smooth((t-4.5)/1.5);
    x=lerp(628,642,u);z=74;gait="grab";spd=0;crouch=smooth(Math.min(1,(t-4.5)/0.7));
    hp=lerp(0.15,0.95,smooth(clamp((t-4.5)/0.8,0,1)));
    mouth=Math.sin(clamp((t-4.6)/0.9,0,1)*Math.PI)*0.55;
    grip=smooth(clamp((t-5.15)/0.7,0,1));
    headY=0;
  }else if(t<8){
    const u=smooth((t-6)/2);
    x=lerp(642,392,u);z=lerp(74,82,u);gait="back";spd=0.78;hp=0.22;mouth=0.06;grip=1;headY=0;
  }else if(t<10){
    const u=smooth((t-8)/2);
    x=lerp(392,430,u);z=lerp(82,118,u);gait="turn";spd=0.55;hp=lerp(0.18,0.02,u);mouth=0.04;grip=1;headY=lerp(0, -0.04,u);
  }else{
    const b=Math.sin((t-10)*2.4);
    x=430;z=118;gait="stand";spd=0;hp=0.03+b*0.02;mouth=0.03;grip=1;headY=-0.02;crouch=0;
  }
  const sc=lerp(1.05,1.28,clamp((z-78)/40,0,1));
  return {t,x,z,y:0,yaw:yawOf(t),gait,spd,crouch,hp,mouth,grip,headY,alert,sc,
    bob: (gait==="run"?Math.abs(Math.sin(t*11))*3.2:gait==="walk"||gait==="back"?Math.abs(Math.sin(t*7))*1.6:gait==="turn"?Math.abs(Math.sin(t*6))*1.1:0)};
}

function fluff(cx,cy,rx,ry,rot,c0,c1,n,seed,alpha){
  ctx.save();ctx.translate(cx,cy);ctx.rotate(rot);
  const g=ctx.createRadialGradient(-rx*0.35,-ry*0.45,Math.min(rx,ry)*0.15,0,0,Math.max(rx,ry));
  g.addColorStop(0,c0);g.addColorStop(1,c1);
  ctx.fillStyle=g;ctx.globalAlpha=alpha==null?1:alpha;
  ctx.beginPath();ctx.ellipse(0,0,rx,ry,0,0,7);ctx.fill();
  ctx.globalAlpha=0.45;ctx.strokeStyle=c1;ctx.lineWidth=1;
  for(let i=0;i<n;i++){
    const a=hash(seed+i*1.7)*6.283, rad=0.55+hash(seed+i*3.1)*0.5;
    const x1=Math.cos(a)*rx*rad, y1=Math.sin(a)*ry*rad;
    const x2=Math.cos(a)*(rx*(rad+0.16)), y2=Math.sin(a)*(ry*(rad+0.2));
    ctx.beginPath();ctx.moveTo(x1,y1);ctx.lineTo(x2,y2);ctx.stroke();
  }
  ctx.restore();
}

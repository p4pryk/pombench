function clamp(v,a,b){return v<a?a:v>b?b:v;}
function lerp(a,b,u){return a+(b-a)*u;}
function smooth(u){u=clamp(u,0,1);return u*u*(3-2*u);}
function hash(n){const s=Math.sin(n*127.1+311.7)*43758.5453;return s-Math.floor(s);}
function pose(time){
  const t=time<0?0:time>12?12:time;
  let x=188,z=102,gait="idle",crouch=0,mouth=0,grip=0,headPitch=0.16,look=0.55,alert=0,yaw=0.04;
  if(t<1){
    alert=smooth(t); look=lerp(0.62,0.02,alert); headPitch=lerp(0.18,0.03,alert);
  }else if(t<3){
    const u=smooth((t-1)/2);
    x=lerp(188,498,u); z=lerp(102,98,u); gait="run"; headPitch=0.05; look=0.02;
  }else if(t<4.5){
    const u=smooth((t-3)/1.5);
    x=lerp(498,668,u); z=lerp(98,96,u); gait="walk"; headPitch=0.08; look=0;
  }else if(t<6){
    const u=smooth((t-4.5)/1.5);
    x=lerp(668,692,u); z=96; gait="grab";
    crouch=smooth(clamp((t-4.5)/0.5,0,1))*(1-smooth(clamp((t-5.55)/0.45,0,1))*0.25);
    headPitch=lerp(0.12,1.02,smooth(clamp((t-4.5)/0.65,0,1)));
    mouth=Math.sin(clamp((t-4.65)/0.95,0,1)*Math.PI)*0.66;
    grip=smooth(clamp((t-5.2)/0.5,0,1));
    look=0;
  }else if(t<8){
    const u=smooth((t-6)/2);
    x=lerp(692,404,u); z=lerp(96,112,u); gait="back";
    headPitch=0.2; mouth=0.05; grip=1; look=0;
  }else if(t<10){
    const u=smooth((t-8)/2);
    x=lerp(404,456,u); z=lerp(112,158,u); gait="turn";
    yaw=lerp(0.04,1.5708,u);
    headPitch=lerp(0.18,0.04,u); mouth=0.04; grip=1; look=lerp(0,-0.04,u);
  }else{
    const b=Math.sin((t-10)*2.4);
    x=456; z=158; yaw=1.5708; gait="stand";
    headPitch=0.03+b*0.012; mouth=0.03; grip=1; look=-0.02;
  }
  if(t<8) yaw=0.04;
  const sc=lerp(1.12,1.46,clamp((z-94)/72,0,1));
  const bob=gait==="run"?Math.abs(Math.sin(t*10.2))*3.2:gait==="walk"?Math.abs(Math.sin(t*6.6))*1.6:gait==="back"?Math.abs(Math.sin(t*5.8))*1.4:gait==="turn"?Math.abs(Math.sin(t*4.4))*1.1:gait==="stand"?Math.sin(t*2.4)*0.55:Math.sin(t*2)*0.35;
  return {t,x,z,yaw,gait,crouch,mouth,grip,headPitch,look,alert,sc,bob};
}
function groundY(z){return 322+(z-100)*1.02;}
function project(px,pz,yaw,fx,rz,up,sc){
  const c=Math.cos(yaw), s=Math.sin(yaw);
  const wx=px+(fx*c-rz*s)*sc;
  const wz=pz+(fx*s+rz*c)*sc*0.46;
  return {wx,wz,sx:wx,sy:groundY(wz)-up*sc,d:wz+up*0.015};
}
function ell(sx,sy,rx,ry,rot,c0,c1){
  ctx.save(); ctx.translate(sx,sy); ctx.rotate(rot);
  const g=ctx.createRadialGradient(-rx*0.32,-ry*0.38,1,0,0,Math.max(rx,ry)*1.05);
  g.addColorStop(0,c0); g.addColorStop(1,c1);
  ctx.fillStyle=g;
  ctx.beginPath(); ctx.ellipse(0,0,Math.max(0.6,rx),Math.max(0.6,ry),0,0,6.2832); ctx.fill();
  ctx.restore();
}

function boneState(p){
  const rest=project(690,100,0,0,0,7,1);
  const mouth=project(p.x,p.z,p.yaw,42+p.headPitch*3,0,30-p.crouch*8-p.headPitch*12+p.bob*0.2,p.sc);
  const g=p.grip;
  const sx=lerp(rest.sx,mouth.sx,g), sy=lerp(rest.sy,mouth.sy,g);
  const ang=lerp(0.15, p.yaw>1? -0.15:0.35, g);
  const inside = g<0.5 && sx>610 && sx<736;
  return {sx,sy,ang,sc:lerp(1,p.sc*0.72,g),inside,d:lerp(rest.d,mouth.d,g)};
}

(() => {
  const c = document.getElementById('stage');
  const ctx = c.getContext('2d', { alpha:false });
  const W=960,H=540;

  const clamp=(v,a,b)=>v<a?a:v>b?b:v;
  const lerp=(a,b,t)=>a+(b-a)*t;
  const smooth=(t)=>t*t*(3-2*t);
  const smoothstep=(a,b,t)=>smooth(clamp((t-a)/(b-a),0,1));
  const sgn=(x)=>x<0?-1:1;

  // Deterministic texture
  function hash(n){ n = Math.imul(n^0x9e3779b9, 2654435761); n ^= n>>>16; return (n>>>0)/4294967295; }
  function rand2(i,j){ return hash((i*1315423911) ^ (j*2654435761)); }

  // Fur palette
  const fur1 = '#f7c88b'; // cream/orange
  const fur2 = '#e89b57'; // warm orange
  const fur3 = '#c8783b'; // deep orange
  const fur4 = '#fff1d1'; // highlights
  const fur5 = '#8b4e22'; // shadow
  const dark = '#2a1a12';

  // Yard
  const yard = {
    groundY: 420,
    left: 0,
    right: W,
    grassTop: 300
  };

  const house = {
    x: 640,
    y: 260,
    w: 250,
    h: 260,
    roofH: 80,
    doorW: 90,
    doorH: 145,
    doorX: 705,
    doorY: 310
  };

  // Puddle plane (still water)
  const puddle = {
    x1: 320, x2: 450,
    yTop: 404, // water surface line
    yBot: 430, // thickness/extent
    wobble: 0
  };

  function drawGround(time){
    // base ground
    const g = ctx.createLinearGradient(0,300,0,H);
    g.addColorStop(0,'#87c070');
    g.addColorStop(0.55,'#6cad57');
    g.addColorStop(1,'#5a9c4b');
    ctx.fillStyle=g;
    ctx.fillRect(0,280,W,H-280);

    // worn dirt path
    ctx.save();
    ctx.globalAlpha=0.95;
    ctx.fillStyle='#6a6a3a';
    ctx.beginPath();
    ctx.moveTo(0,410);
    ctx.quadraticCurveTo(300,375, 560,395);
    ctx.quadraticCurveTo(720,415, 960,390);
    ctx.lineTo(960,540);
    ctx.lineTo(0,540);
    ctx.closePath();
    ctx.fill();

    // subtle texture dots
    ctx.globalAlpha=0.2;
    for(let i=0;i<130;i++){
      const x = (i*73)%W;
      const y = 320 + ((i*97)%200);
      const r = 0.8 + (rand2(i,17)*2.2);
      if(y<405 || y>520) continue;
      ctx.fillStyle = (hash(i*13)^hash(i*7))>0.5?'#2f5b23':'#2a4f1f';
      ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2); ctx.fill();
    }
    ctx.restore();

    // shadow in front of house
    ctx.save();
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#1d1d14';
    ctx.beginPath();
    ctx.ellipse(735,426,70,18,0,0,Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  function drawHouse(){
    // main wall
    ctx.save();
    const {x,y,w,h,roofH,doorX,doorY,doorW,doorH}=house;

    // roof
    const roof = ctx.createLinearGradient(0,y,0,y+roofH);
    roof.addColorStop(0,'#6f4b2d');
    roof.addColorStop(1,'#4e361f');
    ctx.fillStyle=roof;
    ctx.beginPath();
    ctx.moveTo(x-10,y+roofH);
    ctx.lineTo(x+w/2,y-10);
    ctx.lineTo(x+w+10,y+roofH);
    ctx.lineTo(x+w,y+roofH);
    ctx.lineTo(x+w/2,y-2);
    ctx.lineTo(x,y+roofH);
    ctx.closePath();
    ctx.fill();

    // roof edge highlight
    ctx.globalAlpha=0.25;
    ctx.fillStyle='#c58b5e';
    ctx.beginPath();
    ctx.moveTo(x+w/2-60,y+roofH-10);
    ctx.quadraticCurveTo(x+w/2,y+roofH-30,x+w/2+60,y+roofH-10);
    ctx.lineTo(x+w/2+60,y+roofH);
    ctx.lineTo(x+w/2-60,y+roofH);
    ctx.closePath();
    ctx.fill();
    ctx.globalAlpha=1;

    // walls
    const wall = ctx.createLinearGradient(x,y,x+w,y+h);
    wall.addColorStop(0,'#caa27b');
    wall.addColorStop(0.5,'#b88f67');
    wall.addColorStop(1,'#9b724f');
    ctx.fillStyle=wall;
    ctx.fillRect(x,y,w,h);

    // vertical grooves
    ctx.globalAlpha=0.22;
    ctx.fillStyle='#3b2a1e';
    for(let i=0;i<10;i++){
      const gx=x+18+i*(w-36)/9;
      ctx.fillRect(gx,y+10,2, h-20);
    }
    ctx.globalAlpha=1;

    // base trim
    ctx.fillStyle='#7a5639';
    ctx.fillRect(x,y+h-18,w,18);

    // door frame
    ctx.save();
    ctx.fillStyle='#5b3f2a';
    ctx.globalAlpha=0.95;
    // outer frame
    ctx.fillRect(doorX-14, doorY-8, doorW+28, doorH+20);

    // inner frame (rim)
    ctx.globalAlpha=1;
    ctx.fillStyle='#9a6e4c';
    ctx.fillRect(doorX-6, doorY, doorW+12, doorH);

    // doorway interior
    // occlusion: paint "depth"
    const interior = ctx.createLinearGradient(doorX,doorY,doorX+doorW,doorY+doorH);
    interior.addColorStop(0,'#2b2016');
    interior.addColorStop(1,'#1a1310');
    ctx.fillStyle=interior;
    ctx.fillRect(doorX, doorY+10, doorW, doorH-10);

    // door shadow opening edge
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#000';
    ctx.fillRect(doorX,doorY+10,8,doorH-10);
    ctx.fillRect(doorX+doorW-8,doorY+10,8,doorH-10);
    ctx.globalAlpha=1;

    // door lip
    ctx.globalAlpha=0.2;
    ctx.fillStyle='#fff0d1';
    ctx.fillRect(doorX+6,doorY+2,doorW-12,6);
    ctx.globalAlpha=1;

    // doorway "floor" hint
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#3a2b21';
    ctx.beginPath();
    ctx.roundRect(doorX+10, doorY+doorH-18, doorW-20, 12, 6);
    ctx.fill();
    ctx.globalAlpha=1;

    ctx.restore();

    // house front silhouette for occlusion correctness
    // (we'll use door frame rectangles as occluders in draw order)
    ctx.restore();
  }

  // Draw puddle surface and reflection boundary
  function drawPuddleSurface(){
    ctx.save();
    // water body
    const waterGrad = ctx.createLinearGradient(0,puddle.yTop,0,puddle.yBot);
    waterGrad.addColorStop(0,'rgba(80,140,190,0.65)');
    waterGrad.addColorStop(1,'rgba(25,60,100,0.78)');
    ctx.fillStyle=waterGrad;
    ctx.strokeStyle='rgba(180,230,255,0.35)';
    ctx.lineWidth=2;

    // rounded puddle shape
    const rx = (puddle.x2-puddle.x1)/2;
    const mid = (puddle.x1+puddle.x2)/2;
    const topY=puddle.yTop, botY=puddle.yBot;

    ctx.beginPath();
    ctx.moveTo(mid-rx, topY);
    ctx.quadraticCurveTo(mid-rx*0.98, (topY+botY)/2, mid, botY);
    ctx.quadraticCurveTo(mid+rx*0.98, (topY+botY)/2, mid+rx, topY);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    // water specular highlights line
    ctx.globalAlpha=0.45;
    ctx.strokeStyle='rgba(210,245,255,0.55)';
    ctx.beginPath();
    ctx.moveTo(puddle.x1+8, puddle.yTop+5);
    ctx.quadraticCurveTo(mid, puddle.yTop+18, puddle.x2-8, puddle.yTop+5);
    ctx.stroke();

    ctx.globalAlpha=1;
    ctx.restore();
  }

  function drawBone(x,y,depth,grabbed,holdTangent){
    // depth: used to scale/occlusion (fake)
    const s = depth;
    const w = 22*s, h=40*s;
    const rot = holdTangent || 0;
    ctx.save();
    ctx.translate(x,y);
    ctx.rotate(rot);
    // bone body
    const body = ctx.createLinearGradient(-w/2,0,w/2,0);
    body.addColorStop(0,'#f6f0dc');
    body.addColorStop(0.5,'#e8dfc6');
    body.addColorStop(1,'#cdbf9b');
    ctx.fillStyle=body;
    ctx.strokeStyle='rgba(40,30,20,0.35)';
    ctx.lineWidth=Math.max(1,1.2*s);

    // simplified dog-bone: bar + two pads
    ctx.beginPath();
    ctx.roundRect(-w*0.45,-h*0.5,w*0.9,h, w*0.35);
    ctx.fill();
    ctx.stroke();

    // grooves
    ctx.globalAlpha=0.5;
    ctx.fillStyle='#b9ad8a';
    ctx.beginPath();
    ctx.roundRect(-w*0.35,-h*0.15,w*0.7,h*0.7, w*0.25);
    ctx.fill();

    ctx.globalAlpha=1;
    // small shadow holes
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#6a5a40';
    ctx.beginPath();
    ctx.roundRect(-w*0.55,-h*0.12,w*0.36,h*0.22, w*0.15);
    ctx.roundRect(w*0.19,-h*0.12,w*0.36,h*0.22, w*0.15);
    ctx.fill();
    ctx.globalAlpha=1;

    // scale squish for mouth hold
    ctx.restore();
  }

  function drawPomeranian(p){
    // p: {x,y,yJump,yLean,yaw,pitch,runT,phase,air,frontMode,scale,zOrder}
    const {x,y,yaw,yJump,air,runT,phase,frontMode,scale,enterDepth,pitch} = p;
    // frontMode: toward viewer (0 side, 1 more frontal)
    const s = scale;

    // helper: shaded ellipse
    function blob(ax,ay,rx,ry,color,shadowAlpha){
      ctx.save();
      ctx.translate(ax,ay);
      ctx.fillStyle=color;
      if(shadowAlpha!=null){
        ctx.globalAlpha=shadowAlpha;
        ctx.fill();
      }else ctx.fill();
      ctx.restore();
    }

    // compute approximate body orientation
    const yaw01 = clamp((yaw+0.7)/1.4,0,1); // map yaw in [-0.7..0.7]
    const bodyFront = frontMode; // already smooth

    // scale based on y (fake perspective)
    const persp = 1 + (y-400)*0.0009;
    const k = s*persp;

    // Determine components offsets relative to body anchor (x,y is mid-hip approx)
    const dx = Math.sin(yaw)*18*k;
    const dz = Math.cos(yaw)*10*k;

    // Fur ruff and silhouette
    // Tail: plumed and curled over back. Keep curve with yaw.
    const tailCurl = 1 - bodyFront*0.1;

    // Motion-based tilt/lean
    const lean = p.yLean || 0;
    const headBob = p.headBob || 0;

    // draw order for body parts: back tail -> body -> head/ruff -> legs -> tail tip details
    ctx.save();
    ctx.translate(x,y - yJump*k);

    // vertical body pitch
    ctx.rotate(lean);

    // shadow under body (contact)
    // only if not entering water; but always draw small oval shadow
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#0e0e0b';
    ctx.beginPath();
    ctx.ellipse(dx*0.25, 70*k, 38*k, 12*k, yaw*0.3, 0, Math.PI*2);
    ctx.fill();
    ctx.globalAlpha=1;

    // tail (plume behind)
    const tailAng = -2.0 + yaw*0.8;
    ctx.save();
    ctx.rotate(tailAng);
    ctx.translate(26*k, -14*k);
    ctx.globalAlpha=1;
    const plumeGrad = ctx.createLinearGradient(-10,0,20,-40);
    plumeGrad.addColorStop(0,'#2c1b12');
    plumeGrad.addColorStop(0.15,'#d88a45');
    plumeGrad.addColorStop(0.55,'#f2c47f');
    plumeGrad.addColorStop(1,'#9e5a2a');
    ctx.fillStyle=plumeGrad;
    ctx.strokeStyle='rgba(60,35,20,0.4)';
    ctx.lineWidth=1.1*k;

    ctx.beginPath();
    // curled plume
    ctx.moveTo(0,0);
    ctx.quadraticCurveTo(18*k,-8*k, 20*k,-28*k);
    ctx.quadraticCurveTo(18*k,-50*k, -8*k,-50*k);
    ctx.quadraticCurveTo(-24*k,-44*k, -18*k,-24*k);
    ctx.quadraticCurveTo(-12*k,-12*k, 0,0);
    ctx.closePath();
    ctx.fill();
    ctx.globalAlpha=0.75;
    ctx.beginPath();
    ctx.moveTo(2*k,0);
    ctx.quadraticCurveTo(16*k,-10*k, 18*k,-30*k);
    ctx.strokeStyle='rgba(255,235,200,0.55)';
    ctx.stroke();
    ctx.restore();

    // body
    const bodyGrad = ctx.createLinearGradient(-30*k,-60*k,30*k,80*k);
    bodyGrad.addColorStop(0,'#ffd9a3');
    bodyGrad.addColorStop(0.22,fur2);
    bodyGrad.addColorStop(0.6,fur1);
    bodyGrad.addColorStop(1,fur5);

    // back
    ctx.fillStyle=bodyGrad;
    ctx.strokeStyle='rgba(60,30,15,0.35)';
    ctx.lineWidth=1.2*k;

    // torso volume
    const bodyRx = 46*k, bodyRy = 34*k;
    const bodyYaw = yaw*0.35;
    ctx.save();
    ctx.rotate(bodyYaw);
    ctx.beginPath();
    ctx.ellipse(0,0, bodyRx, bodyRy, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.stroke();

    // ruff + neck / chest volume
    const ruff = ctx.createLinearGradient(-20*k,-10*k,60*k,50*k);
    ruff.addColorStop(0,'#fff2cf');
    ruff.addColorStop(0.3,'#f1b574');
    ruff.addColorStop(1,'#a75a2d');
    ctx.fillStyle=ruff;
    ctx.globalAlpha=1;
    // chest fluff
    ctx.beginPath();
    ctx.ellipse(10*k,-30*k, 34*k, 26*k, -0.25, 0, Math.PI*2);
    ctx.fill();

    // head/ruff base
    ctx.fillStyle=ruff;
    ctx.beginPath();
    ctx.ellipse(54*k,-42*k, 28*k, 22*k, -0.15, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();

    // Legs
    // gait: two legs contact alternation
    const stride = phase; // 0..1
    const step = (u)=>Math.sin((u+0.0)*Math.PI*2);
    // contact pattern
    const fwd = (Math.sin(runT*6 + 0.2) > 0 ? 1 : -1);

    function legPose(side){ // side -1 back, +1 front
      // compute which legs tuck (front-right/back-left vs other)
      // Use phase to pick fore/hind
      const tA = (runT*2.2 + phase*1.0);
      const lift = air ? 1 : 0;
      // Determine alternating lift for left/right
      const alt = Math.sin(tA*2 + side*1.1 + (frontMode*0.5));
      const liftAmt = air ? clamp(0.55 + 0.45*alt,0,1) : clamp(0.15 + 0.85*Math.max(0,alt),0,1);

      const baseX = (side>0 ? 22*k : -12*k);
      const baseY = (side>0 ? 25*k : 40*k);
      // lateral offset from yaw
      const lat = (yaw*0.5) * 18*k;
      const sideLat = (frontMode*0.2 + (yaw>0?1:-1)*0.1);

      // four legs: treat as two pairs
      // approximate: front pair slightly forward, hind pair slightly back
      const frontLift = side>0 ? liftAmt : 0.7*liftAmt;
      const zLift = (air ? 0.55 : 0);
      const legLift = (air ? frontLift : frontLift*0.25) + (air?0.08:0);

      // paw arc
      const arc = (Math.max(0, step(tA*0.5+side*0.3)))*0.8;
      const pawY = baseY + (air ? -legLift*20*k : 0) + (-arc*5*k) + (air?0:0);
      const pawX = baseX + (air ? 10*k*Math.sin(tA*1.4+side) : 6*k*Math.sin(tA*2.1+side*0.2));
      const pawFwd = (side>0 ? 1 : -1);

      return {baseX,baseY,lat,sideLat,pawX,pawY,legLift,pawFwd, frontLift};
    }

    // Draw legs in front/back layering
    const legs = [
      {side:+1, idx:0, lr:-1},
      {side:+1, idx:1, lr:+1},
      {side:-1, idx:2, lr:-1},
      {side:-1, idx:3, lr:+1}
    ];

    // legs back first
    ctx.save();
    ctx.globalAlpha=1;
    for(const L of legs.sort((a,b)=>a.side-b.side)){
      const pose = legPose(L.side);
      const px = pose.pawX + pose.lat*L.lr;
      const py = pose.pawY + (L.side>0 ? -4*k : 2*k) + (frontMode*0.5)*L.lr;

      const upperColor = '#c98546';
      const furColor = L.side>0 ? '#e4a25e' : '#d38a49';
      const shadow = air?0.35:0.55;

      // limb shadow on ground when close to ground (not in house)
      if(!air){
        ctx.globalAlpha=shadow;
        ctx.fillStyle='#0e0e0b';
        ctx.beginPath();
        ctx.ellipse(px, 82*k, 10*k, 5.5*k, yaw*0.2, 0, Math.PI*2);
        ctx.fill();
        ctx.globalAlpha=1;
      }

      // leg cylinder
      const legLen = 46*k*(L.side>0?0.95:1.02);
      const lift = pose.legLift;
      const kneeY = lerp(36*k, 18*k, clamp(lift,0,1));
      const ankleY = py;

      // when airborne, tuck
      const tuck = air ? 0.55*lift : 0;
      const bend = lerp(0.9,0.2,tuck);

      ctx.save();
      ctx.translate(px, ankleY);
      ctx.rotate(-0.2 + (air?0.25*tuck:0) + lean*0.1);

      // upper fur shading
      const legGrad = ctx.createLinearGradient(0,-legLen,0,0);
      legGrad.addColorStop(0,'#8e4a22');
      legGrad.addColorStop(0.2,furColor);
      legGrad.addColorStop(1,'#f3c37c');

      ctx.fillStyle=legGrad;
      ctx.strokeStyle='rgba(60,30,15,0.35)';
      ctx.lineWidth=1.1*k;

      ctx.beginPath();
      ctx.roundRect(-6*k, -legLen*bend, 12*k, legLen, 6*k);
      ctx.fill();
      ctx.stroke();

      // paw
      const pawGrad = ctx.createLinearGradient(-8*k,-4*k,8*k,10*k);
      pawGrad.addColorStop(0,'#f7d9a8');
      pawGrad.addColorStop(1,'#c98749');
      ctx.fillStyle=pawGrad;
      ctx.strokeStyle='rgba(30,18,10,0.35)';
      ctx.lineWidth=1*k;
      ctx.beginPath();
      ctx.roundRect(-10*k, -3*k, 20*k, 8*k + (air?-2*k:0), 6*k);
      ctx.fill();
      ctx.stroke();

      // toe hints
      ctx.globalAlpha=0.55;
      ctx.fillStyle='#3a2418';
      if(!air){
        ctx.fillRect(-6*k,0,3*k,2*k);
        ctx.fillRect(3*k,0,3*k,2*k);
      }
      ctx.globalAlpha=1;

      ctx.restore();
    }
    ctx.restore();

    // Head: ears, eyes, muzzle
    const headX = 52*k;
    const headY = -48*k + headBob*k;
    const headRot = yaw*0.25 + pitch;

    ctx.save();
    ctx.translate(headX,headY);
    ctx.rotate(headRot);

    // ears
    function ear(which){
      ctx.save();
      ctx.translate(which*16*k, -16*k);
      ctx.rotate(which*0.08 + yaw*0.15);
      const earGrad=ctx.createLinearGradient(-5*k,-20*k,10*k,15*k);
      earGrad.addColorStop(0,'#6b3a19');
      earGrad.addColorStop(0.25,fur2);
      earGrad.addColorStop(1,'#ffd9a8');
      ctx.fillStyle=earGrad;
      ctx.strokeStyle='rgba(40,20,10,0.35)';
      ctx.lineWidth=1.1*k;
      ctx.beginPath();
      ctx.moveTo(0,-2*k);
      ctx.quadraticCurveTo(-9*k,-15*k,-5*k,-30*k);
      ctx.quadraticCurveTo(4*k,-40*k, 14*k,-28*k);
      ctx.quadraticCurveTo(20*k,-13*k, 8*k, -2*k);
      ctx.closePath();
      ctx.fill(); ctx.stroke();

      // inner
      ctx.globalAlpha=0.28;
      ctx.fillStyle='#b15b2a';
      ctx.beginPath();
      ctx.moveTo(0,-3*k);
      ctx.quadraticCurveTo(-6*k,-12*k,-2*k,-23*k);
      ctx.quadraticCurveTo(4*k,-31*k, 11*k,-22*k);
      ctx.quadraticCurveTo(15*k,-11*k, 7*k,-3*k);
      ctx.closePath();
      ctx.fill();
      ctx.globalAlpha=1;
      ctx.restore();
    }
    ear(-1); ear(1);

    // eye
    ctx.fillStyle='#13100d';
    const eyeX = (bodyFront>0.35 ? 10*k : 14*k);
    const eyeY = -6*k + (bodyFront*0.2)*(-1*k);
    ctx.beginPath();
    ctx.ellipse(eyeX,eyeY, 4.2*k, 5.1*k, 0,0,Math.PI*2);
    ctx.fill();

    // highlight
    ctx.globalAlpha=0.7;
    ctx.fillStyle='rgba(255,255,255,0.7)';
    ctx.beginPath();
    ctx.arc(eyeX-1.1*k, eyeY-1.8*k, 1.3*k, 0, Math.PI*2);
    ctx.fill();
    ctx.globalAlpha=1;

    // muzzle
    const muGrad=ctx.createLinearGradient(-10*k,-20*k,20*k,10*k);
    muGrad.addColorStop(0,'#f7e1b9');
    muGrad.addColorStop(0.4,'#f1b974');
    muGrad.addColorStop(1,'#a15b2d');
    ctx.fillStyle=muGrad;
    ctx.strokeStyle='rgba(40,20,10,0.35)';
    ctx.lineWidth=1.1*k;

    // muzzle volume
    ctx.beginPath();
    ctx.ellipse(18*k, 5*k, 18*k, 13*k, 0, 0, Math.PI*2);
    ctx.fill(); ctx.stroke();

    // nose
    ctx.fillStyle=dark;
    ctx.beginPath();
    ctx.ellipse(23*k, 8*k, 4.2*k, 3.4*k, 0.3, 0, Math.PI*2);
    ctx.fill();

    // mouth line when holding
    if(p.boneInMouth){
      ctx.globalAlpha=0.9;
      ctx.strokeStyle='rgba(20,10,10,0.6)';
      ctx.lineWidth=1.5*k;
      ctx.beginPath();
      ctx.moveTo(18*k, 10*k);
      ctx.quadraticCurveTo(26*k, 10*k, 31*k, 7.5*k);
      ctx.stroke();
      ctx.globalAlpha=1;
    }

    // ruff highlight
    ctx.globalAlpha=0.35;
    ctx.strokeStyle='rgba(255,250,230,0.65)';
    ctx.lineWidth=2.2*k;
    ctx.beginPath();
    ctx.ellipse(-8*k,-20*k, 24*k, 18*k, -0.2, 0, Math.PI*2);
    ctx.stroke();
    ctx.globalAlpha=1;

    ctx.restore();

    // Fur texture "strands" (deterministic)
    ctx.save();
    ctx.globalAlpha=0.35;
    for(let i=0;i<38;i++){
      const xi = (i*37)%120;
      const yi = (i*53)%90;
      const gx = -70*k + xi*1.2*k;
      const gy = -55*k + yi*1.15*k + (i%2? 10*k:0);
      const ang = (hash(100+i*7)-0.5)*1.2;
      const len = (3+hash(200+i*3)*7)*k;
      ctx.strokeStyle = (i%3==0)?'rgba(255,245,210,0.55)':'rgba(90,45,20,0.35)';
      ctx.lineWidth = 0.7*k;
      ctx.beginPath();
      ctx.moveTo(gx,gy);
      ctx.lineTo(gx+Math.cos(ang)*len, gy+Math.sin(ang)*len);
      ctx.stroke();
    }
    ctx.globalAlpha=1;
    ctx.restore();

    // Tail tip flutter subtle
    ctx.save();
    ctx.globalAlpha=0.9;
    ctx.strokeStyle='rgba(255,240,200,0.45)';
    ctx.lineWidth=2.2*k;
    const flutter = air ? 0.3 : 0.6;
    ctx.beginPath();
    ctx.moveTo(38*k, -74*k);
    ctx.quadraticCurveTo(44*k, -68*k+flutter*k, 42*k, -56*k);
    ctx.stroke();
    ctx.restore();

    ctx.restore();
  }

  // Compute dog pose across time with coherent jump + entry + retrieval + reverse
  function poseAt(t){
    t = clamp(t,0,12);
    // phases
    const xStart=170, xPuddleNear=310, xPuddleFar=500, xDoor=720, xInside=740, xExit=665, xFinal=640;
    const yGround=405, yAirBase=390;

    // yaw transitions
    let yaw=-0.25; // running mostly toward house (side-ish, 3/4)
    // 0-10: mostly right-facing; 8-10: turn to viewer
    if(t<8) yaw = lerp(-0.25, 0.18, smoothstep(6.8,8,t));
    else yaw = lerp(0.18, 0.38, smoothstep(8,10,t));

    // enter/back pose depth scaling
    let scale = 1.0;
    let enter = 0;

    // x movement and jump
    let x=0, y=0, yJump=0, air=false;
    if(t<=1){
      // establish: dog near bone and doorway approach start
      const tt=t/1;
      x = lerp(150, 220, tt);
      y = yGround + Math.sin(tt*Math.PI)*-6;
      air=false;
    } else if(t<=3){
      // run + jump over puddle (near x~320-450)
      const tt = (t-1)/2; // 0..1
      // anticipate: short stride and lower
      const pre = smoothstep(0,0.35,tt);
      const runSpeed = 560; // approx in 2 sec
      const xRun = lerp(220, 615, tt);
      // puddle center x ~385; jump window ~1.55..2.55
      const jumpStart=0.28, jumpEnd=0.72; // in tt terms
      const jp = smoothstep(jumpStart,jumpEnd,tt);
      const jp2 = smoothstep(jumpStart, (jumpStart+jumpEnd)/2, tt);
      air = (tt>jumpStart && tt<jumpEnd);
      // create arc height
      let h=0;
      if(tt>=jumpStart && tt<=jumpEnd){
        const u=(tt-jumpStart)/(jumpEnd-jumpStart); // 0..1
        // peak
        h = Math.sin(Math.PI*u)*1.0;
      }
      yJump = h*52; // pixels up (since we subtract yJump*k in draw)
      // land naturally on dry: bring down before doorway continuation
      const yLand = yGround - (air?0: (tt-jumpEnd)*18);
      y = yLand - pre*3 + (air?0:-2*Math.sin(tt*3.5)*0.7);
      x = xRun;
      // ensure whole body clears far edge: shift tiny forward during peak
      if(air) x += (h*10);
    } else if(t<=4.5){
      // entering doorway head first (3-4.5)
      const tt=(t-3)/1.5; // 0..1.5? actually 1.5
      const inT = smoothstep(0,1, (t-3)/1.5);
      air=false;
      // dog moves into house: x increases slightly and scale slightly reduces due to depth
      x = lerp(620, 735, inT);
      y = yGround - 6*inT;
      enter = inT;
      scale = lerp(1.02, 0.92, inT);
      // also yaw slightly turns to align with doorway (still toward right)
      yaw = lerp(-0.25, -0.05, inT);
    } else if(t<=6){
      // retrieval: lower head, grab bone inside (4.5-6)
      const tt=(t-4.5)/1.5;
      const grabT = smoothstep(0,1,tt);
      air=false;
      x = lerp(735, 728, grabT); // subtle adjustment at threshold
      y = yGround - 7 - 16*grabT;
      // bone grip triggers mouth alignment
      scale = lerp(0.92, 0.9, grabT);
      yaw = lerp(-0.05, -0.02, grabT);
    } else if(t<=8){
      // back out with bone (6-8)
      const tt=(t-6)/2;
      const outT = smoothstep(0,1,tt);
      air=false;
      x = lerp(728, 680, outT);
      y = lerp(yGround-23, yGround-5, outT);
      scale = lerp(0.9, 1.01, outT);
      yaw = lerp(-0.02, 0.02, outT);
    } else if(t<=10){
      // turn toward viewer (8-10)
      const tt=(t-8)/2;
      const turnT = smoothstep(0,1,tt);
      air=false;
      x = lerp(680, 640, turnT);
      y = lerp(yGround-5, yGround-2, turnT);
      scale = lerp(1.01, 1.02, turnT);
      yaw = lerp(0.02, 0.42, turnT); // to more frontal
    } else {
      x = 640;
      y = yGround-2;
      air=false;
      yaw = 0.42;
      scale = 1.02;
    }

    // head bob and lean from velocity
    // derive approximate dx/dt
    const dt=1/60;
    const p2=poseAt(Math.min(12,t+dt)); // recursion risk
    // avoid recursion by simple formula:
    const runPhase = t;
    const vel = (t<3)? 1:0;
    let yLean = (t<3)? -0.08*Math.sin(runPhase*6) : (t>=8&&t<10?0.06*Math.sin((t-8)*5):0.02*Math.sin(t*3));
    // Jump pitch
    let pitch = 0;
    if(air) pitch = -0.06 + 0.07*Math.sin((t-1)*(Math.PI));
    // Phase for gait
    const phase = (t*3.1);
    const runT = t;

    // bone state
    const boneInMouth = (t>=5.0);
    const frontMode = clamp((t-8)/2,0,1); // 0->side,1->front

    // compute pose yLean more coherently using jump arc
    if(t>=1 && t<=3){
      yLean = -0.14*clamp((t-1)/0.4,0,1) + 0.10*(air?1:0) + 0.06*Math.sin(t*10);
    }
    if(t>=3 && t<=4.5) yLean = -0.05 - 0.04*smoothstep(3,4.5,t);
    if(t>=6 && t<=8) yLean = 0.04*smoothstep(6,8,t);
    if(t>=8 && t<=10) yLean = 0.06*smoothstep(8,9.2,t) - 0.02*smoothstep(9.2,10,t);

    return {x,y,yJump,air,yaw,yLean,pitch,phase,runT,frontMode,scale,enterDepth:enter,boneInMouth};
  }

  // Reflection: draw dog mirrored over puddle surface.
  function drawDogWithReflection(t, dogPose){
    // reflection only on/near puddle (during 1..3)
    const show = (t>=1.0 && t<=3.05);
    drawPuddleSurface(); // surface after reflections or before? We'll draw reflection before and then overlay surface.

    if(!show) return;

    // clip to puddle shape and draw reflected dog
    ctx.save();
    // create same rounded shape as water
    const rx = (puddle.x2-puddle.x1)/2;
    const mid = (puddle.x1+puddle.x2)/2;
    const topY=puddle.yTop, botY=puddle.yBot;

    ctx.beginPath();
    ctx.moveTo(mid-rx, topY);
    ctx.quadraticCurveTo(mid-rx*0.98, (topY+botY)/2, mid, botY);
    ctx.quadraticCurveTo(mid+rx*0.98, (topY+botY)/2, mid+rx, topY);
    ctx.closePath();
    ctx.clip();

    // mirrored transform around yTop (water surface)
    const ySurf = puddle.yTop;
    const reflected = {...dogPose};
    reflected.yJump = 0; // dogPose already uses yJump; reflection mirrors final body position.
    // We'll mirror by flipping around ySurf in drawing space:
    // drawPomeranian uses anchor y already. We'll draw by flipping context vertically and adjusting.
    ctx.save();
    // flip around ySurf:
    ctx.translate(0, 2*ySurf);
    ctx.scale(1,-1);

    // damp reflection
    ctx.globalAlpha = 0.58;
    // slight darkening and blur-ish via multiple faint passes
    for(let k=0;k<2;k++){
      ctx.save();
      ctx.globalAlpha = 0.58 - k*0.12;
      ctx.translate((k?2:0), (k?1:0));
      drawPomeranian({
        ...dogPose,
        // reflected anchor y:
        // our flip already handles, so just draw same pose
        boneInMouth:false, // during approach it isn't grabbed yet anyway
        scale: dogPose.scale*0.98
      });
      ctx.restore();
    }

    // crisp-ish boundary: fade with alpha gradient at water line
    ctx.restore();
    ctx.restore();

    // add reflection fade near boundary
    ctx.save();
    ctx.globalAlpha=0.9;
    const grad = ctx.createLinearGradient(0,puddle.yTop-5,0,puddle.yBot);
    grad.addColorStop(0,'rgba(120,200,255,0.0)');
    grad.addColorStop(0.35,'rgba(120,200,255,0.18)');
    grad.addColorStop(1,'rgba(60,120,190,0.35)');
    ctx.fillStyle=grad;
    ctx.beginPath();
    ctx.moveTo(puddle.x1, puddle.yTop);
    ctx.lineTo(puddle.x2, puddle.yTop);
    ctx.lineTo(puddle.x2, puddle.yBot);
    ctx.lineTo(puddle.x1, puddle.yBot);
    ctx.closePath();
    ctx.fill();
    ctx.restore();

    // surface drawn already
  }

  function drawBoneAtTime(t, dogPose){
    // single bone initial resting inside/near house doorway, then grabbed and carried.
    const grabbedT = smoothstep(4.5,6,t);
    const carryT = smoothstep(6,8,t);
    const boneRestX = house.doorX + 34; // inside doorway, visible at rest (discoverable)
    const boneRestY = house.doorY + 95;

    // mouth anchor in dog local space:
    // We'll approximate mouth at (x + 52*scale*persp, y-48*scale*persp) plus yaw effect.
    const k = dogPose.scale;
    const persp = 1 + (dogPose.y-400)*0.0009;
    const mX = dogPose.x + (54*k*persp)*Math.cos(dogPose.yaw*0.3);
    // mouth y:
    const mY = dogPose.y - (48*k*persp);
    // for yaw turning, mouth shifts slightly
    const mX2 = mX + 10*k*persp*Math.sin(dogPose.yaw*0.7);
    const mY2 = mY + 6*k*persp*(t<6?0:1);

    // bone depth scale
    const d = lerp(0.98,1.03, smoothstep(6,10,t));

    let bx=boneRestX, by=boneRestY, rot=0.0;
    if(t<4.5){
      bx=boneRestX; by=boneRestY; rot = -0.2;
    } else if(t<6){
      // interpolate from resting to mouth
      const u = grabbedT;
      bx = lerp(boneRestX, mX2, u);
      by = lerp(boneRestY, mY2, u);
      rot = lerp(-0.1, 0.35, u);
    } else if(t<8){
      const u = carryT;
      bx = lerp(mX2, mX2 - (70)*(0.0), 0) ; // keep tied to mouth
      // We'll tie directly to mouth each frame:
      bx = mX2;
      by = mY2 + 2*k*persp*Math.sin(u*2.2);
      rot = 0.35 + 0.18*Math.sin((t-6)*2.8);
    } else if(t<10){
      bx = mX2;
      by = mY2;
      rot = 0.35 - 0.2*smoothstep(8,10,t);
    } else {
      bx = mX2;
      by = mY2;
      rot = 0.12;
    }

    // mouth grip occlusion: bone should remain between viewer and house when outside.
    // We'll render bone after dog for t>=5 (it should appear in mouth).
    const inHouse = (t>=3 && t<6) || (t>=3 && t<4.5);
    ctx.save();
    // depth scale and slight squash when in mouth
    const depthScale = d*(t>=5?1.0:0.98);
    ctx.translate(bx,by);
    ctx.rotate(rot);
    // draw with scale
    ctx.scale(depthScale,depthScale);
    // Bone should look rigid (no deform); just rigid rotation.
    // During inside, partially occluded by doorway rim will be handled by draw order.
    drawBone(0,0,1,true,0);
    ctx.restore();

    return {bx,by,rot};
  }

  // Main renderer
  window.renderFrame = function(timeSeconds){
    const t = timeSeconds < 0 ? 0 : timeSeconds;
    const tt = t>=12 ? 12 : t;

    ctx.fillStyle='#bfe5ff';
    ctx.fillRect(0,0,W,H);

    // sky gradient
    const sky = ctx.createLinearGradient(0,0,0,260);
    sky.addColorStop(0,'#bfe7ff');
    sky.addColorStop(1,'#d5f2ff');
    ctx.fillStyle=sky;
    ctx.fillRect(0,0,W,260);

    // far fence / trees
    ctx.save();
    ctx.globalAlpha=0.65;
    // distant hedge
    ctx.fillStyle='#3f8a34';
    ctx.beginPath();
    ctx.moveTo(0,290);
    for(let x=0;x<=W;x+=20){
      const y=300 + Math.sin(x*0.03)*8 + (hash(x*11)%5);
      ctx.lineTo(x,y);
    }
    ctx.lineTo(W,340);
    ctx.lineTo(0,340);
    ctx.closePath();
    ctx.fill();

    ctx.globalAlpha=1;
    // fence slats
    ctx.fillStyle='#9a6c43';
    for(let i=0;i<11;i++){
      const x=40+i*85;
      ctx.fillRect(x,282,8,70);
      ctx.globalAlpha=0.8;
      ctx.fillRect(x+2,292,4,60);
      ctx.globalAlpha=1;
    }
    ctx.restore();

    drawGround(tt);

    // puddle appears on approach (but remains visible entire film)
    // We'll draw reflection only when appropriate, and always draw puddle surface.
    // Bone resting inside visible from start
    drawHouse();

    // puddle surface (always on)
    drawPuddleSurface();

    // dog pose
    const dogPose = poseAt(tt);

    // reflection pass (before dog and bone for correct layering)
    // We'll reflect only during run/jump across puddle, and the puddle is already drawn.
    // But to keep boundary crisp, draw reflection over ground but under surface.
    // Our drawPuddleSurface is already drawn; so do reflection after and then re-draw surface.
    // We'll reorder: easiest: draw reflection now, then redraw surface to overlay boundary.
    drawDogWithReflection(tt, dogPose);
    // redraw surface on top of reflection
    drawPuddleSurface();

    // Draw bone and dog with correct occlusion for house doorway and entrance
    // We'll use door rim as an occluder by drawing it after dog/bone during entrance.
    // However house itself already drew rim; to ensure occlusion we add a door mask after drawing dog.
    // Order:
    // - dog (outside/threshold)
    // - if bone is held and dog is outside, bone should appear in mouth.
    // - if inside, bone might be partly hidden by doorway rim; we apply door mask.
    // Determine outside/inside
    const entering = (tt>=3 && tt<6);

    // For occlusion, draw dog first, then bone when grabbed, then overlay door occluder.
    // For t<4.5 bone is on ground inside doorway (we still render it, may be partially visible through opening)
    // Ensure doorway occlusion: draw door mask after bone for entering range.
    // Dog should be seen in front of some parts.
    drawPomeranian({
      ...dogPose,
      boneInMouth: tt>=5.0
    });

    // Bone
    // Place bone resting: if before grab, it should appear within doorway opening (visible through).
    // We'll draw bone always, but mask it with door occluder when entering.
    const boneInfo = drawBoneAtTime(tt, dogPose);

    // Door occlusion mask when dog enters and bone is in opening
    if(entering){
      ctx.save();
      // Mask the doorway interior rim: paint a dark frame overlay.
      // This will occlude parts of dog/bone that would cross "solid" walls.
      ctx.fillStyle='#9a6e4c';
      ctx.globalAlpha=1;
      // draw rim thick
      ctx.fillRect(house.doorX-6, house.doorY, house.doorW+12, house.doorH);
      // inner dark to hide through wall
      ctx.fillStyle='#2b2016';
      ctx.fillRect(house.doorX, house.doorY+10, house.doorW, house.doorH-10);

      // edge shine
      ctx.globalAlpha=0.25;
      ctx.fillStyle='#fff0d1';
      ctx.fillRect(house.doorX+6,house.doorY+2, house.doorW-12, 6);
      ctx.globalAlpha=1;

      ctx.restore();
    }

    // Add subtle contact shadows for the dog when on ground and outside house
    if(tt<3 || tt>=8){
      ctx.save();
      ctx.globalAlpha=0.35;
      ctx.fillStyle='#0e0e0b';
      const s = dogPose.scale;
      const px = dogPose.x;
      const py = 470 - (tt<3?0:0);
      ctx.beginPath();
      ctx.ellipse(px, 425, 55*s, 16*s, dogPose.yaw*0.2, 0, Math.PI*2);
      ctx.fill();
      ctx.restore();
    }

    // Bone resting discoverable at start (make it slightly brighter and show through doorway)
    if(tt<4.5){
      ctx.save();
      ctx.globalAlpha=0.95;
      // subtle ground shadow under bone
      ctx.fillStyle='rgba(0,0,0,0.25)';
      ctx.beginPath();
      ctx.ellipse(house.doorX+45, house.doorY+145, 22, 6, 0,0,Math.PI*2);
      ctx.fill();
      ctx.restore();
    }

    // Final framing subtle breathing (done by poseAt time; already slight)
    // For t>=12, poseAt holds at 12.
  };

  // opening state at t<0 requirement: use opening state; recorder calls renderFrame(0) at startup.
  window.renderFrame(0);
})();

<script>
(() => {
  const c = document.getElementById('stage');
  const ctx = c.getContext('2d', { alpha:false });
  const W=960,H=540;

  const clamp=(v,a,b)=>v<a?a:v>b?b:v;
  const lerp=(a,b,t)=>a+(b-a)*t;
  const smooth=(t)=>t*t*(3-2*t);
  const smoothstep=(a,b,t)=>smooth(clamp((t-a)/(b-a),0,1));
  const hash=(n)=>{ n=Math.imul(n^0x9e3779b9,2654435761); n^=n>>>16; return (n>>>0)/4294967295; };

  // Colors
  const fur1 = '#f7c88b';
  const fur2 = '#e89b57';
  const fur3 = '#c8783b';
  const fur4 = '#fff1d1';
  const fur5 = '#8b4e22';
  const dark = '#2a1a12';

  const house = { x: 640, y: 260, w: 250, h: 260, roofH: 80, doorW: 90, doorH: 145, doorX: 705, doorY: 310 };
  const puddle = { x1: 320, x2: 450, yTop: 404, yBot: 430 };

  function rectRound(x,y,w,h,r){
    ctx.beginPath();
    ctx.roundRect(x,y,w,h,r);
    ctx.closePath();
  }

  function drawGround(){
    // sky is outside
    // grass base
    const g = ctx.createLinearGradient(0,280,0,H);
    g.addColorStop(0,'#87c070');
    g.addColorStop(0.55,'#6cad57');
    g.addColorStop(1,'#5a9c4b');
    ctx.fillStyle=g;
    ctx.fillRect(0,280,W,H-280);

    // dirt path
    ctx.save();
    ctx.globalAlpha=0.95;
    ctx.fillStyle='#6a6a3a';
    ctx.beginPath();
    ctx.moveTo(0,410);
    ctx.quadraticCurveTo(300,375, 560,395);
    ctx.quadraticCurveTo(720,415, 960,390);
    ctx.lineTo(960,540);
    ctx.lineTo(0,540);
    ctx.closePath();
    ctx.fill();

    // sparse grass specks deterministic
    ctx.globalAlpha=0.2;
    for(let i=0;i<130;i++){
      const x = (i*73)%W;
      const y = 320 + ((i*97)%200);
      const r = 0.8 + hash(i*17)*2.2;
      if(y<405||y>520) continue;
      ctx.fillStyle = ((i*13)%2===0)?'#2f5b23':'#2a4f1f';
      ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2); ctx.fill();
    }
    ctx.restore();

    // shadow under house front
    ctx.save();
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#1d1d14';
    ctx.beginPath();
    ctx.ellipse(735,426,70,18,0,0,Math.PI*2);
    ctx.fill();
    ctx.restore();

    // faint ground contact shading near puddle
    ctx.save();
    ctx.globalAlpha=0.12;
    ctx.fillStyle='#0e0e0b';
    ctx.beginPath();
    ctx.ellipse((puddle.x1+puddle.x2)/2, (puddle.yTop+puddle.yBot)/2+10, (puddle.x2-puddle.x1)/2+10, 10, 0, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  function drawHouse(){
    const {x,y,w,h,roofH,doorX,doorY,doorW,doorH}=house;

    // roof
    ctx.save();
    const roof = ctx.createLinearGradient(0,y,0,y+roofH);
    roof.addColorStop(0,'#6f4b2d');
    roof.addColorStop(1,'#4e361f');
    ctx.fillStyle=roof;
    ctx.beginPath();
    ctx.moveTo(x-10,y+roofH);
    ctx.lineTo(x+w/2,y-10);
    ctx.lineTo(x+w+10,y+roofH);
    ctx.lineTo(x+w,y+roofH);
    ctx.lineTo(x+w/2,y-2);
    ctx.lineTo(x,y+roofH);
    ctx.closePath();
    ctx.fill();

    // roof highlight
    ctx.globalAlpha=0.25;
    ctx.fillStyle='#c58b5e';
    ctx.beginPath();
    ctx.moveTo(x+w/2-60,y+roofH-10);
    ctx.quadraticCurveTo(x+w/2,y+roofH-30,x+w/2+60,y+roofH-10);
    ctx.lineTo(x+w/2+60,y+roofH);
    ctx.lineTo(x+w/2-60,y+roofH);
    ctx.closePath();
    ctx.fill();
    ctx.globalAlpha=1;

    // walls
    const wall = ctx.createLinearGradient(x,y,x+w,y+h);
    wall.addColorStop(0,'#caa27b');
    wall.addColorStop(0.5,'#b88f67');
    wall.addColorStop(1,'#9b724f');
    ctx.fillStyle=wall;
    ctx.fillRect(x,y,w,h);

    // vertical grooves
    ctx.globalAlpha=0.22;
    ctx.fillStyle='#3b2a1e';
    for(let i=0;i<10;i++){
      const gx=x+18+i*(w-36)/9;
      ctx.fillRect(gx,y+10,2, h-20);
    }
    ctx.globalAlpha=1;

    // base trim
    ctx.fillStyle='#7a5639';
    ctx.fillRect(x,y+h-18,w,18);

    // outer frame occluder base
    ctx.globalAlpha=0.95;
    ctx.fillStyle='#5b3f2a';
    ctx.fillRect(doorX-14, doorY-8, doorW+28, doorH+20);

    // inner rim
    ctx.globalAlpha=1;
    ctx.fillStyle='#9a6e4c';
    ctx.fillRect(doorX-6, doorY, doorW+12, doorH);

    // doorway interior
    const interior = ctx.createLinearGradient(doorX,doorY,doorX+doorW,doorY+doorH);
    interior.addColorStop(0,'#2b2016');
    interior.addColorStop(1,'#1a1310');
    ctx.fillStyle=interior;
    ctx.fillRect(doorX, doorY+10, doorW, doorH-10);

    // inner depth sides
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#000';
    ctx.fillRect(doorX,doorY+10,8,doorH-10);
    ctx.fillRect(doorX+doorW-8,doorY+10,8,doorH-10);
    ctx.globalAlpha=1;

    // lip shine
    ctx.globalAlpha=0.2;
    ctx.fillStyle='#fff0d1';
    ctx.fillRect(doorX+6,doorY+2,doorW-12,6);
    ctx.globalAlpha=1;

    // doorway floor hint
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#3a2b21';
    ctx.beginPath();
    // use round rect without relying on roundRect availability? modern supports.
    ctx.roundRect(doorX+10, doorY+doorH-18, doorW-20, 12, 6);

  }

  function drawPuddleSurface(){
    // water body
    const waterGrad = ctx.createLinearGradient(0,puddle.yTop,0,puddle.yBot);
    waterGrad.addColorStop(0,'rgba(80,140,190,0.65)');
    waterGrad.addColorStop(1,'rgba(25,60,100,0.78)');
    ctx.fillStyle=waterGrad;
    ctx.strokeStyle='rgba(180,230,255,0.35)';
    ctx.lineWidth=2;

    const rx = (puddle.x2-puddle.x1)/2;
    const mid = (puddle.x1+puddle.x2)/2;
    const topY=puddle.yTop, botY=puddle.yBot;

    ctx.beginPath();
    ctx.moveTo(mid-rx, topY);
    ctx.quadraticCurveTo(mid-rx*0.98, (topY+botY)/2, mid, botY);
    ctx.quadraticCurveTo(mid+rx*0.98, (topY+botY)/2, mid+rx, topY);
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    // specular highlight
    ctx.globalAlpha=0.45;
    ctx.strokeStyle='rgba(210,245,255,0.55)';
    ctx.beginPath();
    ctx.moveTo(puddle.x1+8, puddle.yTop+5);
    ctx.quadraticCurveTo(mid, puddle.yTop+18, puddle.x2-8, puddle.yTop+5);
    ctx.stroke();
    ctx.globalAlpha=1;
  }

    const w = 22*s, h=40*s;
    ctx.save();

    body.addColorStop(0,'#f6f0dc');
    body.addColorStop(0.5,'#e8dfc6');
    body.addColorStop(1,'#cdbf9b');
    ctx.fillStyle=body;
    ctx.strokeStyle='rgba(40,30,20,0.35)';
    ctx.lineWidth=Math.max(1,1.2*s);

    // bar
    ctx.beginPath();
    ctx.roundRect(-w*0.45,-h*0.5,w*0.9,h, w*0.35);
    ctx.fill(); ctx.stroke();

    // pad
    ctx.globalAlpha=0.5;
    ctx.fillStyle='#b9ad8a';
    ctx.beginPath();
    ctx.roundRect(-w*0.35,-h*0.15,w*0.7,h*0.7, w*0.25);
    ctx.fill();
    ctx.globalAlpha=1;

    // holes
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#6a5a40';
    ctx.beginPath();
    ctx.roundRect(-w*0.55,-h*0.12,w*0.36,h*0.22, w*0.15);
    ctx.roundRect(w*0.19,-h*0.12,w*0.36,h*0.22, w*0.15);
    ctx.fill();
    ctx.globalAlpha=1;
    ctx.restore();
  }

  function drawPomeranian(p){
    // p: {x,y,yJump, yaw, yLean, pitch, phase, runT, frontMode, scale, boneInMouth, inHouse, turnBoost}
    const {x,y,yJump,yaw,yLean,pitch,phase,runT,frontMode,scale,boneInMouth}=p;
    const k = scale*(1+(y-400)*0.0009);

    ctx.save();
    ctx.translate(x, y - yJump*k);
    ctx.rotate(yLean);

    // ground shadow
    ctx.save();
    ctx.globalAlpha=0.35;
    ctx.fillStyle='#0e0e0b';
    ctx.beginPath();
    ctx.ellipse(0, 70*k, 38*k, 12*k, yaw*0.25, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();

    const bodyFront = frontMode;
    // tail plume (curled over back)
    ctx.save();
    const tailAng = -2.1 + yaw*0.9;
    ctx.rotate(tailAng);
    ctx.translate(26*k, -14*k);
    const plumeGrad = ctx.createLinearGradient(-10,0,20,-40);
    plumeGrad.addColorStop(0,'#2c1b12');
    plumeGrad.addColorStop(0.15,'#d88a45');
    plumeGrad.addColorStop(0.55,'#f2c47f');
    plumeGrad.addColorStop(1,'#9e5a2a');
    ctx.fillStyle=plumeGrad;
    ctx.strokeStyle='rgba(60,35,20,0.4)';
    ctx.lineWidth=1.1*k;
    ctx.beginPath();
    ctx.moveTo(0,0);
    ctx.quadraticCurveTo(18*k,-8*k, 20*k,-28*k);
    ctx.quadraticCurveTo(18*k,-50*k, -8*k,-50*k);
    ctx.quadraticCurveTo(-24*k,-44*k, -18*k,-24*k);
    ctx.quadraticCurveTo(-12*k,-12*k, 0,0);
    ctx.closePath();
    ctx.fill();
    ctx.globalAlpha=0.75;
    ctx.beginPath();
    ctx.moveTo(2*k,0);
    ctx.quadraticCurveTo(16*k,-10*k, 18*k,-30*k);
    ctx.strokeStyle='rgba(255,235,200,0.55)';
    ctx.stroke();
    ctx.globalAlpha=1;
    ctx.restore();

    // torso and ruff volumes
    const yawLocal = yaw*0.35;
    ctx.save();
    ctx.rotate(yawLocal);

    const bodyGrad = ctx.createLinearGradient(-30*k,-60*k,30*k,80*k);
    bodyGrad.addColorStop(0,'#ffd9a3');
    bodyGrad.addColorStop(0.22,fur2);
    bodyGrad.addColorStop(0.6,fur1);
    bodyGrad.addColorStop(1,fur5);

    ctx.fillStyle=bodyGrad;
    ctx.strokeStyle='rgba(60,30,15,0.35)';
    ctx.lineWidth=1.2*k;
    ctx.beginPath();
    ctx.ellipse(0,0, 46*k, 34*k, 0, 0, Math.PI*2);
    ctx.fill(); ctx.stroke();

    // chest fluff + neck ruff
    const ruff = ctx.createLinearGradient(-20*k,-10*k,60*k,50*k);
    ruff.addColorStop(0,'#fff2cf');
    ruff.addColorStop(0.3,'#f1b574');
    ruff.addColorStop(1,'#a75a2d');
    ctx.fillStyle=ruff;
    ctx.globalAlpha=1;
    ctx.beginPath();
    ctx.ellipse(10*k,-30*k, 34*k, 26*k, -0.25, 0, Math.PI*2);
    ctx.fill();

    // ruff base around head
    ctx.beginPath();
    ctx.ellipse(54*k,-42*k, 28*k, 22*k, -0.15, 0, Math.PI*2);
    ctx.fill();
    ctx.restore();

    // legs with gait (deterministic)
    const isAir = p.isAir||false;
    const altPhase = phase;
    const step = Math.sin(altPhase*2.2);

    function leg(side){ // side -1 back, +1 front
      const liftBase = isAir ? 0.65 : 0.08;
      const alt = Math.sin(runT*6 + (side>0?0.6:-0.6) + (yaw*0.2)) ;
      const lift = clamp(liftBase + 0.5*alt*(isAir?1:0),0,1);

      const baseX = (side>0? 22*k : -12*k);
      const baseY = (side>0? 25*k : 40*k);
      const lat = yaw*0.55*k*(side>0?1:0.6) + (frontMode*0.15*k);
      const pawSwing = Math.sin(runT*7 + (side>0?1.1:0.2))*0.9;
      const arc = Math.max(0, step*(side>0?1:-1))*0.35;

      const px = baseX + (side>0? 8*k: -4*k) + pawSwing*(6*k) + lat*(side>0?1:-1);
      const py = baseY + (isAir ? -lift*22*k : 0) + (-arc*4*k);
      const bend = lerp(0.85,0.2, isAir?lift:clamp(alt,0,1));
      const forwardSign = side>0?1:-1;
      return {px,py,bend,forwardSign,lift};
    }

    // front/back layering: draw back legs first (approx)
    const legs = [{side:-1, z:-1},{side:+1,z:1},{side:-1,z:0.2},{side:+1,z:-0.2}];
    const pairs = [
      {side:+1, off:-1},{side:+1, off:1},
      {side:-1, off:-1},{side:-1, off:1}
    ];

    ctx.save();
    for(const L of pairs){
      const pose = leg(L.side);
      const px = pose.px + L.off*(3.5*k);
      const py = pose.py + (L.side>0?2*k:0);
      // shadow on ground when grounded
      if(!isAir){
        ctx.save();
        ctx.globalAlpha=0.48;
        ctx.fillStyle='#0e0e0b';
        ctx.beginPath();
        ctx.ellipse(px, 82*k, 10*k, 5.5*k, yaw*0.2, 0, Math.PI*2);
        ctx.fill(); ctx.restore();
      }

      const upperColor = '#c98546';
      const furColor = (L.side>0)?'#e4a25e':'#d38a49';

      // leg cylinder
      const legLen = 46*k*(L.side>0?0.95:1.02);
      const kneeY = lerp(36*k, 18*k, clamp(pose.lift,0,1));
      ctx.save();
      ctx.translate(px, py);
      const rot = -0.15 + (isAir?0.25*pose.lift:0) + yLean*0.06;
      ctx.rotate(rot);

      const legGrad = ctx.createLinearGradient(0,-legLen,0,0);
      legGrad.addColorStop(0,'#8e4a22');
      legGrad.addColorStop(0.2,furColor);
      legGrad.addColorStop(1,'#f3c37c');
      ctx.fillStyle=legGrad;
      ctx.strokeStyle='rgba(60,30,15,0.35)';
      ctx.lineWidth=1.1*k;

      ctx.beginPath();
      ctx.roundRect(-6*k, -legLen*pose.bend, 12*k, legLen, 6*k);
      ctx.fill(); ctx.stroke();

      // paw
      const pawGrad = ctx.createLinearGradient(-8*k,-4*k,8*k,10*k);
      pawGrad.addColorStop(0,'#f7d9a8');
      pawGrad.addColorStop(1,'#c98749');
      ctx.fillStyle=pawGrad;
      ctx.strokeStyle='rgba(30,18,10,0.35)';
      ctx.lineWidth=1*k;
      ctx.beginPath();
      ctx.roundRect(-10*k,-3*k, 20*k, 8*k + (isAir?-2*k:0), 6*k);
      ctx.fill(); ctx.stroke();

      ctx.restore();
    }
    ctx.restore();

    // head & ears
    const headX = 52*k, headY = -48*k;
    const headRot = yaw*0.25 + pitch;
    ctx.save();
    ctx.translate(headX, headY);
    ctx.rotate(headRot);

    function ear(which){
      ctx.save();
      ctx.translate(which*16*k, -16*k);
      ctx.rotate(which*0.08 + yaw*0.15);
      const earGrad=ctx.createLinearGradient(-5*k,-20*k,10*k,15*k);
      earGrad.addColorStop(0,'#6b3a19');
      earGrad.addColorStop(0.25,fur2);
      earGrad.addColorStop(1,'#ffd9a8');
      ctx.fillStyle=earGrad;
      ctx.strokeStyle='rgba(40,20,10,0.35)';
      ctx.lineWidth=1.1*k;
      ctx.beginPath();
      ctx.moveTo(0,-2*k);
      ctx.quadraticCurveTo(-9*k,-15*k,-5*k,-30*k);
      ctx.quadraticCurveTo(4*k,-40*k, 14*k,-28*k);
      ctx.quadraticCurveTo(20*k,-13*k, 8*k, -2*k);
      ctx.closePath();
      ctx.fill(); ctx.stroke();

      ctx.globalAlpha=0.28;
      ctx.fillStyle='#b15b2a';
      ctx.beginPath();
      ctx.moveTo(0,-3*k);
      ctx.quadraticCurveTo(-6*k,-12*k,-2*k,-23*k);
      ctx.quadraticCurveTo(4*k,-31*k, 11*k,-22*k);
      ctx.quadraticCurveTo(15*k,-11*k, 7*k,-3*k);
      ctx.closePath();
      ctx.fill();
      ctx.globalAlpha=1;
      ctx.restore();
    }
    ear(-1); ear(1);

    // eye
    const eyeX = (frontMode>0.35 ? 10*k : 14*k);
    const eyeY = -6*k;
    ctx.fillStyle='#13100d';
    ctx.beginPath();
    ctx.ellipse(eyeX,eyeY, 4.2*k, 5.1*k, 0,0,Math.PI*2);
    ctx.fill();
    ctx.globalAlpha=0.7;
    ctx.fillStyle='rgba(255,255,255,0.7)';
    ctx.beginPath();
    ctx.arc(eyeX-1.1*k, eyeY-1.8*k, 1.3*k, 0, Math.PI*2);
    ctx.fill();
    ctx.globalAlpha=1;

    // muzzle
    const muGrad=ctx.createLinearGradient(-10*k,-20*k,20*k,10*k);
    muGrad.addColorStop(0,'#f7e1b9');
    muGrad.addColorStop(0.4,'#f1b974');
    muGrad.addColorStop(1,'#a15b2d');
    ctx.fillStyle=muGrad;
    ctx.strokeStyle='rgba(40,20,10,0.35)';
    ctx.lineWidth=1.1*k;
    ctx.beginPath();
    ctx.ellipse(18*k, 5*k, 18*k, 13*k, 0, 0, Math.PI*2);
    ctx.fill(); ctx.stroke();

    // nose
    ctx.fillStyle=dark;
    ctx.beginPath();
    ctx.ellipse(23*k, 8*k, 4.2*k, 3.4*k, 0.3, 0, Math.PI*2);
    ctx.fill();

    if(boneInMouth){
      ctx.globalAlpha=0.9;
      ctx.strokeStyle='rgba(20,10,10,0.6)';
      ctx.lineWidth=1.5*k;
      ctx.beginPath();
      ctx.moveTo(18*k,10*k);
      ctx.quadraticCurveTo(26*k,10*k,31*k,7.5*k);
      ctx.stroke();
      ctx.globalAlpha=1;
    }

    // ruff highlight
    ctx.globalAlpha=0.35;
    ctx.strokeStyle='rgba(255,250,230,0.65)';
    ctx.lineWidth=2.2*k;
    ctx.beginPath();
    ctx.ellipse(-8*k,-20*k, 24*k, 18*k, -0.2, 0, Math.PI*2);
    ctx.stroke();
    ctx.globalAlpha=1;

    ctx.restore();
  }

  function poseAt(t){
    t = clamp(t,0,12);
    const yGround=405;

    // movement along approach -> jump over puddle -> enter -> retrieve -> back out -> face viewer
    let x=0,y=yGround, yJump=0, yaw=-0.25, isAir=false, scale=1.0;
    let yLean=0, pitch=0, frontMode=0.0;
    let enter=0, boneInMouth=false;

    if(t<=1){
      const tt=t/1;
      x=lerp(170,220,tt);
      y=yGround - 6*Math.sin(tt*Math.PI);
      yaw=-0.25;
    } else if(t<=3){
      const tt=(t-1)/2; //0..1
      const pre = smoothstep(0,0.35,tt);
      x=lerp(220,615,tt);
      const jumpStart=0.28, jumpEnd=0.72;
      isAir = (tt>jumpStart && tt<jumpEnd);
      const u = clamp((tt-jumpStart)/(jumpEnd-jumpStart),0,1);
      const h = (tt>=jumpStart && tt<=jumpEnd) ? Math.sin(Math.PI*u) : 0;
      yJump = h*52;
      const yLand = yGround;
      y = yLand - pre*3 + (isAir?0:-2*Math.sin(tt*3.5)*0.7);
      if(isAir) x += h*10;

      // lower body anticipation
      yLean = -0.14*clamp((tt-0.15)/0.35,0,1) + (isAir?0.10:0) + 0.06*Math.sin(t*10);
      pitch = isAir ? (-0.06 + 0.07*Math.sin((t-1)*Math.PI)) : 0;
      frontMode = 0.0 + 0.05*tt;
      yaw = lerp(-0.25,-0.05,smoothstep(0.55,1.0,tt));
    } else if(t<=4.5){
      const inT = smoothstep(0,1,(t-3)/1.5);
      isAir=false;
      x=lerp(620,735,inT);
      y=yGround - 6*inT;
      scale=lerp(1.02,0.92,inT);
      yaw=lerp(-0.25,-0.05,inT);
      yLean = -0.05 - 0.04*smoothstep(3,4.5,t);
      pitch = 0;
      frontMode = 0.05;
      enter=inT;
    } else if(t<=6){
      const u=smoothstep(0,1,(t-4.5)/1.5);
      x=lerp(735,728,u);
      y=yGround - 7 - 16*u;
      scale=lerp(0.92,0.90,u);
      yaw=lerp(-0.05,-0.02,u);
      yLean = -0.05 - 0.06*u;
      pitch = 0.02*u;
      frontMode = 0.08;
      boneInMouth=true;
    } else if(t<=8){
      const u=smoothstep(0,1,(t-6)/2);
      x=lerp(728,680,u);
      y=lerp(yGround-23,yGround-5,u);
      scale=lerp(0.90,1.01,u);
      yaw=lerp(-0.02,0.02,u);
      yLean = 0.04*smoothstep(6,8,t);
      pitch = 0;
      frontMode = 0.15;
      boneInMouth=true;
    } else if(t<=10){
      const u=smoothstep(0,1,(t-8)/2);
      x=lerp(680,640,u);
      y=lerp(yGround-5,yGround-2,u);
      scale=lerp(1.01,1.02,u);
      yaw=lerp(0.02,0.42,u);
      yLean = 0.05*smoothstep(8,10,t);
      pitch = 0;
      frontMode = u; // gradually frontal
      boneInMouth=true;
    } else {
      x=640; y=yGround-2; scale=1.02;
      yaw=0.42;
      yLean=0.02*Math.sin(t*3);
      pitch=0;
      frontMode=1;
      boneInMouth=true;
    }

    const phase=t*3.1;
    const runT=t;
    return {x,y,yJump,yaw,yLean,pitch,phase,runT,frontMode,scale,isAir,boneInMouth,enterDepth:enter};
  }

  function drawBoneAndMouth(t, dogPose){
    // Bone path: rests inside doorway before grab, then rigidly tied to mouth once grabbed
    const grabbed = smoothstep(4.5,6,t);
    const carry = smoothstep(6,8,t);

    const boneRestX = house.doorX + 34;
    const boneRestY = house.doorY + 95;

    // mouth anchor in screen space computed from dog pose roughly
    const k = dogPose.scale*(1+(dogPose.y-400)*0.0009);
    const persp = 1+(dogPose.y-400)*0.0009;

    const mX = dogPose.x + (54*k*persp)*Math.cos(dogPose.yaw*0.3) + 10*k*persp*Math.sin(dogPose.yaw*0.7);
    const mY = dogPose.y - (48*k*persp) + (t>=5? 0 : 0);

    let bx=boneRestX, by=boneRestY, rot=-0.2;
    if(t<4.5){
      bx=boneRestX; by=boneRestY; rot=-0.2;
    } else if(t<6){
      bx=lerp(boneRestX,mX,grabbed);
      by=lerp(boneRestY,mY,grabbed);
      rot=lerp(-0.1,0.35,grabbed);
    } else {
      bx=mX;
      by=mY + 2*k*persp*Math.sin((t-6)*2.2);
      rot=0.35 + 0.18*Math.sin((t-6)*2.8);
    }

    // draw bone rigidly with depth scaling
    const depthScale = lerp(0.98,1.03, smoothstep(6,10,t));
    const squash = (t>=5 && t<6) ? (0.99) : 1.0;

    ctx.save();
    ctx.translate(bx,by);
    ctx.rotate(rot);
    ctx.scale(depthScale*squash, depthScale*squash);
    drawBoneLocal(1,0); // bone local draws at its own scale (we scale by ctx already)
    ctx.restore();
  }

  function dogReflection(t, dogPose){
    // Reflection only during approach/jump over puddle
    const show = (t>=1.0 && t<=3.05);
    if(!show) return;

    // clip to puddle rounded region
    const rx = (puddle.x2-puddle.x1)/2;
    const mid = (puddle.x1+puddle.x2)/2;
    const topY=puddle.yTop, botY=puddle.yBot;

    ctx.save();
    ctx.beginPath();
    ctx.moveTo(mid-rx, topY);
    ctx.quadraticCurveTo(mid-rx*0.98, (topY+botY)/2, mid, botY);
    ctx.quadraticCurveTo(mid+rx*0.98, (topY+botY)/2, mid+rx, topY);
    ctx.closePath();
    ctx.clip();

    // mirror about y=topY
    ctx.save();
    ctx.translate(0, 2*topY);
    ctx.scale(1,-1);

    ctx.globalAlpha=0.58;

    // During reflection, bone not yet in mouth
    drawPomeranian({
      ...dogPose,
      boneInMouth:false,
      isAir:dogPose.isAir,
      frontMode:dogPose.frontMode
    });

    ctx.globalAlpha=0.35;
    drawPomeranian({
      ...dogPose,
      boneInMouth:false,
      isAir:dogPose.isAir,
      frontMode:dogPose.frontMode
    });

    ctx.restore();
    ctx.restore();

    // boundary fade
    ctx.save();
    const grad = ctx.createLinearGradient(0,puddle.yTop-4,0,puddle.yBot);
    grad.addColorStop(0,'rgba(120,200,255,0)');
    grad.addColorStop(0.35,'rgba(120,200,255,0.18)');
    grad.addColorStop(1,'rgba(60,120,190,0.35)');
    ctx.fillStyle=grad;
    ctx.beginPath();
    ctx.moveTo(puddle.x1, puddle.yTop);
    ctx.lineTo(puddle.x2, puddle.yTop);
    ctx.lineTo(puddle.x2, puddle.yBot);
    ctx.lineTo(puddle.x1, puddle.yBot);
    ctx.closePath();
    ctx.fill();
    ctx.restore();
  }

  window.renderFrame = function(timeSeconds){
    const t = timeSeconds < 0 ? 0 : (timeSeconds>=12?12:timeSeconds);

    // background
    ctx.fillStyle='#bfe5ff';
    ctx.fillRect(0,0,W,H);

    const sky = ctx.createLinearGradient(0,0,0,260);
    sky.addColorStop(0,'#bfe7ff');
    sky.addColorStop(1,'#d5f2ff');
    ctx.fillStyle=sky;
    ctx.fillRect(0,0,W,260);

    // distant hedge/fence
    ctx.save();
    ctx.globalAlpha=0.65;
    ctx.fillStyle='#3f8a34';
    ctx.beginPath();
    ctx.moveTo(0,290);
    for(let x=0;x<=W;x+=20){
      const y=300 + Math.sin(x*0.03)*8 + (hash(x*11)%5);
      ctx.lineTo(x,y);
    }
    ctx.lineTo(W,340);
    ctx.lineTo(0,340);
    ctx.closePath();
    ctx.fill();

    ctx.globalAlpha=0.95;
    ctx.fillStyle='#9a6c43';
    for(let i=0;i<11;i++){
      const xx=40+i*85;
      ctx.fillRect(xx,282,8,70);
      ctx.globalAlpha=0.8;
      ctx.fillRect(xx+2,292,4,60);
      ctx.globalAlpha=1;
    }
    ctx.restore();

    drawGround();
    drawHouse();
    drawPuddleSurface();

    const dogPose = poseAt(t);

    // reflection under dog during puddle crossing (under surface)
    dogReflection(t,dogPose);
    // redraw puddle surface on top boundary
    drawPuddleSurface();

    // Draw dog (outside + threshold)
    const entering = (t>=3 && t<4.5) || (t>=4.5 && t<6); // in doorway span
    const onGroundOutside = (t<3 || t>=8);

    // Bone: we draw before door occlusion mask when needed
    // draw dog
    drawPomeranian({...dogPose, boneInMouth:dogPose.boneInMouth});

    // draw bone
    drawBoneAndMouth(t, dogPose);

    // door occlusion mask to ensure correct threshold occlusion when inside
    if(t>=3 && t<6.2){
      ctx.save();
      // rim thick dark overlay + inner depth to hide through-wall overlap
      ctx.fillStyle='#9a6e4c';
      ctx.fillRect(house.doorX-6, house.doorY, house.doorW+12, house.doorH);
      ctx.fillStyle='#2b2016';
      ctx.fillRect(house.doorX, house.doorY+10, house.doorW, house.doorH-10);

      ctx.globalAlpha=0.25;
      ctx.fillStyle='#fff0d1';
      ctx.fillRect(house.doorX+6,house.doorY+2, house.doorW-12, 6);
      ctx.globalAlpha=1;
      ctx.restore();
    }

    // Contact shadow for dog when outside
    if(onGroundOutside){
      ctx.save();
      ctx.globalAlpha=0.32;
      ctx.fillStyle='#0e0e0b';
      const k = dogPose.scale;
      ctx.beginPath();
      ctx.ellipse(dogPose.x, 425, 55*k, 16*k, dogPose.yaw*0.2, 0, Math.PI*2);
      ctx.fill();
      ctx.restore();
    }

    // Keep bone discoverable early (slight ground shadow under resting bone)
    if(t<4.5){
      ctx.save();
      ctx.globalAlpha=0.55;
      ctx.fillStyle='rgba(0,0,0,0.25)';
      ctx.beginPath();
      ctx.ellipse(house.doorX+45, house.doorY+145, 22, 6, 0,0,Math.PI*2);
      ctx.fill();
      ctx.restore();
    }

    // final tiny breathing motion is already in poseAt via yLean sin when t>=10
  };

  renderFrame(0);
  window.renderFrame(0);

})();

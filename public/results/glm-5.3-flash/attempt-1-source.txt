function IK(ax,ay,bx,by,l1,l2,bend){
 var dx=bx-ax,dy=by-ay,d=Math.sqrt(dx*dx+dy*dy);
 d=Math.max(Math.abs(l1-l2)+0.01,Math.min(d,l1+l2-0.01));
 var a1=Math.acos(Math.max(-1,Math.min(1,(l1*l1+d*d-l2*l2)/(2*l1*d))));
 var base=Math.atan2(dy,dx),ang=base+a1*bend;
 return [ax+Math.cos(ang)*l1,ay+Math.sin(ang)*l1];
}

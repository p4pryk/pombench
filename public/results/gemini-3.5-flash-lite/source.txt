<body>
<canvas id="stage" width="960" height="540"></canvas>
<script>
const canvas = document.getElementById('stage');
const ctx = canvas.getContext('2d');

function renderFrame(timeSeconds) {
  let t = Math.max(0, Math.min(12, timeSeconds));
  
  // Clear screen (backyard gradient: grass and fence/sky)
  ctx.clearRect(0, 0, 960, 540);
  
  // Sky
  let gradSky = ctx.createLinearGradient(0, 0, 0, 250);
  gradSky.addColorStop(0, '#8cb8e2');
  gradSky.addColorStop(1, '#d4e6f1');
  ctx.fillStyle = gradSky;
  ctx.fillRect(0, 0, 960, 250);

  // Wooden fence background
  ctx.fillStyle = '#8b6b43';
  ctx.fillRect(0, 180, 960, 70);
  ctx.fillStyle = '#654826';
  for(let x=0; x<960; x+=40) {
    ctx.fillRect(x, 180, 4, 70);
  }

  // Grass ground
  let gradGrass = ctx.createLinearGradient(0, 250, 0, 540);
  gradGrass.addColorStop(0, '#567d46');
  gradGrass.addColorStop(1, '#334d28');
  ctx.fillStyle = gradGrass;
  ctx.fillRect(0, 250, 960, 290);

  // Puddle (centered around x: 320, y: 390)
  ctx.fillStyle = '#3a6073';
  ctx.beginPath();
  ctx.ellipse(320, 390, 80, 30, 0, 0, Math.PI * 2);
  ctx.fill();
  ctx.strokeStyle = '#85c1e9';
  ctx.lineWidth = 2;
  ctx.stroke();

  // Doghouse (positioned at x: 650, y: 220)
  // Base
  ctx.fillStyle = '#a0522d';
  ctx.fillRect(600, 300, 220, 180);
  // Doorway
  ctx.fillStyle = '#2b1700';
  ctx.beginPath();
  ctx.roundRect(670, 360, 80, 120, [40, 40, 0, 0]);
  ctx.fill();
  // Roof
  ctx.fillStyle = '#78281f';
  ctx.beginPath();
  ctx.moveTo(580, 300);
  ctx.lineTo(710, 190);
  ctx.arcTo(715, 185, 720, 190, 5);
  ctx.lineTo(840, 300);
  ctx.closePath();
  ctx.fill();

  // Bone inside doghouse initially (resting at 700, 440)
  let boneInHouse = (t < 4.5);
  let boneX = 700, boneY = 440;
  
  // Dog state calculation based on t (0-12s)
  // Path: Start at x=100, y=400. Run to puddle (x=240 -> 400, jump over 320), reach house at x=670 by t=3.5.
  // Enter house: 3.5 to 4.5. Grab bone: 4.5 to 6. Back out: 6 to 8 (reach x=580). Turn and face viewer: 8 to 10. Settle: 10-12.
  
  let dogX = 100;
  let dogY = 400;
  let bodyAngle = 0;
  let inHouse = false;
  let hasBone = false;
  let facingFront = false;
  let tailWag = Math.sin(t * 10) * 0.2;

  if (t < 1.0) {
    // 0-1s: Establish, notice goal
    dogX = 100;
    dogY = 400;
  } else if (t < 3.2) {
    // 1-3.2s: Run and jump puddle
    let runProgress = (t - 1.0) / 2.2;
    dogX = 100 + runProgress * 520; // moves to ~620
    
    // Jump over puddle (x: 240 to 400)
    if (dogX > 230 && dogX < 410) {
      let jumpT = (dogX - 230) / 180;
      let arc = Math.sin(jumpT * Math.PI) * 70;
      dogY = 400 - arc;
      
      // Reflection in puddle
      let refAlpha = 0.4 * (1 - Math.abs(dogX - 320) / 90);
      if (refAlpha > 0) {
        ctx.save();
        ctx.globalAlpha = refAlpha;
        ctx.translate(dogX, 470 - (dogY - 330));
        ctx.scale(1, -0.6);
        drawPomeranian(ctx, 0, 0, 0, false, false, tailWag);
        ctx.restore();
      }
    } else {
      dogY = 400 + Math.sin(t * 25) * 4; // running bounce
    }
  } else if (t < 4.5) {
    // 3.2-4.5s: Entering house
    let enterProg = (t - 3.2) / 1.3;
    dogX = 620 + enterProg * 50;
    dogY = 400;
    inHouse = true;
  } else if (t < 6.0) {
    // 4.5-6s: Grab bone inside
    dogX = 670;
    dogY = 400;
    inHouse = true;
    hasBone = true;
    boneInHouse = false;
  } else if (t < 8.0) {
    // 6-8s: Backing out with bone
    let backProg = (t - 6.0) / 2.0;
    dogX = 670 - backProg * 90; // backs out to x=580
    dogY = 400;
    hasBone = true;
  } else if (t < 10.0) {
    // 8-10s: Turn to face viewer
    dogX = 580;
    dogY = 420;
    hasBone = true;
    facingFront = true;
  } else {
    // 10-12s: Settle front-facing pose
    dogX = 580;
    dogY = 420;
    hasBone = true;
    facingFront = true;
  }

  // Draw bone if in house
  if (boneInHouse) {
    drawBone(ctx, 690, 430, 0);
  }

  // Draw Dog (if not fully occluded inside)
  if (!inHouse || t < 3.5 || t > 7.5) {
    ctx.save();
    ctx.translate(dogX, dogY);
    if (facingFront) {
      drawPomeranianFront(ctx, 0, 0, hasBone, tailWag);
    } else {
      drawPomeranian(ctx, 0, 0, bodyAngle, hasBone, t >= 6.0 && t < 8.0, tailWag);
    }
    ctx.restore();
  }

  // If dog is partially in doorway during entry/exit, draw door frame overlay for occlusion
  if (inHouse && t >= 3.2 && t <= 8.0) {
    // Redraw doorway arch front edge to occlude rear if needed, or handle via layer order
  }
}

function drawBone(ctx, x, y, angle) {
  ctx.save();
  ctx.translate(x, y);
  ctx.rotate(angle);
  ctx.fillStyle = '#fdfefe';
  ctx.strokeStyle = '#d5dbdb';
  ctx.lineWidth = 1;
  // Simple bone shape
  ctx.beginPath();
  ctx.roundRect(-15, -4, 30, 8, 3);
  ctx.fill();
  ctx.stroke();
  // Knobs
  ctx.beginPath();
  ctx.arc(-15, -6, 4, 0, Math.PI*2);
  ctx.arc(-15, 6, 4, 0, Math.PI*2);
  ctx.arc(15, -6, 4, 0, Math.PI*2);
  ctx.arc(15, 6, 4, 0, Math.PI*2);
  ctx.fillStyle = '#fdfefe';
  ctx.fill();
  ctx.restore();
}

function drawPomeranian(ctx, x, y, angle, hasBone, backingUp, tailWag) {
  ctx.save();
  ctx.translate(x, y);
  ctx.rotate(angle);
  
  // Fluffy orange/cream double coat (Side Profile)
  
  // Tail (Plumed, curled over back)
  ctx.fillStyle = '#e67e22';
  ctx.beginPath();
  ctx.arc(-35, -25 + tailWag*5, 18, 0, Math.PI*2);
  ctx.fill();
  ctx.fillStyle = '#f39c12';
  ctx.beginPath();
  ctx.arc(-30, -20 + tailWag*5, 12, 0, Math.PI*2);
  ctx.fill();

  // Body ruff & torso
  ctx.fillStyle = '#d35400';
  ctx.beginPath();
  ctx.ellipse(-10, -5, 30, 20, 0, 0, Math.PI*2);
  ctx.fill();

  ctx.fillStyle = '#e67e22';
  ctx.beginPath();
  ctx.ellipse(0, -10, 25, 22, 0, 0, Math.PI*2);
  ctx.fill();

  // Legs
  ctx.fillStyle = '#ba4a00';
  ctx.fillRect(-20, 10, 6, 20);
  ctx.fillRect(-10, 12, 6, 18);
  ctx.fillRect(10, 10, 6, 20);
  ctx.fillRect(18, 12, 6, 18);

  // Head
  ctx.fillStyle = '#f39c12';
  ctx.beginPath();
  ctx.arc(20, -18, 16, 0, Math.PI*2);
  ctx.fill();

  // Chest ruff
  ctx.fillStyle = '#f8c471';
  ctx.beginPath();
  ctx.ellipse(10, -2, 18, 14, 0.3, 0, Math.PI*2);
  ctx.fill();

  // Muzzle & Ears
  ctx.fillStyle = '#e67e22';
  // Snout
  ctx.beginPath();
  ctx.moveTo(32, -18);
  ctx.lineTo(42, -14);
  ctx.lineTo(32, -10);
  ctx.closePath();
  ctx.fill();

  // Nose
  ctx.fillStyle = '#111';
  ctx.beginPath();
  ctx.arc(42, -14, 3, 0, Math.PI*2);
  ctx.fill();

  // Eyes
  ctx.beginPath();
  ctx.arc(26, -22, 2.5, 0, Math.PI*2);
  ctx.fill();

  // Ears
  ctx.fillStyle = '#ba4a00';
  ctx.beginPath();
  ctx.moveTo(16, -30);
  ctx.lineTo(22, -38);
  ctx.lineTo(26, -30);
  ctx.closePath();
  ctx.fill();

  // Bone in mouth
  if (hasBone) {
    drawBone(ctx, 40, -10, backingUp ? 0.2 : -0.2);
  }

  ctx.restore();
}

function drawPomeranianFront(ctx, x, y, hasBone, tailWag) {
  ctx.save();
  ctx.translate(x, y);

  // Tail behind body (curled top)
  ctx.fillStyle = '#e67e22';
  ctx.beginPath();
  ctx.arc(0, -45 + tailWag*3, 22, 0, Math.PI*2);
  ctx.fill();

  // Massive neck ruff (Pomeranian signature)
  ctx.fillStyle = '#f8c471';
  ctx.beginPath();
  ctx.ellipse(0, -10, 35, 28, 0, 0, Math.PI*2);
  ctx.fill();
  ctx.fillStyle = '#e67e22';
  ctx.beginPath();
  ctx.ellipse(0, -5, 28, 22, 0, 0, Math.PI*2);
  ctx.fill();

  // Body torso
  ctx.fillStyle = '#d35400';
  ctx.beginPath();
  ctx.ellipse(0, 10, 24, 20, 0, 0, Math.PI*2);
  ctx.fill();

  // Front legs planted
  ctx.fillStyle = '#ba4a00';
  ctx.fillRect(-16, 20, 8, 20);
  ctx.fillRect(8, 20, 8, 20);

  // Head (Front view)
  ctx.fillStyle = '#f39c12';
  ctx.beginPath();
  ctx.arc(0, -22, 22, 0, Math.PI*2);
  ctx.fill();

  // Ears (Upright triangular)
  ctx.fillStyle = '#ba4a00';
  ctx.beginPath();
  ctx.moveTo(-16, -34);
  ctx.lineTo(-24, -48);
  ctx.lineTo(-8, -38);
  ctx.closePath();
  ctx.fill();

  ctx.beginPath();
  ctx.moveTo(16, -34);
  ctx.lineTo(24, -48);
  ctx.lineTo(8, -38);
  ctx.closePath();
  ctx.fill();

  // Eyes (Dark, expressive)
  ctx.fillStyle = '#111';
  ctx.beginPath();
  ctx.arc(-8, -24, 3, 0, Math.PI*2);
  ctx.arc(8, -24, 3, 0, Math.PI*2);
  ctx.fill();
  // Eye highlights
  ctx.fillStyle = '#fff';
  ctx.fillRect(-9, -26, 1.5, 1.5);
  ctx.fillRect(7, -26, 1.5, 1.5);

  // Cute pointed muzzle
  ctx.fillStyle = '#e67e22';
  ctx.beginPath();
  ctx.ellipse(0, -16, 8, 6, 0, 0, Math.PI*2);
  ctx.fill();
  
  // Nose
  ctx.fillStyle = '#000';
  ctx.beginPath();
  ctx.arc(0, -17, 3.5, 0, Math.PI*2);
  ctx.fill();

  // Proudly holding bone in mouth horizontally
  if (hasBone) {
    drawBone(ctx, 0, -10, 0);
  }

  ctx.restore();
}

// Initial call
renderFrame(0);
</script>

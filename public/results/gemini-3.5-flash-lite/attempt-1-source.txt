<script>
const canvas = document.getElementById('stage');
const ctx = canvas.getContext('2d');

// Deterministic PRNG for consistent fur/grass details
function pseudoRandom(seed) {
  let x = Math.sin(seed++) * 10000;
  return x - Math.floor(x);
}

// Draw environment (backyard, grass, fence, doghouse)
function drawEnvironment(t) {
  // Sky
  const gradSky = ctx.createLinearGradient(0, 0, 0, 300);
  gradSky.addColorStop(0, '#87CEEB');
  gradSky.addColorStop(1, '#E0F6FF');
  ctx.fillStyle = gradSky;
  ctx.fillRect(0, 0, 960, 300);

  // Distant fence / trees
  ctx.fillStyle = '#6B8E23';
  ctx.fillRect(0, 240, 960, 60);
  ctx.fillStyle = '#556B2F';
  for (let i = 0; i < 960; i += 40) {
    ctx.beginPath();
    ctx.moveTo(i, 240);
    ctx.lineTo(i + 20, 200);
    ctx.lineTo(i + 40, 240);
    ctx.fill();
  }

  // Backyard ground (grass)
  const gradGround = ctx.createLinearGradient(0, 300, 0, 540);
  gradGround.addColorStop(0, '#7CFC00');
  gradGround.addColorStop(1, '#556B2F');
  ctx.fillStyle = gradGround;
  ctx.fillRect(0, 300, 960, 240);

  // Grass blades details (deterministic)
  ctx.strokeStyle = '#387c17';
  ctx.lineWidth = 1.5;
  for (let i = 0; i < 960; i += 15) {
    let h = 8 + (i % 7);
    ctx.beginPath();
    ctx.moveTo(i, 300);
    ctx.lineTo(i + (i % 3 - 1) * 3, 300 - h);
    ctx.stroke();
  }

  // Doghouse structure (positioned on the right side, angled)
  // House coordinates: center around x=720, y=320
  
  // Shadow under doghouse
  ctx.fillStyle = 'rgba(0,0,0,0.3)';
  ctx.beginPath();
  ctx.ellipse(720, 420, 140, 40, 0, 0, Math.PI * 2);
  ctx.fill();

  // Back/Side wall
  ctx.fillStyle = '#DEB887';
  ctx.fillRect(630, 260, 180, 140);

  // Roof
  ctx.fillStyle = '#8B4513';
  ctx.beginPath();
  ctx.moveTo(610, 260);
  ctx.setInt = true;
  ctx.lineTo(720, 170);
  ctx.lineTo(830, 260);
  ctx.closePath();
  ctx.fill();

  // Roof overhang shadow
  ctx.fillStyle = '#A0522D';
  ctx.beginPath();
  ctx.moveTo(630, 260);
  ctx.lineTo(720, 185);
  ctx.lineTo(810, 260);
  ctx.closePath();
  ctx.fill();

  // Doorway cutout (dark interior)
  ctx.fillStyle = '#2d1810';
  ctx.fillRect(685, 300, 70, 100);

  // Doorway arch
  ctx.beginPath();
  ctx.arc(720, 300, 35, Math.PI, 0, false);
  ctx.fill();
}

// Draw single bone item
function drawBone(x, y, angle, scale) {
  ctx.save();
  ctx.translate(x, y);
  ctx.rotate(angle);
  ctx.scale(scale, scale);

  ctx.fillStyle = '#F5F5DC';
  ctx.strokeStyle = '#D2B48C';
  ctx.lineWidth = 2;

  // Center shaft
  ctx.beginPath();
  ctx.roundRect(-15, -4, 30, 8, 4);
  ctx.fill();
  ctx.stroke();

  // Knobs
  [-15, 15].forEach(kx => {
    [-6, 6].forEach(ky => {
      ctx.beginPath();
      ctx.arc(kx, ky, 5, 0, Math.PI * 2);
      ctx.fillStyle = '#FFF8DC';
      ctx.fill();
      ctx.stroke();
    });
  });

  ctx.restore();
}

// Draw Pomeranian dog with procedural furry silhouette and articulation
function drawPomeranian(state) {
  const { x, y, bodyAngle, headAngle, legAnim, tailWag, inHouse, hasBone, facingFront } = state;

  ctx.save();
  ctx.translate(x, y);

  // Contact shadow
  if (!inHouse) {
    ctx.fillStyle = 'rgba(0,0,0,0.35)';
    ctx.beginPath();
    ctx.ellipse(0, 45, 35, 12, 0, 0, Math.PI * 2);
    ctx.fill();
  }

  // If fully inside house (t between 4.5 and 6), clip or don't draw fully
  if (inHouse && !facingFront) {
    // Only rear visible
    ctx.restore();
    return;
  }

  // Fur base color (Orange/Cream double coat)
  const furGradient = ctx.createRadialGradient(0, 0, 10, 0, 0, 50);
  furGradient.addColorStop(0, '#FF8C00');
  furGradient.addColorStop(0.7, '#FFA500');
  furGradient.addColorStop(1, '#FFD700');

  if (facingFront) {
    // --- FRONT FACING PRESENTATION (10-12s) ---
    // Body ruff (large fluffy collar)
    ctx.fillStyle = '#FFA500';
    ctx.beginPath();
    ctx.arc(0, 10, 45, 0, Math.PI * 2);
    ctx.fill();

    // Chest fur details
    for (let i = 0; i < 12; i++) {
      let angle = (i / 12) * Math.PI * 2;
      ctx.beginPath();
      ctx.moveTo(Math.cos(angle) * 35, 10 + Math.sin(angle) * 35);
      ctx.lineTo(Math.cos(angle) * 52, 15 + Math.sin(angle) * 52);
      ctx.strokeStyle = '#FF8C00';
      ctx.lineWidth = 4;
      ctx.stroke();
    }

    // Head
    ctx.beginPath();
    ctx.arc(0, -15, 25, 0, Math.PI * 2);
    ctx.fillStyle = '#FFA500';
    ctx.fill();
    ctx.stroke();

    // Small upright triangular ears
    [[ -18, -32, -0.2 ], [ 18, -32, 0.2 ]].forEach(([ex, ey, rot]) => {
      ctx.save();
      ctx.translate(ex, ey);
      ctx.rotate(rot);
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.lineTo(-8, -15);
      ctx.lineTo(8, -15);
      ctx.closePath();
      ctx.fillStyle = '#FF8C00';
      ctx.fill();
      ctx.fillStyle = '#FFB6C1'; // inner ear
      ctx.beginPath();
      ctx.moveTo(0, -3);
      ctx.lineTo(-4, -12);
      ctx.lineTo(4, -12);
      ctx.closePath();
      ctx.fill();
      ctx.restore();
    });

    // Muzzle & nose
    ctx.beginPath();
    ctx.arc(0, -8, 10, 0, Math.PI * 2);
    ctx.fillStyle = '#FFE4B5';
    ctx.fill();

    ctx.beginPath();
    ctx.arc(0, -10, 4, 0, Math.PI * 2);
    ctx.fillStyle = '#222';
    ctx.fill();

    // Dark expressive eyes
    [-10, 10].forEach(ex => {
      ctx.beginPath();
      ctx.arc(ex, -20, 3.5, 0, Math.PI * 2);
      ctx.fillStyle = '#111';
      ctx.fill();
      // highlight
      ctx.beginPath();
      ctx.arc(ex - 1, -21, 1, 0, Math.PI * 2);
      ctx.fillStyle = '#FFF';
      ctx.fill();
    });

    // Front paws planted
    [-18, 18].forEach(px => {
      ctx.beginPath();
      ctx.arc(px, 40, 7, 0, Math.PI * 2);
      ctx.fillStyle = '#FFA500';
      ctx.fill();
    });

    // Holding bone in mouth proudly
    if (hasBone) {
      drawBone(0, -5, 0, 1.2);
    }

  } else {
    // --- SIDE / PROFILE VIEW (Running, backing out) ---
    ctx.rotate(bodyAngle);

    // Plumed tail curled over back
    ctx.save();
    ctx.translate(-30, -15);
    ctx.rotate(-0.5 + Math.sin(tailWag) * 0.1);
    ctx.beginPath();
    ctx.arc(0, 0, 20, 0, Math.PI * 2);
    ctx.fillStyle = '#FF8C00';
    ctx.fill();
    ctx.restore();

    // Body (compact oval with fur spikes)
    ctx.beginPath();
    ctx.ellipse(0, 10, 32, 22, 0, 0, Math.PI * 2);
    ctx.fillStyle = furGradient;
    ctx.fill();

    // Neck ruff
    ctx.beginPath();
    ctx.arc(18, 0, 18, 0, Math.PI * 2);
    ctx.fillStyle = '#FFA500';
    ctx.fill();

    // Legs (articulated)
    const l1 = Math.sin(legAnim) * 12;
    const l2 = Math.cos(legAnim) * 12;

    // Back legs
    ctx.strokeStyle = '#FF8C00';
    ctx.lineWidth = 8;
    ctx.lineCap = 'round';
    ctx.beginPath();
    ctx.moveTo(-15, 20);
    ctx.lineTo(-15 + l1, 42);
    ctx.stroke();

    // Front legs
    ctx.beginPath();
    ctx.moveTo(15, 20);
    ctx.lineTo(15 - l1, 42);
    ctx.stroke();

    // Head
    ctx.save();
    ctx.translate(25, -5);
    ctx.rotate(headAngle);

    // Head base
    ctx.beginPath();
    ctx.arc(0, 0, 16, 0, Math.PI * 2);
    ctx.fillStyle = '#FFA500';
    ctx.fill();

    // Snout
    ctx.beginPath();
    ctx.moveTo(10, -2);
    ctx.lineTo(20, 2);
    ctx.lineTo(10, 6);
    ctx.closePath();
    ctx.fillStyle = '#FFA500';
    ctx.fill();

    // Nose
    ctx.beginPath();
    ctx.arc(19, 2, 3, 0, Math.PI * 2);
    ctx.fillStyle = '#111';
    ctx.fill();

    // Ear
    ctx.beginPath();
    ctx.moveTo(-2, -12);
    ctx.lineTo(6, -24);
    ctx.lineTo(10, -12);
    ctx.closePath();
    ctx.fillStyle = '#FF8C00';
    ctx.fill();

    // Eye
    ctx.beginPath();
    ctx.arc(2, -3, 3, 0, Math.PI * 2);
    ctx.fillStyle = '#111';
    ctx.fill();

    // Bone in mouth if carrying
    if (hasBone) {
      drawBone(18, 5, 0.2, 1.0);
    }

    ctx.restore();
  }

  ctx.restore();
}

// Main render frame function as per exact delivery contract
window.renderFrame = function(timeSeconds) {
  let t = Math.max(0, Math.min(12, timeSeconds));

  ctx.clearRect(0, 0, 960, 540);

  // Draw background and doghouse
  drawEnvironment(t);

  // Bone initial position inside the doghouse: x=720, y=360
  let boneX = 720;
  let boneY = 360;
  let boneVisible = true;
  let hasBone = false;

  // Dog state variables
  let dogX = 150;
  let dogY = 370;
  let bodyAngle = 0;
  let headAngle = 0;
  let legAnim = 0;
  let tailWag = 0;
  let inHouse = false;
  let facingFront = false;

  if (t < 1.0) {
    // 0-1s: Establish backyard, dog notices goal
    dogX = 150;
    legAnim = 0;
    headAngle = -0.1;
  } else if (t >= 1.0 && t < 3.0) {
    // 1-3s: Run toward doghouse (x: 150 -> 660)
    let p = (t - 1.0) / 2.0;
    dogX = 150 + p * 510;
    legAnim = t * 15;
    tailWag = t * 10;
    headAngle = Math.sin(t * 8) * 0.05;
  } else if (t >= 3.0 && t < 4.5) {
    // 3-4.5s: Enters through doorway (x: 660 -> 710)
    let p = (t - 3.0) / 1.5;
    dogX = 660 + p * 50;
    legAnim = t * 10;
    inHouse = true;
  } else if (t >= 4.5 && t < 6.0) {
    // 4.5-6s: Lowers head, grips bone inside house
    dogX = 715;
    inHouse = true;
    hasBone = true;
    boneVisible = false; // gripped in mouth
  } else if (t >= 6.0 && t < 8.0) {
    // 6-8s: Backs out through doorway with bone (x: 715 -> 580)
    let p = (t - 6.0) / 2.0;
    dogX = 715 - p * 135;
    legAnim = t * 15;
    hasBone = true;
    boneVisible = false;
    bodyAngle = Math.PI; // facing backward while backing out
  } else if (t >= 8.0 && t < 10.0) {
    // 8-10s: Smooth turn to face viewer (pivot at x=580)
    dogX = 580;
    dogY = 370;
    let p = (t - 8.0) / 2.0;
    // interpolate bodyAngle from PI to 0 or switch to frontal view
    if (p < 0.5) {
      bodyAngle = Math.PI * (1 - p * 2);
    } else {
      facingFront = true;
    }
    hasBone = true;
    boneVisible = false;
  } else {
    // 10-12s: Clear front-facing presentation
    dogX = 580;
    dogY = 370;
    facingFront = true;
    hasBone = true;
    boneVisible = false;
    // Subtle breathing / idle
    tailWag = Math.sin(t * 3) * 0.2;
  }

  // Draw stationary bone inside house if not yet grabbed
  if (boneVisible) {
    drawBone(boneX, boneY, 0.1, 1.0);
  }

  // Draw dog
  drawPomeranian({
    x: dogX,
    y: dogY,
    bodyAngle: bodyAngle,
    headAngle: headAngle,
    legAnim: legAnim,
    tailWag: tailWag,
    inHouse: inHouse,
    hasBone: hasBone,
    facingFront: facingFront
  });
};

// Initial frame render
window.renderFrame(0);
</script>

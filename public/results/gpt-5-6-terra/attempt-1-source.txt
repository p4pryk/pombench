  let u=(t-1)/2;px=245+440*ease(u);phase=(t-1)*10;
  let j=(t-1.55)/.78;
  if(j>0&&j<1){air=Math.sin(j*Math.PI)*57;py=413;phase=12+j*2}
 }else if(t<4.5){px=685+(t-3)*42;phase=20;enter=true}
 else if(t<6){px=748;phase=20;enter=true}

 else if(t<10){px=572-(t-8)*12;phase=41;bone=true;turn=smooth((t-8)/2)}
 else {px=548;phase=41;bone=true;turn=1}
 return {t,px,py,air,phase,tail:Math.sin(t*5),bone,enter,turn}

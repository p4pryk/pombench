    for(var i=0;i<2;i++){
     var nx=(e2.x-e1.x),ny=(e2.y-e1.y),ln=Math.hypot(nx,ny)||1;nx/=ln;ny/=ln;
     ctx.beginPath();ctx.arc(ends[i].x-ny*r*0.8,ends[i].y+nx*r*0.8,r,0,7);ctx.fill();
     ctx.beginPath();ctx.arc(ends[i].x+ny*r*0.8,ends[i].y-nx*r*0.8,r,0,7);ctx.fill();
    }
    ctx.strokeStyle='rgba(150,135,105,0.5)';ctx.lineWidth=1;
    ctx.beginPath();ctx.moveTo(e1.x,e1.y);ctx.lineTo(e2.x,e2.y);ctx.stroke();
   });
  }
 })();
 parts.sort(function(a,b){return a.d-b.d;});
 for(var i=0;i<parts.length;i++)parts[i].f();
}

function dogState(t){
 var KX=[[0,120],[1.0,122],[1.78,336],[1.9,372],[2.5,578],[2.72,614],[3.3,700],[4.4,788],[6.0,788],[8.0,676],[9.7,620],[12,620]];
 var x=key(t,KX);
 var fwd=sm(seg(t,8.2,10.0));
 var gy=groundY(Math.min(x,800))+fwd*34;
 var sc=scaleAt(Math.min(x,800))*(1+fwd*0.30);
 // jump
 var air=0,ja=0,airS=0;
 if(t>1.88&&t<2.54){var s=cl((t-1.9)/0.6,0,1);ja=4*74*s*(1-s);air=sm(cl((t-1.88)/0.08,0,1))*sm(cl((2.54-t)/0.09,0,1));airS=s;}
 // body height / crouch / landing compression
 var Hh=58;
 Hh-=10*sm(seg(t,1.62,1.88))*(1-sm(seg(t,1.88,1.98)));
 Hh-=13*sm(seg(t,2.46,2.62))*(1-sm(seg(t,2.64,2.9)));
 var grab=sm(seg(t,4.35,5.15))*(1-sm(seg(t,5.6,6.1)));
 Hh-=17*grab;
 if(t>=10)Hh+=1.4*Math.sin((t-10)*2.3);
 // body pitch
 var bp=0;
 bp+=0.30*Math.cos(Math.PI*cl((t-1.9)/0.6,0,1))*air;
 bp-=0.14*sm(seg(t,2.5,2.66))*(1-sm(seg(t,2.72,3.0)));
 bp+=0.10*sm(seg(t,1.6,1.88))*(1-sm(seg(t,1.9,2.0)));
 bp-=0.22*grab;
 // head pitch
 var hp=0.95;
 hp=lerp(hp,-1.45,grab);
 hp+=0.12*Math.sin(t*3.2)*(t<1?1:0);
 if(t>=10)hp+=0.05*Math.sin((t-10)*1.7);
 // head yaw lead
 var turn=sm(seg(t,8.2,9.9));
 var yaw=turn*Math.PI/2;
 var hy=0.42*Math.sin(Math.PI*cl((t-8.0)/2.0,0,1));
 if(t<1)hy=0.25*sm(seg(t,0.25,0.75));
 if(t>=10)hy+=0.07*Math.sin((t-10)*1.1);
 // mouth
 var mouth=0;
 mouth=sm(seg(t,4.5,5.0))*(1-sm(seg(t,5.25,5.55)));
 mouth=Math.max(mouth,0.18*(t>=5.55?1:0));
 if(t<1)mouth=Math.max(mouth,0.12);
 var hasBone=(t>=5.45);
 // gait phase from travelled distance
 var phase=(x-120)/(48*0.55);
 phase+=0.55*sm(seg(t,8.3,9.6));
 var run=1;
 if(t<1.0)run=0.0;
 if(t>=1.0&&t<1.2)run=sm(seg(t,1.0,1.2));
 if(t>3.3)run=1-0.5*sm(seg(t,3.3,4.4));
 if(t>9.6)run=0.0*0+ (1-sm(seg(t,9.6,10.0)));
 var footLift=0;
 var tv=Math.sin(t*2.4)*0.5+(t>=10?Math.sin((t-10)*2.0)*0.6:0);
 return {t:t,x:x,y:gy,sc:sc,yaw:yaw,hy:hy,hp:hp,bp:bp,H:Hh,ja:ja,air:air,airS:airS,
  phase:phase,run:run,footLift:footLift,mouth:mouth,hasBone:hasBone,tv:tv,show:true,fwd:fwd};
}

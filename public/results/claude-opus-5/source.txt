/* ---------- motion ---------- */
function posX(t){
  if(t<1)return -1.25;
  if(t<3)return -1.25+1.97*sstep(1,3,t);
  if(t<4.5)return 0.72+0.40*sstep(3,4.5,t);
  if(t<6)return 1.12;
  if(t<8)return 1.12-1.33*sstep(6,8,t);
  if(t<10)return -0.21-0.06*sstep(8,10,t);
  return -0.27;
}
function distT(t){
  var d=0;
  d+=1.97*sstep(1,3,Math.min(t,3));
  if(t>3)d+=0.40*sstep(3,4.5,Math.min(t,4.5));
  if(t>6)d+=1.33*sstep(6,8,Math.min(t,8));
  if(t>8)d+=0.06*sstep(8,10,Math.min(t,10))+0.30*sstep(8,10,Math.min(t,10));
  return d;
}
function headingT(t){return -1.5708*sstep(8.1,9.9,t);}
function state(t){
  t=cl(t,0,12);
  var hl=sstep(4.5,5.25,t);
  var up=sstep(5.75,6.4,t);
  var crouch=-0.075*(hl-up*0.85);
  var hp=0.66*hl-0.42*up-0.10*sstep(8.6,10,t);
  var mo=sstep(4.95,5.22,t)-sstep(5.30,5.5,t);
  var grab=sstep(5.28,5.52,t);
  var yaw=0;
  if(t<1)yaw=0.10*Math.sin(t*2.0);
  else if(t>10)yaw=0.06*Math.sin((t-10)*1.1);
  return {t:t,x:posX(t),h:headingT(t),crouch:crouch,hp:hp,mo:mo,grab:grab,yaw:yaw};
}

/* ---------- dog ---------- */
var LEGS=[{u:0.128,v:0.072,w:0.205,ph:0.0,fr:1},
          {u:0.128,v:-0.072,w:0.205,ph:0.5,fr:1},
          {u:-0.128,v:0.082,w:0.215,ph:0.52,fr:0},
          {u:-0.128,v:-0.082,w:0.215,ph:0.02,fr:0}];
var STRIDE=0.34,DUTY=0.6,L1=0.118,L2=0.118;

function buildDog(t){
  var S=state(t),dt=0.025;
  var xm=posX(cl(t-dt,0,12)),xp=posX(cl(t+dt,0,12));
  var spd=(xp-xm)/(2*dt);
  var hm=headingT(cl(t-dt,0,12)),hpP=headingT(cl(t+dt,0,12));
  var om=(hpP-hm)/(2*dt);
  var acc=(posX(cl(t+2*dt,0,12))-2*S.x+posX(cl(t-2*dt,0,12)))/(4*dt*dt);
  var pitch=cl(-0.035*acc,-0.12,0.12);
  var ph=distT(t)/STRIDE;
  var moving=Math.abs(spd)>0.05||Math.abs(om)>0.05;
  var bounce=moving?0.016*Math.min(1,Math.abs(spd)*0.9)*(0.5-0.5*Math.cos(2*TAU*ph)):0;
  bounce+=0.005*Math.sin(t*2.2)+S.crouch;

  var gx=S.x,gz=0.82;
  var P=V(gx,0.012+bounce,gz);
  var ch=Math.cos(S.h),sh=Math.sin(S.h),cpi=Math.cos(pitch),spi=Math.sin(pitch);
  var F=V(ch*cpi,spi,sh*cpi),Lv=V(-sh,0,ch),U=nrm(cross(Lv,F));
  function BP(u,v,w){return V(P.x+F.x*u+Lv.x*v+U.x*w,P.y+F.y*u+Lv.y*v+U.y*w,P.z+F.z*u+Lv.z*v+U.z*w);}

  // shadow
  var sc=proj(gx,0.002,gz);
  prim(sc.d+2.2,-9,function(){
    X.save();X.globalAlpha=0.3;X.fillStyle='#1b2a12';
    X.beginPath();X.ellipse(sc.x+12,sc.y+2,0.26*sc.s,0.09*sc.s,0,0,TAU);X.fill();X.restore();
  });

  // legs
  var strideU=cl(spd*DUTY*STRIDE/Math.max(0.3,Math.abs(spd))*Math.sign(spd||1)*Math.min(0.26,Math.abs(spd)*0.22),-0.26,0.26);
  strideU=cl(spd*0.20,-0.24,0.24);
  for(var i=0;i<4;i++){
    var lg=LEGS[i];
    var strideV=cl(om*lg.u*0.55,-0.12,0.12);
    var cyc=(ph+lg.ph)%1;if(cyc<0)cyc+=1;
    var fu,fv,lift=0;
    var amp=Math.hypot(strideU,strideV);
    if(!moving){fu=lg.u;fv=lg.v;}
    else if(cyc<DUTY){var ta=cyc/DUTY;fu=lg.u+strideU*(0.5-ta);fv=lg.v+strideV*(0.5-ta);}
    else{var tb=(cyc-DUTY)/(1-DUTY),e=0.5-0.5*Math.cos(Math.PI*tb);
      fu=lg.u+strideU*(-0.5+e);fv=lg.v+strideV*(-0.5+e);
      lift=Math.sin(Math.PI*tb)*Math.min(0.072,0.30*amp);}
    var hip=BP(lg.u,lg.v,lg.w);
    var foot=V(gx+F.x*fu+Lv.x*fv,lift+0.028,gz+F.z*fu+Lv.z*fv);
    var d=sub(foot,hip),dist=len3(d),mx=L1+L2-0.006;
    if(dist>mx){d=nrm(d);foot=add(hip,d,mx);dist=mx;d=sub(foot,hip);}
    if(dist<0.085){d=nrm(d);foot=add(hip,d,0.085);dist=0.085;d=sub(foot,hip);}
    var mid=V(hip.x+d.x*0.5,hip.y+d.y*0.5,hip.z+d.z*0.5);
    var perp=nrm(cross(Lv,d));
    var kk=Math.sqrt(Math.max(0.0002,L1*L1-dist*dist*0.25));
    var knee=add(mid,perp,kk*(lg.fr?-1:1));
    var seed=120+i*41,sd=lg.v>0?1:0.87;
    function cc(c){return [c[0]*sd,c[1]*sd,c[2]*sd];}
    for(var k=0;k<=3;k++){
      var tt=k/3;
      var pA=V(hip.x+(knee.x-hip.x)*tt,hip.y+(knee.y-hip.y)*tt,hip.z+(knee.z-hip.z)*tt);
      ball(pA,0.050-0.017*tt,cc(OR_L),cc(OR),cc(OR_D),0.45,seed+k);
    }
    for(var k3=1;k3<=3;k3++){
      var t2=k3/3;
      var pB=V(knee.x+(foot.x-knee.x)*t2,knee.y+(foot.y-knee.y)*t2,knee.z+(foot.z-knee.z)*t2);
      ball(pB,0.030-0.006*t2,cc(mix(OR_L,CR_L,t2*0.7)),cc(mix(OR,CR,t2*0.7)),cc(mix(OR_D,CR_D,t2*0.55)),0.26,seed+11+k3);
    }
    ball(V(foot.x,foot.y-0.006,foot.z),0.029,cc(CR_L),cc(CR),cc(CR_D),0.18,seed+31);
    var fs=proj(foot.x,0.003,foot.z);
    prim(fs.d+1.6,-9,function(){X.save();X.globalAlpha=0.28;X.fillStyle='#17260f';
      X.beginPath();X.ellipse(fs.x+3,fs.y,0.05*fs.s,0.022*fs.s,0,0,TAU);X.fill();X.restore();});
  }

  // body
  var body=[[-0.175,0.192,0.085],[-0.115,0.212,0.100],[-0.030,0.224,0.106],[0.055,0.224,0.104],[0.125,0.216,0.097]];
  for(var b=0;b<body.length;b++)ball(BP(body[b][0],0,body[b][1]),body[b][2],OR_L,OR,OR_D,0.44,20+b*13);
  ball(BP(0.095,0,0.150),0.070,CR_L,CR,CR_D,0.34,61);
  ball(BP(0.005,0,0.145),0.068,mix(CR_L,OR_L,0.45),mix(CR,OR,0.4),CR_D,0.34,62);
  // ruff
  ball(BP(0.175,0,0.250),0.112,OR_L,OR,OR_D,0.58,70);
  ball(BP(0.185,0.058,0.212),0.082,mix(CR_L,OR_L,0.3),mix(CR,OR,0.3),CR_D,0.52,71);
  ball(BP(0.185,-0.058,0.212),0.082,mix(CR_L,OR_L,0.3),mix(CR,OR,0.3),CR_D,0.52,72);
  ball(BP(0.195,0,0.160),0.070,CR_L,CR,CR_D,0.46,73);

  // tail plume over back
  var tb=BP(-0.175,0,0.215);
  var wag=0.09*Math.sin(t*3.1)+(moving?0.12*Math.sin(ph*TAU*0.5):0);
  for(var ti=0;ti<7;ti++){
    var a=ti/6;
    var u=-0.175+0.185*a*a, w=0.215+0.115*Math.sin(a*2.1)+0.03*a;
    var v=0.02+0.075*a*Math.sin(wag*3+a*0.6);
    var r=0.045+0.036*Math.sin(a*2.6);
    ball(BP(u,v,w),r,CR_L,mix(OR_L,CR,0.5),mix(OR_D,CR_D,0.4),0.80,200+ti*17);
  }

  // head
  var neck=BP(0.168,0,0.252);
  var hy=S.h+S.yaw;
  var cy=Math.cos(hy),sy=Math.sin(hy),cq=Math.cos(S.hp),sq=Math.sin(S.hp);
  var fh=nrm(V(cy*cq,-sq,sy*cq));
  var sh2=nrm(cross(UP,fh));
  var uh=nrm(cross(fh,sh2));
  var head=add(add(neck,fh,0.135),uh,0.040);
  ball(head,0.097,OR_L,OR,OR_D,0.42,300);
  ball(add(add(head,fh,-0.02),uh,-0.045),0.070,mix(CR_L,OR_L,0.4),mix(CR,OR,0.45),CR_D,0.48,301);
  // cheek ruff
  ball(add(add(head,sh2,0.065),fh,-0.025),0.062,CR_L,mix(CR,OR,0.35),CR_D,0.62,302);
  ball(add(add(head,sh2,-0.065),fh,-0.025),0.062,CR_L,mix(CR,OR,0.35),CR_D,0.62,303);
  // muzzle
  var mz=add(add(head,fh,0.082),uh,-0.012);
  ball(mz,0.047,mix(CR_L,OR_L,0.55),mix(CR,OR,0.5),CR_D,0.17,304);
  var nose=add(add(head,fh,0.118),uh,0.000);
  var qn=proj(nose.x,nose.y,nose.z);
  prim(qn.d-0.03,head.x,function(){
    var R=0.021*qn.s;
    var g=X.createRadialGradient(qn.x-R*0.3,qn.y-R*0.4,R*0.1,qn.x,qn.y,R);
    g.addColorStop(0,'#5a4c48');g.addColorStop(0.5,'#241c1a');g.addColorStop(1,'#120d0c');
    X.fillStyle=g;X.beginPath();X.ellipse(qn.x,qn.y,R,R*0.82,0,0,TAU);X.fill();
  });
  // mouth
  if(S.mo>0.02){
    var mp=add(add(head,fh,0.095),uh,-0.032);
    var qm=proj(mp.x,mp.y,mp.z);
    var moAmt=S.mo;
    prim(qm.d-0.028,head.x,function(){
      var R=0.030*qm.s;
      X.fillStyle='rgba(50,18,22,0.92)';
      X.beginPath();X.ellipse(qm.x,qm.y,R*0.85,R*0.75*moAmt+R*0.12,0,0,TAU);X.fill();
    });
  }
  // eyes
  for(var e=-1;e<=1;e+=2){
    (function(e){
      var ep=add(add(add(head,fh,0.060),sh2,0.040*e),uh,0.022);
      var q=proj(ep.x,ep.y,ep.z);
      prim(q.d-0.03,head.x,function(){
        var R=0.0145*q.s;
        X.fillStyle='#120c09';X.beginPath();X.ellipse(q.x,q.y,R,R*0.9,0,0,TAU);X.fill();
        X.fillStyle='rgba(255,255,255,0.75)';
        X.beginPath();X.arc(q.x-R*0.3,q.y-R*0.35,R*0.3,0,TAU);X.fill();
        X.strokeStyle='rgba(40,24,14,0.5)';X.lineWidth=Math.max(1,R*0.3);
        X.beginPath();X.arc(q.x,q.y,R*1.15,0,TAU);X.stroke();
      });
    })(e);
  }
  // ears
  for(var ei=-1;ei<=1;ei+=2){
    (function(ei){
      var flick=0.03*Math.sin(t*2.7+ei);
      var base=add(add(add(head,uh,0.055),sh2,0.055*ei),fh,-0.025);
      var tip=add(add(add(base,uh,0.085+flick),sh2,0.022*ei),fh,-0.012);
      var b1=add(add(base,fh,0.034),sh2,0.006*ei);
      var b2=add(add(base,fh,-0.030),sh2,-0.006*ei);
      var qa=proj(b1.x,b1.y,b1.z),qb=proj(b2.x,b2.y,b2.z),qt=proj(tip.x,tip.y,tip.z);
      prim(qt.d-0.02,head.x,function(){
        X.beginPath();X.moveTo(qa.x,qa.y);X.lineTo(qt.x,qt.y);X.lineTo(qb.x,qb.y);X.closePath();
        var g=X.createLinearGradient(qa.x,qa.y,qt.x,qt.y);
        g.addColorStop(0,rgb(OR));g.addColorStop(1,rgb(OR_D));
        X.fillStyle=g;X.fill();
        X.beginPath();X.moveTo((qa.x+qb.x)/2,(qa.y+qb.y)/2);
        X.lineTo(qt.x*0.82+qa.x*0.18,qt.y*0.82+qa.y*0.18);
        X.lineTo((qa.x*0.55+qb.x*0.45),(qa.y*0.55+qb.y*0.45));X.closePath();
        X.fillStyle='rgba(120,70,58,0.55)';X.fill();
        X.strokeStyle=rgb(CR,0.55);X.lineWidth=Math.max(1,qt.s*0.004);
        for(var f=0;f<4;f++){
          var tt=0.2+f*0.2;
          var px1=qa.x+(qb.x-qa.x)*tt,py1=qa.y+(qb.y-qa.y)*tt;
          X.beginPath();X.moveTo(px1,py1);X.lineTo(px1+(qt.x-px1)*0.5,py1+(qt.y-py1)*0.5);X.stroke();
        }
      });
    })(ei);
  }

  // bone
  var mouth=add(add(head,fh,0.108),uh,-0.030);
  var bc=lerpV(BONE_REST,mouth,S.grab);
  var baxis=nrm(lerpV(V(0,0,1),sh2,S.grab));
  var qbn=proj(bc.x,bc.y,bc.z);
  prim(qbn.d-0.05,bc.x,function(){drawBone(bc,baxis);});
  if(S.grab<0.15){
    var bs=proj(BONE_REST.x,0.004,BONE_REST.z);
    prim(bs.d+1.0,BONE_REST.x,function(){X.save();X.globalAlpha=0.4;X.fillStyle='#120b06';
      X.beginPath();X.ellipse(bs.x,bs.y,0.085*bs.s,0.028*bs.s,0,0,TAU);X.fill();X.restore();});
  }
}

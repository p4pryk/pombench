}

/* ---------- state ---------- */
var HEND=-1.60;
function raw(t){
  t=cl(t,0,12);
  var a;
  if(t<1)a=-1.5;
  else if(t<4.5)a=-1.5+2.38*sstep(1,4.5,t);
  else if(t<6)a=0.88;
  else if(t<8)a=0.88-1.43*sstep(6,8,t);
  else if(t<10)a=-0.55-0.23*sstep(8,10,t);
  else a=-0.78;
  var h=AY+(HEND-AY)*sstep(8,10,t);
  var hl=sstep(4.45,5.3,t)-sstep(5.55,6.2,t);
  var mo=sstep(4.75,5.1,t)-sstep(5.28,5.52,t);
  return {a:a,h:h,hl:hl,mo:mo,held:t>=5.4,t:t};
}

var LEGS=[
 {u:0.135,v:0.075,ph:0.0,fr:1},
 {u:0.135,v:-0.075,ph:0.5,fr:1},

var GT=0.42,DUTY=0.58,L1=0.117,L2=0.117,HIPH=0.215;

function drawDog(t){
  var S=raw(t),dt=0.02;
  var s0=raw(t-dt),s1=raw(t+dt);
  var spd=(s1.a-s0.a)/(2*dt);
  var sm=raw(t-2*dt),sp2=raw(t+2*dt);
  var acc=(sp2.a-2*S.a+sm.a)/(4*dt*dt);
  var om=(s1.h-s0.h)/(2*dt);
  var pitch=cl(0.05*acc,-0.13,0.13);
  var spdAbs=Math.abs(spd);
  var bounce=Math.min(0.03,0.028*spdAbs)*(0.5-0.5*Math.cos(2*TAU*(t/GT)));
  bounce+=0.006*Math.sin(t*2.1)-0.03*S.hl;
  var base=HP(S.a,0,0);
  var P=V(base.x,bounce,base.z);
  var ch=Math.cos(S.h),sh2=Math.sin(S.h),cpi=Math.cos(pitch),spi=Math.sin(pitch);
  var F=V(ch*cpi,spi,sh2*cpi);
  var L=V(-sh2,0,ch);
  var U=cross(L,F);
  function BP(u,v,w){return V(P.x+F.x*u+L.x*v+U.x*w, P.y+F.y*u+L.y*v+U.y*w, P.z+F.z*u+L.z*v+U.z*w);}

  // ---- legs ----
  var strideU=cl(spd*DUTY*GT,-0.30,0.30);
  for(var i=0;i<4;i++){
    var lg=LEGS[i];
    var strideV=cl(om*lg.u*DUTY*GT,-0.12,0.12);
    var cyc=(t/GT+lg.ph)%1; if(cyc<0)cyc+=1;
    var fu,fv,lift=0;
    if(cyc<DUTY){
      var tau=cyc/DUTY;
      fu=lg.u+strideU*(0.5-tau);fv=lg.v+strideV*(0.5-tau);
    }else{
      var tb=(cyc-DUTY)/(1-DUTY),e=0.5-0.5*Math.cos(Math.PI*tb);
      fu=lg.u+strideU*(-0.5+e);fv=lg.v+strideV*(-0.5+e);
      lift=Math.sin(Math.PI*tb)*Math.min(0.075,0.22*Math.hypot(strideU,strideV));
    }
    var hip=BP(lg.u,lg.v,HIPH+(lg.fr?0.01:0.015));
    var grd=HP(S.a,0,0);
    var foot=V(grd.x+F.x*fu+L.x*fv, lift+0.028, grd.z+F.z*fu+L.z*fv);
    var d=sub(foot,hip),dist=len3(d);
    if(dist>L1+L2-0.004)  {d=nrm(d);var k=(L1+L2-0.004);foot=add(hip,d,k);dist=k;d=sub(foot,hip);}
    if(dist<0.08){d=nrm(d);foot=add(hip,d,0.08);dist=0.08;d=sub(foot,hip);}
    var mid=V(hip.x+d.x*0.5,hip.y+d.y*0.5,hip.z+d.z*0.5);
    var perp=nrm(cross(L,d));
    var kk=Math.sqrt(Math.max(0.0001,L1*L1-dist*dist*0.25));
    var sgn=lg.fr?-1:1;
    var knee=add(mid,perp,kk*sgn);
    var seed=100+i*37;
    var shade=lg.v>0?1:0.86;
    function legCol(c){return [c[0]*shade,c[1]*shade,c[2]*shade];}
    for(var k2=0;k2<=3;k2++){
      var tt=k2/3;
      var pA=V(hip.x+(knee.x-hip.x)*tt,hip.y+(knee.y-hip.y)*tt,hip.z+(knee.z-hip.z)*tt);
      ball(pA,0.052-0.018*tt,legCol(OR_L),legCol(OR),legCol(OR_D),0.42,seed+k2);
    }
    for(var k3=1;k3<=3;k3++){
      var tt2=k3/3;
      var pB=V(knee.x+(foot.x-knee.x)*tt2,knee.y+(foot.y-knee.y)*tt2,knee.z+(foot.z-knee.z)*tt2);
      var c1=mix(OR,CR,tt2*0.65),c0=mix(OR_L,CR_L,tt2*0.65),c2=mix(OR_D,CR_D,tt2*0.5);
      ball(pB,0.030-0.006*tt2,legCol(c0),legCol(c1),legCol(c2),0.28,seed+10+k3);
    }
    ball(V(foot.x,foot.y-0.006,foot.z),0.030,legCol(CR_L),legCol(CR),legCol(CR_D),0.2,seed+30);
  }

  // ---- body ----
  var body=[
    [-0.175,0.195,0.086,OR,0.42],
    [-0.115,0.215,0.101,OR,0.42],
    [-0.030,0.228,0.107,OR,0.40],
    [ 0.055,0.228,0.105,OR,0.40],
    [ 0.135,0.220,0.098,OR,0.42]
  ];
  for(var b=0;b<body.length;b++){
    var bb=body[b],pt=BP(bb[0],0,bb[1]);
    ball(pt,bb[2],OR_L,OR,OR_D,bb[4],20+b*11);
  }
  // belly / chest cream
  ball(BP(0.10,0,0.155),0.072,CR_L,CR,CR_D,0.33,61);
  ball(BP(0.00,0,0.150),0.070,mix(CR_L,OR_L,0.4),mix(CR,OR,0.35),CR_D,0.33,62);
  // ruff
  ball(BP(0.185,0,0.255),0.118,OR_L,OR,OR_D,0.55,70);
  ball(BP(0.195,0.055,0.215),0.085,mix(CR_L,OR_L,0.3),mix(CR,OR,0.3),CR_D,0.5,71);
  ball(BP(0.195,-0.055,0.215),0.085,mix(CR_L,OR_L,0.3),mix(CR,OR,0.3),CR_D,0.5,72);
  ball(BP(0.205,0,0.165),0.072,CR_L,CR,CR_D,0.45,73);

  // ---- head ----

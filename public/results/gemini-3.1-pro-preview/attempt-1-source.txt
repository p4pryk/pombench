<canvas id="stage" width="960" height="540"></canvas>
<script>
const canvas = document.getElementById('stage');
const ctx = canvas.getContext('2d');

function drawScene(t) {
    ctx.clearRect(0, 0, 960, 540);

    // Sky and ground
    ctx.fillStyle = '#87CEEB';
    ctx.fillRect(0, 0, 960, 200);
    ctx.fillStyle = '#4CAF50';
    ctx.fillRect(0, 200, 960, 340);

    // Doghouse
    ctx.fillStyle = '#8B4513';
    ctx.fillRect(600, 100, 200, 200);
    ctx.fillStyle = '#654321';
    ctx.beginPath();
    ctx.moveTo(580, 100);
    ctx.lineTo(700, 20);
    ctx.lineTo(820, 100);
    ctx.fill();
    
    // Doorway
    ctx.fillStyle = '#000';
    ctx.fillRect(660, 180, 80, 120);

    // Puddle
    ctx.fillStyle = '#4682B4';
    ctx.beginPath();
    ctx.ellipse(400, 350, 60, 20, 0, 0, Math.PI * 2);
    ctx.fill();

    // Animation logic
    let dogX = 100;
    let dogY = 320;
    let dogScale = 1;
    let dogAngle = 0;
    let facingFront = false;
    let boneX = 700;
    let boneY = 280;
    let boneInMouth = false;

    if (t < 1) {
        // Establish
    } else if (t < 3) {
        // Run and jump
        let progress = (t - 1) / 2;
        dogX = 100 + progress * 560; // moves to 660
        
        // Jump over puddle (center at 400)
        let jumpDist = Math.abs(dogX - 400);
        if (jumpDist < 80) {
            dogY = 320 - Math.cos((jumpDist/80) * Math.PI/2) * 50;
        }
    } else if (t < 4.5) {
        // Enter house
        dogX = 660 + ((t - 3) / 1.5) * 40;
        dogY = 320;
    } else if (t < 6) {
        // Grab bone
        dogX = 700;
        boneInMouth = true;
    } else if (t < 8) {
        // Back out
        dogX = 700 - ((t - 6) / 2) * 100;
        boneInMouth = true;
    } else if (t < 10) {
        // Turn
        dogX = 600;
        boneInMouth = true;
        facingFront = true;
    } else {
        // Front pose
        dogX = 600;
        boneInMouth = true;
        facingFront = true;
    }

    if (boneInMouth) {
        boneX = dogX + (facingFront ? 0 : 40);
        boneY = dogY + (facingFront ? 20 : 0);
    }

    // Draw Bone
    ctx.fillStyle = '#FFF';
    ctx.beginPath();
    ctx.arc(boneX - 10, boneY, 5, 0, Math.PI*2);
    ctx.arc(boneX + 10, boneY, 5, 0, Math.PI*2);
    ctx.rect(boneX - 10, boneY - 3, 20, 6);
    ctx.fill();

    // Draw Dog (Pomeranian)
    ctx.save();
    ctx.translate(dogX, dogY);
    
    // Body (fluffy orange)
    ctx.fillStyle = '#FFA500';
    ctx.beginPath();
    ctx.arc(0, 0, 30, 0, Math.PI * 2);
    if (!facingFront) {
        ctx.arc(-20, 0, 25, 0, Math.PI * 2); // rear
    }
    ctx.fill();

    // Head
    ctx.fillStyle = '#FF8C00';
    if (facingFront) {
        ctx.beginPath();
        ctx.arc(0, -20, 20, 0, Math.PI * 2);
        ctx.fill();
        // Eyes
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(-8, -25, 3, 0, Math.PI*2);
        ctx.arc(8, -25, 3, 0, Math.PI*2);
        ctx.fill();
        // Snout
        ctx.fillStyle = '#FFE4B5';
        ctx.beginPath();
        ctx.arc(0, -15, 10, 0, Math.PI*2);
        ctx.fill();
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(0, -18, 2, 0, Math.PI*2);
        ctx.fill();
    } else {
        ctx.beginPath();
        ctx.arc(20, -15, 18, 0, Math.PI * 2);
        ctx.fill();
        // Eye
        ctx.fillStyle = '#000';
        ctx.beginPath();
        ctx.arc(25, -20, 3, 0, Math.PI*2);
        ctx.fill();
    }

    // Legs
    ctx.strokeStyle = '#D2691E';
    ctx.lineWidth = 6;
    if (facingFront) {
        ctx.beginPath(); ctx.moveTo(-15, 25); ctx.lineTo(-15, 45); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(15, 25); ctx.lineTo(15, 45); ctx.stroke();
    } else {
        // Simple leg animation
        let legOffset = Math.sin(t * 10) * 10;
        if (t < 1 || (t > 4.5 && t < 6) || t >= 10) legOffset = 0; // stop moving
        
        ctx.beginPath(); ctx.moveTo(10, 25); ctx.lineTo(10 + legOffset, 45); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(-10, 25); ctx.lineTo(-10 - legOffset, 45); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(25, 20); ctx.lineTo(25 + legOffset, 40); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(-25, 20); ctx.lineTo(-25 - legOffset, 40); ctx.stroke();
    }

    ctx.restore();
}

window.renderFrame = function(t) {
    drawScene(Math.max(0, Math.min(t, 12)));
};

window.renderFrame(0);
</script>
